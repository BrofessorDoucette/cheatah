5// Uniform engine adapters so every benchmark body and parity check is engine-generic:
6// cheatah::regex, std::regex, boost::regex and Google RE2 behind one static interface.
7//
8// TIMED RE2 rows use Re2Def — RE2's out-of-box configuration (UTF-8, leftmost-first), the
9// configuration real RE2 users get, so the timing comparison is honest. OFFSET/COUNT parity
10// is checked against Re2Longest (longest_match + Latin-1), the configuration that matches
11// cheatah's documented leftmost-LONGEST byte semantics; Re2Longest is never timed.
12// std::regex and boost::regex are leftmost-first, so they participate in boolean parity only.
13

28// The uniform static interface every adapter satisfies. `compile` may throw (std/boost report
29// bad patterns that way); `ok` is the post-construction validity check (cheatah/RE2 never throw).
30// `count_all` counts non-overlapping matches left to right, advancing one byte past an empty match.
31template <class E>
32concept Engine = requires(const typename E::Re& re, std::string_view t, std::size_t& b, std::size_t& e) {
33 { E::name } -> std::convertible_to<const char*>;
34 { E::compile(std::string{}) } -> std::same_as<typename E::Re>;
35 { E::ok(re) } -> std::same_as<bool>;
36 { E::search(re, t) } -> std::same_as<bool>;
37 { E::full(re, t) } -> std::same_as<bool>;
38 { E::find(re, t, b, e) } -> std::same_as<bool>;
39 { E::count_all(re, t) } -> std::same_as<std::size_t>;
40};
41
42struct Cheatah {
43 static constexpr const char* name = "cheatah";
44 using Re = cheatah::regex::Pattern;
45 static Re compile(const std::string& p) { return cheatah::regex::compile(p); }
46 static bool ok(const Re& r) { return r.ok; }
47 static bool search(const Re& r, std::string_view t) { return cheatah::regex::search(r, t); }
48 static bool full(const Re& r, std::string_view t) { return cheatah::regex::full_match(r, t); }
49 static bool find(const Re& r, std::string_view t, std::size_t& b, std::size_t& e) {
50 auto m = cheatah::regex::find(r, t);
51 if (!m.matched) return false;
52 b = m.begin;
53 e = m.end;
54 return true;
55 }
56 static std::size_t count_all(const Re& r, std::string_view t) {
57 std::size_t n = 0, pos = 0;
58 while (pos <= t.size()) {
59 auto m = cheatah::regex::find(r, t.substr(pos));
60 if (!m.matched) break;
61 ++n;
62 pos += (m.end > m.begin) ? m.end : m.begin + 1;
63 }
64 return n;
65 }
66};

68struct Std {
69 static constexpr const char* name = "std";
70 using Re = std::regex;
71 static Re compile(const std::string& p) { return std::regex(p, std::regex::optimize); }
72 static bool ok(const Re&) { return true; } // std::regex reports bad patterns by throwing
73 static bool search(const Re& r, std::string_view t) {
74 return std::regex_search(t.data(), t.data() + t.size(), r);
75 }
76 static bool full(const Re& r, std::string_view t) {
77 return std::regex_match(t.data(), t.data() + t.size(), r);
78 }
79 static bool find(const Re& r, std::string_view t, std::size_t& b, std::size_t& e) {
80 std::cmatch m;
81 if (!std::regex_search(t.data(), t.data() + t.size(), m, r)) return false;
82 b = static_cast<std::size_t>(m.position(0));
83 e = b + static_cast<std::size_t>(m.length(0));
84 return true;
85 }
86 static std::size_t count_all(const Re& r, std::string_view t) {
87 std::size_t n = 0;
88 for (std::cregex_iterator it(t.data(), t.data() + t.size(), r), end; it != end; ++it) ++n;
89 return n;
90 }
91};

118// An RE2 options provider: a name for benchmark rows plus the RE2::Options to compile with.
119template <class T>
120concept Re2OptsProvider = requires {
121 { T::name } -> std::convertible_to<const char*>;
122 { T::options() } -> std::same_as<RE2::Options>;
123};
124
125template <Re2OptsProvider Opts>
126struct Re2Engine {
127 static constexpr const char* name = Opts::name;
128 using Re = std::unique_ptr<re2::RE2>; // RE2 objects are non-copyable
129 static Re compile(const std::string& p) { return std::make_unique<re2::RE2>(p, Opts::options()); }
130 static bool ok(const Re& r) { return r->ok(); }
131 static bool search(const Re& r, std::string_view t) { return re2::RE2::PartialMatch(t, *r); }
132 static bool full(const Re& r, std::string_view t) { return re2::RE2::FullMatch(t, *r); }
133 static bool find(const Re& r, std::string_view t, std::size_t& b, std::size_t& e) {
134 std::string_view g;
135 if (!r->Match(t, 0, t.size(), re2::RE2::UNANCHORED, &g, 1)) return false;
136 b = static_cast<std::size_t>(g.data() - t.data());
137 e = b + g.size();
138 return true;
139 }
140 static std::size_t count_all(const Re& r, std::string_view t) {
141 std::size_t n = 0, pos = 0;
142 std::string_view g;
143 while (pos <= t.size() && r->Match(t, pos, t.size(), re2::RE2::UNANCHORED, &g, 1)) {
144 ++n;
145 const std::size_t b = static_cast<std::size_t>(g.data() - t.data());
146 pos = g.empty() ? b + 1 : b + g.size();
147 }
148 return n;
149 }
150};

172using Re2Def = Re2Engine<Re2DefOpts>; // timed: out-of-box RE2

175// Compile through the adapter, absorbing both failure styles (throwing and `!ok`).
176// Returns nullptr when the engine rejects the pattern.
177template <Engine E>
178std::shared_ptr<typename E::Re> try_compile(const std::string& pat) {
179 try {
180 auto re = std::make_shared<typename E::Re>(E::compile(pat));
181 if (!E::ok(*re)) return nullptr;
182 return re;
183 } catch (...) {
184 return nullptr;
185 }
186}
