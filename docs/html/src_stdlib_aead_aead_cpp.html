8// ChaCha20-Poly1305 AEAD (RFC 8439) from scratch. ChaCha20 is the 20-round ARX block
9// function keyed per RFC; Poly1305 is the one-time authenticator over r,s derived from
10// block 0. The AEAD construction MACs aad || pad || ciphertext || pad || lengths and
11// appends the 16-byte tag. The tag comparison on decrypt is constant-time.

19u32 rotl(u32 x, int n) { return (x << n) | (x >> (32 - n)); }

21// One ChaCha quarter round on four state words. @complexity O(1) @alloc none
22// @test CheatahAead.Rfc8439Encrypt
23void quarter(u32& a, u32& b, u32& c, u32& d) {
24 a += b; d ^= a; d = rotl(d, 16);
25 c += d; b ^= c; b = rotl(b, 12);
26 a += b; d ^= a; d = rotl(d, 8);
27 c += d; b ^= c; b = rotl(b, 7);
28}

30// The ChaCha20 block function: 64 bytes of keystream for (key, counter, nonce).
31// @complexity O(1) — 20 rounds @alloc none @test CheatahAead.Rfc8439Encrypt
32void chacha_block(const u32 key[8], u32 counter, const u32 nonce[3], unsigned char out[64]) {
33 u32 s[16] = {0x61707865, 0x3320646e, 0x79622d32, 0x6b206574, // "expand 32-byte k"
34 key[0], key[1], key[2], key[3], key[4], key[5], key[6], key[7],
35 counter, nonce[0], nonce[1], nonce[2]};
36 u32 w[16];
37 std::memcpy(w, s, sizeof w);
38 for (int round = 0; round < 10; ++round) { // 10 double rounds = 20 rounds
39 quarter(w[0], w[4], w[8], w[12]);
40 quarter(w[1], w[5], w[9], w[13]);
41 quarter(w[2], w[6], w[10], w[14]);
42 quarter(w[3], w[7], w[11], w[15]);
43 quarter(w[0], w[5], w[10], w[15]);
44 quarter(w[1], w[6], w[11], w[12]);
45 quarter(w[2], w[7], w[8], w[13]);
46 quarter(w[3], w[4], w[9], w[14]);
47 }
48 for (int i = 0; i < 16; ++i) {
49 const u32 v = w[i] + s[i];
50 out[4 * i] = static_cast<unsigned char>(v);
51 out[4 * i + 1] = static_cast<unsigned char>(v >> 8);
52 out[4 * i + 2] = static_cast<unsigned char>(v >> 16);
53 out[4 * i + 3] = static_cast<unsigned char>(v >> 24);
54 }
55}

57// XOR `data` with the ChaCha20 keystream starting at block `counter0`.
58// @complexity O(n) @alloc the returned string @test CheatahAead.Rfc8439Encrypt
59std::string chacha_xor(const u32 key[8], const u32 nonce[3], u32 counter0, std::string_view data) {
60 std::string out(data);
61 unsigned char block[64];
62 for (std::size_t off = 0; off < out.size(); off += 64) {
63 chacha_block(key, counter0 + static_cast<u32>(off / 64), nonce, block);
64 const std::size_t n = std::min<std::size_t>(64, out.size() - off);
65 for (std::size_t i = 0; i < n; ++i) {
66 out[off + i] = static_cast<char>(static_cast<unsigned char>(out[off + i]) ^ block[i]);
67 }
68 }
69 return out;

72// Poly1305 over the already-assembled MAC input, keyed by (r, s) from ChaCha block 0.
73// 26-bit limbs in u64 lanes — the standard portable shape. @complexity O(n) @alloc none
74// @test CheatahAead.Rfc8439Encrypt
75void poly1305(const unsigned char rs[32], std::string_view msg, unsigned char tag[16]) {
76 u32 r0, r1, r2, r3, r4;
77 { // load and clamp r (RFC 8439 §2.5)
78 u32 t[4];
79 std::memcpy(t, rs, 16);
80 t[0] &= 0x0fffffff; t[1] &= 0x0ffffffc; t[2] &= 0x0ffffffc; t[3] &= 0x0ffffffc;
81 r0 = t[0] & 0x3ffffff;
82 r1 = ((t[0] >> 26) | (t[1] << 6)) & 0x3ffffff;
83 r2 = ((t[1] >> 20) | (t[2] << 12)) & 0x3ffffff;
84 r3 = ((t[2] >> 14) | (t[3] << 18)) & 0x3ffffff;
85 r4 = (t[3] >> 8) & 0x3ffffff;
86 }
87 const u32 s1 = r1 * 5, s2 = r2 * 5, s3 = r3 * 5, s4 = r4 * 5;
88 u32 h0 = 0, h1 = 0, h2 = 0, h3 = 0, h4 = 0;
89
90 std::size_t pos = 0;
91 while (pos < msg.size()) {
92 unsigned char block[17] = {0};
93 const std::size_t n = std::min<std::size_t>(16, msg.size() - pos);
94 std::memcpy(block, msg.data() + pos, n);
95 block[n] = 1; // the high bit of the (padded) block
96 pos += n;
97
98 u32 t[4];
99 std::memcpy(t, block, 16);
100 h0 += t[0] & 0x3ffffff;
101 h1 += ((t[0] >> 26) | (t[1] << 6)) & 0x3ffffff;
102 h2 += ((t[1] >> 20) | (t[2] << 12)) & 0x3ffffff;
103 h3 += ((t[2] >> 14) | (t[3] << 18)) & 0x3ffffff;
104 h4 += (t[3] >> 8) | (static_cast<u32>(block[16]) << 24);
105
106 const u64 d0 = (u64)h0 * r0 + (u64)h1 * s4 + (u64)h2 * s3 + (u64)h3 * s2 + (u64)h4 * s1;
107 const u64 d1 = (u64)h0 * r1 + (u64)h1 * r0 + (u64)h2 * s4 + (u64)h3 * s3 + (u64)h4 * s2;
108 const u64 d2 = (u64)h0 * r2 + (u64)h1 * r1 + (u64)h2 * r0 + (u64)h3 * s4 + (u64)h4 * s3;
109 const u64 d3 = (u64)h0 * r3 + (u64)h1 * r2 + (u64)h2 * r1 + (u64)h3 * r0 + (u64)h4 * s4;
110 u64 d4 = (u64)h0 * r4 + (u64)h1 * r3 + (u64)h2 * r2 + (u64)h3 * r1 + (u64)h4 * r0;
111
112 u64 c = d0 >> 26; h0 = d0 & 0x3ffffff;
113 const u64 e1 = d1 + c; c = e1 >> 26; h1 = e1 & 0x3ffffff;
114 const u64 e2 = d2 + c; c = e2 >> 26; h2 = e2 & 0x3ffffff;
115 const u64 e3 = d3 + c; c = e3 >> 26; h3 = e3 & 0x3ffffff;
116 d4 += c; c = d4 >> 26; h4 = d4 & 0x3ffffff;
117 h0 += static_cast<u32>(c * 5); c = h0 >> 26; h0 &= 0x3ffffff;
118 h1 += static_cast<u32>(c);
119 }
120
121 // final reduction mod 2^130 - 5, then the trial subtraction (constant-time select)
122 u32 c = h1 >> 26; h1 &= 0x3ffffff; h2 += c;
123 c = h2 >> 26; h2 &= 0x3ffffff; h3 += c;
124 c = h3 >> 26; h3 &= 0x3ffffff; h4 += c;
125 c = h4 >> 26; h4 &= 0x3ffffff; h0 += c * 5;
126 c = h0 >> 26; h0 &= 0x3ffffff; h1 += c;
127
128 u32 g0 = h0 + 5; c = g0 >> 26; g0 &= 0x3ffffff;
129 u32 g1 = h1 + c; c = g1 >> 26; g1 &= 0x3ffffff;
130 u32 g2 = h2 + c; c = g2 >> 26; g2 &= 0x3ffffff;
131 u32 g3 = h3 + c; c = g3 >> 26; g3 &= 0x3ffffff;
132 const u32 g4 = h4 + c - (1u << 26);
133
134 const u32 mask = (g4 >> 31) - 1; // all-ones when h >= p (take g), zero otherwise
135 h0 = (h0 & ~mask) | (g0 & mask);
136 h1 = (h1 & ~mask) | (g1 & mask);
137 h2 = (h2 & ~mask) | (g2 & mask);
138 h3 = (h3 & ~mask) | (g3 & mask);
139 h4 = (h4 & ~mask) | (g4 & mask);
140
141 // h += s (the second 16 bytes of rs), little-endian, then serialize
142 const u64 f0 = ((h0) | (h1 << 26)) & 0xffffffffull;
143 const u64 f1 = ((h1 >> 6) | (h2 << 20)) & 0xffffffffull;
144 const u64 f2 = ((h2 >> 12) | (h3 << 14)) & 0xffffffffull;
145 const u64 f3 = ((h3 >> 18) | (h4 << 8)) & 0xffffffffull;
146 u32 s_part[4];
147 std::memcpy(s_part, rs + 16, 16);
148 const u64 f[4] = {f0, f1, f2, f3};
149 u64 carry_word = 0;
150 for (int i = 0; i < 4; ++i) {
151 const u64 sum = f[i] + s_part[i] + carry_word; // 32-bit lanes with carry between them
152 carry_word = sum >> 32;
153 tag[4 * i] = static_cast<unsigned char>(sum);
154 tag[4 * i + 1] = static_cast<unsigned char>(sum >> 8);
155 tag[4 * i + 2] = static_cast<unsigned char>(sum >> 16);
156 tag[4 * i + 3] = static_cast<unsigned char>(sum >> 24);
157 }
158}

160// Assemble the AEAD MAC input (aad || pad16 || ct || pad16 || len(aad) || len(ct)) and tag it.
161// @complexity O(n) @alloc the assembled buffer @test CheatahAead.Rfc8439Encrypt
162void aead_tag(const u32 key[8], const u32 nonce[3], std::string_view aad, std::string_view ct,
163 unsigned char tag[16]) {
164 unsigned char block0[64];
165 chacha_block(key, 0, nonce, block0); // rs = the first 32 bytes of block 0
166
167 std::string mac_input;
168 mac_input.reserve(aad.size() + ct.size() + 32);
169 mac_input.append(aad);
170 mac_input.append((16 - aad.size() % 16) % 16, '\0');
171 mac_input.append(ct);
172 mac_input.append((16 - ct.size() % 16) % 16, '\0');
173 unsigned char lens[16];
174 const u64 alen = aad.size(), clen = ct.size();
175 for (int i = 0; i < 8; ++i) {
176 lens[i] = static_cast<unsigned char>(alen >> (8 * i));
177 lens[8 + i] = static_cast<unsigned char>(clen >> (8 * i));
178 }
179 mac_input.append(reinterpret_cast<const char*>(lens), 16);
180 poly1305(block0, mac_input, tag);
181}

183// hex -> n bytes (false on malformed). @complexity O(n) @alloc none @test CheatahAead.RejectsTamper
184bool hex_bytes(std::string_view hex, unsigned char* out, std::size_t n) {
185 if (hex.size() != 2 * n) return false;
186 for (std::size_t i = 0; i < n; ++i) {
187 unsigned v = 0;
188 for (int k = 0; k < 2; ++k) {
189 const char ch = hex[2 * i + k];
190 v <<= 4;
191 if (ch >= '0' && ch <= '9') v |= static_cast<unsigned>(ch - '0');
192 else if (ch >= 'a' && ch <= 'f') v |= static_cast<unsigned>(ch - 'a' + 10);
193 else if (ch >= 'A' && ch <= 'F') v |= static_cast<unsigned>(ch - 'A' + 10);
194 else return false;
195 }
196 out[i] = static_cast<unsigned char>(v);
197 }
198 return true;
199}

232// Expand a 16-byte AES-128 key into 11 round keys (176 bytes). State/round-key byte layout is
233// column-major: byte (row r, col c) at index 4*c + r.
234void aes128_key_expand(const unsigned char key[16], unsigned char rk[176]) {
235 static const unsigned char rcon[10] = {0x01,0x02,0x04,0x08,0x10,0x20,0x40,0x80,0x1b,0x36};
236 std::memcpy(rk, key, 16);
237 int r = 0;
238 for (int i = 16; i < 176; i += 4) {
239 unsigned char t[4] = {rk[i - 4], rk[i - 3], rk[i - 2], rk[i - 1]};
240 if (i % 16 == 0) { // RotWord + SubWord + Rcon on the first word of each round key
241 const unsigned char a0 = t[0];
242 t[0] = static_cast<unsigned char>(kSbox[t[1]] ^ rcon[r++]);
243 t[1] = kSbox[t[2]];
244 t[2] = kSbox[t[3]];
245 t[3] = kSbox[a0];
246 }
247 for (int j = 0; j < 4; ++j) rk[i + j] = static_cast<unsigned char>(rk[i - 16 + j] ^ t[j]);
248 }
249}

251unsigned char xtime(unsigned char x) {
252 return static_cast<unsigned char>((x << 1) ^ ((x >> 7) * 0x1b)); // ·2 in GF(2^8)

255// Encrypt one 16-byte block (AES-128, 10 rounds).
256void aes128_encrypt_block(const unsigned char rk[176], const unsigned char in[16],
257 unsigned char out[16]) {
258 unsigned char s[16];
259 for (int i = 0; i < 16; ++i) s[i] = static_cast<unsigned char>(in[i] ^ rk[i]); // round 0
260 for (int round = 1; round <= 10; ++round) {
261 for (int i = 0; i < 16; ++i) s[i] = kSbox[s[i]]; // SubBytes
262 unsigned char t; // ShiftRows
263 t = s[1]; s[1] = s[5]; s[5] = s[9]; s[9] = s[13]; s[13] = t;
264 t = s[2]; s[2] = s[10]; s[10] = t; t = s[6]; s[6] = s[14]; s[14] = t;
265 t = s[15]; s[15] = s[11]; s[11] = s[7]; s[7] = s[3]; s[3] = t;
266 if (round != 10) { // MixColumns
267 for (int c = 0; c < 4; ++c) {
268 unsigned char* col = s + 4 * c;
269 const unsigned char a0 = col[0], a1 = col[1], a2 = col[2], a3 = col[3];
270 col[0] = static_cast<unsigned char>(xtime(a0) ^ (xtime(a1) ^ a1) ^ a2 ^ a3);
271 col[1] = static_cast<unsigned char>(a0 ^ xtime(a1) ^ (xtime(a2) ^ a2) ^ a3);
272 col[2] = static_cast<unsigned char>(a0 ^ a1 ^ xtime(a2) ^ (xtime(a3) ^ a3));
273 col[3] = static_cast<unsigned char>((xtime(a0) ^ a0) ^ a1 ^ a2 ^ xtime(a3));
274 }
275 }
276 const unsigned char* r_k = rk + 16 * round; // AddRoundKey
277 for (int i = 0; i < 16; ++i) s[i] = static_cast<unsigned char>(s[i] ^ r_k[i]);
278 }
279 std::memcpy(out, s, 16);
280}

282// GF(2^128) multiply (SP 800-38D): out = X · Y, big-endian bit order, reduction poly R = 0xe1<<120.
283void gf_mult(const unsigned char X[16], const unsigned char Y[16], unsigned char out[16]) {
284 unsigned char Z[16] = {0}, V[16];
285 std::memcpy(V, Y, 16);
286 for (int i = 0; i < 128; ++i) {
287 if ((X[i / 8] >> (7 - (i % 8))) & 1)
288 for (int j = 0; j < 16; ++j) Z[j] ^= V[j];
289 const unsigned char lsb = V[15] & 1;
290 for (int j = 15; j > 0; --j) V[j] = static_cast<unsigned char>((V[j] >> 1) | ((V[j - 1] & 1) << 7));
291 V[0] >>= 1;
292 if (lsb) V[0] ^= 0xe1;
293 }
294 std::memcpy(out, Z, 16);
295}

297// GHASH-accumulate the zero-padded `data` into Y (Y := (Y XOR block)·H per 16-byte block).
298void ghash_blocks(unsigned char Y[16], const unsigned char H[16], const unsigned char* p,
299 std::size_t n) {
300 for (std::size_t off = 0; off < n; off += 16) {
301 unsigned char b[16] = {0};
302 const std::size_t m = std::min<std::size_t>(16, n - off);
303 std::memcpy(b, p + off, m);
304 for (int i = 0; i < 16; ++i) Y[i] ^= b[i];
305 unsigned char t[16];
306 gf_mult(Y, H, t);
307 std::memcpy(Y, t, 16);
308 }
309}

311// The GCM tag: GHASH(AAD || pad || C || pad || [len(AAD)bits]_64 || [len(C)bits]_64) XOR E(J0).
312void gcm_tag(const unsigned char rk[176], const unsigned char H[16], const unsigned char J0[16],
313 std::string_view aad, std::string_view ct, unsigned char tag[16]) {
314 unsigned char Y[16] = {0};
315 ghash_blocks(Y, H, reinterpret_cast<const unsigned char*>(aad.data()), aad.size());
316 ghash_blocks(Y, H, reinterpret_cast<const unsigned char*>(ct.data()), ct.size());
317 unsigned char lb[16] = {0};
318 const u64 abits = static_cast<u64>(aad.size()) * 8, cbits = static_cast<u64>(ct.size()) * 8;
319 for (int i = 0; i < 8; ++i) {
320 lb[7 - i] = static_cast<unsigned char>(abits >> (8 * i));
321 lb[15 - i] = static_cast<unsigned char>(cbits >> (8 * i));
322 }
323 for (int i = 0; i < 16; ++i) Y[i] ^= lb[i];
324 unsigned char t[16];
325 gf_mult(Y, H, t);
326 unsigned char ej0[16];
327 aes128_encrypt_block(rk, J0, ej0);
328 for (int i = 0; i < 16; ++i) tag[i] = static_cast<unsigned char>(t[i] ^ ej0[i]);
329}

336// GCTR: XOR `data` in place with the AES-CTR keystream starting at counter `ctr` (advanced).
337void gctr(const unsigned char rk[176], unsigned char ctr[16], std::string& data) {
338 unsigned char ks[16];
339 for (std::size_t off = 0; off < data.size(); off += 16) {
340 aes128_encrypt_block(rk, ctr, ks);
341 const std::size_t n = std::min<std::size_t>(16, data.size() - off);
342 for (std::size_t i = 0; i < n; ++i)
343 data[off + i] = static_cast<char>(static_cast<unsigned char>(data[off + i]) ^ ks[i]);
344 inc32(ctr);
345 }

362std::string chacha20poly1305_encrypt(std::string_view key_hex, std::string_view nonce_hex,
363 std::string_view aad, std::string_view plaintext) {
364 u32 key[8], nonce[3];
365 if (!load_key_nonce(key_hex, nonce_hex, key, nonce)) return "";
366 std::string ct = chacha_xor(key, nonce, 1, plaintext); // counter starts at 1 (0 keys the MAC)
367 unsigned char tag[16];
368 aead_tag(key, nonce, aad, ct, tag);
369 ct.append(reinterpret_cast<const char*>(tag), 16);
370 return ct;
371}
372
373std::string chacha20poly1305_decrypt(std::string_view key_hex, std::string_view nonce_hex,
374 std::string_view aad, std::string_view ciphertext) {
375 u32 key[8], nonce[3];
376 if (!load_key_nonce(key_hex, nonce_hex, key, nonce) || ciphertext.size() < 16) return "";
377 const std::string_view ct = ciphertext.substr(0, ciphertext.size() - 16);
378 const std::string_view given = ciphertext.substr(ciphertext.size() - 16);
379 unsigned char tag[16];
380 aead_tag(key, nonce, aad, ct, tag);
381 unsigned char diff = 0; // constant-time compare: never early-exit on a mismatching byte
382 for (int i = 0; i < 16; ++i) {
383 diff |= tag[i] ^ static_cast<unsigned char>(given[i]);
384 }
385 if (diff != 0) return "";
386 return chacha_xor(key, nonce, 1, ct);
387}

389std::string aes128gcm_encrypt(std::string_view key_hex, std::string_view nonce_hex,
390 std::string_view aad, std::string_view plaintext) {
391 unsigned char kb[16], nb[12];
392 if (!hex_bytes(key_hex, kb, 16) || !hex_bytes(nonce_hex, nb, 12)) return "";
393 if (aes_gcm_use_hw()) return accel::aes128gcm_encrypt(kb, nb, aad, plaintext);
394 // --- portable reference path (identical output; this is how AES-128-GCM works) ---
395 unsigned char rk[176];
396 aes128_key_expand(kb, rk);
397 unsigned char H[16], zero[16] = {0};
398 aes128_encrypt_block(rk, zero, H); // hash subkey H = E(0)
399 unsigned char J0[16] = {0};
400 std::memcpy(J0, nb, 12);
401 J0[15] = 1; // J0 = nonce || 0x00000001
402 std::string ct(plaintext);
403 unsigned char ctr[16];
404 std::memcpy(ctr, J0, 16);
405 inc32(ctr); // CTR starts at inc32(J0)
406 gctr(rk, ctr, ct);
407 unsigned char tag[16];
408 gcm_tag(rk, H, J0, aad, ct, tag);
409 ct.append(reinterpret_cast<const char*>(tag), 16);
410 return ct;
411}

413std::string aes128gcm_decrypt(std::string_view key_hex, std::string_view nonce_hex,
414 std::string_view aad, std::string_view ciphertext) {
415 unsigned char kb[16], nb[12];
416 if (!hex_bytes(key_hex, kb, 16) || !hex_bytes(nonce_hex, nb, 12) || ciphertext.size() < 16)
417 return "";
418 if (aes_gcm_use_hw()) return accel::aes128gcm_decrypt(kb, nb, aad, ciphertext);
419 // --- portable reference path (identical output; this is how AES-128-GCM works) ---
420 unsigned char rk[176];
421 aes128_key_expand(kb, rk);
422 unsigned char H[16], zero[16] = {0};
423 aes128_encrypt_block(rk, zero, H);
424 unsigned char J0[16] = {0};
425 std::memcpy(J0, nb, 12);
426 J0[15] = 1;
427 const std::string_view ct = ciphertext.substr(0, ciphertext.size() - 16);
428 const std::string_view given = ciphertext.substr(ciphertext.size() - 16);
429 unsigned char tag[16];
430 gcm_tag(rk, H, J0, aad, ct, tag);
431 unsigned char diff = 0; // constant-time tag compare
432 for (int i = 0; i < 16; ++i) diff |= tag[i] ^ static_cast<unsigned char>(given[i]);
433 if (diff != 0) return "";
434 std::string pt(ct);
435 unsigned char ctr[16];
436 std::memcpy(ctr, J0, 16);
437 inc32(ctr);
438 gctr(rk, ctr, pt);
439 return pt;
440}
