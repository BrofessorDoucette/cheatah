1// System-level end-to-end test for the cheatah `linalg` stdlib module.
2//
3// Unlike the per-function compile-run tests (linalg_cr_test.cpp), this is ONE
4// cohesive linear-algebra program that exercises EVERY purr-callable routine in
5// stdlib/linalg/routines.hpp in a single run: the 7 matrix/vector products
6// (dot, vdot, inner, outer, matmul, matrix_power, kron), the 3 decompositions
7// (cholesky, qr, svd), the 4 eigen routines (eig, eigvals, eigh, eigvalsh), the
8// 6 "numbers" routines (norm, cond, det, matrix_rank, slogdet, trace), and the
9// 4 solve/invert routines (solve, lstsq, inv, pinv) — 24 functions total.
10//
11// It builds a few fixed operands (two vectors, an SPD diagonal matrix, a general
12// 2x2, and the 2x2 identity) and runs a sequence of operations over them. All
13// printed values are integer-valued or exactly representable so io.print's
14// formatting is deterministic. Struct-returning routines print a derived single
15// field (qr().r, svd().s, eig().values, eigh().values, slogdet().sign).
16#include "e2e_harness.hpp"
17
18TEST(StdlibE2E, Linalg) {
19 e2e::expect_e2e("linalg_sys", R"PURR(import io
20import ndarray
21import linalg
22
23# Fixed building blocks: a couple of small vectors and matrices.
24let u = ndarray.array([1.0, 2.0, 3.0])
25let v = ndarray.array([4.0, 5.0, 6.0])
26
27# A symmetric positive-definite diagonal matrix and a general 2x2.
28let d = ndarray.reshape(ndarray.array([4.0, 0.0, 0.0, 9.0]), [2, 2])
29let g = ndarray.reshape(ndarray.array([4.0, 3.0, 6.0, 3.0]), [2, 2])
30let id2 = ndarray.reshape(ndarray.array([1.0, 0.0, 0.0, 1.0]), [2, 2])
31
32# ---- products ----
33io.print(linalg.dot(u, v))
34io.print(linalg.vdot(u, v))
35io.print(linalg.inner(u, v))
36io.print(ndarray.to_string(linalg.outer(ndarray.array([1.0, 2.0]), ndarray.array([3.0, 4.0]))))
37io.print(ndarray.to_string(linalg.matmul(d, id2)))
38io.print(ndarray.to_string(linalg.matrix_power(d, 2)))
39io.print(ndarray.to_string(linalg.kron(id2, ndarray.reshape(ndarray.array([1.0, 2.0, 3.0, 4.0]), [2, 2]))))
40
41# ---- decompositions (print derived scalars / single fields) ----
42io.print(ndarray.to_string(linalg.cholesky(d)))
43io.print(ndarray.to_string(linalg.qr(d).r))
44io.print(ndarray.to_string(linalg.svd(d).s))
45
46# ---- eigenvalues ----
47io.print(ndarray.to_string(linalg.eig(d).values))
48io.print(ndarray.to_string(linalg.eigvals(d)))
49io.print(ndarray.to_string(linalg.eigh(d).values))
50io.print(ndarray.to_string(linalg.eigvalsh(d)))
51
52# ---- norms and numbers ----
53io.print(linalg.norm(ndarray.array([3.0, 4.0])))
54io.print(linalg.cond(d))
55io.print(linalg.det(g))
56io.print(linalg.matrix_rank(d))
57io.print(linalg.slogdet(d).sign)
58io.print(linalg.trace(g))
59
60# ---- solving and inverting ----
61io.print(ndarray.to_string(linalg.solve(g, ndarray.array([10.0, 12.0]))))
62io.print(ndarray.to_string(linalg.lstsq(id2, ndarray.reshape(ndarray.array([5.0, 7.0]), [2, 1]))))
63io.print(ndarray.to_string(linalg.inv(d)))
64io.print(ndarray.to_string(linalg.pinv(d)))
65)PURR",

67 "32\n"
68 "32\n"
69 "[[3, 4], [6, 8]]\n"
70 "[[4, 0], [0, 9]]\n"
71 "[[16, 0], [0, 81]]\n"
72 "[[1, 2, 0, 0], [3, 4, 0, 0], [0, 0, 1, 2], [0, 0, 3, 4]]\n"

92// Complex linear algebra exercised together end-to-end: build complex vectors and a
93// matrix, take bilinear (dot) and Hermitian (vdot) inner products, the conjugate
94// transpose, and a complex matmul. M·Mᴴ is Hermitian; vdot(a,a) is the real ‖a‖².
95TEST(StdlibE2E, LinalgComplex) {
96 e2e::expect_e2e("linalg_complex_sys", R"PURR(import io
97import ndarray
98import linalg
99
100let a = ndarray.complex(ndarray.array([1.0, 3.0]), ndarray.array([2.0, -1.0]))
101let b = ndarray.complex(ndarray.array([0.0, 2.0]), ndarray.array([1.0, 0.0]))
102let M = ndarray.reshape(ndarray.complex(ndarray.array([1.0, 2.0, 0.0, 3.0]), ndarray.array([1.0, 0.0, 0.0, -1.0])), [2, 2])
103let H = ndarray.reshape(ndarray.complex(ndarray.array([2.0, 1.0, 1.0, 3.0]), ndarray.array([0.0, 1.0, -1.0, 0.0])), [2, 2])
104
105io.print(linalg.dot(a, b))
106io.print(linalg.vdot(a, b))
107io.print(linalg.vdot(a, a))
108io.print(ndarray.to_string(linalg.conj_transpose(M)))
109io.print(ndarray.to_string(linalg.matmul(M, linalg.conj_transpose(M))))
110io.print(ndarray.to_string(linalg.eigvalsh(H)))
111io.print(ndarray.to_string(linalg.eigh(H).values))
112)PURR",
113 "4-1j\n"
114 "8+3j\n"
115 "15+0j\n"
116 "[[1-1j, 0+0j], [2+0j, 3+1j]]\n"
117 "[[6+0j, 6+2j], [6-2j, 10+0j]]\n"
118 "[4, 1]\n"
119 "[4, 1]\n");
120}
