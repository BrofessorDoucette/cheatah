9namespace dt = cheatah::datetime;

11namespace {
12// Components use local time; pin the process timezone to UTC so they're deterministic.
13void use_utc() {
14 setenv("TZ", "UTC", 1);
15 tzset();
16}
17} // namespace
18
19TEST(CheatahDatetime, ComponentsOfKnownEpoch) {
20 use_utc();
21 const double e = 1609459200.0; // 2021-01-01 00:00:00 UTC
22 EXPECT_EQ(dt::year(e), 2021);
23 EXPECT_EQ(dt::month(e), 1);
24 EXPECT_EQ(dt::day(e), 1);
25 EXPECT_EQ(dt::hour(e), 0);
26 EXPECT_EQ(dt::minute(e), 0);
27 EXPECT_EQ(dt::second(e), 0);
28 EXPECT_EQ(dt::weekday(e), 4); // 2021-01-01 was a Friday (Mon=0 … Sun=6)
29}
30
31TEST(CheatahDatetime, EpochZero) {
32 use_utc();
33 EXPECT_EQ(dt::year(0.0), 1970);
34 EXPECT_EQ(dt::month(0.0), 1);
35 EXPECT_EQ(dt::day(0.0), 1);
36 EXPECT_EQ(dt::weekday(0.0), 3); // 1970-01-01 was a Thursday
37}
38
39TEST(CheatahDatetime, TimeOfDayComponents) {
40 use_utc();
41 const double e = 1609459200.0 + 13 * 3600 + 37 * 60 + 5; // 13:37:05 UTC
42 EXPECT_EQ(dt::hour(e), 13);
43 EXPECT_EQ(dt::minute(e), 37);
44 EXPECT_EQ(dt::second(e), 5);
45}

47TEST(CheatahDatetime, Format) {
48 use_utc();
49 const double e = 1609459200.0;
50 EXPECT_EQ(dt::format(e, "%Y-%m-%d"), "2021-01-01");
51 EXPECT_EQ(dt::format(e, "%H:%M:%S"), "00:00:00");
52}

54TEST(CheatahDatetime, NowTodayUtcnowTimestamp) {
55 EXPECT_GT(dt::timestamp(), 1.6e9); // well after 2020
56 EXPECT_EQ(dt::now().size(), 19u); // "YYYY-MM-DD HH:MM:SS"
57 EXPECT_EQ(dt::today().size(), 10u); // "YYYY-MM-DD"
58 const std::string u = dt::utcnow(); // "YYYY-MM-DDTHH:MM:SSZ"
59 ASSERT_EQ(u.size(), 20u);
60 EXPECT_EQ(u.back(), 'Z');
61 EXPECT_EQ(u[10], 'T');
