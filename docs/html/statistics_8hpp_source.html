3/**
4 * @file statistics.hpp
5 * @brief cheatah `statistics` — descriptive statistics over numeric sequences,
6 * mirroring Python's `statistics` module. `import statistics` to use it.
7 *
8 * Header-only: every function is a template constrained by the `NumericRange`
9 * concept, so it accepts any iterable of arithmetic values (`list[float]`,
10 * `array[int]`, …) and is instantiated at the call site. Unit tests:
11 * `stdlib/tests/statistics_test.cpp`; the suite runs under AddressSanitizer (the
12 * `asan` preset) and Valgrind (`security/run-valgrind.sh`) on every QA-gate run.
13 *
14 * @note `n` below is the element count. Single-pass reductions are O(n) and
15 * allocation-free; `median` is the exception (it materializes and sorts a
16 * copy). Results are scalar `double`/`std::size_t` — no heap on return.
17 */

27/// NumericRange<R>: an iterable of arithmetic values (list[float], array[int], …).
28template <typename R>
29concept NumericRange =
30 std::ranges::input_range<R> && std::is_arithmetic_v<std::ranges::range_value_t<R>>;

32/**
33 * Sum of the elements.
34 * @param data the numeric range.
35 * @return Σ@p data as `double`.
36 * @note O(n) single pass; no heap.
37 * @test CheatahStatistics.SumCountMean
38 */
39template <NumericRange R>
40double sum(const R& data) {
41 double s = 0.0;
42 for (const auto& x : data) s += static_cast<double>(x);
43 return s;
44}

46/**

53template <NumericRange R>

63/**
64 * Arithmetic mean.
65 * @param data the numeric range.
66 * @return the mean, or 0.0 if empty.
67 * @note O(n) (two passes: count + sum); no heap.
68 * @test CheatahStatistics.SumCountMean
69 */
70template <NumericRange R>
71double mean(const R& data) {
72 const std::size_t n = count(data);
73 return n == 0 ? 0.0 : sum(data) / static_cast<double>(n);
74}

76/**
77 * Population variance (divide by N).
78 * @param data the numeric range.
79 * @return the variance, or 0.0 if empty.
80 * @note O(n); no heap.
81 * @test CheatahStatistics.PopulationVarianceAndStdev
82 */
83template <NumericRange R>
84double pvariance(const R& data) {
85 const double m = mean(data);
86 double s = 0.0;
87 std::size_t n = 0;
88 for (const auto& x : data) {
89 const double d = static_cast<double>(x) - m;
90 s += d * d;
91 ++n;
92 }
93 return n == 0 ? 0.0 : s / static_cast<double>(n);
94}
95/**
96 * Population standard deviation.
97 * @param data the numeric range.
98 * @return √pvariance(@p data).
99 * @note O(n); no heap.
100 * @test CheatahStatistics.PopulationVarianceAndStdev
101 */
102template <NumericRange R>
103double pstdev(const R& data) { return std::sqrt(pvariance(data)); }

105/**
106 * Sample variance (divide by N−1).
107 * @param data the numeric range.
108 * @return the variance, or 0.0 if fewer than 2 elements.
109 * @note O(n); no heap.
110 * @test CheatahStatistics.SampleVarianceAndStdev
111 */
112template <NumericRange R>
113double variance(const R& data) {
114 const double m = mean(data);
115 double s = 0.0;
116 std::size_t n = 0;
117 for (const auto& x : data) {
118 const double d = static_cast<double>(x) - m;
119 s += d * d;
120 ++n;
121 }
122 return n > 1 ? s / static_cast<double>(n - 1) : 0.0;
123}
124/**
125 * Sample standard deviation.
126 * @param data the numeric range.
127 * @return √variance(@p data).
128 * @note O(n); no heap.
129 * @test CheatahStatistics.SampleVarianceAndStdev
130 */
131template <NumericRange R>
132double stdev(const R& data) { return std::sqrt(variance(data)); }

134/**
135 * Median (mean of the two middle values when the count is even).
136 * @param data the numeric range.
137 * @return the median, or 0.0 if empty.
138 * @note O(n log n) — copies the elements into a vector and sorts; allocates a temporary
139 * `std::vector<double>`.
140 * @test CheatahStatistics.MedianOddAndEven
141 */
142template <NumericRange R>
143double median(const R& data) {
144 std::vector<double> v;
145 for (const auto& x : data) v.push_back(static_cast<double>(x));
146 if (v.empty()) return 0.0;
147 std::sort(v.begin(), v.end());
148 const std::size_t n = v.size();
149 return (n % 2 == 1) ? v[n / 2] : (v[n / 2 - 1] + v[n / 2]) / 2.0;
150}
