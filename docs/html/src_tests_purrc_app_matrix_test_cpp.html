tests/purrc/app_matrix_test.cpp

1// System-level "application" test: a small linear-algebra solver written in
2// cheatah that only passes if FOUR stdlib modules cooperate end to end —
3// * ndarray : array / reshape / sub (data construction + elementwise)
4// * linalg : solve / matmul / det / trace / norm (the numerical core)
5// * math : sqrt (scalar check on a linalg result)
6// * io : print (deterministic reporting)
7//
8// The program solves A x = b for a 3x3 system engineered to have the exact
9// integer solution x = [1, 2, 3] (so to_string is byte-stable), then verifies
10// the answer independently: it reconstructs A·x via linalg.matmul, forms the
11// residual r = A·x − b with ndarray.sub, and confirms ||r|| ≈ 0. det and trace
12// are printed as integer-valued cross-checks. The whole pipeline runs as a
13// compiled .purr module under the cheatah runtime; this test just compiles it
14// with purrc, runs it, and asserts the exact stdout (verified by hand-running
15// the program before hard-coding the expectation).
16
17#include "e2e_harness.hpp"
18
19TEST(SystemApps, LinearSolve) {
20 e2e::expect_e2e("app_matrix", R"PURR(import io
21import ndarray
22import linalg
23import math
24
25# A small linear-algebra "solver app": solve A x = b for a 3x3 system whose
26# exact solution is the integer vector x = [1, 2, 3], then independently verify
27# the result via det / trace and a residual norm computed by hand.
28let A = ndarray.reshape(ndarray.array([2.0, 1.0, 1.0, 1.0, 3.0, 2.0, 1.0, 0.0, 0.0]), [3, 3])
29let b = ndarray.reshape(ndarray.array([7.0, 13.0, 1.0]), [3, 1])
30
31# Solve, then reshape the solution to a column vector for matmul.
32let x = linalg.solve(A, b)
33let xcol = ndarray.reshape(x, [3, 1])
34
35# Residual r = A x - b; ||r|| should be ~0 if everything cooperated.
36let resid = ndarray.sub(linalg.matmul(A, xcol), b)
37let rnorm = linalg.norm(resid)
38
39# math.sqrt(rnorm*rnorm) == |rnorm|, exercising math on a linalg scalar.
40io.print("solution", ndarray.to_string(x))
41io.print("det", linalg.det(A))
42io.print("trace", linalg.trace(A))
43io.print("residual_ok", math.sqrt(rnorm * rnorm) < 0.000001)
44)PURR",
45 "solution [1, 2, 3]\n"
46 "det -1\n"
47 "trace 5\n"
48 "residual_ok True\n");
49}
