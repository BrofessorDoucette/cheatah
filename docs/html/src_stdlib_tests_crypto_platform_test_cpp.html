1// Platform-sensitive crypto validation. These are the checks most likely to expose an
2// architecture / OS / compiler-specific bug, gathered into one self-contained, self-reporting
3// test so the SAME correctness bar can be confirmed on each target (x86-64, ARM/Apple Silicon,
4// …) simply by running the suite there. It prints the detected arch/OS and whether the AES-GCM
5// hardware path is active, then:
6// • cross-checks the hardware path against the portable scalar reference, byte-for-byte,
7// across every block-boundary size with random keys/nonces/AAD/plaintext (the check that
8// a NEW hardware path on a new architecture must pass);
9// • round-trips AES-GCM and ChaCha20-Poly1305 (+ tamper rejection) over the same corpus;
10// • re-verifies canonical NIST/RFC known-answer vectors, so a miscompiled primitive on any
11// platform fails here regardless of which code path is taken.
12// The random corpus is generated from a FIXED seed, so a failure reproduces identically
13// everywhere. NOTE: results are only "verified" on a platform once this has actually been run
14// and passed there — see docs/performance.md ("Where the cryptography is verified").

29// Tiny deterministic PRNG (xorshift64) — reproducible corpus across platforms.
30struct Rng {
31 std::uint64_t s;
32 std::uint64_t next() {
33 s ^= s << 13;
34 s ^= s >> 7;
35 s ^= s << 17;
36 return s;
37 }
38 std::size_t below(std::size_t n) { return static_cast<std::size_t>(next() % n); }
39 std::string bytes(std::size_t n) {
40 std::string out(n, '\0');
41 for (char& c : out) c = static_cast<char>(next() & 0xFF);
42 return out;
43 }
44};

46std::string to_hex(const std::string& b) {
47 static constexpr char H[] = "0123456789abcdef";
48 std::string o;
49 o.reserve(b.size() * 2);
50 for (unsigned char c : b) {
51 o.push_back(H[c >> 4]);
52 o.push_back(H[c & 0xF]);
53 }
54 return o;
55}

57const char* arch() {
58#if defined(__aarch64__) || defined(_M_ARM64)
59 return "arm64";
60#elif defined(__arm__) || defined(_M_ARM)
61 return "arm";
62#elif defined(__x86_64__) || defined(_M_X64)
63 return "x86-64";
64#elif defined(__i386__) || defined(_M_IX86)
65 return "x86";

67 return "unknown";

89// Every size that straddles a block (16), the 4-wide (64) and 8-wide (128) group boundaries,
90// plus a few larger buffers — where a CTR/GHASH tail bug would hide.
91const std::vector<std::size_t> kSizes = {0, 1, 15, 16, 17, 31, 32, 63, 64, 65,
92 80, 127, 128, 129, 200, 255, 256, 512, 1000};

96TEST(CryptoPlatform, Report) {
97 std::printf("[crypto-platform] arch=%s os=%s | AES-GCM hardware path active: %s\n", arch(),
98 os_name(), ae::crypto_hardware_active() ? "YES (AES-NI + PCLMULQDQ)"
99 : "no (portable scalar reference)");
100 std::fflush(stdout);
101 // An x86 build MUST take the hardware path (else CPUID/dispatch is broken). On other
102 // architectures the scalar reference runs and this is informational only.
103 if (is_x86()) {
104 EXPECT_TRUE(ae::crypto_hardware_active())
105 << "x86 build is not using the AES-NI/PCLMULQDQ path";
106 }
107 SUCCEED();
108}

110// The key cross-architecture check: the hardware path and the portable scalar reference must
111// agree byte-for-byte, and both must round-trip, over the whole size corpus with random inputs.
112TEST(CryptoPlatform, AesGcmHardwareEqualsPortableAndRoundTrips) {
113 Rng r{0x1234567890abcdefULL};
114 for (std::size_t n : kSizes) {
115 for (int trial = 0; trial < 4; ++trial) {
116 const std::string key = to_hex(r.bytes(16));
117 const std::string nonce = to_hex(r.bytes(12));
118 const std::string aad = r.bytes(r.below(40));
119 const std::string pt = r.bytes(n);
120
121 ae::set_force_portable_crypto(false);
122 const std::string hw = ae::aes128gcm_encrypt(key, nonce, aad, pt);
123 ae::set_force_portable_crypto(true);
124 const std::string sw = ae::aes128gcm_encrypt(key, nonce, aad, pt);
125 ae::set_force_portable_crypto(false);
126
127 ASSERT_EQ(hw.size(), pt.size() + 16) << "arch=" << arch() << " size=" << n;
128 ASSERT_EQ(hw, sw) << "hardware != portable, arch=" << arch() << " size=" << n;
129 EXPECT_EQ(ae::aes128gcm_decrypt(key, nonce, aad, hw), pt); // hardware decrypt
130 ae::set_force_portable_crypto(true);
131 EXPECT_EQ(ae::aes128gcm_decrypt(key, nonce, aad, hw), pt); // portable decrypt
132 ae::set_force_portable_crypto(false);
133 }
134 }
135}

137TEST(CryptoPlatform, ChaCha20Poly1305RoundTripAndTamper) {
138 Rng r{0xfeedfacecafebeefULL};
139 for (std::size_t n : kSizes) {
140 const std::string key = to_hex(r.bytes(32));
141 const std::string nonce = to_hex(r.bytes(12));
142 const std::string aad = r.bytes(r.below(40));
143 const std::string pt = r.bytes(n);
144 const std::string ct = ae::chacha20poly1305_encrypt(key, nonce, aad, pt);
145 ASSERT_EQ(ct.size(), pt.size() + 16) << "arch=" << arch() << " size=" << n;
146 EXPECT_EQ(ae::chacha20poly1305_decrypt(key, nonce, aad, ct), pt);
147 std::string tampered = ct;
148 tampered[tampered.size() - 1] = static_cast<char>(tampered.back() ^ 0x01);
149 EXPECT_EQ(ae::chacha20poly1305_decrypt(key, nonce, aad, tampered), "");
150 }
151}

153// Canonical known-answer vectors — independent of which code path is taken, so a miscompiled
154// primitive on any platform is caught here too.
155TEST(CryptoPlatform, KnownAnswerVectors) {
156 // SHA-2 (NIST FIPS 180-4).
157 EXPECT_EQ(hl::sha256("abc"),
158 "ba7816bf8f01cfea414140de5dae2223b00361a396177a9cb410ff61f20015ad");
159 EXPECT_EQ(hl::sha512("abc"),
160 "ddaf35a193617abacc417349ae20413112e6fa4e89a97ea20a9eeee64b55d39a"
161 "2192992a274fc1a836ba3c23a3feebbd454d4423643ce80e2a9ac94fa54ca49f");
162
163 // AES-128-GCM (NIST GCM test case 4): K, IV, AAD, P -> C || T.
164 auto unhex = [](const std::string& h) {
165 auto nib = [](char c) { return c <= '9' ? c - '0' : (c | 0x20) - 'a' + 10; };
166 std::string o;
167 for (std::size_t i = 0; i + 1 < h.size(); i += 2)
168 o.push_back(static_cast<char>((nib(h[i]) << 4) | nib(h[i + 1])));
169 return o;
170 };
171 const std::string key = "feffe9928665731c6d6a8f9467308308";
172 const std::string iv = "cafebabefacedbaddecaf888";
173 const std::string aad = unhex("feedfacedeadbeeffeedfacedeadbeefabaddad2");
174 const std::string p = unhex(
175 "d9313225f88406e5a55909c5aff5269a86a7a9531534f7da2e4c303d8a318a721c3c0c95956809532fcf0e2"
176 "449a6b525b16aedf5aa0de657ba637b39");
177 const std::string ct = ae::aes128gcm_encrypt(key, iv, aad, p);
178 const std::string expected =
179 "42831ec2217774244b7221b784d0d49ce3aa212f2c02a4e035c17e2329aca12e21d514b25466931c7d8f6a5"
180 "aac84aa051ba30b396a0aac973d58e091" // ciphertext
181 "5bc94fbc3221a5db94fae95ae7121a47"; // 16-byte tag
182 EXPECT_EQ(to_hex(ct), expected);
183 EXPECT_EQ(ae::aes128gcm_decrypt(key, iv, aad, ct), p); // round trip
184}
