1// Copyright (c) 2026 BigBrain LLC. MIT-licensed (see LICENSE).
2// Original work; see ACKNOWLEDGMENTS.md for the open-source ideas we build upon.
3// Platform-sensitive crypto validation. These are the checks most likely to expose an
4// architecture / OS / compiler-specific bug, gathered into one self-contained, self-reporting
5// test so the SAME correctness bar can be confirmed on each target (x86-64, ARM/Apple Silicon,
6// …) simply by running the suite there. It prints the detected arch/OS and whether the AES-GCM
7// hardware path is active, then:
8// • cross-checks the hardware path against the portable scalar reference, byte-for-byte,
9// across every block-boundary size with random keys/nonces/AAD/plaintext (the check that
10// a NEW hardware path on a new architecture must pass);
11// • round-trips AES-GCM and ChaCha20-Poly1305 (+ tamper rejection) over the same corpus;
12// • re-verifies canonical NIST/RFC known-answer vectors, so a miscompiled primitive on any
13// platform fails here regardless of which code path is taken.
14// The random corpus is generated from a FIXED seed, so a failure reproduces identically
15// everywhere. NOTE: results are only "verified" on a platform once this has actually been run
16// and passed there — see docs/performance.md ("Where the cryptography is verified").

31// Tiny deterministic PRNG (xorshift64) — reproducible corpus across platforms.
32struct Rng {
33 std::uint64_t s;
34 std::uint64_t next() {
35 s ^= s << 13;
36 s ^= s >> 7;
37 s ^= s << 17;
38 return s;
39 }
40 std::size_t below(std::size_t n) { return static_cast<std::size_t>(next() % n); }
41 std::string bytes(std::size_t n) {
42 std::string out(n, '\0');
43 for (char& c : out) c = static_cast<char>(next() & 0xFF);
44 return out;
45 }
46};

48std::string to_hex(const std::string& b) {
49 static constexpr char H[] = "0123456789abcdef";
50 std::string o;
51 o.reserve(b.size() * 2);
52 for (unsigned char c : b) {
53 o.push_back(H[c >> 4]);
54 o.push_back(H[c & 0xF]);
55 }
56 return o;
57}

59const char* arch() {

61 return "arm64";
62#elif defined(__arm__) || defined(_M_ARM)
63 return "arm";

65 return "x86-64";

91// Every size that straddles a block (16), the 4-wide (64) and 8-wide (128) group boundaries,
92// plus a few larger buffers — where a CTR/GHASH tail bug would hide.
93const std::vector<std::size_t> kSizes = {0, 1, 15, 16, 17, 31, 32, 63, 64, 65,
94 80, 127, 128, 129, 200, 255, 256, 512, 1000};

98TEST(CryptoPlatform, Report) {
99 std::printf("[crypto-platform] arch=%s os=%s | AES-GCM hardware path active: %s\n", arch(),
100 os_name(), ae::crypto_hardware_active() ? "YES (AES-NI + PCLMULQDQ)"
101 : "no (portable scalar reference)");
102 std::fflush(stdout);
103 // An x86 build MUST take the hardware path (else CPUID/dispatch is broken). On other
104 // architectures the scalar reference runs and this is informational only.
105 if (is_x86()) {
106 EXPECT_TRUE(ae::crypto_hardware_active())
107 << "x86 build is not using the AES-NI/PCLMULQDQ path";
108 }
109 SUCCEED();
110}

112// The key cross-architecture check: the hardware path and the portable scalar reference must
113// agree byte-for-byte, and both must round-trip, over the whole size corpus with random inputs.
114TEST(CryptoPlatform, AesGcmHardwareEqualsPortableAndRoundTrips) {
115 Rng r{0x1234567890abcdefULL};
116 for (std::size_t n : kSizes) {
117 for (int trial = 0; trial < 4; ++trial) {
118 const std::string key = to_hex(r.bytes(16));
119 const std::string nonce = to_hex(r.bytes(12));
120 const std::string aad = r.bytes(r.below(40));
121 const std::string pt = r.bytes(n);
122
123 ae::set_force_portable_crypto(false);
124 const std::string hw = ae::aes128gcm_encrypt(key, nonce, aad, pt);
125 ae::set_force_portable_crypto(true);
126 const std::string sw = ae::aes128gcm_encrypt(key, nonce, aad, pt);
127 ae::set_force_portable_crypto(false);
128
129 ASSERT_EQ(hw.size(), pt.size() + 16) << "arch=" << arch() << " size=" << n;
130 ASSERT_EQ(hw, sw) << "hardware != portable, arch=" << arch() << " size=" << n;
131 EXPECT_EQ(ae::aes128gcm_decrypt(key, nonce, aad, hw), pt); // hardware decrypt
132 ae::set_force_portable_crypto(true);
133 EXPECT_EQ(ae::aes128gcm_decrypt(key, nonce, aad, hw), pt); // portable decrypt
134 ae::set_force_portable_crypto(false);
135 }
136 }
137}

139TEST(CryptoPlatform, ChaCha20Poly1305RoundTripAndTamper) {
140 Rng r{0xfeedfacecafebeefULL};
141 for (std::size_t n : kSizes) {
142 const std::string key = to_hex(r.bytes(32));
143 const std::string nonce = to_hex(r.bytes(12));
144 const std::string aad = r.bytes(r.below(40));
145 const std::string pt = r.bytes(n);
146 const std::string ct = ae::chacha20poly1305_encrypt(key, nonce, aad, pt);
147 ASSERT_EQ(ct.size(), pt.size() + 16) << "arch=" << arch() << " size=" << n;
148 EXPECT_EQ(ae::chacha20poly1305_decrypt(key, nonce, aad, ct), pt);
149 std::string tampered = ct;
150 tampered[tampered.size() - 1] = static_cast<char>(tampered.back() ^ 0x01);
151 EXPECT_EQ(ae::chacha20poly1305_decrypt(key, nonce, aad, tampered), "");
152 }
153}

155// Canonical known-answer vectors — independent of which code path is taken, so a miscompiled
156// primitive on any platform is caught here too.
157TEST(CryptoPlatform, KnownAnswerVectors) {
158 // SHA-2 (NIST FIPS 180-4).
159 EXPECT_EQ(hl::sha256("abc"),
160 "ba7816bf8f01cfea414140de5dae2223b00361a396177a9cb410ff61f20015ad");
161 EXPECT_EQ(hl::sha512("abc"),
162 "ddaf35a193617abacc417349ae20413112e6fa4e89a97ea20a9eeee64b55d39a"
163 "2192992a274fc1a836ba3c23a3feebbd454d4423643ce80e2a9ac94fa54ca49f");
164
165 // AES-128-GCM (NIST GCM test case 4): K, IV, AAD, P -> C || T.
166 auto unhex = [](const std::string& h) {
167 auto nib = [](char c) { return c <= '9' ? c - '0' : (c | 0x20) - 'a' + 10; };
168 std::string o;
169 for (std::size_t i = 0; i + 1 < h.size(); i += 2)
170 o.push_back(static_cast<char>((nib(h[i]) << 4) | nib(h[i + 1])));
171 return o;
172 };
173 const std::string key = "feffe9928665731c6d6a8f9467308308";
174 const std::string iv = "cafebabefacedbaddecaf888";
175 const std::string aad = unhex("feedfacedeadbeeffeedfacedeadbeefabaddad2");
176 const std::string p = unhex(
177 "d9313225f88406e5a55909c5aff5269a86a7a9531534f7da2e4c303d8a318a721c3c0c95956809532fcf0e2"
178 "449a6b525b16aedf5aa0de657ba637b39");
179 const std::string ct = ae::aes128gcm_encrypt(key, iv, aad, p);
180 const std::string expected =
181 "42831ec2217774244b7221b784d0d49ce3aa212f2c02a4e035c17e2329aca12e21d514b25466931c7d8f6a5"
182 "aac84aa051ba30b396a0aac973d58e091" // ciphertext
183 "5bc94fbc3221a5db94fae95ae7121a47"; // 16-byte tag
184 EXPECT_EQ(to_hex(ct), expected);
185 EXPECT_EQ(ae::aes128gcm_decrypt(key, iv, aad, ct), p); // round trip
186}
