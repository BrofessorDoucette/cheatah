5// ec_core.hpp — the width-generic short-Weierstrass ECDSA machinery shared by the `p256`
6// and `p384` modules. NOT a cheatah module itself: an internal implementation header the
7// two curve .cpp files include (purrc resolves modules by `<module>.hpp` name only, so
8// this file is invisible to `import`).
9//
10// Everything is templated on a WeierstrassCurve traits struct carrying the limb count and
11// the curve constants; the field/scalar Montgomery contexts are still DERIVED from the
12// modulus at startup (no hand-transcribed Montgomery magic). A value is uint64_t[kLimbs],
13// LEAST-significant limb first; points are Jacobian (X:Y:Z) with the curve a = -3 (true
14// of every NIST prime curve). The algorithms are limb-count-independent copies of the
15// battle-tested p256 versions — only bounds changed, from 4/256/32 to
16// kLimbs/kBits/kBytes.
17

31// The curve-traits concept every template below is constrained by: a NIST-style
32// short-Weierstrass curve (a = -3) over a prime field, its size in 64-bit limbs plus the
33// field prime P, group order N, coefficient B and base point (GX, GY) as little-endian
34// limb arrays.
35template <class C>
36concept WeierstrassCurve = requires {
37 { C::kLimbs } -> std::convertible_to<std::size_t>;
38 requires C::kLimbs >= 4 && C::kLimbs <= 8;
39 requires std::same_as<std::remove_cvref_t<decltype(C::P)>, std::array<u64, C::kLimbs>>;
40 requires std::same_as<std::remove_cvref_t<decltype(C::N)>, std::array<u64, C::kLimbs>>;
41 requires std::same_as<std::remove_cvref_t<decltype(C::B)>, std::array<u64, C::kLimbs>>;
42 requires std::same_as<std::remove_cvref_t<decltype(C::GX)>, std::array<u64, C::kLimbs>>;
43 requires std::same_as<std::remove_cvref_t<decltype(C::GY)>, std::array<u64, C::kLimbs>>;
44};

52inline constexpr std::size_t kBytes = C::kLimbs * 8; // big-endian byte size (32 / 48)

54// ---- plain multi-limb helpers -----------------------------------------------------------------
55template <WeierstrassCurve C>
56bool is_zero(const fe<C>& a) {
57 u64 acc = 0;
58 for (const u64 limb : a) acc |= limb;
59 return acc == 0;
60}
61template <WeierstrassCurve C>
62bool geq(const fe<C>& a, const fe<C>& b) { // a >= b
63 for (int i = static_cast<int>(C::kLimbs) - 1; i >= 0; --i)
64 if (a[i] != b[i]) return a[i] > b[i];
65 return true;
66}
67// a - b (mod 2^kBits), returns borrow.
68template <WeierstrassCurve C>
69u64 sub_borrow(fe<C>& r, const fe<C>& a, const fe<C>& b) {
70 u128 br = 0;
71 for (std::size_t i = 0; i < C::kLimbs; ++i) {
72 u128 d = (u128)a[i] - b[i] - br;
73 r[i] = (u64)d;
74 br = (d >> 64) & 1;
75 }
76 return (u64)br;
77}
78// a + b (mod 2^kBits), returns carry.
79template <WeierstrassCurve C>
80u64 add_carry(fe<C>& r, const fe<C>& a, const fe<C>& b) {
81 u128 c = 0;
82 for (std::size_t i = 0; i < C::kLimbs; ++i) {
83 u128 s = (u128)a[i] + b[i] + c;
84 r[i] = (u64)s;
85 c = s >> 64;
86 }
87 return (u64)c;
88}

90// ---- Montgomery context for one modulus (R = 2^kBits) ------------------------------------------
91/// Montgomery context for one modulus (R = 2^kBits) — all constants derived at startup.
92template <WeierstrassCurve C>
93struct Mont {
94 fe<C> m; ///< the modulus
95 fe<C> rr; ///< R^2 mod m
96 fe<C> one; ///< R mod m (Montgomery form of 1)
97 u64 n0; ///< -m^{-1} mod 2^64
98};

100// CIOS Montgomery multiplication: r = a*b*R^-1 mod m.
101template <WeierstrassCurve C>
102void mont_mul(fe<C>& r, const fe<C>& a, const fe<C>& b, const Mont<C>& M) {
103 constexpr std::size_t L = C::kLimbs;
104 u64 t[L + 1] = {};
105 for (std::size_t i = 0; i < L; ++i) {
106 // t += a * b[i]
107 u128 carry = 0;
108 for (std::size_t j = 0; j < L; ++j) {
109 u128 p = (u128)a[j] * b[i] + t[j] + carry;
110 t[j] = (u64)p;
111 carry = p >> 64;
112 }
113 u128 s = (u128)t[L] + carry;
114 t[L] = (u64)s;
115 u64 top = (u64)(s >> 64);
116 // m_mul = t[0] * n0 mod 2^64; t += m_mul * m; then shift right one limb
117 u64 mmul = (u64)((u128)t[0] * M.n0);
118 carry = 0;
119 {
120 u128 p = (u128)mmul * M.m[0] + t[0];
121 carry = p >> 64; // low limb becomes 0
122 }
123 for (std::size_t j = 1; j < L; ++j) {
124 u128 p = (u128)mmul * M.m[j] + t[j] + carry;
125 t[j - 1] = (u64)p;
126 carry = p >> 64;
127 }
128 u128 s2 = (u128)t[L] + carry;
129 t[L - 1] = (u64)s2;
130 t[L] = top + (u64)(s2 >> 64);
131 }
132 fe<C> res;
133 for (std::size_t i = 0; i < L; ++i) res[i] = t[i];
134 // final conditional subtraction (t may be in [0, 2m))
135 if (t[L] != 0 || geq<C>(res, M.m)) {
136 fe<C> tmp;
137 sub_borrow<C>(tmp, res, M.m);
138 res = tmp;
139 }
140 r = res;
141}
142template <WeierstrassCurve C>
143void mont_add(fe<C>& r, const fe<C>& a, const fe<C>& b, const Mont<C>& M) {
144 fe<C> s;
145 u64 c = add_carry<C>(s, a, b);
146 if (c || geq<C>(s, M.m)) {
147 fe<C> t;
148 sub_borrow<C>(t, s, M.m);
149 s = t;
150 }
151 r = s;
152}
153template <WeierstrassCurve C>
154void mont_sub(fe<C>& r, const fe<C>& a, const fe<C>& b, const Mont<C>& M) {
155 fe<C> d;
156 u64 br = sub_borrow<C>(d, a, b);
157 if (br) {
158 fe<C> t;
159 add_carry<C>(t, d, M.m);
160 d = t;
161 }
162 r = d;
163}
164template <WeierstrassCurve C>
165void to_mont(fe<C>& r, const fe<C>& a, const Mont<C>& M) {
166 mont_mul<C>(r, a, M.rr, M);
167}
168template <WeierstrassCurve C>
169void from_mont(fe<C>& r, const fe<C>& a, const Mont<C>& M) {
170 fe<C> one{};
171 one[0] = 1;
172 mont_mul<C>(r, a, one, M);
173}
174// r = a^-1 mod m, via Fermat: a^(m-2). (m is prime for both p and n.)
175template <WeierstrassCurve C>
176void mont_inv(fe<C>& r, const fe<C>& a, const Mont<C>& M) {
177 fe<C> two{};
178 two[0] = 2;
179 fe<C> exp;
180 sub_borrow<C>(exp, M.m, two); // m - 2
181 fe<C> result = M.one; // Montgomery 1
182 fe<C> base = a;
183 for (std::size_t i = 0; i < kBits<C>; ++i) {
184 if ((exp[i / 64] >> (i % 64)) & 1) mont_mul<C>(result, result, base, M);
185 mont_mul<C>(base, base, base, M);
186 }
187 r = result;
188}

190inline u64 inv64(u64 a) { // a^-1 mod 2^64 (a odd), Newton's iteration
191 u64 x = a; // correct to 3 bits
192 for (int i = 0; i < 5; ++i) x *= 2 - a * x;
193 return x;
194}
195template <WeierstrassCurve C>
196Mont<C> make_mont(const fe<C>& m) {
197 Mont<C> M;
198 M.m = m;
199 M.n0 = 0 - inv64(m[0]);
200 // rr = 2^(2*kBits) mod m, by 2*kBits doublings of 1 with conditional subtract.
201 fe<C> x{};
202 x[0] = 1;
203 for (std::size_t i = 0; i < 2 * kBits<C>; ++i) {
204 fe<C> d;
205 u64 c = add_carry<C>(d, x, x);
206 if (c || geq<C>(d, m)) {
207 fe<C> t;
208 sub_borrow<C>(t, d, m);
209 d = t;
210 }
211 x = d;
212 }
213 M.rr = x;
214 // one = R mod m = 2^kBits mod m -> to_mont(1)
215 fe<C> oneN{};
216 oneN[0] = 1;
217 mont_mul<C>(M.one, oneN, M.rr, M);
218 return M;
219}

233// ---- bytes <-> limbs (kBytes big-endian bytes) --------------------------------------------------
234template <WeierstrassCurve C>
235fe<C> be_to_fe(const unsigned char* b) {
236 fe<C> r{};
237 for (std::size_t limb = 0; limb < C::kLimbs; ++limb) {
238 u64 v = 0;
239 const unsigned char* p = b + (C::kLimbs - 1 - limb) * 8; // most-significant 8 bytes last
240 for (int k = 0; k < 8; ++k) v = (v << 8) | p[k];
241 r[limb] = v;
242 }
243 return r;
244}
245template <WeierstrassCurve C>
246void fe_to_be(unsigned char* out, const fe<C>& a) {
247 for (std::size_t limb = 0; limb < C::kLimbs; ++limb) {
248 u64 v = a[limb];
249 unsigned char* p = out + (C::kLimbs - 1 - limb) * 8;
250 for (int k = 7; k >= 0; --k) {
251 p[k] = (unsigned char)(v & 0xFF);
252 v >>= 8;
253 }
254 }
255}

258/// A curve point in Jacobian projective coordinates (x = X/Z^2, y = Y/Z^3), Montgomery form.
259template <WeierstrassCurve C>
260struct Jac {
261 fe<C> X; ///< projective X
262 fe<C> Y; ///< projective Y
263 fe<C> Z; ///< projective Z (0 also encodes the point at infinity)
264 bool inf; ///< explicit point-at-infinity flag
265};
266template <WeierstrassCurve C>
267Jac<C> jac_infinity() {
268 return Jac<C>{Fp<C>().one, Fp<C>().one, fe<C>{}, true};
269}

271template <WeierstrassCurve C>
272void jac_double(Jac<C>& r, const Jac<C>& q) {
273 const Mont<C>& F = Fp<C>();
274 if (q.inf || is_zero<C>(q.Z)) {
275 r = jac_infinity<C>();
276 return;
277 }
278 fe<C> A, B, Cc, D, t1, t2;
279 mont_mul<C>(A, q.X, q.X, F); // X^2
280 mont_mul<C>(B, q.Y, q.Y, F); // Y^2
281 mont_mul<C>(Cc, B, B, F); // Y^4
282 // D = 2*((X+B)^2 - A - C)
283 mont_add<C>(t1, q.X, B, F);
284 mont_mul<C>(t1, t1, t1, F);
285 mont_sub<C>(t1, t1, A, F);
286 mont_sub<C>(t1, t1, Cc, F);
287 mont_add<C>(D, t1, t1, F);
288 // ZZ = Z^2 ; E = 3*(X - ZZ)*(X + ZZ) [uses a = -3]
289 fe<C> ZZ;
290 mont_mul<C>(ZZ, q.Z, q.Z, F);
291 mont_sub<C>(t1, q.X, ZZ, F);
292 mont_add<C>(t2, q.X, ZZ, F);
293 mont_mul<C>(t1, t1, t2, F);
294 fe<C> E;
295 mont_add<C>(E, t1, t1, F);
296 mont_add<C>(E, E, t1, F); // 3*(...)
297 // F2 = E^2 ; X3 = F2 - 2D
298 fe<C> X3;
299 mont_mul<C>(X3, E, E, F);
300 mont_sub<C>(X3, X3, D, F);
301 mont_sub<C>(X3, X3, D, F);
302 // Y3 = E*(D - X3) - 8C
303 fe<C> Y3, eight;
304 mont_sub<C>(t1, D, X3, F);
305 mont_mul<C>(Y3, E, t1, F);
306 mont_add<C>(eight, Cc, Cc, F);
307 mont_add<C>(eight, eight, eight, F);
308 mont_add<C>(eight, eight, eight, F); // 8C
309 mont_sub<C>(Y3, Y3, eight, F);
310 // Z3 = 2*Y*Z
311 fe<C> Z3;
312 mont_mul<C>(Z3, q.Y, q.Z, F);
313 mont_add<C>(Z3, Z3, Z3, F);
314 r = Jac<C>{X3, Y3, Z3, false};
315}

317template <WeierstrassCurve C>
318void jac_add(Jac<C>& r, const Jac<C>& a, const Jac<C>& b) {
319 const Mont<C>& F = Fp<C>();
320 if (a.inf || is_zero<C>(a.Z)) {
321 r = b;
322 return;
323 }
324 if (b.inf || is_zero<C>(b.Z)) {
325 r = a;
326 return;
327 }
328 fe<C> Z1Z1, Z2Z2, U1, U2, S1, S2;
329 mont_mul<C>(Z1Z1, a.Z, a.Z, F);
330 mont_mul<C>(Z2Z2, b.Z, b.Z, F);
331 mont_mul<C>(U1, a.X, Z2Z2, F);
332 mont_mul<C>(U2, b.X, Z1Z1, F);
333 fe<C> t;
334 mont_mul<C>(t, b.Z, Z2Z2, F);
335 mont_mul<C>(S1, a.Y, t, F);
336 mont_mul<C>(t, a.Z, Z1Z1, F);
337 mont_mul<C>(S2, b.Y, t, F);
338 fe<C> H, Rr;
339 mont_sub<C>(H, U2, U1, F);
340 mont_sub<C>(Rr, S2, S1, F);
341 if (is_zero<C>(H)) {
342 if (is_zero<C>(Rr)) {
343 jac_double<C>(r, a);
344 return;
345 }
346 r = jac_infinity<C>();
347 return;
348 }
349 fe<C> HH, HHH, V;
350 mont_mul<C>(HH, H, H, F);
351 mont_mul<C>(HHH, HH, H, F);
352 mont_mul<C>(V, U1, HH, F);
353 fe<C> X3;
354 mont_mul<C>(X3, Rr, Rr, F);
355 mont_sub<C>(X3, X3, HHH, F);
356 mont_sub<C>(X3, X3, V, F);
357 mont_sub<C>(X3, X3, V, F);
358 fe<C> Y3;
359 mont_sub<C>(t, V, X3, F);
360 mont_mul<C>(Y3, Rr, t, F);
361 fe<C> s1hhh;
362 mont_mul<C>(s1hhh, S1, HHH, F);
363 mont_sub<C>(Y3, Y3, s1hhh, F);
364 fe<C> Z3;
365 mont_mul<C>(Z3, a.Z, b.Z, F);
366 mont_mul<C>(Z3, Z3, H, F);
367 r = Jac<C>{X3, Y3, Z3, false};
368}

370// Strauss-Shamir: u1*A + u2*B with ONE doubling chain (kBits doublings total)
371// instead of two separate scalar multiplications. A 2-bit window over both
372// scalars uses a 16-entry combined table [i*A + j*B] so it also halves the adds.
373template <WeierstrassCurve C>
374void jac_double_mul(Jac<C>& r, const fe<C>& u1, const Jac<C>& A, const fe<C>& u2, const Jac<C>& B) {
375 Jac<C> tbl[4][4]; // tbl[i][j] = i*A + j*B, i,j in {0..3}
376 tbl[0][0] = jac_infinity<C>();
377 tbl[1][0] = A;
378 jac_double<C>(tbl[2][0], A);
379 jac_add<C>(tbl[3][0], tbl[2][0], A);
380 tbl[0][1] = B;
381 jac_double<C>(tbl[0][2], B);
382 jac_add<C>(tbl[0][3], tbl[0][2], B);
383 for (int i = 1; i < 4; ++i)
384 for (int j = 1; j < 4; ++j) jac_add<C>(tbl[i][j], tbl[i][0], tbl[0][j]);
385
386 Jac<C> acc = jac_infinity<C>();
387 for (int i = static_cast<int>(kBits<C>) - 2; i >= 0; i -= 2) { // kBits is even
388 Jac<C> t;
389 jac_double<C>(t, acc);
390 acc = t;
391 jac_double<C>(t, acc);
392 acc = t;
393 const unsigned a = (u1[i / 64] >> (i % 64)) & 0x3;
394 const unsigned b = (u2[i / 64] >> (i % 64)) & 0x3;
395 if (a || b) {
396 jac_add<C>(t, acc, tbl[a][b]);
397 acc = t;
398 }
399 }
400 r = acc;
401}

403// affine x-coordinate (normal form) of a Jacobian point: x = X / Z^2.
404template <WeierstrassCurve C>
405fe<C> jac_affine_x(const Jac<C>& q) {
406 const Mont<C>& F = Fp<C>();
407 fe<C> zinv, zinv2, x;
408 mont_inv<C>(zinv, q.Z, F);
409 mont_mul<C>(zinv2, zinv, zinv, F);
410 mont_mul<C>(x, q.X, zinv2, F);
411 fe<C> out;
412 from_mont<C>(out, x, F);
413 return out;
414}

432// Fixed-base comb for k*G. G is constant, so we precompute (once) the 2^kLimbs-entry
433// table T[s] = sum over set bits i of s of (2^(64*i) * G). Then k*G is just 64
434// doublings + 64 adds (vs kBits doublings for a generic window) — the big win for
435// the per-message signing path. Selector at step j is bit j of each 64-bit limb.
436template <WeierstrassCurve C>
437const std::array<Jac<C>, (1u << C::kLimbs)>& g_comb() {
438 static const std::array<Jac<C>, (1u << C::kLimbs)> tbl = [] {
439 constexpr std::size_t L = C::kLimbs;
440 Jac<C> gi[L];
441 gi[0] = affine_to_jac<C>(C::GX, C::GY);
442 for (std::size_t i = 1; i < L; ++i) {
443 Jac<C> acc = gi[i - 1];
444 for (int b = 0; b < 64; ++b) { // gi[i] = 2^64 * gi[i-1]
445 Jac<C> t;
446 jac_double<C>(t, acc);
447 acc = t;
448 }
449 gi[i] = acc;
450 }
451 std::array<Jac<C>, (1u << L)> t;
452 t[0] = jac_infinity<C>();
453 for (unsigned s = 1; s < (1u << L); ++s) {
454 Jac<C> acc = jac_infinity<C>();
455 for (std::size_t i = 0; i < L; ++i)
456 if (s & (1u << i)) {
457 Jac<C> r;
458 jac_add<C>(r, acc, gi[i]);
459 acc = r;
460 }
461 t[s] = acc;
462 }
463 return t;
464 }();
465 return tbl;
466}

468// ---- constant-time point ops for the SECRET-scalar path (signing k*G, keygen d*G) --------------
469// jac_double_mul (verify) operates on PUBLIC data and stays branchy; the fixed-base comb below,
470// which multiplies the secret nonce/key, must not branch or index on secret bits. These helpers
471// give it branch-free doubling, addition, and table selection. They are differentially tested
472// against the branchy jac_double/jac_add over random + edge inputs (CheatahP256.ConstantTimeAddMatches).

474inline u64 ct_mask(bool c) { return u64(0) - static_cast<u64>(c); } // false->0, true->all-ones

476template <WeierstrassCurve C>
477inline void fe_cmov(fe<C>& r, const fe<C>& a, u64 m) { // r = m ? a : r (per-limb, constant-time)
478 for (std::size_t i = 0; i < C::kLimbs; ++i) r[i] = (r[i] & ~m) | (a[i] & m);
479}
480template <WeierstrassCurve C>
481inline void jac_cmov(Jac<C>& r, const Jac<C>& a, u64 m) {
482 fe_cmov<C>(r.X, a.X, m);
483 fe_cmov<C>(r.Y, a.Y, m);
484 fe_cmov<C>(r.Z, a.Z, m);
485 r.inf = is_zero<C>(r.Z); // infinity is encoded by Z==0 throughout the CT path
486}

488// Point doubling WITHOUT the is-infinity early return: the formula's Z3 = 2*Y*Z is already 0 when
489// the input is infinity (Z==0), so it self-encodes infinity, and a prime-order curve has no
490// finite 2-torsion point that could double TO infinity — so no branch is needed.
491template <WeierstrassCurve C>
492void jac_double_ct(Jac<C>& r, const Jac<C>& q) {
493 const Mont<C>& F = Fp<C>();
494 fe<C> A, B, Cc, D, t1, t2;
495 mont_mul<C>(A, q.X, q.X, F);
496 mont_mul<C>(B, q.Y, q.Y, F);
497 mont_mul<C>(Cc, B, B, F);
498 mont_add<C>(t1, q.X, B, F);
499 mont_mul<C>(t1, t1, t1, F);
500 mont_sub<C>(t1, t1, A, F);
501 mont_sub<C>(t1, t1, Cc, F);
502 mont_add<C>(D, t1, t1, F);
503 fe<C> ZZ;
504 mont_mul<C>(ZZ, q.Z, q.Z, F);
505 mont_sub<C>(t1, q.X, ZZ, F);
506 mont_add<C>(t2, q.X, ZZ, F);
507 mont_mul<C>(t1, t1, t2, F);
508 fe<C> E;
509 mont_add<C>(E, t1, t1, F);
510 mont_add<C>(E, E, t1, F);
511 fe<C> X3;
512 mont_mul<C>(X3, E, E, F);
513 mont_sub<C>(X3, X3, D, F);
514 mont_sub<C>(X3, X3, D, F);
515 fe<C> Y3, eight;
516 mont_sub<C>(t1, D, X3, F);
517 mont_mul<C>(Y3, E, t1, F);
518 mont_add<C>(eight, Cc, Cc, F);
519 mont_add<C>(eight, eight, eight, F);
520 mont_add<C>(eight, eight, eight, F);
521 mont_sub<C>(Y3, Y3, eight, F);
522 fe<C> Z3;
523 mont_mul<C>(Z3, q.Y, q.Z, F);
524 mont_add<C>(Z3, Z3, Z3, F);
525 r = Jac<C>{X3, Y3, Z3, is_zero<C>(Z3)};
526}

528// Point addition, branch-free. It always computes the general add formula, then constant-time-
529// selects the correct result over the special cases via masks: a==inf -> b, b==inf -> a,
530// a==b -> double(a), a==-b -> infinity. Precedence is enforced by cmov ORDER (a==inf last / highest).
531template <WeierstrassCurve C>
532void jac_add_ct(Jac<C>& r, const Jac<C>& a, const Jac<C>& b) {
533 const Mont<C>& F = Fp<C>();
534 fe<C> Z1Z1, Z2Z2, U1, U2, S1, S2;
535 mont_mul<C>(Z1Z1, a.Z, a.Z, F);
536 mont_mul<C>(Z2Z2, b.Z, b.Z, F);
537 mont_mul<C>(U1, a.X, Z2Z2, F);
538 mont_mul<C>(U2, b.X, Z1Z1, F);
539 fe<C> t;
540 mont_mul<C>(t, b.Z, Z2Z2, F);
541 mont_mul<C>(S1, a.Y, t, F);
542 mont_mul<C>(t, a.Z, Z1Z1, F);
543 mont_mul<C>(S2, b.Y, t, F);
544 fe<C> H, Rr;
545 mont_sub<C>(H, U2, U1, F);
546 mont_sub<C>(Rr, S2, S1, F);
547 fe<C> HH, HHH, V;
548 mont_mul<C>(HH, H, H, F);
549 mont_mul<C>(HHH, HH, H, F);
550 mont_mul<C>(V, U1, HH, F);
551 fe<C> X3;
552 mont_mul<C>(X3, Rr, Rr, F);
553 mont_sub<C>(X3, X3, HHH, F);
554 mont_sub<C>(X3, X3, V, F);
555 mont_sub<C>(X3, X3, V, F);
556 fe<C> Y3;
557 mont_sub<C>(t, V, X3, F);
558 mont_mul<C>(Y3, Rr, t, F);
559 fe<C> s1hhh;
560 mont_mul<C>(s1hhh, S1, HHH, F);
561 mont_sub<C>(Y3, Y3, s1hhh, F);
562 fe<C> Z3;
563 mont_mul<C>(Z3, a.Z, b.Z, F);
564 mont_mul<C>(Z3, Z3, H, F);
565 r = Jac<C>{X3, Y3, Z3, false}; // start = the general-case result
566
567 const u64 ma = ct_mask(is_zero<C>(a.Z)); // a is infinity
568 const u64 mb = ct_mask(is_zero<C>(b.Z)); // b is infinity
569 const u64 hz = ct_mask(is_zero<C>(H));
570 const u64 rz = ct_mask(is_zero<C>(Rr));
571 Jac<C> dbl;
572 jac_double_ct<C>(dbl, a);
573 const Jac<C> infp = jac_infinity<C>();
574 jac_cmov<C>(r, dbl, hz & rz); // a == b -> 2a
575 jac_cmov<C>(r, infp, hz & ~rz); // a == -b -> infinity
576 jac_cmov<C>(r, a, mb); // b == infinity -> a
577 jac_cmov<C>(r, b, ma); // a == infinity -> b (highest precedence, applied last)
578 r.inf = is_zero<C>(r.Z);
579}

581// Constant-time table lookup: scan every entry, copying the one whose index == sel via a mask, so
582// the memory-access pattern (and timing) is independent of the secret selector.
583template <WeierstrassCurve C, std::size_t N>
584void ct_select(Jac<C>& out, const std::array<Jac<C>, N>& tbl, unsigned sel) {
585 out = jac_infinity<C>();
586 for (unsigned i = 0; i < N; ++i) jac_cmov<C>(out, tbl[i], ct_mask(i == sel));
587}

589// k*G for a SECRET scalar k, in constant time: 64 doublings + 64 unconditional adds over the
590// fixed-base comb table. The old form skipped the add when the window was zero and indexed the
591// table by the secret selector — both leaked bits of k. Here every step does the same work
592// (branch-free double, masked table select, unconditional branch-free add — add of the T[0]=infinity
593// entry when the window is zero is a no-op via the CT add's masks).
594template <WeierstrassCurve C>
595void jac_mul_base(Jac<C>& r, const fe<C>& k) {
596 const auto& T = g_comb<C>();
597 Jac<C> acc = jac_infinity<C>();
598 for (int j = 63; j >= 0; --j) {
599 Jac<C> t;
600 jac_double_ct<C>(t, acc);
601 acc = t;
602 unsigned sel = 0;
603 for (std::size_t i = 0; i < C::kLimbs; ++i) sel |= static_cast<unsigned>((k[i] >> j) & 1u) << i;
604 Jac<C> add;
605 ct_select<C>(add, T, sel);
606 jac_add_ct<C>(t, acc, add);
607 acc = t;
608 }
609 r = acc;
610}

612// Differential self-check for the constant-time point ops. A TEMPLATE, instantiated ONLY by the
613// p256/p384 test seam (so there is no such code in a production build), it confirms jac_add_ct /
614// jac_double_ct agree with the branchy reference jac_add / jac_double on the general case AND every
615// special case — a==b, a==-b, and infinity operands — which the signing path exercises rarely or
616// never, so this both proves correctness and drives those branches for coverage. Returns true iff
617// all match.
618template <WeierstrassCurve C>
619bool ct_add_selfcheck() {
620 const Mont<C>& F = Fp<C>();
621 auto affine_eq = [&](const Jac<C>& u, const Jac<C>& v) -> bool {
622 const bool ui = is_zero<C>(u.Z), vi = is_zero<C>(v.Z);
623 if (ui || vi) return ui == vi; // both infinity, or neither
624 auto affine = [&](const Jac<C>& p, fe<C>& x, fe<C>& y) {
625 fe<C> zi, zi2, zi3, xm, ym;
626 mont_inv<C>(zi, p.Z, F);
627 mont_mul<C>(zi2, zi, zi, F);
628 mont_mul<C>(zi3, zi2, zi, F);
629 mont_mul<C>(xm, p.X, zi2, F);
630 mont_mul<C>(ym, p.Y, zi3, F);
631 from_mont<C>(x, xm, F);
632 from_mont<C>(y, ym, F);
633 };
634 fe<C> ux, uy, vx, vy;
635 affine(u, ux, uy);
636 affine(v, vx, vy);
637 return ux == vx && uy == vy;
638 };
639 // Reference points via the branchy ops: P = 3G, Q = 5G, and -P.
640 const Jac<C>& G = base_point<C>();
641 Jac<C> P, Q, tmp;
642 jac_double<C>(tmp, G); // 2G
643 jac_add<C>(P, tmp, G); // 3G
644 jac_double<C>(tmp, tmp); // 4G
645 jac_add<C>(Q, tmp, G); // 5G
646 fe<C> zero{};
647 Jac<C> negP = P;
648 mont_sub<C>(negP.Y, zero, P.Y, F); // -P = (X, -Y, Z)
649 const Jac<C> inf = jac_infinity<C>();
650
651 Jac<C> ct, ref;
652 bool ok = true;
653 jac_add_ct<C>(ct, P, Q); jac_add<C>(ref, P, Q); ok &= affine_eq(ct, ref); // general
654 jac_add_ct<C>(ct, P, P); jac_double<C>(ref, P); ok &= affine_eq(ct, ref); // a == b
655 jac_add_ct<C>(ct, P, negP); ok &= is_zero<C>(ct.Z); // a == -b -> infinity
656 jac_add_ct<C>(ct, inf, P); ok &= affine_eq(ct, P); // a == infinity
657 jac_add_ct<C>(ct, P, inf); ok &= affine_eq(ct, P); // b == infinity
658 jac_add_ct<C>(ct, inf, inf); ok &= is_zero<C>(ct.Z); // inf + inf
659 jac_double_ct<C>(ct, P); jac_double<C>(ref, P); ok &= affine_eq(ct, ref); // double general
660 jac_double_ct<C>(ct, inf); ok &= is_zero<C>(ct.Z); // double infinity
661 return ok;
662}

664// Reduce a scalar already known to be < 2n into [0, n): a single conditional
665// subtraction of the group order n. Used for the FIPS 186-4 hash truncation and
666// for folding a curve x-coordinate (which lives in [0, p) < 2n) into a scalar.
667template <WeierstrassCurve C>
668fe<C> reduce_mod_n(const fe<C>& v) {
669 if (geq<C>(v, C::N)) {
670 fe<C> t;
671 sub_borrow<C>(t, v, C::N);
672 return t;
673 }
674 return v;
675}

677// Reduce a big-endian hash to a scalar in [0, n). A hash of at least kBytes keeps its
678// leftmost kBytes (the FIPS 186-4 leftmost-bits truncation); a SHORTER hash is the whole
679// value (X9.62 bits2int — right-aligned), e.g. a SHA-256 signature under a P-384 key.
680template <WeierstrassCurve C>
681fe<C> hash_to_scalar(const std::string& h) {
682 unsigned char buf[kBytes<C>] = {0};
683 if (h.size() >= kBytes<C>)
684 std::memcpy(buf, h.data(), kBytes<C>);
685 else
686 std::memcpy(buf + (kBytes<C> - h.size()), h.data(), h.size());
687 return reduce_mod_n<C>(be_to_fe<C>(buf));
688}

690// ---- minimal DER helpers ------------------------------------------------------------------------
691// Parse SEQUENCE{INTEGER r, INTEGER s} -> kBytes big-endian r and s. false on error.
692// Short-form lengths only: both curves' SEQUENCE stays under 128 bytes (P-384: <= ~104).
693template <WeierstrassCurve C>
694bool der_to_rs(const std::string& der, unsigned char* r, unsigned char* s) {
695 const unsigned char* p = (const unsigned char*)der.data();
696 std::size_t n = der.size(), i = 0;
697 auto read_int = [&](unsigned char* out) -> bool {
698 if (i >= n || p[i++] != 0x02) return false;
699 if (i >= n) return false;
700 std::size_t len = p[i++];
701 if (len & 0x80) return false; // curve-order ints are short-form
702 if (i + len > n || len == 0) return false;
703 const unsigned char* v = p + i;
704 // strip a leading zero (sign byte)
705 while (len > 1 && v[0] == 0) {
706 ++v;
707 --len;
708 }
709 if (len > kBytes<C>) return false;
710 std::memset(out, 0, kBytes<C>);
711 std::memcpy(out + (kBytes<C> - len), v, len);
712 i += (std::size_t)(v - (p + i)) + len; // advance past the original field
713 return true;
714 };
715 if (i >= n || p[i++] != 0x30) return false;
716 if (i >= n) return false;
717 std::size_t seqlen = p[i++];
718 if (seqlen & 0x80) return false;
719 if (i + seqlen != n) return false;
720 return read_int(r) && read_int(s);
721}

723// Is (x, y) on the curve y^2 = x^3 - 3x + b (mod p)? Rejects an off-curve /
724// invalid-curve public key — SP 800-56A / FIPS 186 point validation, which the plain
725// coordinate-range check (x,y < p) does not catch.
726template <WeierstrassCurve C>
727bool on_curve(const fe<C>& x, const fe<C>& y) {
728 const Mont<C>& F = Fp<C>();
729 fe<C> xm, ym, x2, x3, tx, rhs, bm, y2;
730 to_mont<C>(xm, x, F);
731 to_mont<C>(ym, y, F);
732 mont_mul<C>(x2, xm, xm, F); // x^2
733 mont_mul<C>(x3, x2, xm, F); // x^3
734 mont_add<C>(tx, xm, xm, F); // 2x
735 mont_add<C>(tx, tx, xm, F); // 3x
736 mont_sub<C>(rhs, x3, tx, F); // x^3 - 3x
737 to_mont<C>(bm, C::B, F);
738 mont_add<C>(rhs, rhs, bm, F); // x^3 - 3x + b
739 mont_mul<C>(y2, ym, ym, F); // y^2
740 return std::memcmp(y2.data(), rhs.data(), sizeof(fe<C>)) == 0;
741}

743// ECDSA verification over raw byte forms: pubkey = 2*kBytes X||Y, sig = 2*kBytes r||s.
744template <WeierstrassCurve C>
745bool verify_raw(const std::string& pubkey_xy, const std::string& msg_hash,
746 const std::string& sig_raw) {
747 if (pubkey_xy.size() != 2 * kBytes<C> || sig_raw.size() != 2 * kBytes<C>) return false;
748 fe<C> r = be_to_fe<C>((const unsigned char*)sig_raw.data());
749 fe<C> s = be_to_fe<C>((const unsigned char*)sig_raw.data() + kBytes<C>);
750 if (is_zero<C>(r) || is_zero<C>(s) || geq<C>(r, C::N) || geq<C>(s, C::N)) return false;
751
752 const Mont<C>& Fnn = Fn<C>();
753 fe<C> e = hash_to_scalar<C>(msg_hash);
754 fe<C> sm, em, rm, w, u1, u2;
755 to_mont<C>(sm, s, Fnn);
756 mont_inv<C>(w, sm, Fnn); // w = s^-1 (Montgomery)
757 to_mont<C>(em, e, Fnn);
758 to_mont<C>(rm, r, Fnn);
759 fe<C> u1m, u2m;
760 mont_mul<C>(u1m, em, w, Fnn);
761 mont_mul<C>(u2m, rm, w, Fnn);
762 from_mont<C>(u1, u1m, Fnn);
763 from_mont<C>(u2, u2m, Fnn);
764
765 fe<C> qx = be_to_fe<C>((const unsigned char*)pubkey_xy.data());
766 fe<C> qy = be_to_fe<C>((const unsigned char*)pubkey_xy.data() + kBytes<C>);
767 if (geq<C>(qx, C::P) || geq<C>(qy, C::P)) return false;
768 if (!on_curve<C>(qx, qy)) return false; // reject off-curve / invalid-curve public keys
769 Jac<C> Q = affine_to_jac<C>(qx, qy);
770
771 Jac<C> R;
772 jac_double_mul<C>(R, u1, base_point<C>(), u2, Q); // u1*G + u2*Q, one doubling chain
773 if (R.inf || is_zero<C>(R.Z)) return false;
774 fe<C> x = reduce_mod_n<C>(jac_affine_x<C>(R));
775 return std::memcmp(x.data(), r.data(), sizeof(fe<C>)) == 0;
776}

778// ECDSA verification of the DER form (SEQUENCE{INTEGER r, INTEGER s} — TLS/X.509).
779template <WeierstrassCurve C>
780bool verify_der(const std::string& pubkey_xy, const std::string& msg_hash,
781 const std::string& sig_der) {
782 unsigned char r[kBytes<C>], s[kBytes<C>];
783 if (!der_to_rs<C>(sig_der, r, s)) return false;
784 std::string raw;
785 raw.resize(2 * kBytes<C>);
786 std::memcpy(raw.data(), r, kBytes<C>);
787 std::memcpy(raw.data() + kBytes<C>, s, kBytes<C>);
788 return verify_raw<C>(pubkey_xy, msg_hash, raw);
789}
