5// ec_core.hpp — the width-generic short-Weierstrass ECDSA machinery shared by the `p256`
6// and `p384` modules. NOT a cheatah module itself: an internal implementation header the
7// two curve .cpp files include (purrc resolves modules by `<module>.hpp` name only, so
8// this file is invisible to `import`).
9//
10// Everything is templated on a WeierstrassCurve traits struct carrying the limb count and
11// the curve constants; the field/scalar Montgomery contexts are still DERIVED from the
12// modulus at startup (no hand-transcribed Montgomery magic). A value is uint64_t[kLimbs],
13// LEAST-significant limb first; points are Jacobian (X:Y:Z) with the curve a = -3 (true
14// of every NIST prime curve). The algorithms are limb-count-independent copies of the
15// battle-tested p256 versions — only bounds changed, from 4/256/32 to
16// kLimbs/kBits/kBytes.
17

31// The curve-traits concept every template below is constrained by: a NIST-style
32// short-Weierstrass curve (a = -3) over a prime field, its size in 64-bit limbs plus the
33// field prime P, group order N, coefficient B and base point (GX, GY) as little-endian
34// limb arrays.
35template <class C>
36concept WeierstrassCurve = requires {
37 { C::kLimbs } -> std::convertible_to<std::size_t>;
38 requires C::kLimbs >= 4 && C::kLimbs <= 8;
39 requires std::same_as<std::remove_cvref_t<decltype(C::P)>, std::array<u64, C::kLimbs>>;
40 requires std::same_as<std::remove_cvref_t<decltype(C::N)>, std::array<u64, C::kLimbs>>;
41 requires std::same_as<std::remove_cvref_t<decltype(C::B)>, std::array<u64, C::kLimbs>>;
42 requires std::same_as<std::remove_cvref_t<decltype(C::GX)>, std::array<u64, C::kLimbs>>;
43 requires std::same_as<std::remove_cvref_t<decltype(C::GY)>, std::array<u64, C::kLimbs>>;
44};

46template <WeierstrassCurve C>
47using fe = std::array<u64, C::kLimbs>; // one field/scalar value, limb[0] = least significant

49template <WeierstrassCurve C>

51template <WeierstrassCurve C>
52inline constexpr std::size_t kBytes = C::kLimbs * 8; // big-endian byte size (32 / 48)

54// ---- plain multi-limb helpers -----------------------------------------------------------------
55template <WeierstrassCurve C>
56bool is_zero(const fe<C>& a) {
57 u64 acc = 0;
58 for (const u64 limb : a) acc |= limb;
59 return acc == 0;
60}
61template <WeierstrassCurve C>
62bool geq(const fe<C>& a, const fe<C>& b) { // a >= b
63 for (int i = static_cast<int>(C::kLimbs) - 1; i >= 0; --i)
64 if (a[i] != b[i]) return a[i] > b[i];
65 return true;
66}
67// a - b (mod 2^kBits), returns borrow.
68template <WeierstrassCurve C>
69u64 sub_borrow(fe<C>& r, const fe<C>& a, const fe<C>& b) {
70 u128 br = 0;
71 for (std::size_t i = 0; i < C::kLimbs; ++i) {
72 u128 d = (u128)a[i] - b[i] - br;
73 r[i] = (u64)d;
74 br = (d >> 64) & 1;
75 }
76 return (u64)br;
77}
78// a + b (mod 2^kBits), returns carry.
79template <WeierstrassCurve C>
80u64 add_carry(fe<C>& r, const fe<C>& a, const fe<C>& b) {
81 u128 c = 0;
82 for (std::size_t i = 0; i < C::kLimbs; ++i) {
83 u128 s = (u128)a[i] + b[i] + c;
84 r[i] = (u64)s;
85 c = s >> 64;
86 }
87 return (u64)c;
88}

90// ---- Montgomery context for one modulus (R = 2^kBits) ------------------------------------------
91/// Montgomery context for one modulus (R = 2^kBits) — all constants derived at startup.
92template <WeierstrassCurve C>
93struct Mont {
94 fe<C> m; ///< the modulus
95 fe<C> rr; ///< R^2 mod m
96 fe<C> one; ///< R mod m (Montgomery form of 1)
97 u64 n0; ///< -m^{-1} mod 2^64
98};

100// CIOS Montgomery multiplication: r = a*b*R^-1 mod m.
101template <WeierstrassCurve C>
102void mont_mul(fe<C>& r, const fe<C>& a, const fe<C>& b, const Mont<C>& M) {
103 constexpr std::size_t L = C::kLimbs;
104 u64 t[L + 1] = {};
105 for (std::size_t i = 0; i < L; ++i) {
106 // t += a * b[i]
107 u128 carry = 0;
108 for (std::size_t j = 0; j < L; ++j) {
109 u128 p = (u128)a[j] * b[i] + t[j] + carry;
110 t[j] = (u64)p;
111 carry = p >> 64;
112 }
113 u128 s = (u128)t[L] + carry;
114 t[L] = (u64)s;
115 u64 top = (u64)(s >> 64);
116 // m_mul = t[0] * n0 mod 2^64; t += m_mul * m; then shift right one limb
117 u64 mmul = (u64)((u128)t[0] * M.n0);
118 carry = 0;
119 {
120 u128 p = (u128)mmul * M.m[0] + t[0];
121 carry = p >> 64; // low limb becomes 0
122 }
123 for (std::size_t j = 1; j < L; ++j) {
124 u128 p = (u128)mmul * M.m[j] + t[j] + carry;
125 t[j - 1] = (u64)p;
126 carry = p >> 64;
127 }
128 u128 s2 = (u128)t[L] + carry;
129 t[L - 1] = (u64)s2;
130 t[L] = top + (u64)(s2 >> 64);
131 }
132 fe<C> res;
133 for (std::size_t i = 0; i < L; ++i) res[i] = t[i];
134 // final conditional subtraction (t may be in [0, 2m))
135 if (t[L] != 0 || geq<C>(res, M.m)) {
136 fe<C> tmp;
137 sub_borrow<C>(tmp, res, M.m);
138 res = tmp;
139 }
140 r = res;
141}
142template <WeierstrassCurve C>
143void mont_add(fe<C>& r, const fe<C>& a, const fe<C>& b, const Mont<C>& M) {
144 fe<C> s;
145 u64 c = add_carry<C>(s, a, b);
146 if (c || geq<C>(s, M.m)) {
147 fe<C> t;
148 sub_borrow<C>(t, s, M.m);
149 s = t;
150 }
151 r = s;
152}
153template <WeierstrassCurve C>
154void mont_sub(fe<C>& r, const fe<C>& a, const fe<C>& b, const Mont<C>& M) {
155 fe<C> d;
156 u64 br = sub_borrow<C>(d, a, b);
157 if (br) {
158 fe<C> t;
159 add_carry<C>(t, d, M.m);
160 d = t;
161 }
162 r = d;
163}
164template <WeierstrassCurve C>
165void to_mont(fe<C>& r, const fe<C>& a, const Mont<C>& M) {
166 mont_mul<C>(r, a, M.rr, M);
167}
168template <WeierstrassCurve C>
169void from_mont(fe<C>& r, const fe<C>& a, const Mont<C>& M) {
170 fe<C> one{};
171 one[0] = 1;
172 mont_mul<C>(r, a, one, M);
173}
174// r = a^-1 mod m, via Fermat: a^(m-2). (m is prime for both p and n.)
175template <WeierstrassCurve C>
176void mont_inv(fe<C>& r, const fe<C>& a, const Mont<C>& M) {
177 fe<C> two{};
178 two[0] = 2;
179 fe<C> exp;
180 sub_borrow<C>(exp, M.m, two); // m - 2
181 fe<C> result = M.one; // Montgomery 1
182 fe<C> base = a;
183 for (std::size_t i = 0; i < kBits<C>; ++i) {
184 if ((exp[i / 64] >> (i % 64)) & 1) mont_mul<C>(result, result, base, M);
185 mont_mul<C>(base, base, base, M);
186 }
187 r = result;
188}

190inline u64 inv64(u64 a) { // a^-1 mod 2^64 (a odd), Newton's iteration
191 u64 x = a; // correct to 3 bits
192 for (int i = 0; i < 5; ++i) x *= 2 - a * x;
193 return x;
194}
195template <WeierstrassCurve C>
196Mont<C> make_mont(const fe<C>& m) {
197 Mont<C> M;
198 M.m = m;
199 M.n0 = 0 - inv64(m[0]);
200 // rr = 2^(2*kBits) mod m, by 2*kBits doublings of 1 with conditional subtract.
201 fe<C> x{};
202 x[0] = 1;
203 for (std::size_t i = 0; i < 2 * kBits<C>; ++i) {
204 fe<C> d;
205 u64 c = add_carry<C>(d, x, x);
206 if (c || geq<C>(d, m)) {
207 fe<C> t;
208 sub_borrow<C>(t, d, m);
209 d = t;
210 }
211 x = d;
212 }
213 M.rr = x;
214 // one = R mod m = 2^kBits mod m -> to_mont(1)
215 fe<C> oneN{};
216 oneN[0] = 1;
217 mont_mul<C>(M.one, oneN, M.rr, M);
218 return M;
219}

233// ---- bytes <-> limbs (kBytes big-endian bytes) --------------------------------------------------
234template <WeierstrassCurve C>
235fe<C> be_to_fe(const unsigned char* b) {
236 fe<C> r{};
237 for (std::size_t limb = 0; limb < C::kLimbs; ++limb) {
238 u64 v = 0;
239 const unsigned char* p = b + (C::kLimbs - 1 - limb) * 8; // most-significant 8 bytes last
240 for (int k = 0; k < 8; ++k) v = (v << 8) | p[k];
241 r[limb] = v;
242 }
243 return r;
244}
245template <WeierstrassCurve C>
246void fe_to_be(unsigned char* out, const fe<C>& a) {
247 for (std::size_t limb = 0; limb < C::kLimbs; ++limb) {
248 u64 v = a[limb];
249 unsigned char* p = out + (C::kLimbs - 1 - limb) * 8;
250 for (int k = 7; k >= 0; --k) {
251 p[k] = (unsigned char)(v & 0xFF);
252 v >>= 8;
253 }
254 }
255}

257// ---- Jacobian points (coordinates in Montgomery form, mod p) ------------------------------------
258/// A curve point in Jacobian projective coordinates (x = X/Z^2, y = Y/Z^3), Montgomery form.
259template <WeierstrassCurve C>
260struct Jac {
261 fe<C> X; ///< projective X
262 fe<C> Y; ///< projective Y
263 fe<C> Z; ///< projective Z (0 also encodes the point at infinity)
264 bool inf; ///< explicit point-at-infinity flag
265};
266template <WeierstrassCurve C>
267Jac<C> jac_infinity() {
268 return Jac<C>{Fp<C>().one, Fp<C>().one, fe<C>{}, true};
269}

271template <WeierstrassCurve C>
272void jac_double(Jac<C>& r, const Jac<C>& q) {
273 const Mont<C>& F = Fp<C>();
274 if (q.inf || is_zero<C>(q.Z)) {
275 r = jac_infinity<C>();
276 return;
277 }
278 fe<C> A, B, Cc, D, t1, t2;
279 mont_mul<C>(A, q.X, q.X, F); // X^2
280 mont_mul<C>(B, q.Y, q.Y, F); // Y^2
281 mont_mul<C>(Cc, B, B, F); // Y^4
282 // D = 2*((X+B)^2 - A - C)
283 mont_add<C>(t1, q.X, B, F);
284 mont_mul<C>(t1, t1, t1, F);
285 mont_sub<C>(t1, t1, A, F);
286 mont_sub<C>(t1, t1, Cc, F);
287 mont_add<C>(D, t1, t1, F);
288 // ZZ = Z^2 ; E = 3*(X - ZZ)*(X + ZZ) [uses a = -3]
289 fe<C> ZZ;
290 mont_mul<C>(ZZ, q.Z, q.Z, F);
291 mont_sub<C>(t1, q.X, ZZ, F);
292 mont_add<C>(t2, q.X, ZZ, F);
293 mont_mul<C>(t1, t1, t2, F);
294 fe<C> E;
295 mont_add<C>(E, t1, t1, F);
296 mont_add<C>(E, E, t1, F); // 3*(...)
297 // F2 = E^2 ; X3 = F2 - 2D
298 fe<C> X3;
299 mont_mul<C>(X3, E, E, F);
300 mont_sub<C>(X3, X3, D, F);
301 mont_sub<C>(X3, X3, D, F);
302 // Y3 = E*(D - X3) - 8C
303 fe<C> Y3, eight;
304 mont_sub<C>(t1, D, X3, F);
305 mont_mul<C>(Y3, E, t1, F);
306 mont_add<C>(eight, Cc, Cc, F);
307 mont_add<C>(eight, eight, eight, F);
308 mont_add<C>(eight, eight, eight, F); // 8C
309 mont_sub<C>(Y3, Y3, eight, F);
310 // Z3 = 2*Y*Z
311 fe<C> Z3;
312 mont_mul<C>(Z3, q.Y, q.Z, F);
313 mont_add<C>(Z3, Z3, Z3, F);
314 r = Jac<C>{X3, Y3, Z3, false};
315}

317template <WeierstrassCurve C>
318void jac_add(Jac<C>& r, const Jac<C>& a, const Jac<C>& b) {
319 const Mont<C>& F = Fp<C>();
320 if (a.inf || is_zero<C>(a.Z)) {
321 r = b;
322 return;
323 }
324 if (b.inf || is_zero<C>(b.Z)) {
325 r = a;
326 return;
327 }
328 fe<C> Z1Z1, Z2Z2, U1, U2, S1, S2;
329 mont_mul<C>(Z1Z1, a.Z, a.Z, F);
330 mont_mul<C>(Z2Z2, b.Z, b.Z, F);
331 mont_mul<C>(U1, a.X, Z2Z2, F);
332 mont_mul<C>(U2, b.X, Z1Z1, F);
333 fe<C> t;
334 mont_mul<C>(t, b.Z, Z2Z2, F);
335 mont_mul<C>(S1, a.Y, t, F);
336 mont_mul<C>(t, a.Z, Z1Z1, F);
337 mont_mul<C>(S2, b.Y, t, F);
338 fe<C> H, Rr;
339 mont_sub<C>(H, U2, U1, F);
340 mont_sub<C>(Rr, S2, S1, F);
341 if (is_zero<C>(H)) {
342 if (is_zero<C>(Rr)) {
343 jac_double<C>(r, a);
344 return;
345 }
346 r = jac_infinity<C>();
347 return;
348 }
349 fe<C> HH, HHH, V;
350 mont_mul<C>(HH, H, H, F);
351 mont_mul<C>(HHH, HH, H, F);
352 mont_mul<C>(V, U1, HH, F);
353 fe<C> X3;
354 mont_mul<C>(X3, Rr, Rr, F);
355 mont_sub<C>(X3, X3, HHH, F);
356 mont_sub<C>(X3, X3, V, F);
357 mont_sub<C>(X3, X3, V, F);
358 fe<C> Y3;
359 mont_sub<C>(t, V, X3, F);
360 mont_mul<C>(Y3, Rr, t, F);
361 fe<C> s1hhh;
362 mont_mul<C>(s1hhh, S1, HHH, F);
363 mont_sub<C>(Y3, Y3, s1hhh, F);
364 fe<C> Z3;
365 mont_mul<C>(Z3, a.Z, b.Z, F);
366 mont_mul<C>(Z3, Z3, H, F);
367 r = Jac<C>{X3, Y3, Z3, false};
368}

370// Strauss-Shamir: u1*A + u2*B with ONE doubling chain (kBits doublings total)
371// instead of two separate scalar multiplications. A 2-bit window over both
372// scalars uses a 16-entry combined table [i*A + j*B] so it also halves the adds.
373template <WeierstrassCurve C>
374void jac_double_mul(Jac<C>& r, const fe<C>& u1, const Jac<C>& A, const fe<C>& u2, const Jac<C>& B) {
375 Jac<C> tbl[4][4]; // tbl[i][j] = i*A + j*B, i,j in {0..3}
376 tbl[0][0] = jac_infinity<C>();
377 tbl[1][0] = A;
378 jac_double<C>(tbl[2][0], A);
379 jac_add<C>(tbl[3][0], tbl[2][0], A);
380 tbl[0][1] = B;
381 jac_double<C>(tbl[0][2], B);
382 jac_add<C>(tbl[0][3], tbl[0][2], B);
383 for (int i = 1; i < 4; ++i)
384 for (int j = 1; j < 4; ++j) jac_add<C>(tbl[i][j], tbl[i][0], tbl[0][j]);
385
386 Jac<C> acc = jac_infinity<C>();
387 for (int i = static_cast<int>(kBits<C>) - 2; i >= 0; i -= 2) { // kBits is even
388 Jac<C> t;
389 jac_double<C>(t, acc);
390 acc = t;
391 jac_double<C>(t, acc);
392 acc = t;
393 const unsigned a = (u1[i / 64] >> (i % 64)) & 0x3;
394 const unsigned b = (u2[i / 64] >> (i % 64)) & 0x3;
395 if (a || b) {
396 jac_add<C>(t, acc, tbl[a][b]);
397 acc = t;
398 }
399 }
400 r = acc;
401}

403// affine x-coordinate (normal form) of a Jacobian point: x = X / Z^2.
404template <WeierstrassCurve C>
405fe<C> jac_affine_x(const Jac<C>& q) {
406 const Mont<C>& F = Fp<C>();
407 fe<C> zinv, zinv2, x;
408 mont_inv<C>(zinv, q.Z, F);
409 mont_mul<C>(zinv2, zinv, zinv, F);
410 mont_mul<C>(x, q.X, zinv2, F);
411 fe<C> out;
412 from_mont<C>(out, x, F);
413 return out;
414}

416template <WeierstrassCurve C>
417Jac<C> affine_to_jac(const fe<C>& x, const fe<C>& y) {

420 to_mont<C>(p.X, x, F);

432// Fixed-base comb for k*G. G is constant, so we precompute (once) the 2^kLimbs-entry
433// table T[s] = sum over set bits i of s of (2^(64*i) * G). Then k*G is just 64
434// doublings + 64 adds (vs kBits doublings for a generic window) — the big win for
435// the per-message signing path. Selector at step j is bit j of each 64-bit limb.
436template <WeierstrassCurve C>
437const std::array<Jac<C>, (1u << C::kLimbs)>& g_comb() {
438 static const std::array<Jac<C>, (1u << C::kLimbs)> tbl = [] {
439 constexpr std::size_t L = C::kLimbs;
440 Jac<C> gi[L];
441 gi[0] = affine_to_jac<C>(C::GX, C::GY);
442 for (std::size_t i = 1; i < L; ++i) {
443 Jac<C> acc = gi[i - 1];
444 for (int b = 0; b < 64; ++b) { // gi[i] = 2^64 * gi[i-1]
445 Jac<C> t;
446 jac_double<C>(t, acc);
447 acc = t;
448 }
449 gi[i] = acc;
450 }
451 std::array<Jac<C>, (1u << L)> t;
452 t[0] = jac_infinity<C>();
453 for (unsigned s = 1; s < (1u << L); ++s) {
454 Jac<C> acc = jac_infinity<C>();
455 for (std::size_t i = 0; i < L; ++i)
456 if (s & (1u << i)) {
457 Jac<C> r;
458 jac_add<C>(r, acc, gi[i]);
459 acc = r;
460 }
461 t[s] = acc;
462 }
463 return t;
464 }();
465 return tbl;
466}

468template <WeierstrassCurve C>
469void jac_mul_base(Jac<C>& r, const fe<C>& k) {
470 const auto& T = g_comb<C>();
471 Jac<C> acc = jac_infinity<C>();
472 for (int j = 63; j >= 0; --j) {
473 Jac<C> t;
474 jac_double<C>(t, acc);
475 acc = t;
476 unsigned sel = 0;
477 for (std::size_t i = 0; i < C::kLimbs; ++i) sel |= ((k[i] >> j) & 1u) << i;
478 if (sel) {
479 jac_add<C>(t, acc, T[sel]);
480 acc = t;
481 }
482 }
483 r = acc;
484}

486// Reduce a scalar already known to be < 2n into [0, n): a single conditional
487// subtraction of the group order n. Used for the FIPS 186-4 hash truncation and
488// for folding a curve x-coordinate (which lives in [0, p) < 2n) into a scalar.
489template <WeierstrassCurve C>
490fe<C> reduce_mod_n(const fe<C>& v) {
491 if (geq<C>(v, C::N)) {
492 fe<C> t;
493 sub_borrow<C>(t, v, C::N);
494 return t;
495 }
496 return v;
497}

499// Reduce a big-endian hash to a scalar in [0, n). A hash of at least kBytes keeps its
500// leftmost kBytes (the FIPS 186-4 leftmost-bits truncation); a SHORTER hash is the whole
501// value (X9.62 bits2int — right-aligned), e.g. a SHA-256 signature under a P-384 key.
502template <WeierstrassCurve C>
503fe<C> hash_to_scalar(const std::string& h) {
504 unsigned char buf[kBytes<C>] = {0};
505 if (h.size() >= kBytes<C>)
506 std::memcpy(buf, h.data(), kBytes<C>);
507 else
508 std::memcpy(buf + (kBytes<C> - h.size()), h.data(), h.size());
509 return reduce_mod_n<C>(be_to_fe<C>(buf));
510}

512// ---- minimal DER helpers ------------------------------------------------------------------------
513// Parse SEQUENCE{INTEGER r, INTEGER s} -> kBytes big-endian r and s. false on error.
514// Short-form lengths only: both curves' SEQUENCE stays under 128 bytes (P-384: <= ~104).
515template <WeierstrassCurve C>
516bool der_to_rs(const std::string& der, unsigned char* r, unsigned char* s) {
517 const unsigned char* p = (const unsigned char*)der.data();
518 std::size_t n = der.size(), i = 0;
519 auto read_int = [&](unsigned char* out) -> bool {
520 if (i >= n || p[i++] != 0x02) return false;
521 if (i >= n) return false;
522 std::size_t len = p[i++];
523 if (len & 0x80) return false; // curve-order ints are short-form
524 if (i + len > n || len == 0) return false;
525 const unsigned char* v = p + i;
526 // strip a leading zero (sign byte)
527 while (len > 1 && v[0] == 0) {
528 ++v;
529 --len;
530 }
531 if (len > kBytes<C>) return false;
532 std::memset(out, 0, kBytes<C>);
533 std::memcpy(out + (kBytes<C> - len), v, len);
534 i += (std::size_t)(v - (p + i)) + len; // advance past the original field
535 return true;
536 };
537 if (i >= n || p[i++] != 0x30) return false;
538 if (i >= n) return false;
539 std::size_t seqlen = p[i++];
540 if (seqlen & 0x80) return false;
541 if (i + seqlen != n) return false;
542 return read_int(r) && read_int(s);
543}

545// Is (x, y) on the curve y^2 = x^3 - 3x + b (mod p)? Rejects an off-curve /
546// invalid-curve public key — SP 800-56A / FIPS 186 point validation, which the plain
547// coordinate-range check (x,y < p) does not catch.
548template <WeierstrassCurve C>
549bool on_curve(const fe<C>& x, const fe<C>& y) {
550 const Mont<C>& F = Fp<C>();
551 fe<C> xm, ym, x2, x3, tx, rhs, bm, y2;
552 to_mont<C>(xm, x, F);
553 to_mont<C>(ym, y, F);
554 mont_mul<C>(x2, xm, xm, F); // x^2
555 mont_mul<C>(x3, x2, xm, F); // x^3
556 mont_add<C>(tx, xm, xm, F); // 2x
557 mont_add<C>(tx, tx, xm, F); // 3x
558 mont_sub<C>(rhs, x3, tx, F); // x^3 - 3x
559 to_mont<C>(bm, C::B, F);
560 mont_add<C>(rhs, rhs, bm, F); // x^3 - 3x + b
561 mont_mul<C>(y2, ym, ym, F); // y^2
562 return std::memcmp(y2.data(), rhs.data(), sizeof(fe<C>)) == 0;
563}

565// ECDSA verification over raw byte forms: pubkey = 2*kBytes X||Y, sig = 2*kBytes r||s.
566template <WeierstrassCurve C>
567bool verify_raw(const std::string& pubkey_xy, const std::string& msg_hash,
568 const std::string& sig_raw) {
569 if (pubkey_xy.size() != 2 * kBytes<C> || sig_raw.size() != 2 * kBytes<C>) return false;
570 fe<C> r = be_to_fe<C>((const unsigned char*)sig_raw.data());
571 fe<C> s = be_to_fe<C>((const unsigned char*)sig_raw.data() + kBytes<C>);
572 if (is_zero<C>(r) || is_zero<C>(s) || geq<C>(r, C::N) || geq<C>(s, C::N)) return false;
573
574 const Mont<C>& Fnn = Fn<C>();
575 fe<C> e = hash_to_scalar<C>(msg_hash);
576 fe<C> sm, em, rm, w, u1, u2;
577 to_mont<C>(sm, s, Fnn);
578 mont_inv<C>(w, sm, Fnn); // w = s^-1 (Montgomery)
579 to_mont<C>(em, e, Fnn);
580 to_mont<C>(rm, r, Fnn);
581 fe<C> u1m, u2m;
582 mont_mul<C>(u1m, em, w, Fnn);
583 mont_mul<C>(u2m, rm, w, Fnn);
584 from_mont<C>(u1, u1m, Fnn);
585 from_mont<C>(u2, u2m, Fnn);
586
587 fe<C> qx = be_to_fe<C>((const unsigned char*)pubkey_xy.data());
588 fe<C> qy = be_to_fe<C>((const unsigned char*)pubkey_xy.data() + kBytes<C>);
589 if (geq<C>(qx, C::P) || geq<C>(qy, C::P)) return false;
590 if (!on_curve<C>(qx, qy)) return false; // reject off-curve / invalid-curve public keys
591 Jac<C> Q = affine_to_jac<C>(qx, qy);
592
593 Jac<C> R;
594 jac_double_mul<C>(R, u1, base_point<C>(), u2, Q); // u1*G + u2*Q, one doubling chain
595 if (R.inf || is_zero<C>(R.Z)) return false;
596 fe<C> x = reduce_mod_n<C>(jac_affine_x<C>(R));
597 return std::memcmp(x.data(), r.data(), sizeof(fe<C>)) == 0;
598}

600// ECDSA verification of the DER form (SEQUENCE{INTEGER r, INTEGER s} — TLS/X.509).
601template <WeierstrassCurve C>
602bool verify_der(const std::string& pubkey_xy, const std::string& msg_hash,
603 const std::string& sig_der) {
604 unsigned char r[kBytes<C>], s[kBytes<C>];
605 if (!der_to_rs<C>(sig_der, r, s)) return false;
606 std::string raw;
607 raw.resize(2 * kBytes<C>);
608 std::memcpy(raw.data(), r, kBytes<C>);
609 std::memcpy(raw.data() + kBytes<C>, s, kBytes<C>);
610 return verify_raw<C>(pubkey_xy, msg_hash, raw);
611}
