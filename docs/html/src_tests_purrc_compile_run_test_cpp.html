26std::string run_capture(const std::string& cmd, int& exit_code) {
27 std::string out;
28 FILE* pipe = popen(cmd.c_str(), "r");
29 if (pipe == nullptr) {
30 exit_code = -1;
31 return out;
32 }
33 std::array<char, 256> buf{};
34 while (std::fgets(buf.data(), static_cast<int>(buf.size()), pipe) != nullptr) {
35 out += buf.data();
36 }
37 const int status = pclose(pipe);
38 exit_code = WIFEXITED(status) ? WEXITSTATUS(status) : -1;
39 return out;
40}

44TEST(PurrcPipeline, CompilesAndRunsHelloMeow) {
45 const std::string tmp = PURR_TEST_TMP;
46 const std::string purr = tmp + "/hellomeow_it.purr";
47 const std::string mod = tmp + "/hellomeow_it.so";
48
49 {
50 std::ofstream f(purr);
51 f << "import io\n";
52 f << "io.print(\"meow\")\n";
53 }
54
55 // 1) Compile with purrc.
56 const std::string compile =
57 std::string(PURRC_PATH) + " \"" + purr + "\" -o \"" + mod + "\"";
58 ASSERT_EQ(std::system(compile.c_str()), 0) << "purrc failed to compile the program";
59
60 // 2) Run it with the cheatah runtime and capture stdout.
61 int rc = -1;
62 const std::string out = run_capture(std::string(CHEATAH_RUNTIME_PATH) + " \"" + mod + "\"", rc);
63 EXPECT_EQ(rc, 0);
64 EXPECT_EQ(out, "meow\n");
65}
