16TEST(CheatahCodegen, EmitsPurrMainWithNamespaceQualifiedCall) {
17 const ParseResult pr = parse_source("import io\nio.print(\"meow\")\n");
18 ASSERT_TRUE(pr.ok());
19
20 const CodegenResult cg = codegen(pr.program);
21 ASSERT_TRUE(cg.ok());
22 EXPECT_TRUE(contains(cg.source, "#include \"io.hpp\""));
23 // The prelude (std headers + built-ins + the PURR_EXPORT macro) comes in through
24 // one consolidated include rather than a dozen repeated #includes per file.
25 EXPECT_TRUE(contains(cg.source, "#include \"cheatah.hpp\""));
26 // purr_main is emitted through the portable export macro (defined in the prelude:
27 // C linkage everywhere, dllexport on Windows so the DLL exposes the entry point).
28 EXPECT_TRUE(contains(cg.source, "PURR_EXPORT void purr_main()"));
29 // The program lives in a dedicated namespace and the exported entry is a trampoline
30 // into it — this is what keeps the module aliases from clashing with global symbols.
31 EXPECT_TRUE(contains(cg.source, "namespace cheatah_program {"));
32 EXPECT_TRUE(contains(cg.source,
33 "PURR_EXPORT void purr_main() { cheatah_program::purr_main(); }"));
34 // Each imported module gets its own short namespace alias, so the call reads
35 // `io::print` (not `cheatah::io::print`). io.print also only streams its args, so a
36 // string literal passes as a bare const char* — no throwaway std::string.
37 EXPECT_TRUE(contains(cg.source, "namespace io = ::cheatah::io;"));
38 EXPECT_TRUE(contains(cg.source, "io::print(\"meow\");"));
39
40 ASSERT_EQ(cg.modules.size(), 1u);
41 EXPECT_EQ(cg.modules[0], "io");
42}

44TEST(CheatahCodegen, StreamingCallsTakeBareStringLiterals) {
45 // io.print streams its args and io.format takes a std::string_view format — both
46 // accept a literal as a bare const char*, no intermediate std::string. A literal
47 // bound to a variable or concatenated still becomes a std::string (it must, to be a
48 // first-class cheatah string).
49 // Lowering test: keep the unused `let s` by disabling dead-variable removal (DCE has
50 // its own tests) so the bound-literal -> std::string lowering is observable.
51 const CodegenResult cg = codegen(
52 parse_source("import io\nio.print(\"a\", io.format(\"{}\", 1))\nlet s = \"b\"\n").program,
53 "", false);
54 ASSERT_TRUE(cg.ok());
55 EXPECT_TRUE(contains(cg.source, "io::print(\"a\""));
56 EXPECT_TRUE(contains(cg.source, "io::format(\"{}\""));
57 EXPECT_FALSE(contains(cg.source, "std::string(\"a\")"));
58 EXPECT_TRUE(contains(cg.source, "auto s = std::string(\"b\");")); // bound var stays std::string
59}

73TEST(CheatahCodegen, NamespaceAliasSkippedWhenNameCollides) {
74 // A program identifier named like an imported module (here a `struct os`) would
75 // clash with `namespace os = cheatah::os;`, so the codegen must NOT alias os — it
76 // stays fully qualified. A non-colliding module (io) is still aliased.
77 const CodegenResult cg = codegen(parse_source(
78 "import os\nimport io\nstruct os { n: int }\nlet p = os(1)\n"
79 "io.print(os.path.join(\"a\", \"b\"), p.n)\n").program);
80 ASSERT_TRUE(cg.ok());
81 EXPECT_TRUE(contains(cg.source, "namespace io = ::cheatah::io;")); // safe -> aliased
82 EXPECT_FALSE(contains(cg.source, "namespace os = ::cheatah::os;")); // collides -> explicit
83 EXPECT_TRUE(contains(cg.source, "::cheatah::os::path::join")); // stays fully qualified
84 EXPECT_TRUE(contains(cg.source, "struct os {"));
85}

123TEST(CheatahCodegen, SelfAppendChainsIntoOneStatement) {
124 // `x = x + "lit" + y` lowers to a single CHAINED in-place append —
125 // `(x += "lit") += y;` — not three `x += …` lines and not an intermediate
126 // std::string. The literal appends as a bare const char*; operator+= chains
127 // because it returns a reference to x.
128 const ParseResult pr =
129 parse_source("let head = \"a\"\nlet y = \"b\"\nhead = head + \"lit\" + y\n");
130 ASSERT_TRUE(pr.ok());
131 const CodegenResult cg = codegen(pr.program);
132 ASSERT_TRUE(cg.ok());
133 EXPECT_TRUE(contains(cg.source, "(head += \"lit\") += y;")); // one chained statement
134 EXPECT_FALSE(contains(cg.source, "std::string(\"lit\")")); // no temporary
135 // A single appended operand stays a plain `x += e;` (nothing to chain).
136 const CodegenResult one = codegen(parse_source("let s = \"a\"\ns = s + \"b\"\n").program);
137 ASSERT_TRUE(one.ok());
138 EXPECT_TRUE(contains(one.source, "s += \"b\";"));
139}

334TEST(CheatahCodegen, CppBlockTopLevelIsFileScopeNestedIsInline) {
335 const ParseResult pr = parse_source(
336 "cpp {\nstatic int helper() { return 7; }\n}\n"
337 "fn f() {\ncpp { int x = 1; }\n}\n");
338 ASSERT_TRUE(pr.ok()) << (pr.diagnostics.empty() ? "" : pr.diagnostics.front().message);
339 const CodegenResult cg = codegen(pr.program);
340 ASSERT_TRUE(cg.ok());
341 // Raw bodies are emitted verbatim.
342 EXPECT_TRUE(contains(cg.source, "static int helper() { return 7; }"));
343 EXPECT_TRUE(contains(cg.source, "int x = 1;"));
344 // The top-level block lands at FILE SCOPE — before purr_main.
345 const auto helper_pos = cg.source.find("static int helper()");
346 const auto main_pos = cg.source.find("purr_main");
347 ASSERT_NE(helper_pos, std::string::npos);
348 ASSERT_NE(main_pos, std::string::npos);
349 EXPECT_LT(helper_pos, main_pos);
350 // The nested block lands INSIDE the function f().
351 const auto f_pos = cg.source.find("auto f(");
352 const auto inline_pos = cg.source.find("int x = 1;");
353 ASSERT_NE(f_pos, std::string::npos);
354 EXPECT_GT(inline_pos, f_pos);
