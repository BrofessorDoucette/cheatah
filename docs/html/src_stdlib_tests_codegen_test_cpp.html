18TEST(CheatahCodegen, EmitsPurrMainWithNamespaceQualifiedCall) {
19 const ParseResult pr = parse_source("import io\nio.print(\"meow\")\n");
20 ASSERT_TRUE(pr.ok());
21
22 const CodegenResult cg = codegen(pr.program);
23 ASSERT_TRUE(cg.ok());
24 EXPECT_TRUE(contains(cg.source, "#include \"io.hpp\""));
25 // The prelude (std headers + built-ins + the PURR_EXPORT macro) comes in through
26 // one consolidated include rather than a dozen repeated #includes per file.
27 EXPECT_TRUE(contains(cg.source, "#include \"cheatah.hpp\""));
28 // purr_main is emitted through the portable export macro (defined in the prelude:
29 // C linkage everywhere, dllexport on Windows so the DLL exposes the entry point).
30 EXPECT_TRUE(contains(cg.source, "PURR_EXPORT void purr_main()"));
31 // The program lives in a dedicated namespace and the exported entry is a trampoline
32 // into it — this is what keeps the module aliases from clashing with global symbols.
33 EXPECT_TRUE(contains(cg.source, "namespace cheatah_program {"));
34 EXPECT_TRUE(contains(cg.source,
35 "PURR_EXPORT void purr_main() { cheatah_program::purr_main(); }"));
36 // Each imported module gets its own short namespace alias, so the call reads
37 // `io::print` (not `cheatah::io::print`). io.print also only streams its args, so a
38 // string literal passes as a bare const char* — no throwaway std::string.
39 EXPECT_TRUE(contains(cg.source, "namespace io = ::cheatah::io;"));
40 EXPECT_TRUE(contains(cg.source, "io::print(\"meow\");"));
41
42 ASSERT_EQ(cg.modules.size(), 1u);
43 EXPECT_EQ(cg.modules[0], "io");
44}

105TEST(CheatahCodegen, DottedImportDoesNotDoubleTheSubmoduleSegment) {

107 // full — `a.b.fn(...)` — qualifies to `a::b::fn`, NOT `a::b::b::fn`. (Regression: binding

113 EXPECT_FALSE(contains(cg.source, "os::path::path::join")); // the doubled form must not appear

176 // `x = x + "lit" + y` lowers to a single CHAINED in-place append —
177 // `(x += "lit") += y;` — not three `x += …` lines and not an intermediate
178 // std::string. The literal appends as a bare const char*; operator+= chains

187 // A single appended operand stays a plain `x += e;` (nothing to chain).
188 const CodegenResult one = codegen(parse_source("let s = \"a\"\ns = s + \"b\"\n").program);

423TEST(CheatahCodegen, CppBlockTopLevelIsFileScopeNestedIsInline) {
424 const ParseResult pr = parse_source(
425 "cpp {\nstatic int helper() { return 7; }\n}\n"
426 "fn f() {\ncpp { int x = 1; }\n}\n");
427 ASSERT_TRUE(pr.ok()) << (pr.diagnostics.empty() ? "" : pr.diagnostics.front().message);
428 const CodegenResult cg = codegen(pr.program);
429 ASSERT_TRUE(cg.ok());
430 // Raw bodies are emitted verbatim.
431 EXPECT_TRUE(contains(cg.source, "static int helper() { return 7; }"));
432 EXPECT_TRUE(contains(cg.source, "int x = 1;"));
433 // The top-level block lands at FILE SCOPE — before purr_main.
434 const auto helper_pos = cg.source.find("static int helper()");
435 const auto main_pos = cg.source.find("purr_main");
436 ASSERT_NE(helper_pos, std::string::npos);
437 ASSERT_NE(main_pos, std::string::npos);
