115 ASSERT_TRUE(pr.ok());

171TEST(CheatahCodegen, CppBlockTopLevelIsFileScopeNestedIsInline) {
172 const ParseResult pr = parse_source(
173 "cpp {\nstatic int helper() { return 7; }\n}\n"
174 "fn f() {\ncpp { int x = 1; }\n}\n");
175 ASSERT_TRUE(pr.ok()) << (pr.diagnostics.empty() ? "" : pr.diagnostics.front().message);
176 const CodegenResult cg = codegen(pr.program);
177 ASSERT_TRUE(cg.ok());
178 // Raw bodies are emitted verbatim.
179 EXPECT_TRUE(contains(cg.source, "static int helper() { return 7; }"));
180 EXPECT_TRUE(contains(cg.source, "int x = 1;"));
181 // The top-level block lands at FILE SCOPE — before purr_main.
182 const auto helper_pos = cg.source.find("static int helper()");
183 const auto main_pos = cg.source.find("purr_main");
184 ASSERT_NE(helper_pos, std::string::npos);
185 ASSERT_NE(main_pos, std::string::npos);
186 EXPECT_LT(helper_pos, main_pos);
187 // The nested block lands INSIDE the function f().
188 const auto f_pos = cg.source.find("auto f(");
189 const auto inline_pos = cg.source.find("int x = 1;");
190 ASSERT_NE(f_pos, std::string::npos);
191 EXPECT_GT(inline_pos, f_pos);
