20// Build a raw (unmasked, server-style) WebSocket frame: FIN|opcode, a length encoding, payload.
21std::string frame(unsigned char b0, const std::string& payload) {
22 std::string f(1, static_cast<char>(b0));
23 const std::uint64_t n = payload.size();
24 if (n < 126) {
25 f.push_back(static_cast<char>(n));
26 } else {
27 f.push_back(static_cast<char>(126));
28 f.push_back(static_cast<char>((n >> 8) & 0xFF));
29 f.push_back(static_cast<char>(n & 0xFF));
30 }
31 return f + payload;
32}
33
34// A frame HEADER that DECLARES a length via the 64-bit field but carries no payload — the shape a
35// malicious server uses to overflow `header + len`. The cap must reject it before any read/copy.
36std::string len64_header(unsigned char b0, std::uint64_t declared) {
37 std::string f(1, static_cast<char>(b0));
38 f.push_back(static_cast<char>(127));
39 for (int sh = 56; sh >= 0; sh -= 8) f.push_back(static_cast<char>((declared >> sh) & 0xFF));
40 return f;
41}
42
43// A control-frame header declaring a length via the 16-bit field, no payload.
44std::string len16_header(unsigned char b0, std::uint64_t declared) {
45 std::string f(1, static_cast<char>(b0));
46 f.push_back(static_cast<char>(126));
47 f.push_back(static_cast<char>((declared >> 8) & 0xFF));
48 f.push_back(static_cast<char>(declared & 0xFF));
49 return f;
50}
51
52// Drive recv() over a synthetic session pre-loaded with `bytes` (frame-parser fuzzing seam), then
53// free it. Returns recv()'s result; rethrows whatever recv throws (after freeing the session).
54std::string recv_bytes(const std::string& bytes, std::uint64_t max_frame = 0,
55 std::uint64_t max_message = 0) {
56 const long long h = ws::testonly::session_from_bytes(bytes, max_frame, max_message);
57 std::string out;
58 try {
59 out = ws::recv(h);
60 } catch (...) {
61 ws::close(h);
62 throw;
63 }
64 ws::close(h);
65 return out;
66}

70// === Red-team: a malicious server must not corrupt memory, OOM, or bypass RFC 6455 framing. ===
71
72// F1 (CRITICAL): a 64-bit length near 2^64 must be rejected BEFORE it overflows `header + len`
73// and drives an out-of-bounds unmask/copy. This is the memory-corruption bug.
74TEST(CheatahWebSocket, RejectsOverflowingFrameLength) {
75 // opcode 0x2 (binary), FIN; declared length = 0xFFFFFFFFFFFFFFFF (would wrap header+len).
76 EXPECT_THROW(recv_bytes(len64_header(0x82, 0xFFFFFFFFFFFFFFFFull)), std::runtime_error);
77 // A merely-huge (non-wrapping) length is rejected the same way (would otherwise OOM).
78 EXPECT_THROW(recv_bytes(len64_header(0x82, 8ull << 30)), std::runtime_error); // 8 GiB
79 // Even the MASKED path (the actual out-of-bounds-write trigger) is refused before unmasking.
80 std::string masked = len64_header(0x82, 1ull << 40);
81 masked[1] = static_cast<char>(0x80 | 127); // set the MASK bit in b1
82 EXPECT_THROW(recv_bytes(masked), std::runtime_error);
83}
84
85// F1: a frame just over the (here-tiny) cap is rejected; one at the cap is accepted.
86TEST(CheatahWebSocket, FramePayloadCapEnforced) {
87 EXPECT_THROW(recv_bytes(frame(0x82, std::string(101, 'x')), /*max_frame=*/100), std::runtime_error);
88 EXPECT_EQ(recv_bytes(frame(0x82, std::string(100, 'x')), /*max_frame=*/100), std::string(100, 'x'));
89}
90
91// F2 (OOM): a fragmented message that would exceed the reassembly cap is rejected.
92TEST(CheatahWebSocket, ReassembledMessageCapEnforced) {
93 // First fragment (text, FIN=0) of 5 bytes, then a continuation of 5 more; cap = 8 < 10.
94 const std::string frames = frame(0x01, "aaaaa") + frame(0x80, "bbbbb");
95 EXPECT_THROW(recv_bytes(frames, /*max_frame=*/0, /*max_message=*/8), std::runtime_error);
96}

98// F3: control frames must be <=125 bytes and MUST NOT be fragmented (RFC 6455 §5.5).
99TEST(CheatahWebSocket, RejectsOversizedControlFrame) {
100 EXPECT_THROW(recv_bytes(len16_header(0x89, 200)), std::runtime_error); // ping, 200 bytes
101}
102TEST(CheatahWebSocket, RejectsFragmentedControlFrame) {
103 EXPECT_THROW(recv_bytes(frame(0x09, "abc")), std::runtime_error); // ping, FIN=0
104}

106// F4: reserved bits set (no extension negotiated) and undefined opcodes fail the connection.
107TEST(CheatahWebSocket, RejectsReservedBits) {
108 EXPECT_THROW(recv_bytes(frame(0xC1, "")), std::runtime_error); // RSV1 | text | FIN
109}
110TEST(CheatahWebSocket, RejectsUnknownOpcode) {
111 EXPECT_THROW(recv_bytes(frame(0x83, "")), std::runtime_error); // opcode 0x3 (undefined)
112}

114// Fragmentation state machine: a stray continuation, or a new data frame mid-message, is invalid.
115TEST(CheatahWebSocket, RejectsContinuationWithNoMessage) {
116 EXPECT_THROW(recv_bytes(frame(0x80, "x")), std::runtime_error); // continuation, FIN, no msg
117}
118TEST(CheatahWebSocket, RejectsNewDataFrameDuringFragment) {
119 const std::string frames = frame(0x01, "ab") + frame(0x81, "cd"); // text(FIN=0) then text(FIN)
120 EXPECT_THROW(recv_bytes(frames), std::runtime_error);
121}

123// === Blue-team: the valid paths still work (single frame, fragmentation, an interleaved pong). ===
124TEST(CheatahWebSocket, AcceptsValidSingleFrame) {
125 EXPECT_EQ(recv_bytes(frame(0x81, "hello")), "hello"); // text, FIN
126 const std::string bin("\x00\x01\x02", 3); // NUL-containing binary payload
127 EXPECT_EQ(recv_bytes(frame(0x82, bin)), bin); // binary, FIN — byte-safe
128}
129TEST(CheatahWebSocket, AcceptsFragmentedMessage) {
130 const std::string frames = frame(0x01, "he") + frame(0x00, "l") + frame(0x80, "lo");
131 EXPECT_EQ(recv_bytes(frames), "hello");
132}
133TEST(CheatahWebSocket, SkipsPongThenReturnsData) {
134 const std::string frames = frame(0x8A, "") + frame(0x81, "ok"); // pong (ignored), then text
135 EXPECT_EQ(recv_bytes(frames), "ok");
136}

138TEST(CheatahWebSocket, ConnectUrlRejectsNonWss) {
139 // Only wss:// is supported; a non-wss scheme fails fast, before any socket work. open_url()
140 // is the cheatah-facing guard factory (it delegates to the C++-only connect_url()).
141 EXPECT_THROW(ws::open_url("ws://example.com/"), std::runtime_error);
142 EXPECT_THROW(ws::open_url("https://example.com/"), std::runtime_error);
143 EXPECT_THROW(ws::open_url("example.com"), std::runtime_error);
144}

146// A default-constructed Client owns nothing: closed, id() == 0, and close() reports -1 without
147// touching a session. open_url() on a non-wss scheme throws before a session is ever created,
148// so the guard is never left holding a bogus handle. (Functional send/recv/close over a live
149// session are covered by WebSocketSys.EchoRoundTrip against a real wss peer.)
150TEST(CheatahWebSocket, ClientDefaultIsClosed) {
151 ws::Client c;
152 EXPECT_FALSE(c.is_open());
153 EXPECT_EQ(c.id(), 0);
154 EXPECT_EQ(c.close(), -1); // nothing to close
155 EXPECT_THROW(ws::open_url("ws://example.com/"), std::runtime_error);
156}
