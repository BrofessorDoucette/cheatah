stdlib/tests/websocket_test.cpp

1// websocket_test — offline unit checks for the websocket client. The framing
2// and live round-trip are covered by the system tests (a real WSS server);
3// here we cover the input validation that needs no network.
4
5#include <stdexcept>
6#include <string>
7
8#include <gtest/gtest.h>
9
10#include "websocket.hpp"
11
12namespace ws = cheatah::websocket;
13
14TEST(CheatahWebSocket, ConnectUrlRejectsNonWss) {
15 // Only wss:// is supported; a non-wss scheme fails fast, before any socket work. open_url()
16 // is the cheatah-facing guard factory (it delegates to the C++-only connect_url()).
17 EXPECT_THROW(ws::open_url("ws://example.com/"), std::runtime_error);
18 EXPECT_THROW(ws::open_url("https://example.com/"), std::runtime_error);
19 EXPECT_THROW(ws::open_url("example.com"), std::runtime_error);
20}
21
22// A default-constructed Client owns nothing: closed, id() == 0, and close() reports -1 without
23// touching a session. open_url() on a non-wss scheme throws before a session is ever created,
24// so the guard is never left holding a bogus handle. (Functional send/recv/close over a live
25// session are covered by WebSocketSys.EchoRoundTrip against a real wss peer.)
26TEST(CheatahWebSocket, ClientDefaultIsClosed) {
27 ws::Client c;
28 EXPECT_FALSE(c.is_open());
29 EXPECT_EQ(c.id(), 0);
30 EXPECT_EQ(c.close(), -1); // nothing to close
31 EXPECT_THROW(ws::open_url("ws://example.com/"), std::runtime_error);
32}
