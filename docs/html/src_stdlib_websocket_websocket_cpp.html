24// A session is heap-allocated and its address IS the handle — so send/recv do a
25// single pointer cast, never a map lookup or lock, on the hot path (cheatah is
26// single-trust; an invalid handle is a caller bug, like a bad pointer in C).
27struct Session {
28 long long fd = -1; // the TCP fd (we own it; close it ourselves)
29 long long tls = -1; // the TLS session riding that fd
30 std::string buf; // read buffer, REUSED across every frame
31 std::size_t pos = 0; // parse offset into buf (consumed prefix is buf[0,pos))
32 bool closed = false;
33 std::uint64_t max_frame = 0; // per-frame payload cap (0 -> the kMaxFramePayload default)
34 std::uint64_t max_message = 0; // reassembled-message cap (0 -> the kMaxMessageBytes default)
35};
36
37// Hard caps a client MUST impose itself — RFC 6455 sets no upper bound on a frame or a
38// reassembled message, so a malicious server could otherwise (a) overflow `header + len`
39// in size_t math and drive an out-of-bounds unmask, or (b) stream an unbounded body to
40// exhaust memory. A single frame's payload and the reassembled message are each capped;
41// anything larger fails the connection. Control frames are bounded to 125 bytes (§5.5).
42constexpr std::uint64_t kMaxFramePayload = 64ull << 20; // 64 MiB per frame
43constexpr std::uint64_t kMaxMessageBytes = 64ull << 20; // 64 MiB reassembled total
44constexpr std::uint64_t kMaxControlPayload = 125; // RFC 6455 §5.5
45
46[[nodiscard]] Session* as_session(long long h) {
47 return reinterpret_cast<Session*>(static_cast<std::uintptr_t>(h));
48}
49[[nodiscard]] long long as_handle(Session* s) {
50 return static_cast<long long>(reinterpret_cast<std::uintptr_t>(s));
51}

53// Standard base64 (for the Sec-WebSocket-Key). Tiny inputs (16/raw bytes).
54[[nodiscard]] std::string base64(const std::string& in) {
55 static const char* T = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";
56 std::string out;
57 out.reserve((in.size() + 2) / 3 * 4);
58 std::size_t i = 0;
59 for (; i + 3 <= in.size(); i += 3) {
60 const std::uint32_t n = (static_cast<unsigned char>(in[i]) << 16) |
61 (static_cast<unsigned char>(in[i + 1]) << 8) |
62 static_cast<unsigned char>(in[i + 2]);
63 out.push_back(T[(n >> 18) & 63]);
64 out.push_back(T[(n >> 12) & 63]);
65 out.push_back(T[(n >> 6) & 63]);
66 out.push_back(T[n & 63]);
67 }
68 if (i < in.size()) {
69 std::uint32_t n = static_cast<unsigned char>(in[i]) << 16;
70 if (i + 1 < in.size()) n |= static_cast<unsigned char>(in[i + 1]) << 8;
71 out.push_back(T[(n >> 18) & 63]);
72 out.push_back(T[(n >> 12) & 63]);
73 out.push_back(i + 1 < in.size() ? T[(n >> 6) & 63] : '=');
74 out.push_back('=');
75 }
76 return out;
77}

79void destroy(Session* s) {
80 if (s == nullptr) return;
81 if (s->tls >= 0) tls::close(s->tls);
82 if (s->fd >= 0) socket::close(s->fd);
83 delete s;
84}

86// Ensure the buffer holds at least @p need unconsumed bytes (from pos), draining
87// the TLS stream in big chunks. Compacts the consumed prefix only when needed,
88// so steady-state framing does no front-erase per frame.
89void ensure(Session* s, std::size_t need) {
90 while (s->buf.size() - s->pos < need) {
91 if (s->pos > 0 && (s->pos == s->buf.size() || s->pos >= (1u << 16))) {
92 s->buf.erase(0, s->pos);
93 s->pos = 0;
94 }
95 std::string chunk = tls::recv(s->tls, 1 << 16); // 64 KiB drains
96 if (chunk.empty()) throw std::runtime_error("websocket: connection closed by peer");
97 s->buf.append(chunk);
98 }
99}

101// Build one frame header for a client (always-masked) text/control send.
102void put_header(std::string& frame, unsigned char opcode, std::size_t n) {
103 frame.push_back(static_cast<char>(0x80 | opcode)); // FIN | opcode
104 if (n < 126) {
105 frame.push_back(static_cast<char>(0x80 | n)); // MASK | len
106 } else if (n <= 0xFFFF) {
107 frame.push_back(static_cast<char>(0x80 | 126));
108 frame.push_back(static_cast<char>((n >> 8) & 0xFF));
109 frame.push_back(static_cast<char>(n & 0xFF));
110 } else {
111 frame.push_back(static_cast<char>(0x80 | 127));
112 for (int sh = 56; sh >= 0; sh -= 8) frame.push_back(static_cast<char>((n >> sh) & 0xFF));
113 }
114}

116// XOR-mask @p n bytes of @p src into @p dst with the 4-byte @p key, eight bytes
117// per step (the key tiled into a 64-bit word) — the fast masking path.
118void mask_into(char* dst, const char* src, std::size_t n, const unsigned char key[4]) {
119 std::uint64_t k64;
120 unsigned char tile[8] = {key[0], key[1], key[2], key[3], key[0], key[1], key[2], key[3]};
121 std::memcpy(&k64, tile, 8);
122 std::size_t i = 0;
123 for (; i + 8 <= n; i += 8) {
124 std::uint64_t w;
125 std::memcpy(&w, src + i, 8);
126 w ^= k64;
127 std::memcpy(dst + i, &w, 8);
128 }
129 for (; i < n; ++i) dst[i] = static_cast<char>(src[i] ^ key[i & 3]);
130}

132long long send_frame(Session* s, unsigned char opcode, const std::string& payload) {
133 const std::size_t n = payload.size();
134 std::string frame;
135 frame.reserve(n + 14);
136 put_header(frame, opcode, n);
137 const std::string key = os::urandom(4);
138 const unsigned char k[4] = {static_cast<unsigned char>(key[0]), static_cast<unsigned char>(key[1]),
139 static_cast<unsigned char>(key[2]), static_cast<unsigned char>(key[3])};
140 frame.append(reinterpret_cast<const char*>(k), 4);
141 const std::size_t off = frame.size();
142 frame.resize(off + n);
143 mask_into(frame.data() + off, payload.data(), n, k);
144 if (tls::send(s->tls, frame) < 0)
145 throw std::runtime_error("websocket: TLS send failed: " + tls::last_error());
146 return static_cast<long long>(n);
147}

151/// @cond INTERNAL — the C++-only low-level session API (websocket_lowlevel.hpp); cheatah uses the Client guard
152long long connect(const std::string& host, long long port, const std::string& path,
153 const std::string& server_name, bool insecure, const std::string& ca_file) {
154 const long long fd = socket::tcp_connect(host, port);
155 if (fd < 0) throw std::runtime_error("websocket: TCP connect to " + host + " failed");
156 const long long tlss = tls::client_connect(fd, server_name, insecure, ca_file);
157 if (tlss < 0) {
158 socket::close(fd);
159 throw std::runtime_error("websocket: TLS handshake failed: " + tls::last_error());
160 }
161 Session* s = new Session();
162 s->fd = fd;
163 s->tls = tlss;
164
165 const std::string key = base64(os::urandom(16));
166 const std::string req = "GET " + path + " HTTP/1.1\r\n" + "Host: " + server_name + "\r\n" +
167 "Upgrade: websocket\r\n" + "Connection: Upgrade\r\n" +
168 "Sec-WebSocket-Key: " + key + "\r\n" + "Sec-WebSocket-Version: 13\r\n\r\n";
169 if (tls::send(tlss, req) < 0) {
170 const std::string err = tls::last_error();
171 destroy(s);
172 throw std::runtime_error("websocket: upgrade request failed: " + err);
173 }
174
175 // Read the response header block (up to and including the blank line). Any
176 // bytes after it are the start of the WebSocket frame stream and stay in buf.
177 std::size_t hdr_end = std::string::npos;
178 for (;;) {
179 const std::string chunk = tls::recv(tlss, 1 << 12);
180 if (chunk.empty()) {
181 destroy(s);
182 throw std::runtime_error("websocket: connection closed during upgrade");
183 }
184 s->buf.append(chunk);
185 hdr_end = s->buf.find("\r\n\r\n");
186 if (hdr_end != std::string::npos) break;
187 if (s->buf.size() > (1u << 16)) {
188 destroy(s);
189 throw std::runtime_error("websocket: upgrade response too large / not a WebSocket server");
190 }
191 }
192 const std::string status = s->buf.substr(0, s->buf.find("\r\n"));
193 if (status.find(" 101") == std::string::npos) {
194 destroy(s);
195 throw std::runtime_error("websocket: server did not switch protocols: " + status);
196 }
197 s->pos = hdr_end + 4; // frame stream begins after the blank line
198 return as_handle(s);
199}

201long long connect_url(const std::string& url, bool insecure, const std::string& ca_file) {
202 const std::string scheme = "wss://";
203 if (url.compare(0, scheme.size(), scheme) != 0)
204 throw std::runtime_error("websocket: only wss:// URLs are supported: " + url);
205 const std::size_t host_start = scheme.size();
206 const std::size_t slash = url.find('/', host_start);
207 const std::string authority =
208 url.substr(host_start, slash == std::string::npos ? std::string::npos : slash - host_start);
209 const std::string path = slash == std::string::npos ? "/" : url.substr(slash);
210 const std::size_t colon = authority.find(':');
211 const std::string host = authority.substr(0, colon);
212 const long long port = colon == std::string::npos
213 ? 443
214 : static_cast<long long>(std::stol(authority.substr(colon + 1)));
215 return connect(host, port, path, host, insecure, ca_file);
216}

222std::string recv(long long session) {
223 Session* s = as_session(session);
224 if (s->closed) return std::string();
225 const std::uint64_t max_frame = s->max_frame != 0 ? s->max_frame : kMaxFramePayload;
226 const std::uint64_t max_message = s->max_message != 0 ? s->max_message : kMaxMessageBytes;
227 std::string message; // reassembly buffer for fragmented messages
228 bool fragmenting = false; // mid multi-frame message
229 for (;;) {
230 ensure(s, 2);
231 const unsigned char b0 = static_cast<unsigned char>(s->buf[s->pos]);
232 const unsigned char b1 = static_cast<unsigned char>(s->buf[s->pos + 1]);
233 const bool fin = (b0 & 0x80) != 0;
234 const int opcode = b0 & 0x0F;
235 const bool masked = (b1 & 0x80) != 0; // server frames are unmasked
236 std::uint64_t len = b1 & 0x7F;
237 std::size_t header = 2;
238 if (len == 126) {
239 ensure(s, 4);
240 len = (static_cast<std::uint64_t>(static_cast<unsigned char>(s->buf[s->pos + 2])) << 8) |
241 static_cast<unsigned char>(s->buf[s->pos + 3]);
242 header = 4;
243 } else if (len == 127) {
244 ensure(s, 10);
245 len = 0;
246 for (int i = 0; i < 8; ++i)
247 len = (len << 8) | static_cast<unsigned char>(s->buf[s->pos + 2 + i]);
248 header = 10;
249 }
250 // Validate BEFORE trusting `len` in size math or a copy: reserved bits must be
251 // clear (no extension negotiated), a control frame must be <=125 bytes and not
252 // fragmented, and no frame may exceed the cap. Capping `len` here is what makes
253 // `header + len` below unable to overflow size_t.
254 if ((b0 & 0x70) != 0) throw std::runtime_error("websocket: reserved bits set");
255 if ((opcode & 0x08) != 0 && (len > kMaxControlPayload || !fin))
256 throw std::runtime_error("websocket: invalid control frame");
257 if (len > max_frame) throw std::runtime_error("websocket: frame too large");
258
259 std::size_t mask_off = 0;
260 if (masked) {
261 mask_off = header;
262 header += 4;
263 }
264 ensure(s, header + len);
265 char* payload = s->buf.data() + s->pos + header;
266 if (masked) { // protocol-irregular for a server, but unmask defensively
267 const unsigned char k[4] = {static_cast<unsigned char>(s->buf[s->pos + mask_off]),
268 static_cast<unsigned char>(s->buf[s->pos + mask_off + 1]),
269 static_cast<unsigned char>(s->buf[s->pos + mask_off + 2]),
270 static_cast<unsigned char>(s->buf[s->pos + mask_off + 3])};
271 mask_into(payload, payload, len, k);
272 }
273
274 switch (opcode) {
275 case 0x9: { // ping -> pong with the same payload (control, off hot path)
276 send_frame(s, 0xA, std::string(payload, len));
277 s->pos += header + len;
278 continue;
279 }
280 case 0xA: // pong -> ignore
281 s->pos += header + len;
282 continue;
283 case 0x8: { // close -> echo close, mark done, signal EOF
284 send_frame(s, 0x8, std::string());
285 s->pos += header + len;
286 s->closed = true;
287 return std::string();
288 }
289 case 0x1: // text
290 case 0x2: // binary
291 if (fragmenting)
292 throw std::runtime_error("websocket: new data frame during a fragmented message");
293 if (fin) { // single-frame message (the common case) — no reassembly buffer
294 std::string out(payload, len);
295 s->pos += header + len;
296 return out;
297 }
298 message.append(payload, len); // first fragment (frame cap bounds it; total capped below)
299 s->pos += header + len;
300 fragmenting = true;
301 continue;
302 case 0x0: // continuation
303 if (!fragmenting)
304 throw std::runtime_error("websocket: continuation frame with no message in progress");
305 if (message.size() + len > max_message)
306 throw std::runtime_error("websocket: message too large");
307 message.append(payload, len);
308 s->pos += header + len;
309 if (fin) return message;
310 continue;
311 default: // 0x3-0x7, 0xB-0xF are reserved/undefined — fail the connection
312 throw std::runtime_error("websocket: unknown opcode");
313 }
314 }

334long long shutdown(long long session) {
335 // Wake a reader blocked in recv() WITHOUT freeing the session: half-close the
336 // underlying socket so recv returns "". The owner still calls close() after it
337 // has joined the reader. Safe to call from another thread concurrently with the
338 // reader's recv (that is exactly what ::shutdown is for).
339 Session* s = as_session(session);
340 if (s == nullptr || s->tls < 0) return -1;
341 return tls::shutdown(s->tls);
342}
343
344long long close(long long session) {
345 Session* s = as_session(session);
346 if (s == nullptr) return 0;
347 if (!s->closed && s->tls >= 0) {
348 try {
349 send_frame(s, 0x8, std::string());
350 } catch (...) {
351 }
352 }
353 destroy(s);
354 return 0;
355}

358// ---- owning RAII client ----
359// Each method forwards to the handle-based free function above; the guard adds deterministic
360// close() (close frame + TLS + socket teardown) on scope exit, so a `with` block cannot leak.

362Client& Client::operator=(Client&& other) noexcept {
363 if (this != &other) {
364 if (session_ != 0) cheatah::websocket::close(session_);
365 session_ = other.session_;
366 other.session_ = 0;
367 }
368 return *this;
369}
