2// cheatah tls 1.3 client + socket. See websocket.hpp for the contract and the
3// performance design (reused read buffer, unmasked-recv fast path, word-XOR
4// masking on send, in-place header parsing, large TLS drains).

22// A session is heap-allocated and its address IS the handle — so send/recv do a
23// single pointer cast, never a map lookup or lock, on the hot path (cheatah is
24// single-trust; an invalid handle is a caller bug, like a bad pointer in C).
25struct Session {
26 long long fd = -1; // the TCP fd (we own it; close it ourselves)
27 long long tls = -1; // the TLS session riding that fd
28 std::string buf; // read buffer, REUSED across every frame
29 std::size_t pos = 0; // parse offset into buf (consumed prefix is buf[0,pos))
30 bool closed = false;
31};

33[[nodiscard]] Session* as_session(long long h) {
34 return reinterpret_cast<Session*>(static_cast<std::uintptr_t>(h));
35}
36[[nodiscard]] long long as_handle(Session* s) {
37 return static_cast<long long>(reinterpret_cast<std::uintptr_t>(s));

40// Standard base64 (for the Sec-WebSocket-Key). Tiny inputs (16/raw bytes).
41[[nodiscard]] std::string base64(const std::string& in) {
42 static const char* T = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";
43 std::string out;
44 out.reserve((in.size() + 2) / 3 * 4);
45 std::size_t i = 0;
46 for (; i + 3 <= in.size(); i += 3) {
47 const std::uint32_t n = (static_cast<unsigned char>(in[i]) << 16) |
48 (static_cast<unsigned char>(in[i + 1]) << 8) |
49 static_cast<unsigned char>(in[i + 2]);
50 out.push_back(T[(n >> 18) & 63]);
51 out.push_back(T[(n >> 12) & 63]);
52 out.push_back(T[(n >> 6) & 63]);
53 out.push_back(T[n & 63]);
54 }
55 if (i < in.size()) {
56 std::uint32_t n = static_cast<unsigned char>(in[i]) << 16;
57 if (i + 1 < in.size()) n |= static_cast<unsigned char>(in[i + 1]) << 8;
58 out.push_back(T[(n >> 18) & 63]);
59 out.push_back(T[(n >> 12) & 63]);
60 out.push_back(i + 1 < in.size() ? T[(n >> 6) & 63] : '=');
61 out.push_back('=');
62 }
63 return out;
64}

66void destroy(Session* s) {
67 if (s == nullptr) return;
68 if (s->tls >= 0) tls::close(s->tls);
69 if (s->fd >= 0) socket::close(s->fd);
70 delete s;
71}

73// Ensure the buffer holds at least @p need unconsumed bytes (from pos), draining
74// the TLS stream in big chunks. Compacts the consumed prefix only when needed,
75// so steady-state framing does no front-erase per frame.
76void ensure(Session* s, std::size_t need) {
77 while (s->buf.size() - s->pos < need) {
78 if (s->pos > 0 && (s->pos == s->buf.size() || s->pos >= (1u << 16))) {
79 s->buf.erase(0, s->pos);
80 s->pos = 0;
81 }
82 std::string chunk = tls::recv(s->tls, 1 << 16); // 64 KiB drains
83 if (chunk.empty()) throw std::runtime_error("websocket: connection closed by peer");
84 s->buf.append(chunk);
85 }
86}

88// Build one frame header for a client (always-masked) text/control send.
89void put_header(std::string& frame, unsigned char opcode, std::size_t n) {
90 frame.push_back(static_cast<char>(0x80 | opcode)); // FIN | opcode
91 if (n < 126) {
92 frame.push_back(static_cast<char>(0x80 | n)); // MASK | len
93 } else if (n <= 0xFFFF) {
94 frame.push_back(static_cast<char>(0x80 | 126));
95 frame.push_back(static_cast<char>((n >> 8) & 0xFF));
96 frame.push_back(static_cast<char>(n & 0xFF));
97 } else {
98 frame.push_back(static_cast<char>(0x80 | 127));
99 for (int sh = 56; sh >= 0; sh -= 8) frame.push_back(static_cast<char>((n >> sh) & 0xFF));
100 }
101}

103// XOR-mask @p n bytes of @p src into @p dst with the 4-byte @p key, eight bytes
104// per step (the key tiled into a 64-bit word) — the fast masking path.
105void mask_into(char* dst, const char* src, std::size_t n, const unsigned char key[4]) {
106 std::uint64_t k64;
107 unsigned char tile[8] = {key[0], key[1], key[2], key[3], key[0], key[1], key[2], key[3]};
108 std::memcpy(&k64, tile, 8);
109 std::size_t i = 0;
110 for (; i + 8 <= n; i += 8) {
111 std::uint64_t w;
112 std::memcpy(&w, src + i, 8);
113 w ^= k64;
114 std::memcpy(dst + i, &w, 8);
115 }
116 for (; i < n; ++i) dst[i] = static_cast<char>(src[i] ^ key[i & 3]);
117}

119long long send_frame(Session* s, unsigned char opcode, const std::string& payload) {
120 const std::size_t n = payload.size();
121 std::string frame;
122 frame.reserve(n + 14);
123 put_header(frame, opcode, n);
124 const std::string key = os::urandom(4);
125 const unsigned char k[4] = {static_cast<unsigned char>(key[0]), static_cast<unsigned char>(key[1]),
126 static_cast<unsigned char>(key[2]), static_cast<unsigned char>(key[3])};
127 frame.append(reinterpret_cast<const char*>(k), 4);
128 const std::size_t off = frame.size();
129 frame.resize(off + n);
130 mask_into(frame.data() + off, payload.data(), n, k);
131 if (tls::send(s->tls, frame) < 0)
132 throw std::runtime_error("websocket: TLS send failed: " + tls::last_error());
133 return static_cast<long long>(n);
134}

138long long connect(const std::string& host, long long port, const std::string& path,
139 const std::string& server_name) {
140 const long long fd = socket::tcp_connect(host, port);
141 if (fd < 0) throw std::runtime_error("websocket: TCP connect to " + host + " failed");
142 const long long tlss = tls::client_connect(fd, server_name);
143 if (tlss < 0) {
144 socket::close(fd);
145 throw std::runtime_error("websocket: TLS handshake failed: " + tls::last_error());
146 }
147 Session* s = new Session();
148 s->fd = fd;
149 s->tls = tlss;
150
151 const std::string key = base64(os::urandom(16));
152 const std::string req = "GET " + path + " HTTP/1.1\r\n" + "Host: " + server_name + "\r\n" +
153 "Upgrade: websocket\r\n" + "Connection: Upgrade\r\n" +
154 "Sec-WebSocket-Key: " + key + "\r\n" + "Sec-WebSocket-Version: 13\r\n\r\n";
155 if (tls::send(tlss, req) < 0) {
156 const std::string err = tls::last_error();
157 destroy(s);
158 throw std::runtime_error("websocket: upgrade request failed: " + err);
159 }
160
161 // Read the response header block (up to and including the blank line). Any
162 // bytes after it are the start of the WebSocket frame stream and stay in buf.
163 std::size_t hdr_end = std::string::npos;
164 for (;;) {
165 const std::string chunk = tls::recv(tlss, 1 << 12);
166 if (chunk.empty()) {
167 destroy(s);
168 throw std::runtime_error("websocket: connection closed during upgrade");
169 }
170 s->buf.append(chunk);
171 hdr_end = s->buf.find("\r\n\r\n");
172 if (hdr_end != std::string::npos) break;
173 if (s->buf.size() > (1u << 16)) {
174 destroy(s);
175 throw std::runtime_error("websocket: upgrade response too large / not a WebSocket server");
176 }
177 }
178 const std::string status = s->buf.substr(0, s->buf.find("\r\n"));
179 if (status.find(" 101") == std::string::npos) {
180 destroy(s);
181 throw std::runtime_error("websocket: server did not switch protocols: " + status);
182 }
183 s->pos = hdr_end + 4; // frame stream begins after the blank line
184 return as_handle(s);
185}

187long long connect_url(const std::string& url) {
188 const std::string scheme = "wss://";
189 if (url.compare(0, scheme.size(), scheme) != 0)
190 throw std::runtime_error("websocket: only wss:// URLs are supported: " + url);
191 const std::size_t host_start = scheme.size();
192 const std::size_t slash = url.find('/', host_start);
193 const std::string authority =
194 url.substr(host_start, slash == std::string::npos ? std::string::npos : slash - host_start);
195 const std::string path = slash == std::string::npos ? "/" : url.substr(slash);
196 const std::size_t colon = authority.find(':');
197 const std::string host = authority.substr(0, colon);
198 const long long port = colon == std::string::npos
199 ? 443
200 : static_cast<long long>(std::stol(authority.substr(colon + 1)));
201 return connect(host, port, path, host);
202}

208std::string recv(long long session) {
209 Session* s = as_session(session);
210 if (s->closed) return std::string();
211 std::string message; // reassembly buffer for fragmented messages
212 bool fragmenting = false; // mid multi-frame message
213 for (;;) {
214 ensure(s, 2);
215 const unsigned char b0 = static_cast<unsigned char>(s->buf[s->pos]);
216 const unsigned char b1 = static_cast<unsigned char>(s->buf[s->pos + 1]);
217 const bool fin = (b0 & 0x80) != 0;
218 const int opcode = b0 & 0x0F;
219 const bool masked = (b1 & 0x80) != 0; // server frames are unmasked
220 std::uint64_t len = b1 & 0x7F;
221 std::size_t header = 2;
222 if (len == 126) {
223 ensure(s, 4);
224 len = (static_cast<std::uint64_t>(static_cast<unsigned char>(s->buf[s->pos + 2])) << 8) |
225 static_cast<unsigned char>(s->buf[s->pos + 3]);
226 header = 4;
227 } else if (len == 127) {
228 ensure(s, 10);
229 len = 0;
230 for (int i = 0; i < 8; ++i)
231 len = (len << 8) | static_cast<unsigned char>(s->buf[s->pos + 2 + i]);
232 header = 10;
233 }
234 std::size_t mask_off = 0;
235 if (masked) {
236 mask_off = header;
237 header += 4;
238 }
239 ensure(s, header + len);
240 char* payload = s->buf.data() + s->pos + header;
241 if (masked) { // protocol-irregular for a server, but unmask defensively
242 const unsigned char k[4] = {static_cast<unsigned char>(s->buf[s->pos + mask_off]),
243 static_cast<unsigned char>(s->buf[s->pos + mask_off + 1]),
244 static_cast<unsigned char>(s->buf[s->pos + mask_off + 2]),
245 static_cast<unsigned char>(s->buf[s->pos + mask_off + 3])};
246 mask_into(payload, payload, len, k);
247 }
248
249 switch (opcode) {
250 case 0x9: { // ping -> pong with the same payload (control, off hot path)
251 send_frame(s, 0xA, std::string(payload, len));
252 s->pos += header + len;
253 continue;
254 }
255 case 0xA: // pong -> ignore
256 s->pos += header + len;
257 continue;
258 case 0x8: { // close -> echo close, mark done, signal EOF
259 send_frame(s, 0x8, std::string());
260 s->pos += header + len;
261 s->closed = true;
262 return std::string();
263 }
264 case 0x0: // continuation
265 case 0x1: // text
266 case 0x2: // binary
267 default: {
268 // Single-frame message (the common case): copy the payload out of
269 // the buffer, advance, return — no reassembly buffer touched.
270 if (fin && !fragmenting) {
271 std::string out(payload, len);
272 s->pos += header + len;
273 return out;
274 }
275 // Fragmented message: accumulate; the last fragment (fin) returns.
276 message.append(payload, len);
277 s->pos += header + len;
278 if (fin) return message;
279 fragmenting = true;
280 continue;
281 }
282 }
283 }

286long long shutdown(long long session) {
287 // Wake a reader blocked in recv() WITHOUT freeing the session: half-close the
288 // underlying socket so recv returns "". The owner still calls close() after it
289 // has joined the reader. Safe to call from another thread concurrently with the
290 // reader's recv (that is exactly what ::shutdown is for).
291 Session* s = as_session(session);
292 if (s == nullptr || s->tls < 0) return -1;
293 return tls::shutdown(s->tls);
294}
295
296long long close(long long session) {
297 Session* s = as_session(session);
298 if (s == nullptr) return 0;
299 if (!s->closed && s->tls >= 0) {
300 try {
301 send_frame(s, 0x8, std::string());
302 } catch (...) {
303 }
304 }
305 destroy(s);
306 return 0;
307}

309// ---- owning RAII client ----
310// Each method forwards to the handle-based free function above; the guard adds deterministic
311// close() (close frame + TLS + socket teardown) on scope exit, so a `with` block cannot leak.
312
313Client& Client::operator=(Client&& other) noexcept {
314 if (this != &other) {
315 if (session_ != 0) cheatah::websocket::close(session_);
316 session_ = other.session_;
317 other.session_ = 0;
318 }
319 return *this;
320}
321Client::~Client() {
322 if (session_ != 0) cheatah::websocket::close(session_);
323}
