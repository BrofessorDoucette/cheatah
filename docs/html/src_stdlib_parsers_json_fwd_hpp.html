1#pragma once
2
3// cheatah::parsers::json — forward declarations for the whole value model (NOT a second Node;
4// the Node class lives only in node.hpp). These forwards must precede array.hpp / object.hpp so
5// that each token header can stand on its own (and so IntelliSense can compile them in isolation)
6// without pulling in the recursive Node class definition.
7//
8// Node is a forward-declarable class, which is what lets Array AND Object be templated on their
9// storage TYPE with real concepts (ArrayStorage / ObjectStorage), each with an owning backing
10// (std::vector, self-contained) and a viewing backing (std::span, into a Parser's pool).
11
12#include <span>
13#include <type_traits>
14#include <vector>
15
16namespace cheatah::parsers::json {
17
18class Node; // the value type; defined in node.hpp
19
20// ---- arrays ------------------------------------------------------------------
21
22// Backing storage for an Array: owning std::vector<Node> or non-owning std::span<const Node>.
23template <typename S>
24concept ArrayStorage =
25 std::is_same_v<S, std::vector<Node>> || std::is_same_v<S, std::span<const Node>>;
26
27template <ArrayStorage S>
28class Array; // defined in array.hpp
29
30using OwnedArray = Array<std::vector<Node>>; // owns its elements (self-contained)
31using ArrayView = Array<std::span<const Node>>; // views elements in a pool (zero-copy)
32
33// ---- objects -----------------------------------------------------------------
34
35// One object member: a key (a string Node) and its value. A forward-declared STRUCT (not a
36// std::pair alias) so it can be named here while Node is incomplete and defined in node.hpp after
37// Node — std::span<const Member> would otherwise eagerly instantiate pair<Node,Node> (via span's
38// iterator concepts) while Node is still incomplete. (Node itself is forward-declarable, so
39// span<const Node> for ArrayView does not have this problem.)
40struct Member;
41
42// Backing storage for an Object, mirroring ArrayStorage: owning vector or non-owning span.
43template <typename S>
44concept ObjectStorage =
45 std::is_same_v<S, std::vector<Member>> || std::is_same_v<S, std::span<const Member>>;
46
47template <ObjectStorage S>
48class Object; // defined in object.hpp
49
50using OwnedObject = Object<std::vector<Member>>; // owns its members (self-contained)
51using ObjectView = Object<std::span<const Member>>; // views members in a pool (zero-copy)
52
53} // namespace cheatah::parsers::json
