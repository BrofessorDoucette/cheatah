1// Copyright (c) 2026 BigBrain LLC. MIT-licensed (see LICENSE).
2// Original work; see ACKNOWLEDGMENTS.md for the open-source ideas we build upon.
3// System-level tests for cheatah's module-integrity feature, written as worked EXAMPLES
4// of how cheatah keeps a tampered/injected binary from being run. Each test drives the
5// REAL toolchain end to end:
6//
7// purrc --keygen <prefix> generate an Ed25519 signing keypair
8// purrc app.purr -o app.so --sign build a module and sign it (writes app.so.sig +
9// app.so.sha512)
10// cheatah [--verify] app.so the runtime verifies BEFORE it loads/runs
11//
12// The story they tell: a genuine signed module runs; flip a single byte (an injection
13// into the binary) and the runtime REFUSES to load it; demand a signature and an
14// unsigned or untrusted-signer module is refused. This is the same Ed25519 + SHA-512
15// exercised by ed25519_sys_test.cpp / hashlib_sys_test.cpp, now guarding the loader.
16//
17// Suite name ModuleIntegrity (not *CompileRun*) so the QA gate always runs it.
18#include "e2e_harness.hpp"

27// Run a command, returning {exit_code, combined_stdout/stderr}.
28struct Proc { int code; std::string out; };
29Proc run_cmd(const std::string& cmd) {
30 int code = -1;
31 const std::string out = e2e::run_capture(cmd + " 2>&1", code);
32 return {code, out};
33}

42// Flip one byte in the middle of a file (simulate an injection into the binary).
43void flip_a_byte(const std::string& path) {
44 std::fstream f(path, std::ios::in | std::ios::out | std::ios::binary);
45 f.seekg(0, std::ios::end);
46 const std::streamoff size = f.tellg();
47 const std::streamoff at = size / 2;
48 f.seekg(at);
49 char c = 0;
50 f.read(&c, 1);
51 c = static_cast<char>(c ^ 0xFF);
52 f.seekp(at);
53 f.write(&c, 1);
54}

59// Build a module at <name>.so from @p src; returns its path. Optionally sign/checksum it
60// via @p extra (e.g. "--sign key.key", "--checksum").
61std::string build_module_src(const std::string& name, const std::string& src,
62 const std::string& extra = "") {
63 const std::string purr = kTmp + "/" + name + ".purr";
64 const std::string so = kTmp + "/" + name + ".so";
65 write_purr(purr, src);
66 const Proc r = run_cmd(std::string(PURRC_PATH) + " " + q(purr) + " -o " + q(so) + " " + extra);
67 EXPECT_EQ(r.code, 0) << name << ": purrc failed:\n" << r.out;
68 return so;
69}

87// Generate a keypair at <prefix>.{key,pub}; returns the public-key hex.
88std::string keygen(const std::string& prefix) {
89 EXPECT_EQ(run_cmd(std::string(PURRC_PATH) + " --keygen \"" + prefix + "\"").code, 0);
90 std::string pub = read_text(prefix + ".pub");
91 while (!pub.empty() && (pub.back() == '\n' || pub.back() == ' ')) pub.pop_back();
92 return pub;

299// ============================================================================
300// C-runtime validation (suite RuntimeCheck): the <module>.rt build-runtime manifest
301// is checked against the HOST before loading, and is signed by a key SEPARATE from the
302// code-signing key. These run the real toolchain end to end.
303// ============================================================================
304namespace {
305// Replace the "<key> ..." line of a module's .rt manifest with "<key> <val>".
306void rewrite_rt_field(const std::string& so, const std::string& key, const std::string& val) {
307 std::istringstream in(read_text(so + ".rt"));
308 std::ostringstream out;
309 std::string line;
310 while (std::getline(in, line)) {
311 if (line.rfind(key + " ", 0) == 0) out << key << " " << val << "\n";
312 else out << line << "\n";
313 }
314 write_text(so + ".rt", out.str());
315}
316} // namespace

439 const std::string so = build_module("rt_shutdown_app", "--runtime");
440 rewrite_rt_field(so, "libc", "99.9"); // the C runtime fingerprint no longer matches reality
441
442 const Proc r = run_cmd(CH(q(so)));
443 EXPECT_NE(r.code, 0) << r.out;
444 EXPECT_EQ(r.out.find("integrity ok"), std::string::npos) << "ran on an incompatible runtime:\n" << r.out;
445 EXPECT_NE(r.out.find("needs glibc >= 99.9"), std::string::npos) << r.out;
446}

448// And when the C-runtime manifest is SIGNED (separate key) and then tampered, the signature
449// catches it and the runtime shuts down before running — even when the tampered values are
450// otherwise ABI-compatible, so only the signature can detect the edit.
451TEST(RuntimeCheck, SignedCRuntimeTamperShutsDownBeforeRunning) {
452 const std::string codek = kTmp + "/rtsd_code", rtk = kTmp + "/rtsd_rt";
453 keygen(codek);
454 keygen(rtk);
455 const std::string so = build_module("rt_signed_shutdown",
456 "--sign " + q(codek + ".key") + " --sign-runtime " + q(rtk + ".key"));
457 rewrite_rt_field(so, "libcxx", "libstdc++-1"); // ABI-compatible value, but not what was signed
458
459 const Proc r = run_cmd("CHEATAH_VERIFY=strict CHEATAH_TRUST=" + q(codek + ".pub") +
460 " CHEATAH_RT_TRUST=" + q(rtk + ".pub") + " " + CH(q(so)));
461 EXPECT_NE(r.code, 0) << r.out;
462 EXPECT_EQ(r.out.find("integrity ok"), std::string::npos) << r.out;
463 EXPECT_NE(r.out.find("runtime manifest signature does not verify"), std::string::npos) << r.out;
464}

466// Strict mode with a runtime trust configured REQUIRES the manifest: a module without a
467// .rt is refused (an attacker can't strip the runtime check).
468TEST(RuntimeCheck, StrictRequiresRuntimeManifest) {
