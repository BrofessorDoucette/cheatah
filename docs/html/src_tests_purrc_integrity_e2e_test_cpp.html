1// System-level tests for cheatah's module-integrity feature, written as worked EXAMPLES
2// of how cheatah keeps a tampered/injected binary from being run. Each test drives the
3// REAL toolchain end to end:
4//
5// purrc --keygen <prefix> generate an Ed25519 signing keypair
6// purrc app.purr -o app.so --sign build a module and sign it (writes app.so.sig +
7// app.so.sha512)
8// cheatah [--verify] app.so the runtime verifies BEFORE it loads/runs
9//
10// The story they tell: a genuine signed module runs; flip a single byte (an injection
11// into the binary) and the runtime REFUSES to load it; demand a signature and an
12// unsigned or untrusted-signer module is refused. This is the same Ed25519 + SHA-512
13// exercised by ed25519_sys_test.cpp / hashlib_sys_test.cpp, now guarding the loader.
14//
15// Suite name ModuleIntegrity (not *CompileRun*) so the QA gate always runs it.
16#include "e2e_harness.hpp"
17

19#include <string>

25// Run a command, returning {exit_code, combined_stdout/stderr}.
26struct Proc { int code; std::string out; };
27Proc run_cmd(const std::string& cmd) {
28 int code = -1;
29 const std::string out = e2e::run_capture(cmd + " 2>&1", code);
30 return {code, out};
31}

40// Flip one byte in the middle of a file (simulate an injection into the binary).
41void flip_a_byte(const std::string& path) {
42 std::fstream f(path, std::ios::in | std::ios::out | std::ios::binary);
43 f.seekg(0, std::ios::end);
44 const std::streamoff size = f.tellg();
45 const std::streamoff at = size / 2;
46 f.seekg(at);
47 char c = 0;
48 f.read(&c, 1);
49 c = static_cast<char>(c ^ 0xFF);
50 f.seekp(at);
51 f.write(&c, 1);
52}

57// Build a module at <name>.so from @p src; returns its path. Optionally sign/checksum it
58// via @p extra (e.g. "--sign key.key", "--checksum").
59std::string build_module_src(const std::string& name, const std::string& src,
60 const std::string& extra = "") {
61 const std::string purr = kTmp + "/" + name + ".purr";
62 const std::string so = kTmp + "/" + name + ".so";
63 write_purr(purr, src);
64 const Proc r = run_cmd(std::string(PURRC_PATH) + " " + q(purr) + " -o " + q(so) + " " + extra);
65 EXPECT_EQ(r.code, 0) << name << ": purrc failed:\n" << r.out;
66 return so;
67}

85// Generate a keypair at <prefix>.{key,pub}; returns the public-key hex.
86std::string keygen(const std::string& prefix) {
87 EXPECT_EQ(run_cmd(std::string(PURRC_PATH) + " --keygen \"" + prefix + "\"").code, 0);
88 std::string pub = read_text(prefix + ".pub");

297// ============================================================================
298// C-runtime validation (suite RuntimeCheck): the <module>.rt build-runtime manifest
299// is checked against the HOST before loading, and is signed by a key SEPARATE from the
300// code-signing key. These run the real toolchain end to end.
301// ============================================================================
302namespace {
303// Replace the "<key> ..." line of a module's .rt manifest with "<key> <val>".
304void rewrite_rt_field(const std::string& so, const std::string& key, const std::string& val) {
305 std::istringstream in(read_text(so + ".rt"));
306 std::ostringstream out;
307 std::string line;
308 while (std::getline(in, line)) {
309 if (line.rfind(key + " ", 0) == 0) out << key << " " << val << "\n";
310 else out << line << "\n";
311 }
312 write_text(so + ".rt", out.str());
313}
314} // namespace

437TEST(RuntimeCheck, SignedCRuntimeTamperShutsDownBeforeRunning) {
438 const std::string codek = kTmp + "/rtsd_code", rtk = kTmp + "/rtsd_rt";

441 const std::string so = build_module("rt_signed_shutdown",

443 rewrite_rt_field(so, "libcxx", "libstdc++-1"); // ABI-compatible value, but not what was signed

445 const Proc r = run_cmd("CHEATAH_VERIFY=strict CHEATAH_TRUST=" + q(codek + ".pub") +
