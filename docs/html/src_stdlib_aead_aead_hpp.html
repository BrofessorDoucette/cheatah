5/**
6 * @file aead.hpp
7 * @brief cheatah `aead` — ChaCha20-Poly1305 (RFC 8439) and AES-GCM authenticated
8 * encryption, from scratch. `import aead` to use it. The record ciphers of the
9 * from-scratch cheatah TLS 1.3 client (with `x25519` and hashlib's HKDF). No OpenSSL.
10 *
11 * Keys are hex (64-char / 32 bytes for ChaCha20-Poly1305 and AES-256-GCM; 32-char /
12 * 16 bytes for AES-128-GCM); nonces are 24-char hex (12 bytes). Messages and the
13 * additional authenticated data are RAW BYTE strings (binary-safe, like socket::recv).
14 * The 16-byte tag is APPENDED to the ciphertext; decryption verifies it in
15 * constant time and returns "" on ANY mismatch — tampered data never comes back.
16 */
17#include <cstddef>
18#include <string>
19#include <string_view>
20
21namespace cheatah::aead {
22
23/**
24 * Encrypt + authenticate: ChaCha20-Poly1305(key, nonce, aad, plaintext).
25 *
26 * @param key_hex the 64-char hex key (32 bytes).
27 * @param nonce_hex the 24-char hex nonce (12 bytes); MUST be unique per key.
28 * @param aad additional authenticated data — authenticated but not encrypted ("" for none).
29 * @param plaintext the raw-byte message to encrypt (binary-safe).
30 * @return ciphertext with the 16-byte tag appended, or "" on malformed key/nonce hex or a
31 * message over the 64 GiB single-message cap.
32 * @complexity O(|plaintext| + |aad|).
33 * @alloc the returned string plus a temporary Poly1305 MAC-input buffer
34 * (|aad| + |plaintext| + padding).
35 * @warning The nonce MUST never repeat under the same key: nonce reuse leaks the XOR of the
36 * plaintexts and lets an attacker forge tags.
37 * @test CheatahAead.Rfc8439Encrypt
38 * @systest TlsSys.HandshakeAgainstOpenssl
39 */
40std::string chacha20poly1305_encrypt(std::string_view key_hex, std::string_view nonce_hex,
41 std::string_view aad, std::string_view plaintext);
42
43/**
44 * ChaCha20-Poly1305 encryption into a CALLER-PROVIDED buffer — the allocation-free form.
45 *
46 * The string-returning form above allocates its result and a temporary MAC-input buffer, which
47 * makes it unusable where allocation is forbidden or merely unwelcome: signal handlers, embedded
48 * targets, hot loops that own their memory, and any caller that already has the bytes in place.
49 * This form allocates NOTHING — the tag is computed by streaming the AEAD's segments through
50 * Poly1305 rather than concatenating them — and is therefore safe to call from such contexts.
51 * Byte-for-byte identical output to the string form (asserted by an equivalence test).
52 *
53 * @param key the 32-byte key.
54 * @param nonce the 12-byte nonce; MUST be unique per key.
55 * @param aad additional authenticated data (may be null when @p aad_len is 0).
56 * @param aad_len length of @p aad in bytes.
57 * @param plaintext the message to encrypt (may be null when @p plaintext_len is 0).
58 * @param plaintext_len length of @p plaintext in bytes.
59 * @param out receives `plaintext_len + 16` bytes: the ciphertext followed by the tag. May alias
60 * @p plaintext to encrypt in place.
61 * @return false on a null buffer with a nonzero length, or a message over the 64 GiB
62 * single-message cap; true otherwise.
63 * @complexity O(|plaintext| + |aad|).
64 * @alloc none.
65 * @thread any thread; no shared state. Async-signal-safe (no allocation, no locks, no errno use).
66 * @warning The nonce MUST never repeat under the same key: nonce reuse leaks the XOR of the
67 * plaintexts and lets an attacker forge tags.
68 * @test CheatahAead.IntoFormsMatchStringForms
69 */
70bool chacha20poly1305_encrypt_into(const unsigned char key[32], const unsigned char nonce[12],
71 const unsigned char* aad, std::size_t aad_len,
72 const unsigned char* plaintext, std::size_t plaintext_len,
73 unsigned char* out);
74
75/**
76 * Verify + decrypt into a CALLER-PROVIDED buffer — the allocation-free inverse of
77 * @ref chacha20poly1305_encrypt_into. The tag is verified in constant time before any plaintext is
78 * written back to the caller.
79 *
80 * @param key the 32-byte key.
81 * @param nonce the 12-byte nonce used to encrypt.
82 * @param aad the same additional authenticated data (may be null when @p aad_len is 0).
83 * @param aad_len length of @p aad in bytes.
84 * @param ciphertext the ciphertext WITH its trailing 16-byte tag.
85 * @param ciphertext_len total length including the tag; must be >= 16.
86 * @param out receives `ciphertext_len - 16` plaintext bytes. May alias @p ciphertext. May be null
87 * ONLY when the message is tag-only (`ciphertext_len == 16`), i.e. there is no plaintext to
88 * write — so authenticating an empty message needs no buffer.
89 * @return false when the tag does not verify (nothing is written), or on a malformed argument;
90 * true on success.
91 * @complexity O(|ciphertext| + |aad|).
92 * @alloc none.
93 * @thread any thread; no shared state. Async-signal-safe (no allocation, no locks, no errno use).
94 * @test CheatahAead.IntoFormsMatchStringForms
95 */
96bool chacha20poly1305_decrypt_into(const unsigned char key[32], const unsigned char nonce[12],
97 const unsigned char* aad, std::size_t aad_len,
98 const unsigned char* ciphertext, std::size_t ciphertext_len,
99 unsigned char* out);
100
101/**
102 * Verify + decrypt the inverse of chacha20poly1305_encrypt.
103 *
104 * @param key_hex the 64-char hex key (32 bytes) used to encrypt.
105 * @param nonce_hex the 24-char hex nonce (12 bytes) used to encrypt.
106 * @param aad the same additional authenticated data supplied at encryption ("" for none).
107 * @param ciphertext the ciphertext with its 16-byte tag appended.
108 * @return the plaintext, or "" when the tag does not verify (tampering / wrong key or
109 * nonce / malformed input) — the tag check is constant-time.
110 * @complexity O(|ciphertext| + |aad|).
111 * @alloc the returned string plus a temporary Poly1305 MAC-input buffer
112 * (|aad| + |ciphertext| + padding).
113 * @test CheatahAead.Rfc8439Decrypt
114 * @systest TlsSys.HandshakeAgainstOpenssl
115 */
116std::string chacha20poly1305_decrypt(std::string_view key_hex, std::string_view nonce_hex,
117 std::string_view aad, std::string_view ciphertext);

119/**
120 * Encrypt + authenticate: AES-128-GCM(key, nonce, aad, plaintext) — the other TLS 1.3 record cipher
121 * (TLS_AES_128_GCM_SHA256). The nonce is used as the GCM IV (J0 = nonce || 0x00000001); the 16-byte
122 * GCM tag is appended.
123 *
124 * @param key_hex the 32-char hex key (16 bytes, AES-128).
125 * @param nonce_hex the 24-char hex nonce (12 bytes), used as the GCM IV; MUST be unique per key.
126 * @param aad additional authenticated data — authenticated but not encrypted ("" for none).
127 * @param plaintext the raw-byte message to encrypt (binary-safe).
128 * @return ciphertext with the 16-byte tag appended, or "" on malformed key/nonce hex or a
129 * message over the 64 GiB single-message cap.
130 * @complexity O(|plaintext| + |aad|).
131 * @alloc the returned string.
132 * @warning The nonce MUST never repeat under the same key: GCM nonce reuse leaks the XOR of
133 * the plaintexts AND the GHASH authentication key (forgeries follow).
134 * @test CheatahAead.AesGcmNistCase4
135 * @systest TlsSys.HandshakeAes128Gcm
136 */
137std::string aes128gcm_encrypt(std::string_view key_hex, std::string_view nonce_hex,
138 std::string_view aad, std::string_view plaintext);
139
140/**
141 * Verify + decrypt the inverse of aes128gcm_encrypt.
142 *
143 * @param key_hex the 32-char hex key (16 bytes, AES-128) used to encrypt.
144 * @param nonce_hex the 24-char hex nonce (12 bytes) used to encrypt.
145 * @param aad the same additional authenticated data supplied at encryption ("" for none).
146 * @param ciphertext the ciphertext with its 16-byte GCM tag appended.
147 * @return the plaintext, or "" when the tag does not verify (the check is constant-time) or the input
148 * is malformed.
149 * @complexity O(|ciphertext| + |aad|).
150 * @alloc the returned string (the hardware path allocates the candidate plaintext even when
151 * the tag check fails and "" is returned).
152 * @test CheatahAead.AesGcmRejectsTamperAndMalformed
153 */
154std::string aes128gcm_decrypt(std::string_view key_hex, std::string_view nonce_hex,
155 std::string_view aad, std::string_view ciphertext);

157/**
158 * Encrypt + authenticate: AES-256-GCM(key, nonce, aad, plaintext) — the record cipher of TLS 1.3's
159 * TLS_AES_256_GCM_SHA384 suite. The nonce is the GCM IV (J0 = nonce || 0x00000001); the 16-byte tag is
160 * appended. Same runtime dispatch as AES-128-GCM: the hardware path (x86 AES-NI/PCLMULQDQ or ARMv8
161 * AES/PMULL) when present and self-tested, otherwise the portable scalar reference.
162 *
163 * @param key_hex the 64-char hex key (32 bytes, AES-256).
164 * @param nonce_hex the 24-char hex nonce (12 bytes), used as the GCM IV; MUST be unique per key.
165 * @param aad additional authenticated data — authenticated but not encrypted ("" for none).
166 * @param plaintext the raw-byte message to encrypt (binary-safe).
167 * @return ciphertext with the 16-byte tag appended, or "" on malformed key/nonce hex or a
168 * message over the 64 GiB single-message cap.
169 * @complexity O(|plaintext| + |aad|).
170 * @alloc the returned string.
171 * @warning The nonce MUST never repeat under the same key: GCM nonce reuse leaks the XOR of
172 * the plaintexts AND the GHASH authentication key (forgeries follow).
173 * @test CheatahAead.Aes256GcmNistKat
174 * @systest TlsSys.HandshakeAes256GcmSha384
175 */
176std::string aes256gcm_encrypt(std::string_view key_hex, std::string_view nonce_hex,
177 std::string_view aad, std::string_view plaintext);

184 * @param aad the same additional authenticated data supplied at encryption ("" for none).

195/// @cond INTERNAL — a test hook (pins the scalar reference path for cross-checking), not user API
196/**
197 * Force the portable (non-hardware) AES-GCM path on/off (both AES-128 and AES-256). AES-GCM
198 * normally uses the CPU's crypto instructions when present (x86 AES-NI/PCLMULQDQ or ARMv8
199 * AES/PMULL); this hook pins the scalar reference implementation instead — for deterministic
200 * cross-checking that both paths agree, and so the reference stays exercised on hardware that
201 * would otherwise always take the fast path.
202 * @param on true to pin the portable scalar path; false to allow the hardware path again when present.
203 * @complexity O(1). @alloc none.
204 * @concurrency Not thread-safe: writes a plain (non-atomic) global read by every AES-GCM
205 * call — do not flip it while another thread encrypts or decrypts.
206 * @test CheatahAead.AesGcmPortableMatchesHardware
207 */
208void set_force_portable_crypto(bool on);
209/// @endcond
210
211/**
212 * Whether AES-GCM (AES-128 and AES-256) is currently using the CPU's crypto instructions: the
213 * x86 AES-NI + PCLMULQDQ path, or the ARMv8 AES + PMULL path on AArch64 (e.g. Apple Silicon).
214 * True on a capable CPU unless set_force_portable_crypto(true) is in effect; false where neither
215 * ISA is present (the portable scalar reference then runs). Lets a platform test report and assert
216 * which implementation a given machine actually exercised.
217 * @return true iff a hardware AES-GCM path (x86 AES-NI/PCLMULQDQ or ARMv8 AES/PMULL) is active.
218 * @complexity O(1). @alloc none after the first call (the first call may run the one-time
219 * hardware known-answer self-test, which allocates short scratch strings).
220 * @test CryptoPlatform.Report
221 */
222[[nodiscard]] bool crypto_hardware_active();
