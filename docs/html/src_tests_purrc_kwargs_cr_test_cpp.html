tests/purrc/kwargs_cr_test.cpp

1// Compile-run system tests for DEFAULT and KEYWORD arguments:
2// fn f(a, b = expr) { ... } — trailing defaults (lower to forwarding overloads, since a
3// C++ default on an `auto` parameter cannot drive deduction);
4// f(x, b = v) / f(b = v, a = u) — keyword arguments, reordered at the call site against the
5// known signature, missing slots filled from the defaults.
6// Each test compiles a .purr with purrc, runs it, and asserts exact stdout; the negative tests
7// assert that purrc REJECTS the program (bad kwarg name, non-trailing default, dup parameter).
8#include "e2e_harness.hpp"
9
10TEST(KwargsCompileRun, TrailingDefaults) {
11 e2e::expect_e2e("kwargs_defaults", R"PURR(import io
12fn greet(name, greeting = "hello", punct = "!") {
13 io.print(greeting + ", " + name + punct)
14}
15greet("world")
16greet("cheatah", "purr")
17)PURR", "hello, world!\npurr, cheatah!\n");
18}
19
20TEST(KwargsCompileRun, KeywordReorder) {
21 e2e::expect_e2e("kwargs_reorder", R"PURR(import io
22fn greet(name, greeting = "hello", punct = "!") {
23 io.print(greeting + ", " + name + punct)
24}
25greet("kwargs", punct = "?")
26greet(punct = ".", name = "reordered")
27greet("mixed", greeting = "hi", punct = "~")
28)PURR", "hello, kwargs?\nhello, reordered.\nhi, mixed~\n");
29}
30
31TEST(KwargsCompileRun, DefaultsInExpressions) {
32 e2e::expect_e2e("kwargs_exprs", R"PURR(import io
33fn area(w, h = 2) { return w * h }
34io.print(area(3))
35io.print(area(3, h = 5))
36io.print(area(h = 4, w = 2))
37)PURR", "6\n15\n8\n");
38}
39
40// purrc must REJECT: a keyword that names no parameter.
41TEST(KwargsCompileRun, RejectsUnknownKeyword) {
42 e2e::expect_compile_fail("kwargs_unknown", R"PURR(import io
43fn f(a) { io.print(a) }
44f(b = 1)
45)PURR");
46}
47
48// purrc must REJECT: a non-defaulted parameter after a defaulted one.
49TEST(KwargsCompileRun, RejectsNonTrailingDefault) {
50 e2e::expect_compile_fail("kwargs_nontrailing", R"PURR(import io
51fn f(a = 1, b) { io.print(a + b) }
52f(1, 2)
53)PURR");
54}
55
56// purrc must REJECT: the same parameter given positionally AND by keyword.
57TEST(KwargsCompileRun, RejectsDuplicateParameter) {
58 e2e::expect_compile_fail("kwargs_dup", R"PURR(import io
59fn f(a, b = 2) { io.print(a + b) }
60f(1, a = 3)
61)PURR");
62}

64// ---- `%` (Python floor-mod) and membership `in` (added alongside the requests port) ----
65
66TEST(LangFeatures, Modulo) {
67 e2e::expect_e2e("lang_modulo", R"PURR(import io
68io.print(7 % 3)
69io.print(-7 % 3)
70io.print(7.5 % 2)
71io.print(255 % 16)
72)PURR", "1\n2\n1.5\n15\n");
73}
74
75TEST(LangFeatures, InOperator) {
76 e2e::expect_e2e("lang_in", R"PURR(import io
77let d: dict<str, int> = {}
78d["alpha"] = 1
79io.print("alpha" in d)
80io.print("beta" in d)
81let xs = [10, 20, 30]
82io.print(20 in xs)
83io.print(99 in xs)
84io.print("ell" in "hello")
85)PURR", "True\nFalse\nTrue\nFalse\nTrue\n");
86}
