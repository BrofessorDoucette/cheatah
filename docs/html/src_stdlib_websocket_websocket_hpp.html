3/**
4 * @file websocket.hpp
5 * @brief cheatah `websocket` — a from-scratch, low-latency WebSocket CLIENT
6 * (RFC 6455) over the cheatah `tls` 1.3 client and `socket`. `import
7 * websocket` to use it. No external libraries.
8 *
9 * Built for SPEED. The receive path is the hot path and is allocation-quiet:
10 * - one read buffer per session, REUSED across every frame (no per-frame heap);
11 * - server-to-client frames are unmasked by the protocol (RFC 6455 §5.1), so
12 * recv does ZERO unmasking work — it slices the payload straight out of the
13 * buffer;
14 * - frame headers are parsed in place (no header object is materialized);
15 * - the TLS layer is drained in large chunks, so many frames are decoded per
16 * underlying read/decrypt.
17 * The send path masks (clients MUST, §5.3) with a 64-bit-word XOR (8 bytes per
18 * step), but sends are rare (subscribe/control) so they are off the hot path.
19 *
20 * The cheatah-facing API is the owning `websocket::Client` guard, created by
21 * `websocket.open(...)` / `websocket.open_url("wss://...")`: it sends a close frame and tears
22 * down the TLS session + socket automatically when it goes out of scope, so a cheatah program
23 * cannot leak the heap Session. wss:// only (WebSocket over TLS) — the transport is always
24 * encrypted, like the rest of cheatah's net stack. The flat handle-based calls (an integer
25 * session id) live in websocket_lowlevel.hpp (C++ only).
26 *
27 * Threading: a session is single-owner; do not call recv and send for the same
28 * session from two threads at once. Separate sessions are independent.
29 */
30

35// The low-level, handle-based API (connect / connect_url / send_text / recv / close / shutdown,
36// keyed by an integer session id backed by a heap Session) is C++-only and lives in
37// websocket_lowlevel.hpp. It is intentionally NOT part of this cheatah-facing header: a cheatah
38// program cannot reach it, so it cannot leak the heap Session — it uses the owning
39// `websocket::Client` guard + `websocket.open()`/`open_url()` below, which free the Session
40// automatically at scope exit. `websocket::Client` is implemented on top of that low-level API.
41
42// ---- owning RAII client (the `with`-friendly, leak-proof API) ----
43
44/**
45 * @brief An owning WebSocket client — closes the connection (frame + TLS + socket) on destruction.
46 *
47 * The RAII counterpart to the handle-based calls above. A `Client` owns one session; when it is
48 * destroyed (scope exit out of a `with` body, including via exception) or explicitly close()d, the
49 * close frame is sent and the TLS session and TCP socket are torn down, so
50 * `with websocket.open_url(url) as ws { … }` cannot leak the session, its `tls` session, or its
51 * fd. Move-only: the copy operations are deleted and a moved-from `Client` is left closed.
52 */
53class Client {
54public:
55 /**
56 * Construct a closed client (owns nothing).
57 * @complexity O(1).
58 * @alloc none.
59 * @test CheatahWebSocket.ClientDefaultIsClosed
60 */
61 Client() = default;
62 /**
63 * Adopt an existing session handle (e.g. from connect()); the `Client` now owns it.
64 * @param session a session handle to take ownership of (0 for a closed client).
65 * @complexity O(1).
66 * @alloc none.
67 * @test CheatahWebSocket.ClientMoveTransfersOwnership
68 */
69 explicit Client(long long session) : session_(session) {}
70 Client(const Client&) = delete;
71 Client& operator=(const Client&) = delete;
72 /**
73 * Move-construct, taking over @p other's session (the moved-from `Client` becomes closed).
74 * @param other the client to move from.
75 * @complexity O(1).
76 * @alloc none.
77 * @test CheatahWebSocket.ClientMoveTransfersOwnership
78 */
79 Client(Client&& other) noexcept : session_(other.session_) { other.session_ = 0; }
80 /**
81 * Move-assign, closing this session first, then taking over @p other's (which becomes closed).
82 * @param other the client to move from.
83 * @return reference to this client.
84 * @complexity O(1).
85 * @alloc none.
86 * @test CheatahWebSocket.ClientMoveTransfersOwnership
87 */
88 Client& operator=(Client&& other) noexcept;
89 /**
90 * Close the connection if still open (close frame + TLS + socket teardown).
91 * @complexity one TLS write + teardown.
92 * @alloc a small close frame (when still open).
93 * @test CheatahWebSocket.ClientDefaultIsClosed
94 */
95 ~Client();
96
97 /**
98 * Is a connection open?
99 * @return true iff this owns an open session.
100 * @complexity O(1).
101 * @alloc none.
102 * @test CheatahWebSocket.ClientDefaultIsClosed
103 */
104 bool is_open() const { return session_ != 0; }
105 /**
106 * The raw session handle (for the low-level calls).
107 * @return the owned handle, or 0 when closed.
108 * @complexity O(1).
109 * @alloc none.
110 * @test CheatahWebSocket.ClientMoveTransfersOwnership
111 */
112 long long id() const { return session_; }
113 /**
114 * Send one application TEXT message as a single masked frame (see the free send_text()).
115 * @param message the UTF-8 payload.
116 * @return the number of payload bytes sent.
117 * @throws std::runtime_error on a transport error.
118 * @complexity O(message length).
119 * @alloc one frame buffer sized to the message.
120 * @systest WebSocketSys.EchoRoundTrip
121 */
122 long long send_text(const std::string& message);
123 /**
124 * Receive the next application message (see the free recv()).
125 * @return the message payload; "" once the peer has closed.
126 * @throws std::runtime_error on a transport/protocol error.
127 * @complexity O(message length).
128 * @alloc the returned payload.
129 * @systest WebSocketSys.EchoRoundTrip
130 */
131 std::string recv();
132 /**
133 * Wake a reader blocked in recv() WITHOUT freeing the session (see the free shutdown()).
134 * @return 0 on success, -1 on error.
135 * @complexity O(1) + one syscall.
136 * @alloc none.
137 * @systest WebSocketSys.EchoRoundTrip
138 */
139 long long shutdown();
140 /**
141 * Close the connection now (idempotent — the destructor will not close it again).
142 * @return 0 on success, -1 if already closed.
143 * @complexity one TLS write + teardown.
144 * @alloc a small close frame (when still open).
145 * @test CheatahWebSocket.ClientDefaultIsClosed
146 */
147 long long close();
148
149private:
150 long long session_ = 0;
151};

153/**
154 * Open a secure WebSocket connection and return it as an owning Client (the RAII,
155 * `with`-friendly form of connect()).
156 * @param host the server host, e.g. "echo.websocket.org".
157 * @param port the TLS port, normally 443.
158 * @param path the request path, e.g. "/".
159 * @param server_name the TLS SNI / Host (usually == @p host).
160 * @return an owning Client.
161 * @throws std::runtime_error on connect/TLS/upgrade failure.
162 * @complexity one TCP + one TLS handshake + one HTTP round trip.
163 * @alloc the session.
164 * @systest WebSocketSys.EchoRoundTrip
165 */
166Client open(const std::string& host, long long port, const std::string& path,
167 const std::string& server_name);
