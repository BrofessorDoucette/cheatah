1// Copyright (c) 2026 BigBrain LLC. MIT-licensed (see LICENSE).
2// Original work; see ACKNOWLEDGMENTS.md for the open-source ideas we build upon.
3// System-level tests for the generated-code namespace aliasing. The whole program is
4// emitted inside a `namespace cheatah_program { … }`, where each stdlib module gets its
5// OWN short alias (`namespace io = ::cheatah::io;`, `namespace random = …`, …) so the
6// body reads `io::print` / `random::randint` instead of `cheatah::io::print`. The
7// wrapper namespace is what makes this SAFE: an alias like `random`/`time`/`socket`
8// cannot collide with the global C function of the same name.
9//
10// These run the .purr through the REAL purrc -> cheatah pipeline and assert on BOTH:
11// 1. the generated C++ (the feature itself — that each module is aliased distinctly), and
12// 2. the program's runtime output (that the aliased code still compiles and behaves).
13//
14// Named NamespaceAliasing (not *CompileRun*) so the QA gate always runs them.
15#include "e2e_harness.hpp"
16
17static bool has(const std::string& hay, const std::string& needle) {
18 return hay.find(needle) != std::string::npos;
19}
20
21// No collision: every imported module — plus the always-available builtins — gets its
22// own distinct alias, and the body uses the short form. The program still runs.
23TEST(NamespaceAliasing, EachModuleGetsItsOwnAlias) {
24 const std::string gen = e2e::expect_e2e_source("ns_distinct", R"PURR(import io
25import math
26io.print(math.sqrt(16.0), math.floor(2.7), len("hi"))
27)PURR", "4 2 2\n");
28 EXPECT_TRUE(has(gen, "namespace cheatah_program {"));
29 // Distinct alias per module — never `cheatah::io::` / `cheatah::math::` in the body.
30 EXPECT_TRUE(has(gen, "namespace io = ::cheatah::io;"));
31 EXPECT_TRUE(has(gen, "namespace math = ::cheatah::math;"));
32 EXPECT_TRUE(has(gen, "namespace builtins = ::cheatah::builtins;"));
33 EXPECT_TRUE(has(gen, "io::print("));
34 EXPECT_TRUE(has(gen, "math::sqrt("));
35 EXPECT_TRUE(has(gen, "builtins::len("));
36 EXPECT_FALSE(has(gen, "cheatah::io::print"));
37 EXPECT_FALSE(has(gen, "cheatah::math::sqrt"));
38}

40// Modules whose names match a global C library function (random -> ::random,
41// time -> ::time) are STILL aliased — the wrapper namespace prevents the clash. If the
42// alias leaked to global scope this would not compile ("redefinition of 'random'").
43TEST(NamespaceAliasing, LibcNamedModulesAreAliasedSafely) {
44 const std::string gen = e2e::expect_e2e_source("ns_libc", R"PURR(import io
45import random
46random.seed(123)
47io.print(random.randint(1, 1))
48)PURR", "1\n");
49 EXPECT_TRUE(has(gen, "namespace random = ::cheatah::random;"));
50 EXPECT_TRUE(has(gen, "random::seed("));
51 EXPECT_TRUE(has(gen, "random::randint("));
52 EXPECT_FALSE(has(gen, "cheatah::random::randint"));
53}

55// Collision with a PROGRAM identifier: a `struct os` shares the imported `os` module's
56// name, so os must NOT be aliased (it would clash with the struct inside the namespace).
57// It stays explicit (`::cheatah::os::…`) while the non-colliding io is still aliased —
58// and the program compiles and runs.
59TEST(NamespaceAliasing, ProgramNameCollisionStaysExplicit) {
60 const std::string gen = e2e::expect_e2e_source("ns_conflict", R"PURR(import io
61import os
62struct os { n: int }
63let p = os(7)
64io.print(os.path.join("a", "b"), p.n)
65)PURR", "a/b 7\n");
66 EXPECT_TRUE(has(gen, "namespace io = ::cheatah::io;")); // safe -> aliased
67 EXPECT_FALSE(has(gen, "namespace os = ::cheatah::os;")); // collides -> not aliased
68 EXPECT_TRUE(has(gen, "::cheatah::os::path::join")); // stays explicit
69}

71// A program function named like the generated entry point (`fn run`, `fn purr_main`)
72// must NOT collide with the internal trampoline function — the entry name escalates
73// until it's free, so the program compiles and runs.
74TEST(NamespaceAliasing, EntryFunctionNameNeverCollides) {
75 const std::string gen = e2e::expect_e2e_source("ns_entry", R"PURR(import io
76fn run() {
77 io.print("ran")
78}
79fn purr_main() {
80 run()
81}
82purr_main()
83)PURR", "ran\n");
84 // The exported C entry calls the escalated trampoline, NOT the user's run()/purr_main().
85 EXPECT_TRUE(has(gen, "PURR_EXPORT void purr_main() { cheatah_program::purr_main_(); }"));
86 EXPECT_FALSE(has(gen, "cheatah_program::run()"));
87}
88
89// A function parameter named like an imported module shadows it: the bare `math` is the
90// parameter (not the module namespace), aliasing is skipped, and the program still runs.
91TEST(NamespaceAliasing, ParameterNameShadowsModule) {
92 const std::string gen = e2e::expect_e2e_source("ns_param", R"PURR(import io
93import math
94fn bump(math) {
95 return math + 1
96}
97io.print(bump(41))
98)PURR", "42\n");
99 EXPECT_FALSE(has(gen, "namespace math = ::cheatah::math;")); // shadowed -> not aliased
100 EXPECT_TRUE(has(gen, "return (math + 1LL);")); // bare param, not a namespace
101}
