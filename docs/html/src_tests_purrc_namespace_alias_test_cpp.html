tests/purrc/namespace_alias_test.cpp

1// System-level tests for the generated-code namespace aliasing. The whole program is
2// emitted inside a `namespace cheatah_program { … }`, where each stdlib module gets its
3// OWN short alias (`namespace io = ::cheatah::io;`, `namespace random = …`, …) so the
4// body reads `io::print` / `random::randint` instead of `cheatah::io::print`. The
5// wrapper namespace is what makes this SAFE: an alias like `random`/`time`/`socket`
6// cannot collide with the global C function of the same name.
7//
8// These run the .purr through the REAL purrc -> cheatah pipeline and assert on BOTH:
9// 1. the generated C++ (the feature itself — that each module is aliased distinctly), and
10// 2. the program's runtime output (that the aliased code still compiles and behaves).
11//
12// Named NamespaceAliasing (not *CompileRun*) so the QA gate always runs them.
13#include "e2e_harness.hpp"
14
15static bool has(const std::string& hay, const std::string& needle) {
16 return hay.find(needle) != std::string::npos;
17}
18
19// No collision: every imported module — plus the always-available builtins — gets its
20// own distinct alias, and the body uses the short form. The program still runs.
21TEST(NamespaceAliasing, EachModuleGetsItsOwnAlias) {
22 const std::string gen = e2e::expect_e2e_source("ns_distinct", R"PURR(import io
23import math
24io.print(math.sqrt(16.0), math.floor(2.7), len("hi"))
25)PURR", "4 2 2\n");
26 EXPECT_TRUE(has(gen, "namespace cheatah_program {"));
27 // Distinct alias per module — never `cheatah::io::` / `cheatah::math::` in the body.
28 EXPECT_TRUE(has(gen, "namespace io = ::cheatah::io;"));
29 EXPECT_TRUE(has(gen, "namespace math = ::cheatah::math;"));
30 EXPECT_TRUE(has(gen, "namespace builtins = ::cheatah::builtins;"));
31 EXPECT_TRUE(has(gen, "io::print("));
32 EXPECT_TRUE(has(gen, "math::sqrt("));
33 EXPECT_TRUE(has(gen, "builtins::len("));
34 EXPECT_FALSE(has(gen, "cheatah::io::print"));
35 EXPECT_FALSE(has(gen, "cheatah::math::sqrt"));
36}

38// Modules whose names match a global C library function (random -> ::random,
39// time -> ::time) are STILL aliased — the wrapper namespace prevents the clash. If the
40// alias leaked to global scope this would not compile ("redefinition of 'random'").
41TEST(NamespaceAliasing, LibcNamedModulesAreAliasedSafely) {
42 const std::string gen = e2e::expect_e2e_source("ns_libc", R"PURR(import io
43import random
44random.seed(123)
45io.print(random.randint(1, 1))
46)PURR", "1\n");
47 EXPECT_TRUE(has(gen, "namespace random = ::cheatah::random;"));
48 EXPECT_TRUE(has(gen, "random::seed("));
49 EXPECT_TRUE(has(gen, "random::randint("));
50 EXPECT_FALSE(has(gen, "cheatah::random::randint"));
51}

53// Collision with a PROGRAM identifier: a `struct os` shares the imported `os` module's
54// name, so os must NOT be aliased (it would clash with the struct inside the namespace).
55// It stays explicit (`::cheatah::os::…`) while the non-colliding io is still aliased —
56// and the program compiles and runs.
57TEST(NamespaceAliasing, ProgramNameCollisionStaysExplicit) {
58 const std::string gen = e2e::expect_e2e_source("ns_conflict", R"PURR(import io
59import os
60struct os { n: int }
61let p = os(7)
62io.print(os.path.join("a", "b"), p.n)
63)PURR", "a/b 7\n");
64 EXPECT_TRUE(has(gen, "namespace io = ::cheatah::io;")); // safe -> aliased
65 EXPECT_FALSE(has(gen, "namespace os = ::cheatah::os;")); // collides -> not aliased
66 EXPECT_TRUE(has(gen, "::cheatah::os::path::join")); // stays explicit
67}
68
69// A program function named like the generated entry point (`fn run`, `fn purr_main`)
70// must NOT collide with the internal trampoline function — the entry name escalates
71// until it's free, so the program compiles and runs.
72TEST(NamespaceAliasing, EntryFunctionNameNeverCollides) {
73 const std::string gen = e2e::expect_e2e_source("ns_entry", R"PURR(import io
74fn run() {
75 io.print("ran")
76}
77fn purr_main() {
78 run()
79}
80purr_main()
81)PURR", "ran\n");
82 // The exported C entry calls the escalated trampoline, NOT the user's run()/purr_main().
83 EXPECT_TRUE(has(gen, "PURR_EXPORT void purr_main() { cheatah_program::purr_main_(); }"));
84 EXPECT_FALSE(has(gen, "cheatah_program::run()"));
85}
86
87// A function parameter named like an imported module shadows it: the bare `math` is the
88// parameter (not the module namespace), aliasing is skipped, and the program still runs.
89TEST(NamespaceAliasing, ParameterNameShadowsModule) {
90 const std::string gen = e2e::expect_e2e_source("ns_param", R"PURR(import io
91import math
92fn bump(math) {
93 return math + 1
94}
95io.print(bump(41))
96)PURR", "42\n");
97 EXPECT_FALSE(has(gen, "namespace math = ::cheatah::math;")); // shadowed -> not aliased
98 EXPECT_TRUE(has(gen, "return (math + 1LL);")); // bare param, not a namespace
99}
