8#include <cstdint>

10namespace cheatah::memory {
11
12/// How the owner schedules pending writes against reads. Chosen at construction (a stored value).
13enum class policy : std::uint8_t {
14 interleave, ///< (default) readers renew *between* writes; writers + waiting readers are fair.
15 writes_first, ///< drain the ENTIRE write queue before renewing any reader (owner's declared choice).
16};
17/// Shorthand for `policy::interleave` — readers renew between writes (the default, fair) policy.
18inline constexpr policy interleave = policy::interleave;
19/// Shorthand for `policy::writes_first` — drain the whole write queue before renewing any reader.
20inline constexpr policy writes_first = policy::writes_first;
21
22/// The readable spelling of an immediate-write priority. Any negative priority is immediate; this is
23/// simply its name: `x.rwrite<memory::immediate>()` reads better than `x.rwrite<-1>()`.
24inline constexpr long long immediate = -1;
25
26} // namespace cheatah::memory
