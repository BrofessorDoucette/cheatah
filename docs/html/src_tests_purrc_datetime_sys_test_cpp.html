tests/purrc/datetime_sys_test.cpp

1// System-level test for the cheatah `datetime` module: a single cohesive .purr
2// program that exercises EVERY public function in stdlib/datetime/datetime.hpp
3// (timestamp, now, utcnow, today, format, year, month, day, hour, minute,
4// second, weekday), compiles it with purrc into a loadable module, runs it
5// under the cheatah runtime (TZ=UTC), and asserts the exact stdout.
6//
7// The current-time functions (timestamp/now/utcnow/today) are
8// non-deterministic, so the program prints DETERMINISTIC PROPERTIES (sign /
9// string length). The component getters and `format` use the calendar in local
10// time, so they are pinned to a FIXED epoch (0.0 = 1970-01-01 00:00:00 UTC) and
11// run under TZ=UTC (the 4th expect_e2e arg) to be fully deterministic.
12// Complements the in-process unit tests (stdlib/tests/datetime_test.cpp) and
13// the per-function compile-run tests (tests/purrc/datetime_cr_test.cpp).
14#include "e2e_harness.hpp"
15
16TEST(StdlibE2E, Datetime) {
17 e2e::expect_e2e("datetime_sys", R"PURR(import io
18import datetime
19
20# Current-time functions are non-deterministic; assert stable properties.
21io.print(datetime.timestamp() > 0.0)
22io.print(len(datetime.now()) == 19)
23io.print(len(datetime.utcnow()) == 20)
24io.print(len(datetime.today()) == 10)
25
26# Fixed epoch (0.0 = 1970-01-01 00:00:00 UTC) under TZ=UTC is fully deterministic.
27io.print(datetime.format(0.0, "%Y-%m-%d %H:%M:%S"))
28io.print(datetime.year(0.0), datetime.month(0.0), datetime.day(0.0))
29io.print(datetime.hour(0.0), datetime.minute(0.0), datetime.second(0.0))
30io.print(datetime.weekday(0.0))
31)PURR",
32 "True\n"
33 "True\n"
34 "True\n"
35 "True\n"
36 "1970-01-01 00:00:00\n"
37 "1970 1 1\n"
38 "0 0 0\n"
39 "3\n",
40 "TZ=UTC ");
41}
