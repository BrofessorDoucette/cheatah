18// Bug: a call argument / list literal split across multiple lines failed to parse with

21// continuation). This runs the exact (corrected) program from review/c1.purr.

28}

316// Compile `src` with purrc and capture its DIAGNOSTICS (stdout+stderr) so a test can assert the
317// error message itself, not just that compilation failed. Sets `rc` to purrc's exit code.
318static std::string compile_diagnostics(const std::string& name, const std::string& src, int& rc) {
319 const std::string tmp = PURR_TEST_TMP;
320 const std::string purr = tmp + "/" + name + ".purr";
321 const std::string mod = tmp + "/" + name + ".so";
322 {
323 std::ofstream f(purr);
324 f << src;
325 }
326 return e2e::run_capture(
327 std::string(PURRC_PATH) + " \"" + purr + "\" -o \"" + mod + "\" 2>&1", rc);
328}

397// Bug: `regex.search(regex.compile("a*$"), "bbb")` returned False — an empty match at
398// end-of-input was never tried. The `matches_empty` shortcut in the unanchored search path was
399// gated on "no end anchor", so an empty-matchable `$`-anchored pattern fell through to the
400// first-byte candidate scan, which only proposes positions holding a possible first byte ('a'
401// here; "bbb" has none) and can never propose end-of-input. Fixed: a pattern that can match
402// empty always matches an unanchored search — at position 0, or at end-of-input under `$`.
403// `find` already got this right (its start loop runs to i == len), so offsets are pinned too.
404TEST(PreviouslyBroken, RegexEmptyMatchAtEndOfInput) {
405 e2e::expect_e2e("prevbroken_regex_empty_end", R"PURR(import io
406import regex
407let star = regex.compile("a*$")
408io.print(regex.search(star, "bbb"))
409let m = regex.find(star, "bbb")
410io.print(m.matched, m.begin, m.end)
411io.print(regex.search(star, "baab"))
412io.print(regex.search(regex.compile("a+$"), "bbb"))
413)PURR",
414 "True\n" // a*$ matches empty at end of "bbb"
415 "True 3 3\n" // ... and find pins it at [3,3)
416 "True\n" // candidates exist mid-text but die; end still matches
417 "False\n"); // a+ can NOT match empty — must stay a miss

446// `undefined symbol: cheatah::hashlib::hkdf_expand_sha384`. purrc emitted the static archives in
447// the order codegen reported its imports — alphabetical, because codegen keeps roots in a
448// std::set — so libcheatah_hashlib.a was scanned BEFORE libcheatah_tls.a. A .a only contributes
449// members resolving an already-undefined symbol, so hashlib's member was passed over; tls.cpp
450// then referenced hkdf_expand_sha384, and since a shared object may carry undefined symbols the
451// LINK SUCCEEDED and the failure was deferred to load time, far from its cause.
452//
453// `import tls` alone always worked — hashlib was discovered as tls's dependency and appended
454// after it — which is why nothing caught this: no .purr under tests/ imported tls at all.
455//
456// Fixed by making all_modules a post-order DFS (dependents before dependencies) in
457// compiler/purrc.cpp. --start-group/--end-group rode along as a backstop, and on 2026-08-16 it
458// was emitted UNCONDITIONALLY -- which broke every macOS link that resolved an archive-bearing
459// module, because ld64 rejects the option outright. It is now supplied by the platform seam
460// (cmake/Portability.cmake) and is EMPTY on macOS and Windows, whose linkers re-scan archives
461// instead of making one ordered pass.
462//
463// So on those platforms this test is carried by the ORDERING alone -- which is the point: the
464// DFS is the mechanism, the group was only ever a belt for a missing cheatah-deps edge or a
465// future cycle. Verified by building purrc with the group forced empty and re-running this
466// exact case; it links and runs.
467//
468// The import of hashlib is what makes this a regression test rather than a smoke test: remove
469// it and the program links even with the bug present. main() must also CALL tls, or the linker
470// never needs the symbol and the test proves nothing.
471TEST(PreviouslyBroken, TlsLinksWhenHashlibIsAlsoImported) {
472 e2e::expect_e2e("prevbroken_tls_hashlib_link",
473 "import io\n"
474 "import hashlib\n"
475 "import tls\n"
476 "fn main() {\n"
477 " io.print(\"tls-linked\")\n"
478 " io.print(tls.last_error())\n"
479 "}\n"
480 "main()\n",
481 "tls-linked\n\n");
482}

500// Bug: a slice assignment compiled clean and did NOTHING. `xs[1:3] = [9, 9]` reached value-position
501// codegen, emitting `builtins::slice(xs, 1, 3) = ...` — a write into the temporary the slice
502// returned — so the list came back unchanged and the program still exited 0. Silence was the
503// defect, so this asserts the OUTPUT of every form, not merely a successful compile.
504TEST(PreviouslyBroken, SliceAssignmentActuallyWrites) {
505 int rc = -1;
506 const std::string out = e2e::run_purr_file(
507 "prevbroken_slice_assign", std::string(PREVBROKEN_DIR) + "/slice_assign.purr", rc);
508 EXPECT_EQ(rc, 0) << "slice_assign.purr failed to compile or run";
509 EXPECT_EQ(out,
510 "[1, 9, 9, 4]\n" // the exact program from the report
511 "[1, 7, 7, 7, 4]\n" // grows
512 "[1, 5, 4]\n" // shrinks
513 "[1, 4]\n" // deletes
514 "[1, 1, 2, 3, 4, 4]\n" // xs[1:3] = xs
515 "[1, 1, 2, 4]\n" // overlapping source
516 "[1, 9, 9, 4] 4\n" // ndarray copies in, shape unchanged
517 "[1, 99, 9, 4]\n" // and a slice read is a view onto the parent
518 "8 8 3\n"); // fixarray fills its fixed extent
519}
