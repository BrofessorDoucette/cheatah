2// Original work; see ACKNOWLEDGMENTS.md for the open-source ideas we build upon.
3//
4// bench_main — the entry point for cheatah_benchmarks, replacing benchmark::benchmark_main.
5//
6// It exists for the three things a Google Benchmark flag cannot express:
7//
8// 1. PROVENANCE. A published number is worthless without the machine, commit, compiler,
9// competitor versions, and harness settings that produced it. Those go into the
10// benchmark's own context (so they land in --benchmark_out_format=json) and into the
11// stamp block of any generated Markdown table.
12// 2. A PUBLISHABILITY VERDICT. Repetitions stay a FLAG rather than a per-registration
13// ->Repetitions() call, because Benchmark::Repetitions() overrides the flag and would
14// make scripts/bench_smoke.sh — the fast pass inside the QA gate, whose timings are
15// explicitly discarded — pay for statistics it throws away. The cost of that choice is
16// that someone can run this binary bare and read the numbers as if they were rigorous.
17// So the binary judges its own run: fewer than kMinPublishReps repetitions, or no
18// interleaving, or too short a min_time, and it says so on stderr and records
19// cheatah_publishable=false where a script can see it.
20// 3. THE COMPARISON TABLE. Google Benchmark reports rows; it has no idea that
21// BM_dot_cheatah and BM_dot_eigen are two sides of one claim. bench_pairs.hpp supplies
22// that, and CHEATAH_BENCH_TABLE=<path> writes the paired result out as stamped
23// Markdown — the same trick stdlib/regex/bench/rxbench.cpp already uses, generalized.
24//
25// Everything here is additive: the banner goes to stderr only, the exit status is
26// unchanged, and stdout stays byte-for-byte what the default main would have produced, so
27// scripts/bench_smoke.sh (greps stdout for ^BM_) and scripts/bench_gate.sh (parses stdout
28// CSV) are unaffected.

59// A run must clear all three to be publishable. The repetition floor is 5 because that is
60// the smallest odd count where a median is not just "the middle of three"; the min_time
61// floor keeps a case from being timed inside one scheduler slice.
62constexpr int kMinPublishReps = 5;
63constexpr double kMinPublishMinTimeSec = 0.2;
64
65// The same band scripts/bench_gate.sh uses, for the same reason: a ratio on a
66// sub-nanosecond operation is noise, so a difference must clear BOTH a relative and an
67// absolute floor before it is called a win or a loss.
68constexpr double kThreshold = 1.15;
69constexpr double kMinGapNs = 0.25;

73 int reps = 1;
74 double min_time_sec = 0.0; // 0 = flag absent, i.e. Google Benchmark's own default
75 bool interleaved = false;
76 bool aggregates_only = false;

80// Scanned from argv BEFORE benchmark::Initialize, which consumes the flags it recognizes.
81RunSettings scan_settings(int argc, char** argv) {
82 RunSettings s;
83 for (int i = 1; i < argc; ++i) {
84 const char* a = argv[i];
85 if (std::strncmp(a, "--benchmark_repetitions=", 24) == 0) {
86 s.reps = std::atoi(a + 24); // NOLINT(cert-err34-c): a benchmark flag we control, not untrusted input
87 } else if (std::strncmp(a, "--benchmark_min_time=", 21) == 0) {
88 s.min_time_sec = std::strtod(a + 21, nullptr); // "0.2s" -> 0.2; "10x" -> 10
89 if (std::strchr(a + 21, 'x') != nullptr) s.min_time_sec = 0.0; // iteration form
90 } else if (std::strncmp(a, "--benchmark_enable_random_interleaving=", 39) == 0) {
91 s.interleaved = std::strcmp(a + 39, "true") == 0 || std::strcmp(a + 39, "1") == 0;
92 } else if (std::strncmp(a, "--benchmark_report_aggregates_only=", 35) == 0) {
93 s.aggregates_only = std::strcmp(a + 35, "true") == 0 || std::strcmp(a + 35, "1") == 0;
94 } else if (std::strncmp(a, "--benchmark_list_tests", 22) == 0) {
95 s.listing_only = std::strstr(a, "=false") == nullptr;
96 }
97 }
98 return s;
99}

168struct CaseTimes {
169 std::vector<double> raw_ns; // one per repetition, when they are reported
170 double median_ns = 0.0; // the median aggregate, when Google Benchmark reports one
171 double stddev_ns = -1.0; // the stddev aggregate, when present
172 double bytes_per_sec = 0.0; // set only by throughput benchmarks
173 double items_per_sec = 0.0; // SetItemsProcessed — p256 counts operations, not bytes
174};

176double median_of(std::vector<double> v) {
177 if (v.empty()) return 0.0;
178 std::sort(v.begin(), v.end());
179 const std::size_t n = v.size();
180 return (n % 2) ? v[n / 2] : 0.5 * (v[n / 2 - 1] + v[n / 2]);
181}
182
183// Inter-quartile range: the dispersion we publish. Chosen over stddev because a benchmark
184// sample is not normal — one descheduled repetition drags a stddev far more than it drags
185// the middle 50%.
186double iqr_of(std::vector<double> v) {
187 if (v.size() < 4) return -1.0;
188 std::sort(v.begin(), v.end());
189 auto quantile = [&v](double q) {
190 const double pos = q * static_cast<double>(v.size() - 1);
191 const auto lo = static_cast<std::size_t>(pos);
192 const std::size_t hi = std::min(lo + 1, v.size() - 1);
193 return v[lo] + (pos - static_cast<double>(lo)) * (v[hi] - v[lo]);
194 };
195 return quantile(0.75) - quantile(0.25);
196}

205 bool ReportContext(const Context& context) override {

207 num_cpus_ = context.cpu_info.num_cpus;
208 mhz_ = context.cpu_info.cycles_per_second / 1e6;
209 switch (context.cpu_info.scaling) {
210 case benchmark::CPUInfo::ENABLED: scaling_ = "enabled"; break;
211 case benchmark::CPUInfo::DISABLED: scaling_ = "disabled"; break;
212 default: scaling_ = "unknown"; break;

217 void ReportRunsConfig(double min_time, bool has_explicit_iters,
218 benchmark::IterationCount iters) override {
219 inner_->ReportRunsConfig(min_time, has_explicit_iters, iters);
220 }
221
222 void ReportRuns(const std::vector<Run>& report) override {
223 for (const Run& r : report) {
224 if (r.skipped) continue;
225 CaseTimes& c = cases_[r.run_name.str()];
226 const bool is_agg = r.run_type == Run::RT_Aggregate;
227
228 // SetBytesProcessed lands in the counters map, not a named field — and every
229 // aggregate row carries one, including `stddev` and `cv`. Taking the last one
230 // seen would report the standard deviation of the throughput as the throughput.
231 // Only a real measurement (a raw run, or the median aggregate) may set it.
232 if (!is_agg || r.aggregate_name == "median") {
233 if (const auto it = r.counters.find("bytes_per_second"); it != r.counters.end())
234 if (it->second.value > 0.0) c.bytes_per_sec = it->second.value;
235 // p256_bench.cpp uses SetItemsProcessed, not SetBytesProcessed — a signing
236 // operation has no meaningful byte count. Without this the `solo` layout's
237 // throughput column would be silently empty for the one suite that needs it.
238 if (const auto it = r.counters.find("items_per_second"); it != r.counters.end())
239 if (it->second.value > 0.0) c.items_per_sec = it->second.value;
240 }
241
242 if (is_agg) {
243 if (r.aggregate_name == "median") c.median_ns = r.GetAdjustedRealTime();
244 if (r.aggregate_name == "stddev") c.stddev_ns = r.GetAdjustedRealTime();
245 } else {
246 c.raw_ns.push_back(r.GetAdjustedRealTime());
247 }
248 }
249 inner_->ReportRuns(report);
250 }
251
252 void Finalize() override { inner_->Finalize(); }

279// The dispersion we can honestly report for a case. IQR is preferred — a benchmark sample
280// is not normal, and one descheduled repetition moves a standard deviation far more than it
281// moves the middle 50%. But --benchmark_report_aggregates_only (what both gates pass)
282// suppresses the raw per-repetition rows, leaving only Google Benchmark's own stddev. Fall
283// back to that rather than print nothing, and label which one it is.
284struct Spread {
285 double value = -1.0;
286 const char* kind = "";
287};
288
289Spread spread_of(const CaseTimes& c) {
290 if (const double q = iqr_of(c.raw_ns); q >= 0.0) return {q, "IQR"};
291 if (c.stddev_ns >= 0.0) return {c.stddev_ns, "sd"};
292 return {};
293}
294
295std::string fmt_disp(const Spread& s) {
296 if (s.value < 0.0) return "—";
297 return "±" + fmt_ns(s.value) + " " + s.kind;
298}

300// >1 means cheatah is faster. The verdict uses the same two-floor band as bench_gate.sh.
301std::string verdict(double ours, double rival) {
302 if (rival > ours * kThreshold && (rival - ours) > kMinGapNs) return "faster";
303 if (ours > rival * kThreshold && (ours - rival) > kMinGapNs) return "**slower**";
304 return "parity";
305}

307// ---- layouts -------------------------------------------------------------------------
308//
309// One suite, one shape. Rather than let each document invent a table and be hand-typed into
310// place (which is how every published table in this repo ended up unreproducible), the
311// FORMATTER lives here and the document is generated from it. Selected by
312// CHEATAH_BENCH_LAYOUT; unset means `pairs`, whose bytes are unchanged, so scripts/bench_gate.sh
313// and ad-hoc runs are unaffected.
314enum class Layout { Pairs, OpsType, Throughput, Solo, Highlights };
315
316Layout layout_from_env() {
317 const std::string v = env_or("CHEATAH_BENCH_LAYOUT", "pairs");
318 if (v == "pairs") return Layout::Pairs;
319 if (v == "opstype") return Layout::OpsType;
320 if (v == "throughput") return Layout::Throughput;
321 if (v == "solo") return Layout::Solo;
322 if (v == "highlights") return Layout::Highlights;
323 // Never silently fall back to a different shape than asked for: that would publish a
324 // table nobody reviewed under a stamp claiming it was the one they did.
325 static_cast<void>(std::fprintf(stderr, "bench_main: unknown CHEATAH_BENCH_LAYOUT '%s' — using `pairs`\n",
326 v.c_str()));
327 return Layout::Pairs;
328}

330// The command that reproduces this table, rebuilt from the argv captured before
331// benchmark::Initialize ate it, plus the CHEATAH_BENCH_* variables that actually shaped it.
332std::string reproduction_command() {
333 std::vector<std::string> lines;
334 for (const char* key : {"CHEATAH_BENCH_SUITE", "CHEATAH_BENCH_LAYOUT", "CHEATAH_BENCH_ROWS",
335 "CHEATAH_BENCH_WATCH"}) {
336 const std::string v = env_or(key, "");
337 if (!v.empty()) lines.push_back(std::string(key) + "='" + v + "'");
338 }
339 std::string argv_line;
340 for (std::size_t i = 0; i < g_argv.size(); ++i) {
341 argv_line += g_argv[i];
342 if (i + 1 < g_argv.size()) argv_line += " ";
343 }
344 // CHEATAH_BENCH_TABLE is deliberately omitted: it is where THIS run happened to write,
345 // not part of what the table measures, and baking a /tmp path into a published doc is
346 // noise. The reader supplies their own output path.
347 if (!argv_line.empty()) lines.push_back(argv_line);
348 std::string cmd;
349 for (std::size_t i = 0; i < lines.size(); ++i) {
350 cmd += lines[i];
351 if (i + 1 < lines.size()) cmd += " \\\n ";
352 }
353 return cmd;
354}

398OpType split_op_type(const std::string& key) {
399 std::string stem = key;
400 if (stem.rfind("BM_", 0) == 0) stem = stem.substr(3);
401 const std::size_t u = stem.rfind('_');
402 if (u == std::string::npos) return {stem, ""};

413// For THROUGHPUT, higher is better, so the band test runs on the rates rather than on times.
414std::string prose_gap(double ours_bps, double rival_bps) {
415 if (ours_bps <= 0.0 || rival_bps <= 0.0) return "—";
416 char buf[64];
417 if (ours_bps > rival_bps * kThreshold) {
418 static_cast<void>(std::snprintf(buf, sizeof buf, "**%.2f\u00d7 faster**", ours_bps / rival_bps));
419 } else if (rival_bps > ours_bps * kThreshold) {
420 static_cast<void>(std::snprintf(buf, sizeof buf, "%.2f\u00d7 slower", rival_bps / ours_bps));
421 } else {
422 static_cast<void>(std::snprintf(buf, sizeof buf, "**parity** (%.2f\u00d7)",
423 ours_bps > rival_bps ? ours_bps / rival_bps : rival_bps / ours_bps));
424 }
425 return buf;
426}

428// CHEATAH_BENCH_ROWS='key=label;key2=label2' — curation lives in the INVOCATION, which the
429// stamp records verbatim, rather than in a Markdown edit nobody can reproduce. Order is
430// curation too, so rows emit in the order given.
431std::vector<std::pair<std::string, std::string>> parse_row_spec(const std::string& spec) {
432 std::vector<std::pair<std::string, std::string>> out;
433 std::size_t i = 0;
434 while (i < spec.size()) {
435 std::size_t semi = spec.find(';', i);
436 if (semi == std::string::npos) semi = spec.size();
437 const std::string entry = spec.substr(i, semi - i);
438 if (!entry.empty()) {
439 const std::size_t eq = entry.find('=');
440 if (eq == std::string::npos) out.emplace_back(entry, entry);
441 else out.emplace_back(entry.substr(0, eq), entry.substr(eq + 1));
442 }
443 i = semi + 1;
444 }
445 return out;
446}

463// One case -> one number: the median aggregate when Google Benchmark computed one, else the
464// median of the repetitions we saw.
465struct Resolved {
466 std::map<std::string, double> ns, bps, ips;
467 std::map<std::string, Spread> spread;
468};
469
470Resolved resolve(const PairTally& tally) {
471 Resolved r;
472 for (const auto& [name, c] : tally.cases()) {
473 r.ns[name] = (c.median_ns > 0.0) ? c.median_ns : median_of(c.raw_ns);
474 r.spread[name] = spread_of(c);
475 r.bps[name] = c.bytes_per_sec;
476 r.ips[name] = c.items_per_sec;
477 }
478 return r;
479}

497std::vector<PairedRow> collect_pairs(const Resolved& res) {
498 const auto& ns = res.ns;
499 const auto& bps = res.bps;
500 const auto& spread = res.spread;
501
502 // Group by pair key. `ours` and the rivals of one key land in the same slot.
503 struct Slot {
504 std::string ours_name;
505 std::map<std::string, std::string> rivals; // label -> benchmark name
506 };
507 std::map<std::string, Slot> slots;
508 for (const auto& [name, _] : ns) {
509 const cheatah::bench::Pairing p = cheatah::bench::classify_benchmark(name);
510 if (p.side == cheatah::bench::Side::Ours) slots[p.key].ours_name = name;
511 else if (p.side == cheatah::bench::Side::Rival) slots[p.key].rivals[p.label] = name;
512 }
513
514 std::vector<PairedRow> rows;
515 for (const auto& [key, slot] : slots) {
516 if (slot.ours_name.empty()) continue; // a rival with no cheatah twin proves nothing
517 for (const auto& [label, rival_name] : slot.rivals) {
518 PairedRow r;
519 r.key = key;
520 r.rival_label = label;
521 r.ours_ns = ns.at(slot.ours_name);
522 r.ours_spread = spread.at(slot.ours_name);
523 r.rival_ns = ns.at(rival_name);
524 r.rival_spread = spread.at(rival_name);
525 r.ours_bps = bps.at(slot.ours_name);
526 r.rival_bps = bps.at(rival_name);
527 rows.push_back(r);
528 }
529 }
530
531 return rows;
532}

534// The stamp. `layout:` and `watch:` are emitted ONLY when their env var is set, so a caller
535// that sets neither (bench_gate.sh, an ad-hoc run) gets byte-for-byte what this file emitted
536// before layouts existed. `note` is appended verbatim before the closing --> so a layout can
537// add its own reproduction block.
538void emit_stamp(std::FILE* f, const PairTally& tally, const RunSettings& settings,
539 const std::string& note) {
540 const bool ok = publishable(settings);
541 static_cast<void>(std::fprintf(f,
542 "<!-- cheatah-bench-stamp v1\n"
543 " suite: %s\n"
544 " generated: %s\n"
545 " commit: %s\n"
546 " host: %s\n"
547 " cpu-scaling: %s\n"
548 " build: %s, Google Benchmark %s\n"
549 " competitors: %s\n"
550 " harness: %s\n"
551 " statistic: median real time per case; spread = IQR over\n"
552 " repetitions, or `sd` where\n"
553 " --benchmark_report_aggregates_only hid the raw runs\n"
554 " publishable: %s\n",
555 env_or("CHEATAH_BENCH_SUITE", "cheatah_benchmarks").c_str(), today().c_str(),
556 env_or("CHEATAH_BENCH_COMMIT", "unknown").c_str(), tally.host_line().c_str(),
557 tally.scaling().c_str(), compiler_id().c_str(),
558 benchmark::GetBenchmarkVersion().c_str(),
559 competitor_versions().c_str(), harness_line(settings).c_str(),
560 ok ? "true" : "false"));
561
562 const std::string layout = env_or("CHEATAH_BENCH_LAYOUT", "");
563 if (!layout.empty()) static_cast<void>(std::fprintf(f, " layout: %s\n", layout.c_str()));
564 // `watch:` is what scripts/bench_table.purr uses to decide whether this table has gone
565 // stale: it names the source paths whose change invalidates the measurement.
566 const std::string watch = env_or("CHEATAH_BENCH_WATCH", "");
567 if (!watch.empty()) static_cast<void>(std::fprintf(f, " watch: %s\n", watch.c_str()));
568 if (!note.empty()) static_cast<void>(std::fprintf(f, "%s", note.c_str()));
569 static_cast<void>(std::fprintf(f, "-->\n\n"));
570}

572void emit_pairs(std::FILE* f, const std::vector<PairedRow>& rows) {
573 if (rows.empty()) {
574 static_cast<void>(std::fprintf(f, "_No paired cases in this run._\n"));
575 return;
576 }
577
578 static_cast<void>(std::fprintf(f, "| case | cheatah | spread | vs | rival | spread | ratio | verdict |\n"));
579 static_cast<void>(std::fprintf(f, "|---|--:|--:|---|--:|--:|--:|---|\n"));
580 for (const PairedRow& r : rows) {
581 static_cast<void>(std::fprintf(f, "| %s | %s | %s | %s | %s | %s | %.2fx | %s |\n", r.key.c_str(),
582 fmt_ns(r.ours_ns).c_str(), fmt_disp(r.ours_spread).c_str(),
583 r.rival_label.c_str(), fmt_ns(r.rival_ns).c_str(),
584 fmt_disp(r.rival_spread).c_str(),
585 (r.ours_ns > 0.0) ? r.rival_ns / r.ours_ns : 0.0,
586 verdict(r.ours_ns, r.rival_ns).c_str()));
587 }
588
589 // Throughput rows, when the benchmark set bytes_per_second. Reported separately rather
590 // than as another column: only some suites measure it, and a mostly-empty column reads
591 // as missing data rather than as "not applicable".
592 const bool any_bps = std::any_of(rows.begin(), rows.end(), [](const PairedRow& r) {
593 return r.ours_bps > 0.0 && r.rival_bps > 0.0;
594 });
595 if (any_bps) {
596 static_cast<void>(std::fprintf(f, "\n| case | cheatah | vs | rival | ratio |\n|---|--:|---|--:|--:|\n"));
597 for (const PairedRow& r : rows) {
598 if (r.ours_bps <= 0.0 || r.rival_bps <= 0.0) continue;
599 static_cast<void>(std::fprintf(f, "| %s | %s | %s | %s | %.2fx |\n", r.key.c_str(),
600 fmt_gibs(r.ours_bps).c_str(), r.rival_label.c_str(),
601 fmt_gibs(r.rival_bps).c_str(), r.ours_bps / r.rival_bps));
602 }
603 }
604
605 // The tally, per rival — the honest summary line, including losses.
606 std::map<std::string, std::array<int, 3>> tally_by_rival; // label -> {faster, parity, slower}
607 for (const PairedRow& r : rows) {
608 const std::string v = verdict(r.ours_ns, r.rival_ns);
609 auto& t = tally_by_rival[r.rival_label];
610 if (v == "faster") ++t[0];
611 else if (v == "parity") ++t[1];
612 else ++t[2];
613 }
614 static_cast<void>(std::fprintf(f, "\n**Tally** (a difference counts only above %.2fx AND %.2f ns) — ", kThreshold,
615 kMinGapNs));
616 bool first = true;
617 for (const auto& [label, t] : tally_by_rival) {
618 static_cast<void>(std::fprintf(f, "%svs %s: **%d faster / %d parity / %d slower**", first ? "" : "; ",
619 label.c_str(), t[0], t[1], t[2]));
620 first = false;
621 }
622 static_cast<void>(std::fprintf(f, ".\n"));
623
624 for (const PairedRow& r : rows) {
625 if (verdict(r.ours_ns, r.rival_ns) == "**slower**")
626 static_cast<void>(std::fprintf(f, "- Loss vs %s: `%s` — cheatah %s vs %s (%.2fx slower)\n",
627 r.rival_label.c_str(), r.key.c_str(), fmt_ns(r.ours_ns).c_str(),
628 fmt_ns(r.rival_ns).c_str(), r.ours_ns / r.rival_ns));
629 }
630
631}
632
633// ---- opstype: the fixarray-vs-GLM shape ----------------------------------------------
634//
635// RATIO DIRECTION: ours/rival, so LOWER IS BETTER here. That is the opposite of `pairs` and
636// of `highlights` (both rival/ours, higher is better). The inversion is deliberate — this
637// table's column header says "lower is better" — but it is a genuine trap, so if you touch
638// one of the three, check the other two.
639void emit_opstype(std::FILE* f, const std::vector<PairedRow>& rows, const RunSettings& settings) {
640 if (rows.empty()) {
641 static_cast<void>(std::fprintf(f, "_No paired cases in this run._\n"));
642 return;
643 }
644 const char* kind = rows.front().ours_spread.kind;
645 const std::string spread_word =
646 (std::strcmp(kind, "IQR") == 0) ? "IQR" : "sample standard deviation";
647
648 // No blank line between --> and <details>: the rendered page reads as one block.
649 static_cast<void>(std::fprintf(f,
650 "<details><summary><b>Full %s comparison — all %zu operations</b> "
651 "(ns, lower is better; ± is the %s over %d interleaved repetitions)"
652 "</summary>\n\n",
653 cheatah::bench::display_label(rows.front().rival_label).c_str(), rows.size(),
654 spread_word.c_str(), settings.reps));
655
656 struct Group {
657 const char* heading;
658 std::vector<const PairedRow*> rows;
659 };
660 Group vectors{"Vectors", {}}, matrices{"Matrices", {}}, other{"Other", {}};
661 for (const PairedRow& r : rows) {
662 const std::string t = split_op_type(r.key).type;
663 if (t.rfind("mat", 0) == 0) matrices.rows.push_back(&r);
664 else if (t.rfind("vec", 0) == 0) vectors.rows.push_back(&r);
665 else other.rows.push_back(&r);
666 }
667
668 for (const Group* g : {&vectors, &matrices, &other}) {
669 if (g->rows.empty()) continue; // no stray heading over an empty section
670 static_cast<void>(std::fprintf(f, "\n#### %s\n\n", g->heading));
671 static_cast<void>(std::fprintf(f, "| operation | type | `Fixed` | %s | ratio | |\n",
672 cheatah::bench::display_label(g->rows.front()->rival_label).c_str()));
673 static_cast<void>(std::fprintf(f, "|---|---|--:|--:|:--:|---|\n"));
674 for (const PairedRow* r : g->rows) {
675 const OpType ot = split_op_type(r->key);
676 static_cast<void>(std::fprintf(f, "| `%s` | %s | %s %s | %s %s | %s | %s |\n", ot.op.c_str(),
677 ot.type.c_str(), fmt_num2(r->ours_ns).c_str(),
678 fmt_pm2(r->ours_spread).c_str(), fmt_num2(r->rival_ns).c_str(),
679 fmt_pm2(r->rival_spread).c_str(),
680 fmt_ratio(r->rival_ns > 0.0 ? r->ours_ns / r->rival_ns : 0.0).c_str(),
681 emoji_verdict(r->ours_ns, r->rival_ns)));
682 }
683 }
684
685 int faster = 0, parity = 0, slower = 0;
686 for (const PairedRow& r : rows) {
687 const std::string v = verdict(r.ours_ns, r.rival_ns);
688 if (v == "faster") ++faster;
689 else if (v == "parity") ++parity;
690 else ++slower;
691 }
692 static_cast<void>(std::fprintf(f, "\n**%d faster, %d at parity, %d slower** across %zu operations.\n\n",
693 faster, parity, slower, rows.size()));
694 static_cast<void>(std::fprintf(f, "</details>\n"));
695}

697// ---- throughput: the crypto-vs-OpenSSL shape -----------------------------------------
698//
699// Rates, not times: for a cipher the interesting quantity is bytes per second, and the
700// verdict band therefore runs on the RATES (higher better) rather than on the ns.
701void emit_throughput(std::FILE* f, std::vector<PairedRow> rows) {
702 // Only rows where both sides measured a rate. This is what keeps
703 // BM_CryptoChaCha20Poly1305_Cheatah_Into — a variant of ours with no rival — out.
704 rows.erase(std::remove_if(rows.begin(), rows.end(),
705 [](const PairedRow& r) {
706 return r.ours_bps <= 0.0 || r.rival_bps <= 0.0;
707 }),
708 rows.end());
709 if (rows.empty()) {
710 static_cast<void>(std::fprintf(f, "_No throughput pairs in this run._\n"));
711 return;
712 }
713 // Best standing first: the reader should meet the honest headline, then the gaps.
714 std::sort(rows.begin(), rows.end(), [](const PairedRow& a, const PairedRow& b) {
715 return (a.rival_bps / a.ours_bps) < (b.rival_bps / b.ours_bps);
716 });
717
718 static_cast<void>(std::fprintf(f, "| Primitive | cheatah | %s | gap |\n|-----------|--------:|--------:|----:|\n",
719 cheatah::bench::display_label(rows.front().rival_label).c_str()));
720 for (const PairedRow& r : rows) {
721 const std::string gap = prose_gap(r.ours_bps, r.rival_bps);
722 const bool losing = gap.find("slower") != std::string::npos;
723 const std::string ours = fmt_rate(r.ours_bps, 0.0);
724 static_cast<void>(std::fprintf(f, "| %s | %s%s%s | %s | %s |\n",
725 cheatah::bench::label_for(r.key).c_str(), losing ? "" : "**", ours.c_str(),
726 losing ? "" : "**", fmt_rate(r.rival_bps, 0.0).c_str(), gap.c_str()));
727 }
728}

730// ---- solo: a suite with no rival (p256) ----------------------------------------------
731void emit_solo(std::FILE* f, const std::vector<SoloRow>& rows) {
732 if (rows.empty()) {
733 static_cast<void>(std::fprintf(f, "_No cases in this run._\n"));
734 return;
735 }
736 static_cast<void>(std::fprintf(f, "| Op | median | spread | throughput |\n|---|--:|--:|--:|\n"));
737 for (const SoloRow& r : rows)
738 static_cast<void>(std::fprintf(f, "| `%s` | %s | %s | %s |\n", r.key.c_str(), fmt_ns(r.ns).c_str(),
739 fmt_disp(r.spread).c_str(), fmt_rate(r.bps, r.ips).c_str()));
740}

742// ---- highlights: a curated subset, curated in the INVOCATION --------------------------
743//
744// RATIO DIRECTION: rival/ours, higher is better — the opposite of `opstype`. See the note
745// there.
746bool emit_highlights(std::FILE* f, const std::vector<PairedRow>& rows, const std::string& spec) {
747 const auto wanted = parse_row_spec(spec);
748 if (wanted.empty()) {
749 static_cast<void>(std::fprintf(f, "_CHEATAH_BENCH_ROWS was empty — nothing to highlight._\n"));
750 return false;
751 }
752 std::map<std::string, const PairedRow*> by_key;
753 for (const PairedRow& r : rows) by_key[r.key] = &r;
754
755 static_cast<void>(std::fprintf(f, "| operation | `Fixed` | %s | | |\n|-----------|--------:|----:|---|---|\n",
756 rows.empty() ? "rival"
757 : cheatah::bench::display_label(rows.front().rival_label).c_str()));
758 bool complete = true;
759 for (const auto& [key, label] : wanted) {
760 const auto it = by_key.find(key);
761 if (it == by_key.end()) {
762 // Do NOT silently drop it: a curation spec naming a case that did not run is a
763 // broken table, and the em-dashes plus publishable:false make the gate say so.
764 static_cast<void>(std::fprintf(f, "| `%s` | — | — | — | not measured |\n", label.c_str()));
765 static_cast<void>(std::fprintf(stderr, "bench_main: CHEATAH_BENCH_ROWS names '%s', which produced no "
766 "measurement\n", key.c_str()));
767 complete = false;
768 continue;
769 }
770 const PairedRow* r = it->second;
771 const std::string v = verdict(r->ours_ns, r->rival_ns);
772 std::string note = v;
773 if (v == "parity") {
774 char buf[64];
775 static_cast<void>(std::snprintf(buf, sizeof buf, "parity — gap %.2f ns",
776 r->rival_ns > r->ours_ns ? r->rival_ns - r->ours_ns
777 : r->ours_ns - r->rival_ns));
778 note = buf;
779 }
780 const bool win = (v == "faster");
781 static_cast<void>(std::fprintf(f, "| `%s` | %s%s%s %s | %s %s | %s | %s |\n", label.c_str(),
782 win ? "**" : "", fmt_ns(r->ours_ns).c_str(), win ? "**" : "",
783 fmt_pm2(r->ours_spread).c_str(), fmt_ns(r->rival_ns).c_str(),
784 fmt_pm2(r->rival_spread).c_str(),
785 fmt_ratio(r->ours_ns > 0.0 ? r->rival_ns / r->ours_ns : 0.0).c_str(),
786 note.c_str()));
787 }
788 return complete;
789}

791void write_table(const char* path, const PairTally& tally, const RunSettings& settings) {
792 const Layout layout = layout_from_env();
793 const Resolved res = resolve(tally);
794 const std::vector<PairedRow> pairs = collect_pairs(res);
795
796 std::FILE* f = std::fopen(path, "w");
797 if (f == nullptr) {
798 static_cast<void>(std::fprintf(stderr, "bench_main: cannot write table to %s\n", path));
799 return;
800 }
801
802 // Every published table carries its reproduction command into the stamp, so it says exactly
803 // how to regenerate itself — including the curation spec where a layout has one. This used to
804 // skip the `pairs` layout, which left linalg-vs-eigen as the one table a reader could not
805 // reproduce from what it printed.
806 const std::string note = "\n PRODUCED BY:\n " + reproduction_command() + "\n";
807 emit_stamp(f, tally, settings, note);
808
809 switch (layout) {
810 case Layout::Pairs: emit_pairs(f, pairs); break;
811 case Layout::OpsType: emit_opstype(f, pairs, settings); break;
812 case Layout::Throughput: emit_throughput(f, pairs); break;
813 case Layout::Solo: emit_solo(f, collect_solo(res)); break;
814 case Layout::Highlights:
815 emit_highlights(f, pairs, env_or("CHEATAH_BENCH_ROWS", ""));
816 break;
817 }
818
819 static_cast<void>(std::fclose(f));
820 static_cast<void>(std::fprintf(stderr, "bench_main: table written to %s\n", path));
821}

825int main(int argc, char** argv) {
826 const RunSettings settings = scan_settings(argc, argv);
827 remember_argv(argc, argv); // before Initialize, which consumes the flags it recognizes
828
829 benchmark::Initialize(&argc, argv);
830 if (benchmark::ReportUnrecognizedArguments(argc, argv)) return 1;
831
832 const bool ok = publishable(settings);
833 benchmark::AddCustomContext("cheatah_publishable", ok ? "true" : "false");
834 benchmark::AddCustomContext("cheatah_commit", env_or("CHEATAH_BENCH_COMMIT", "unknown"));
835 benchmark::AddCustomContext("cheatah_compiler", compiler_id());
836 benchmark::AddCustomContext("cheatah_competitors", competitor_versions());
837 benchmark::AddCustomContext("cheatah_harness", harness_line(settings));
838 benchmark::AddCustomContext("cheatah_statistic", "median real time; dispersion = IQR");
839
840 // stderr only, and never fatal: a fast unpublishable run is a legitimate thing to want
841 // (that is exactly what the QA gate's smoke pass is). What is not legitimate is reading
842 // its numbers as a benchmark result, so say so where a human will see it.
843 if (!ok && !settings.listing_only && std::getenv("CHEATAH_BENCH_SMOKE") == nullptr) {
844 static_cast<void>(std::fprintf(stderr,
845 "\n[bench_main] NOT PUBLISHABLE — %s.\n"
846 " These timings are indicative only. For numbers that may be "
847 "published,\n"
848 " run through scripts/bench_run.sh publish (>=%d repetitions, "
849 ">=%.1fs min_time,\n"
850 " random interleaving on).\n\n",
851 harness_line(settings).c_str(), kMinPublishReps, kMinPublishMinTimeSec));
852 }
853
854 // CreateDefaultDisplayReporter returns a leaked singleton — do NOT take ownership.
855 PairTally tally(benchmark::CreateDefaultDisplayReporter());
856 benchmark::RunSpecifiedBenchmarks(&tally);
857 benchmark::Shutdown();
858
859 if (const char* path = std::getenv("CHEATAH_BENCH_TABLE")) write_table(path, tally, settings);
860 return 0;
861}
