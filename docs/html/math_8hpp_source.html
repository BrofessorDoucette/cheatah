31/**
32 * Absolute value.
33 * @param x any signed value.
34 * @return |@p x|.
35 * @note O(1) time; no heap allocation.
36 * @test CheatahMath.BuiltinLikeOps
37 */
38template <typename T>
39T abs(T x) { return x < T{} ? -x : x; }

41/**
42 * Smallest of two-or-more values (variadic; the overloads chain to fold extra args).
43 * @param a,b the values to compare (`operator<` required).
44 * @return a reference to the minimum.
45 * @note O(n) in the argument count; no heap allocation.
46 * @test CheatahMath.BuiltinLikeOps
47 */
48template <typename T>
49const T& min(const T& a, const T& b) { return (b < a) ? b : a; }
50/**
51 * Smallest of three-or-more values (folds the extra args onto the two-argument overload).
52 * @param a,b the first two values.
53 * @param rest the remaining values (`operator<` required).
54 * @return a reference to the minimum.
55 * @test CheatahMath.BuiltinLikeOps
56 */
57template <typename T, typename... Rest>
58const T& min(const T& a, const T& b, const Rest&... rest) { return min(min(a, b), rest...); }
59
60/**
61 * Largest of two-or-more values (variadic; the overloads chain to fold extra args).
62 * @param a,b the values to compare (`operator<` required).
63 * @return a reference to the maximum.
64 * @note O(n) in the argument count; no heap allocation.
65 * @test CheatahMath.BuiltinLikeOps
66 */
67template <typename T>
68const T& max(const T& a, const T& b) { return (a < b) ? b : a; }
69/**
70 * Largest of three-or-more values (folds the extra args onto the two-argument overload).
71 * @param a,b the first two values.
72 * @param rest the remaining values (`operator<` required).
73 * @return a reference to the maximum.
74 * @test CheatahMath.BuiltinLikeOps
75 */
76template <typename T, typename... Rest>
77const T& max(const T& a, const T& b, const Rest&... rest) { return max(max(a, b), rest...); }

79/**
80 * Power.
81 * @param base the base.
82 * @param exp the exponent.
83 * @return @p base raised to @p exp (computed as `double`).
84 * @note O(1) time; no heap allocation.
85 * @test CheatahMath.BuiltinLikeOps
86 */
87template <typename Base, typename Exp>
88double pow(Base base, Exp exp) {
89 return std::pow(static_cast<double>(base), static_cast<double>(exp));
