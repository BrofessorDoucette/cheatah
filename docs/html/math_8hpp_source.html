3/**
4 * @file math.hpp
5 * @brief cheatah `math` — scalar math: Python's `math` module plus the
6 * `abs`/`min`/`max`/`pow` built-ins. `import math` to use it.
7 *
8 * Every function here is **pure and allocation-free** (operates on `double` /
9 * `long long` by value). Unit tests: `stdlib/tests/math_test.cpp`. The whole
10 * suite runs under AddressSanitizer (the `asan` preset) and Valgrind
11 * (`security/run-valgrind.sh`) on every QA-gate run.
12 *
13 * Doc convention (see also the other stdlib headers): each function's `@note`
14 * states its runtime complexity (Big-O) and whether it touches the heap, and
15 * `@test` links the unit test that covers it.
16 */

31/** Absolute value.
32 * @param x any signed value.
33 * @return |@p x|.
34 * @note O(1) time; no heap allocation. @test CheatahMath.BuiltinLikeOps */
35template <typename T>
36T abs(T x) { return x < T{} ? -x : x; }
37
38/** Smallest of two-or-more values (variadic; the overloads chain to fold extra args).
39 * @param a,b the values to compare (`operator<` required).
40 * @return a reference to the minimum.
41 * @note O(n) in the argument count; no heap allocation. @test CheatahMath.BuiltinLikeOps */
42template <typename T>
43const T& min(const T& a, const T& b) { return (b < a) ? b : a; }
44template <typename T, typename... Rest>
45const T& min(const T& a, const T& b, const Rest&... rest) { return min(min(a, b), rest...); }
46
47/** Largest of two-or-more values (variadic; the overloads chain to fold extra args).
48 * @param a,b the values to compare (`operator<` required).
49 * @return a reference to the maximum.
50 * @note O(n) in the argument count; no heap allocation. @test CheatahMath.BuiltinLikeOps */
51template <typename T>
52const T& max(const T& a, const T& b) { return (a < b) ? b : a; }
53template <typename T, typename... Rest>
54const T& max(const T& a, const T& b, const Rest&... rest) { return max(max(a, b), rest...); }
55
56/** Power.
57 * @param base the base.
58 * @param exp the exponent.
59 * @return @p base raised to @p exp (computed as `double`).
60 * @note O(1) time; no heap allocation. @test CheatahMath.BuiltinLikeOps */
61template <typename Base, typename Exp>
62double pow(Base base, Exp exp) {
63 return std::pow(static_cast<double>(base), static_cast<double>(exp));
64}
