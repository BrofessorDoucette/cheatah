3// cheatah::parsers::url — a from-scratch parser for the http(s) URL subset the requests module
4// speaks: `scheme://host[:port][/path][?query]`. No allocation beyond the component strings, no
5// regex, no dependencies. Userinfo (`user@`) and fragments (`#...`) are not supported — the first
6// is an obsolete security hazard in http URLs, the second is never sent to the server anyway.
7//
8// Laid out as a cheatah stdlib module: from .purr this is `import parsers.url.Parser as Parser`,
9// mirroring `import parsers.json.Parser as Parser` — each parsers submodule exposes a Parser.
10
11#include <string>
12#include <string_view>
13
14namespace cheatah::parsers::url {
15
16/**
17 * @brief One parsed http(s) URL. @c target is the HTTP request-target — the path plus the original
18 * query, always beginning with '/' (an empty path becomes "/").
19 */
20struct Url {
21 std::string scheme; ///< the lowercased scheme: "http" or "https".
22 std::string host; ///< the host (name or IP); never empty on success.
23 long long port = 0; ///< the explicit port, or the scheme default (80 for http, 443 for https).
24 std::string target; ///< the HTTP request-target "/path?query" (always begins with '/').
25};
26
27/**
28 * @brief The URL parser. Stateless and reusable; a class (not a free function) so the module
29 * surface is symmetric with parsers::json::Parser and imports the same way from cheatah.
30 */
31class Parser {
32public:
33 /**
34 * Parse @p text into @p out. Accepts `scheme://host[:port][/path][?query]` with scheme http or
35 * https (case-insensitive). Rejects empty hosts, non-numeric or out-of-range ports, userinfo,
36 * and fragments. On failure @p out is left unspecified.
37 *
38 * @param text the URL text to parse.
39 * @param out receives the parsed components on success.
40 * @return true iff @p text is a valid accepted URL.
41 * @complexity O(|text|)
42 * @alloc the component strings in @p out
43 * @test UrlParser.Components
44 */
45 [[nodiscard]] bool parse(std::string_view text, Url& out) const {
46 const std::size_t scheme_end = text.find("://");
47 if (scheme_end == std::string_view::npos || scheme_end == 0) {
48 return false;
49 }
50 out.scheme.clear();
51 for (const char ch : text.substr(0, scheme_end)) { // lowercase the scheme as we copy
52 out.scheme.push_back(ch >= 'A' && ch <= 'Z' ? static_cast<char>(ch - 'A' + 'a') : ch);
53 }
54 if (out.scheme != "http" && out.scheme != "https") {
55 return false;
56 }
57
58 std::string_view rest = text.substr(scheme_end + 3);
59 const std::size_t path_start = rest.find('/');
60 const std::size_t query_start = rest.find('?');
61 const std::size_t authority_end = std::min(path_start, query_start);
62 const std::string_view authority = rest.substr(0, authority_end);
63 if (authority.empty() || authority.find('@') != std::string_view::npos ||
64 rest.find('#') != std::string_view::npos) {
65 return false; // empty host, userinfo, and fragments are all rejected
66 }
67
68 const std::size_t colon = authority.rfind(':');
69 if (colon == std::string_view::npos) {
70 out.host = std::string(authority);
71 out.port = (out.scheme == "https") ? 443 : 80;
72 } else {
73 out.host = std::string(authority.substr(0, colon));
74 const std::string_view digits = authority.substr(colon + 1);
75 if (out.host.empty() || digits.empty() || digits.size() > 5) {
76 return false;
77 }
78 long long port = 0;
79 for (const char ch : digits) {
80 if (ch < '0' || ch > '9') {
81 return false;
82 }
83 port = port * 10 + (ch - '0');
84 }
85 if (port < 1 || port > 65535) {
86 return false;
87 }
88 out.port = port;
89 }
90
91 // The request-target: everything from the first '/' on (or "/" when the path is absent,
92 // including the bare-query form "host?x=1" -> "/?x=1").
93 if (path_start != std::string_view::npos) {
94 out.target = std::string(rest.substr(path_start));
95 } else if (query_start != std::string_view::npos) {
96 out.target = "/" + std::string(rest.substr(query_start));
97 } else {
98 out.target = "/";
99 }
100 return true;
101 }
102};
103
104} // namespace cheatah::parsers::url
