5/**
6 * @file xml.hpp
7 * @brief `parsers.xml` — a small, tolerant, from-scratch XML reader. `import parsers.xml`.
8 *
9 * A submodule of `parsers` (alongside `parsers.json`, `parsers.url`, `parsers.html`). It
10 * parses an XML document into a **slab DOM**: every element and text run is a `Node` stored
11 * in one flat vector inside the `Document`, and the tree is navigated by **integer node id**
12 * — never by pointer. That keeps the whole surface value-semantic and safe to drive from
13 * cheatah (ids + lists, no owning handles to leak): a `.purr` program parses once and walks
14 * the tree with `find` / `findall` / `iter` and `attr` / `text`.
15 *
16 * It exists to feed cheatah's own tooling (e.g. reading Doxygen XML), so it targets the
17 * pragmatic subset real tools emit: elements, attributes, text, character references
18 * (`&lt; &gt; &amp; &quot; &apos; &#NN; &#xNN;`), `<!-- comments -->`, `<![CDATA[…]]>`, the
19 * `<?xml …?>` prolog and processing instructions, and `<!DOCTYPE …>`. Parsing is **iterative**
20 * (an explicit open-element stack — no recursion, so no stack overflow at any nesting depth)
21 * and **lenient**: malformed markup never throws; unterminated constructs consume to
22 * end-of-input and a mismatched close tag is tolerated.
23 */
24#include <string>
25#include <string_view>
26#include <vector>
27
28namespace cheatah::parsers::xml {
29
30/// One attribute of an element: `name="value"` (value already reference-decoded).
31struct Attr {
32 std::string name; ///< the attribute name.
33 std::string value; ///< the decoded attribute value.
34};
35
36/// One DOM node: an **element** (`is_element` true — has @ref tag, @ref attrs, @ref children)
37/// or a **text** run (`is_element` false — @ref text holds the decoded character data). Child
38/// order (elements and text interleaved) is preserved via @ref children, which lists child
39/// node **ids** into the owning @ref Document.
40struct Node {
41 bool is_element = false; ///< true: an element; false: a text run.
42 std::string tag; ///< element tag name (empty for text).
43 std::string text; ///< decoded text (for a text node; empty for elements).
44 std::vector<Attr> attrs; ///< element attributes (empty for text).
45 std::vector<int> children; ///< ids of child nodes, in document order.
46};
47
48/// A parsed document: the node slab plus the id of the synthetic **root** (the root's
49/// @ref Node::children are the document's top-level nodes, so a prolog/comment before the
50/// root element is kept as siblings). Navigate everything by id through the functions below.
51struct Document {
52 std::vector<Node> nodes; ///< every node; index == its id.
53 int root = 0; ///< id of the root container node.
54};
55
56/**
57 * Parse @p xml into a @ref Document (slab DOM). Never throws.
58 *
59 * The returned document always has at least the root node (`root`). Malformed input is
60 * tolerated: unterminated tags/comments/CDATA consume to end-of-input, and a `</x>` with no
61 * matching open element is ignored rather than aborting the parse.
62 * @param xml the document text.
63 * @return the parsed document (root-only for empty/blank input).
64 * @complexity O(n) in the length of @p xml for well-formed input; a run of `&` with no `;` after it, or repeated unmatched end tags under deep nesting, degrade to O(n²).
65 * @alloc the node slab and its strings on the heap (owned by the returned Document).
66 * @test ParsersXml.ParsesElementsAttrsAndText
67 * @test ParsersXml.LenientOnMalformedInput
68 */
69Document parse(std::string_view xml);
70
71/**
72 * The document root's node id (its children are the top-level nodes).
73 * @param doc the document.
74 * @return the root node id (its children are the top-level nodes).
75 * @complexity O(1).
76 * @alloc none.
77 * @test ParsersXml.IgnorableVsSignificantWhitespace
78 */
79int root(const Document& doc);
80
81/**
82 * Whether @p id is a valid element node in @p doc.
83 * @param doc the document.
84 * @param id the node id to test.
85 * @return true iff @p id is a valid element node.
86 * @complexity O(1).
87 * @alloc none.
88 * @test ParsersXml.OutOfRangeAndTypeMismatchedIdsAreSafe
89 */
90bool is_element(const Document& doc, int id);
91
92/**
93 * The tag name of element @p id, or "" if @p id is not an element.
94 * @param doc the document.
95 * @param id the element node id.
96 * @return the tag name, or "" if @p id is not an element.
97 * @complexity O(1).
98 * @alloc one result string.
99 * @test ParsersXml.ParsesElementsAttrsAndText
100 */
101std::string tag(const Document& doc, int id);
102
103/**
104 * Value of attribute @p name on element @p id, or "" if absent (or @p id is not an element).
105 * @param doc the document.
106 * @param id the element node id.
107 * @param name the attribute name.
108 * @return the attribute value, or "" if absent.
109 * @complexity O(k) in the attribute count of @p id.
110 * @alloc one result string.
111 * @test ParsersXml.AttributeQuotingForms
112 */
113std::string attr(const Document& doc, int id, std::string_view name);
114
115/**
116 * Whether element @p id carries an attribute named @p name.
117 * @param doc the document.
118 * @param id the element node id.
119 * @param name the attribute name.
120 * @return whether @p id carries attribute @p name.
121 * @complexity O(k) in the attribute count of @p id.
122 * @alloc none.
123 * @test ParsersXml.ParsesElementsAttrsAndText
124 */
125bool has_attr(const Document& doc, int id, std::string_view name);
126
127/**
128 * The concatenated text of node @p id: for a text node, its own text; for an element, all
129 * descendant text in document order (like an XML `.textContent`).
130 * @param doc the document.
131 * @param id the node id.
132 * @return the concatenated descendant text.
133 * @complexity O(m) in the number of descendants of @p id.
134 * @alloc the result string and a transient walk stack.
135 * @test ParsersXml.NestedTextIsCollectedRecursively
136 */
137std::string text(const Document& doc, int id);
138
139/**
140 * The child node ids of @p id, in document order (elements and text).
141 * @param doc the document.
142 * @param id the node id.
143 * @return the child node ids in document order.
144 * @complexity O(c) in the child count of @p id (the id list is copied).
145 * @alloc the id list.
146 * @test ParsersXml.FindFindallIterChildren
147 */
148std::vector<int> children(const Document& doc, int id);
149
150/**
151 * The id of the **first child element** of @p id whose tag equals @p tag, or -1 if none.
152 * (Direct children only — not descendants; use @ref iter for the whole subtree.)
153 * @param doc the document.
154 * @param id the parent node id.
155 * @param tag the child tag to match.
156 * @return the first matching child element id, or -1.
157 * @complexity O(c) in the direct-child count of @p id.
158 * @alloc none.
159 * @test ParsersXml.FindFindallIterChildren
160 */
161int find(const Document& doc, int id, std::string_view tag);
162
163/**
164 * The ids of **all direct child elements** of @p id whose tag equals @p tag, in order.
165 * @param doc the document.
166 * @param id the parent node id.
167 * @param tag the child tag to match.
168 * @return the matching direct-child element ids.
169 * @complexity O(c) in the direct-child count of @p id.
170 * @alloc the result id list.
171 * @test ParsersXml.FindFindallIterChildren
172 */
173std::vector<int> findall(const Document& doc, int id, std::string_view tag);
174
175/**
176 * The ids of **every element in the subtree** rooted at @p id (including @p id itself) whose
177 * tag equals @p tag, in document order — the analogue of an XML tree `.iter(tag)`.
178 * @param doc the document.
179 * @param id the subtree root node id.
180 * @param tag the tag to match.
181 * @return the matching element ids in the subtree, in document order.
182 * @complexity O(m) in the subtree size of @p id.
183 * @alloc the result id list (and a transient walk stack).
184 * @test ParsersXml.FindFindallIterChildren
185 */
186std::vector<int> iter(const Document& doc, int id, std::string_view tag);

188} // namespace cheatah::parsers::xml
