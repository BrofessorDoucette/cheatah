5/**
6 * @file tls.hpp
7 * @brief cheatah `tls` — a from-scratch TLS 1.3 CLIENT (RFC 8446). `import tls` to use it.
8 * Built ENTIRELY on the cheatah crypto modules: `x25519` key exchange, the `aead`
9 * ChaCha20-Poly1305 record cipher, and hashlib's HKDF key schedule. No OpenSSL.
10 *
11 * Scope (v1): TLS 1.3 only, cipher suite TLS_CHACHA20_POLY1305_SHA256, X25519 key share,
12 * SNI. The handshake transcript is fully verified (server Finished MAC), and the server's
13 * CertificateVerify signature is checked for the common leaf-certificate key types: Ed25519
14 * (cheatah's `ed25519`), ECDSA P-256 (`p256`) and P-384 (`p384`), and RSA via
15 * rsa_pss_rsae_sha256 (`rsa_verify.hpp`). Servers using any OTHER certificate algorithm are
16 * REFUSED with a clear error rather than silently left unauthenticated — no unverified
17 * connections. The presented chain then gets full X.509 path validation (hostname, validity
18 * window, signatures up to a trusted CA — see x509.hpp) unless explicitly opted out.
19 *
20 * Sessions ride an already-connected TCP fd (cheatah `socket`). The cheatah-facing API is the
21 * owning `tls::Conn` guard, created by `tls.open(fd, server_name)`: it sends close_notify and
22 * erases the session automatically when it goes out of scope (so a cheatah program cannot leak a
23 * session). The underlying fd is NOT owned by the session — close it with `socket.close()` (or
24 * guard it with a `socket::Conn`). The flat handle-based calls live in tls_lowlevel.hpp (C++ only).
25 */
26#include <string>
27#include <string_view>
28
29namespace cheatah::tls {

31// The low-level, handle-based API (client_connect / send / recv / close / shutdown, keyed by an
32// integer session id) is C++-only and lives in tls_lowlevel.hpp. It is intentionally NOT part of
33// this cheatah-facing header: a cheatah program cannot reach it, so it cannot leak a session — it
34// uses the owning `tls::Conn` guard + `tls.open()` below, which release automatically at scope
35// exit. `tls::Conn` is implemented on top of that low-level API.

49// ---- owning RAII session (the `with`-friendly, leak-proof API) ----
50
51/**
52 * @brief An owning TLS 1.3 client session — closes (and erases) itself on destruction.
53 *
54 * The RAII counterpart to the handle-based calls above. A `Conn` owns one session; when it
55 * is destroyed (scope exit out of a `with` body, including via exception) or explicitly
56 * close()d, the session is torn down and removed from the module's session table, so
57 * `with tls.open(sock.fd(), host) as conn { … }` cannot leak the session. Move-only: the
58 * copy operations are deleted and a moved-from `Conn` is left closed. The underlying TCP fd
59 * is NOT owned here (tls rides a caller-owned socket) — guard it with a socket::Conn.
60 */
61class Conn {
62public:
63 /**
64 * Construct a closed session (owns nothing).
65 * @complexity O(1).
66 * @alloc none.
67 * @test CheatahTls.ConnDefaultIsClosed
68 */
69 Conn() = default;
70 /**
71 * Adopt an existing session handle (e.g. from client_connect()); the `Conn` now owns it.
72 * @param session a session handle to take ownership of (<= 0 for a closed session).
73 * @complexity O(1).
74 * @alloc none.
75 * @test TlsSys.ConnGuardRoundTrip
76 */
77 explicit Conn(long long session) : session_(session) {}
78 Conn(const Conn&) = delete;
79 Conn& operator=(const Conn&) = delete;
80 /**
81 * Move-construct, taking over @p other's session (the moved-from `Conn` becomes closed).
82 * @param other the session to move from.
83 * @complexity O(1).
84 * @alloc none.
85 * @test TlsSys.ConnGuardRoundTrip
86 */
87 Conn(Conn&& other) noexcept : session_(other.session_) { other.session_ = 0; }
88 /**
89 * Move-assign, closing this session first, then taking over @p other's (which becomes closed).
90 * @param other the session to move from.
91 * @return reference to this session.
92 * @complexity O(1).
93 * @alloc none.
94 * @test TlsSys.ConnGuardRoundTrip
95 */
96 Conn& operator=(Conn&& other) noexcept;
97 /**
98 * Send close_notify and forget the session if still open.
99 * @complexity O(log n) lookup + a close_notify write.
100 * @alloc a small close_notify record (when still open).
101 * @test TlsSys.ConnGuardRoundTrip
102 */
103 ~Conn();
104
105 /**
106 * Is a session open?
107 * @return true iff this owns an open session.
108 * @complexity O(1).
109 * @alloc none.
110 * @test CheatahTls.ConnDefaultIsClosed
111 */
112 bool is_open() const { return session_ > 0; }
113 /**
114 * The raw session handle (for the low-level calls).
115 * @return the owned handle, or 0 when closed.
116 * @complexity O(1).
117 * @alloc none.
118 * @test TlsSys.ConnGuardRoundTrip
119 */
120 long long id() const { return session_; }
121 /**
122 * Encrypt and send @p data as TLS application data (see the free send()).
123 * @param data plaintext to send.
124 * @return 0 on success, -1 on error.
125 * @complexity O(|data|).
126 * @alloc the ciphertext record(s).
127 * @test TlsSys.ConnGuardRoundTrip
128 */
129 long long send(const std::string& data);
130 /**
131 * Receive and decrypt up to @p bufsize bytes of application data (see the free recv()).
132 * @param bufsize maximum plaintext bytes to return.
133 * @return the plaintext, or "" on clean close/EOF/error.
134 * @complexity O(record size).
135 * @alloc the returned plaintext.
136 * @test TlsSys.ConnGuardRoundTrip
137 */
138 std::string recv(long long bufsize);
139 /**
140 * Wake a reader blocked in recv() WITHOUT closing the session (see the free shutdown()).
141 * @return 0 on success, -1 on error.
142 * @complexity O(log n) lookup + one syscall.
143 * @alloc none.
144 * @test TlsSys.ConnGuardRoundTrip
145 */
146 long long shutdown();
147 /**
148 * Close the session now (idempotent — the destructor will not close it again).
149 * @return 0 on success, -1 if already closed / unknown.
150 * @complexity O(log n) lookup + a close_notify write.
151 * @alloc a small close_notify record (when still open).
152 * @test TlsSys.ConnGuardRoundTrip
153 */
154 long long close();
155
156private:
157 long long session_ = 0;
158};

160/**
161 * Run the TLS 1.3 client handshake over connected fd @p fd and return an owning Conn (the
162 * RAII, `with`-friendly form of client_connect()). By default the server is AUTHENTICATED:
163 * the certificate chain is built to a trusted CA, the hostname is matched against the leaf's
164 * subjectAltName, and the validity dates are checked — so the connection resists an active MITM.
165 * @param fd a CONNECTED TCP socket (e.g. socket::Conn::fd()).
166 * @param server_name the hostname (SNI + certificate SAN match).
167 * @param insecure when true, skip chain/hostname/expiry validation (leaf-key possession only) —
168 * for a pinned/controlled peer where identity is established out of band. Default false.
169 * @param ca_file a PEM CA bundle to trust instead of the system store (empty = system default).
170 * @return an owning Conn; on handshake or validation failure its is_open() is false (see last_error()).
171 * @complexity one network round trip + O(handshake bytes) crypto (+ a one-time CA-store parse).
172 * @alloc the session state.
173 * @test TlsSys.ConnGuardRoundTrip
174 * @systest TlsSys.HttpsGet
175 */
176Conn open(long long fd, const std::string& server_name, bool insecure = false,
177 const std::string& ca_file = "");
178
179/**
180 * Run the TLS 1.3 SERVER handshake over an accepted TCP fd and return an owning Conn — the
181 * `with`-friendly server counterpart to open(). We present @p cert_pem and prove possession of
182 * its key by signing the handshake, so a client that validates the certificate gets an
183 * authenticated, encrypted channel with **no OpenSSL** anywhere.
184 *
185 * The server certificate must be **Ed25519** (the leaf's key and the private key both Ed25519);
186 * this is the from-scratch signing path cheatah owns end to end. Both suites (ChaCha20-Poly1305
187 * and AES-128-GCM) and X25519 key exchange are supported; the client picks the suite.
188 * @param fd a CONNECTED TCP socket from socket::accept()/Listener (e.g. one client of a server loop).
189 * @param cert_pem the server's leaf certificate, PEM (`-----BEGIN CERTIFICATE-----`).
190 * @param key_pem the matching PKCS#8 Ed25519 private key, PEM (`-----BEGIN PRIVATE KEY-----`).
191 * @return an owning Conn; on handshake failure its is_open() is false (see last_error()).
192 * @complexity one network round trip + O(handshake bytes) crypto.
193 * @alloc the session state.
194 * @systest TlsSys.ServerHandshakeAgainstOpenssl
195 */
196Conn accept(long long fd, const std::string& cert_pem, const std::string& key_pem);
