3// cheatah::parsers::json — the JSON string token (decoded, un-escaped). See json.hpp.
4
5#include <string>
6#include <string_view>
7#include <type_traits>
8#include <utility>
9
10namespace cheatah::parsers::json {
11
12// String holds its characters one of two ways, chosen by the storage type S, which is DEDUCED
13// from the constructor argument (see the deduction guides below):
14//
15// - String<std::string_view> — a non-owning VIEW. Zero-copy. This is what a const char* /
16// string literal or an existing buffer becomes. The viewed characters must outlive the
17// String; string literals live in the binary for the whole program, so String{"x"} can
18// refer to that const data as its own value, safely, forever.
19// - String<std::string> — OWNS its characters (an rvalue std::string moved in).
20//
21// A const char* is ALWAYS wrapped in a std::string_view and is NEVER silently copied into a
22// std::string. S is constrained to exactly those two storage types.
23// Valid backing storage for a String: either owning (std::string) or non-owning (std::string_view).
24template <typename S>
25concept StringStorage = std::is_same_v<S, std::string> || std::is_same_v<S, std::string_view>;
26
27/**
28 * @brief A decoded (un-escaped) JSON string token, storing its characters either owned
29 * (String<std::string>) or as a zero-copy view (String<std::string_view>), chosen by
30 * the storage type @p S via the deduction guides below.
31 * @tparam S the backing storage: std::string (owning) or std::string_view (viewing).
32 */
33template <StringStorage S>
34class String {
35private:
36 S value_;
37
38public:
39 /**
40 * Construct from the backing storage. Owning S=std::string moves the rvalue in; viewing
41 * S=std::string_view just stores the (cheap) view — the characters are never copied here.
42 * @param value the string storage (moved in).
43 */
44 String(S value) : value_(std::move(value)) {}
45 ~String() = default;
46
47 /**
48 * Read the characters uniformly as a view, whether owning or viewing (no setter).
49 * @return a std::string_view over the stored characters.
50 */
51 [[nodiscard]] std::string_view value() const noexcept { return value_; }
52};
53
54// Deduction guides — route each argument to the right storage. A const char* and a
55// std::string_view become VIEWS (never a std::string); an rvalue std::string is OWNED. These
56// user-defined guides are preferred over the implicit one, so String{"lit"} deduces
57// String<std::string_view>, not String<const char*>.
58String(const char*) -> String<std::string_view>;
59String(std::string_view) -> String<std::string_view>;
60String(std::string) -> String<std::string>;
61
62} // namespace cheatah::parsers::json
