25// A from-scratch TLS 1.3 client (RFC 8446), cipher suite TLS_CHACHA20_POLY1305_SHA256 only.
26// The implementation walks the RFC top to bottom: record layer, transcript hash, the HKDF
27// key schedule, then the handshake state machine. Every secret derives through hashlib's
28// HKDF; every record seals/opens through the aead module; the ephemeral key is x25519.

39// ---- hex <-> bytes: the ONE canonical implementation lives in hashlib -----------
40// The crypto modules speak hex for keys; tls feeds from_hex only valid, even-length lowercase
41// x25519 hex, so the canonical from_hex's odd-length/non-hex throws are never reached here.
42using hashlib::to_hex; // bytes -> lowercase hex (string_view / (uint8_t*, n) overloads).
43using hashlib::from_hex; // hex -> bytes (throws on odd length / non-hex).

45// 16/24-bit big-endian helpers for the wire format.
46void put16(std::string& out, unsigned v) {
47 out.push_back(static_cast<char>(v >> 8));
48 out.push_back(static_cast<char>(v));
49}
50void put24(std::string& out, unsigned v) {
51 out.push_back(static_cast<char>(v >> 16));
52 out.push_back(static_cast<char>(v >> 8));
53 out.push_back(static_cast<char>(v));
54}
55unsigned get16(std::string_view s, std::size_t i) {
56 return (static_cast<unsigned char>(s[i]) << 8) | static_cast<unsigned char>(s[i + 1]);
57}
58unsigned get24(std::string_view s, std::size_t i) {
59 return (static_cast<unsigned char>(s[i]) << 16) | (static_cast<unsigned char>(s[i + 1]) << 8) |
60 static_cast<unsigned char>(s[i + 2]);
61}

67// HKDF-Expand-Label(secret, label, context, length) with the "tls13 " prefix.
68std::string expand_label_impl(std::string_view secret, std::string_view label,
69 std::string_view context, unsigned length) {
70 std::string info;
71 put16(info, length);
72 info.push_back(static_cast<char>(6 + label.size()));
73 info += "tls13 ";
74 info += label;
75 info.push_back(static_cast<char>(context.size()));
76 info += context;
77 return hashlib::hkdf_expand(secret, info, length);
78}

83 return expand_label_impl(secret, label, hashlib::sha256_digest(transcript), 32);

88// One traffic direction: AEAD key + iv + record sequence number.
89struct Keys {
90 std::string key_hex; // AEAD key (hex): 32 bytes for ChaCha20-Poly1305, 16 for AES-128-GCM
91 std::string iv; // 12-byte raw iv; per-record nonce = iv XOR seq
92 std::uint64_t seq = 0;
93 bool aes_gcm = false; // record cipher: false = ChaCha20-Poly1305 (0x1303), true = AES-128-GCM (0x1301)
94};
95
96// Both suites use the SHA-256 key schedule and a 12-byte iv; only the record-key length differs
97// (AES-128 = 16 bytes, ChaCha20 = 32). aes_gcm selects which record cipher seal/open_record use.
98Keys traffic_keys(std::string_view secret, bool aes_gcm) {
99 Keys k;
100 k.aes_gcm = aes_gcm;
101 k.key_hex = to_hex(expand_label_impl(secret, "key", "", aes_gcm ? 16 : 32));
102 k.iv = expand_label_impl(secret, "iv", "", 12);
103 return k;
104}
105
106// The per-record nonce: the 12-byte iv with the 8-byte big-endian sequence XORed into its tail.
107std::string nonce_hex(const Keys& k) {
108 std::string n = k.iv;
109 for (int i = 0; i < 8; ++i) {
110 n[4 + i] = static_cast<char>(static_cast<unsigned char>(n[4 + i]) ^
111 static_cast<unsigned char>(k.seq >> (8 * (7 - i))));
112 }
113 return to_hex(n);
114}

133// Read exactly one TLS record (header + payload) from the socket into (type, payload).
134// Blocking, bounded by the fd's socket timeout. False on EOF/short read.
135bool read_record(long long fd, std::string& buffer, unsigned& type, std::string& payload) {
136 while (buffer.size() < 5) {
137 const std::string chunk = sock::recv(fd, 16384);
138 if (chunk.empty()) return false;
139 buffer += chunk;
140 }
141 type = static_cast<unsigned char>(buffer[0]);
142 const unsigned len = get16(buffer, 3);
143 if (len > 16384 + 256) { // RFC bound + AEAD overhead: anything bigger is malformed
144 return false;
145 }
146 while (buffer.size() < 5 + len) {
147 const std::string chunk = sock::recv(fd, 16384);
148 if (chunk.empty()) return false;
149 buffer += chunk;
150 }
151 payload = buffer.substr(5, len);
152 buffer.erase(0, 5 + len);
153 return true;
154}

156bool write_record(long long fd, unsigned type, std::string_view payload) {
157 std::string rec;
158 rec.push_back(static_cast<char>(type));
159 put16(rec, 0x0303); // legacy_record_version
160 put16(rec, static_cast<unsigned>(payload.size()));
161 rec += payload;
162 return sock::sendall(fd, rec) == 0;
163}

165// Seal one application_data record (RFC 8446 §5.2): inner plaintext = content || content_type,
166// AAD = the record header, then ChaCha20-Poly1305.
167bool seal_record(long long fd, Keys& k, unsigned inner_type, std::string_view content) {
168 std::string inner(content);
169 inner.push_back(static_cast<char>(inner_type));
170 std::string aad;
171 aad.push_back(23);
172 put16(aad, 0x0303);
173 put16(aad, static_cast<unsigned>(inner.size() + 16));
174 const std::string ct = k.aes_gcm
175 ? aead::aes128gcm_encrypt(k.key_hex, nonce_hex(k), aad, inner)
176 : aead::chacha20poly1305_encrypt(k.key_hex, nonce_hex(k), aad, inner);
177 ++k.seq;
178 if (ct.empty()) return false;
179 return sock::sendall(fd, aad + ct) == 0;
180}
181
182// Open one encrypted record: returns the inner content and type, false on AEAD failure.
183bool open_record(Keys& k, std::string_view payload, unsigned& inner_type, std::string& content) {
184 std::string aad;
185 aad.push_back(23);
186 put16(aad, 0x0303);
187 put16(aad, static_cast<unsigned>(payload.size()));
188 std::string inner = k.aes_gcm
189 ? aead::aes128gcm_decrypt(k.key_hex, nonce_hex(k), aad, payload)
190 : aead::chacha20poly1305_decrypt(k.key_hex, nonce_hex(k), aad, payload);
191 ++k.seq;
192 if (inner.empty() && payload.size() > 16) return false; // tag mismatch (or empty record)
193 while (!inner.empty() && inner.back() == '\0') inner.pop_back(); // strip padding
194 if (inner.empty()) return false; // a record must carry a content type
195 inner_type = static_cast<unsigned char>(inner.back());
196 content = inner.substr(0, inner.size() - 1);
197 return true;
198}

202std::string random_bytes(std::size_t n) {
203 std::string out(n, '\0');
204 std::size_t got = 0;
205 while (got < n) {
206 const ssize_t r = getrandom(out.data() + got, n - got, 0);
207 if (r <= 0) return ""; // the kernel CSPRNG failing is fatal for key material
208 got += static_cast<std::size_t>(r);
209 }
210 return out;
211}

213// Build the ClientHello handshake MESSAGE (no record header). Fills `client_hello_random`.
214std::string build_client_hello(const std::string& server_name, std::string_view pub_raw) {
215 std::string body;
216 put16(body, 0x0303); // legacy_version
217 body += random_bytes(32); // random
218 body.push_back(32); // legacy_session_id (32 bytes, middlebox compatibility)
219 body += random_bytes(32);
220 put16(body, 4); // cipher_suites: two suites (4 bytes)
221 put16(body, 0x1303); // TLS_CHACHA20_POLY1305_SHA256 (preferred)
222 put16(body, 0x1301); // TLS_AES_128_GCM_SHA256 (some hosts only do AES-GCM)
223 body.push_back(1); // legacy_compression_methods
224 body.push_back(0); // null
225
226 std::string ext;
227 { // server_name (0)
228 std::string names;
229 names.push_back(0); // host_name
230 put16(names, static_cast<unsigned>(server_name.size()));
231 names += server_name;
232 std::string sni;
233 put16(sni, static_cast<unsigned>(names.size()));
234 sni += names;
235 put16(ext, 0);
236 put16(ext, static_cast<unsigned>(sni.size()));
237 ext += sni;
238 }
239 { // supported_groups (10): x25519 only
240 std::string g;
241 put16(g, 2);
242 put16(g, 0x001d);
243 put16(ext, 10);
244 put16(ext, static_cast<unsigned>(g.size()));
245 ext += g;
246 }
247 { // signature_algorithms (13): ed25519 (verifiable) + the common ones so real servers
248 // complete the handshake far enough for our explicit refusal to be diagnosable
249 std::string a;
250 put16(a, 6);
251 put16(a, 0x0807); // ed25519
252 put16(a, 0x0804); // rsa_pss_rsae_sha256 (verified — see rsa_verify.hpp)
253 put16(a, 0x0403); // ecdsa_secp256r1_sha256 (verified — see p256)
254 put16(ext, 13);
255 put16(ext, static_cast<unsigned>(a.size()));
256 ext += a;
257 }
258 { // supported_versions (43): TLS 1.3
259 std::string v;
260 v.push_back(2);
261 put16(v, 0x0304);
262 put16(ext, 43);
263 put16(ext, static_cast<unsigned>(v.size()));
264 ext += v;
265 }
266 { // key_share (51): our X25519 public key
267 std::string entry;
268 put16(entry, 0x001d);
269 put16(entry, 32);
270 entry += pub_raw;
271 std::string ks;
272 put16(ks, static_cast<unsigned>(entry.size()));
273 ks += entry;
274 put16(ext, 51);
275 put16(ext, static_cast<unsigned>(ks.size()));
276 ext += ks;
277 }
278 put16(body, static_cast<unsigned>(ext.size()));
279 body += ext;
280
281 std::string msg;
282 msg.push_back(1); // client_hello
283 put24(msg, static_cast<unsigned>(body.size()));
284 msg += body;
285 return msg;
286}

288// Human-readable TLS alert (RFC 8446 §6) from the 2 alert bytes — so a handshake refusal names its
289// cause (e.g. 40 handshake_failure = no common cipher/group; 70 protocol_version = no TLS 1.3).
290std::string alert_text(std::string_view p) {
291 if (p.size() < 2) return "(empty alert)";
292 const unsigned lvl = static_cast<unsigned char>(p[0]);
293 const unsigned d = static_cast<unsigned char>(p[1]);
294 const char* name = "unknown";
295 switch (d) {
296 case 0: name = "close_notify"; break;
297 case 10: name = "unexpected_message"; break;
298 case 20: name = "bad_record_mac"; break;
299 case 22: name = "record_overflow"; break;
300 case 40: name = "handshake_failure"; break;
301 case 42: name = "bad_certificate"; break;
302 case 43: name = "unsupported_certificate"; break;
303 case 47: name = "illegal_parameter"; break;
304 case 48: name = "unknown_ca"; break;
305 case 49: name = "access_denied"; break;
306 case 50: name = "decode_error"; break;
307 case 51: name = "decrypt_error"; break;
308 case 70: name = "protocol_version"; break;
309 case 71: name = "insufficient_security"; break;
310 case 80: name = "internal_error"; break;
311 case 109: name = "missing_extension"; break;
312 case 110: name = "unsupported_extension"; break;
313 case 112: name = "unrecognized_name"; break;
314 case 116: name = "certificate_required"; break;
315 case 120: name = "no_application_protocol"; break;
316 default: break;
317 }
318 return "alert level=" + std::to_string(lvl) + " description=" + std::to_string(d) + " (" + name + ")";
319}

321// Parse ServerHello: confirm TLS 1.3 + one of our suites, extract the server's X25519 key share and
322// the CHOSEN cipher suite (0x1303 ChaCha20-Poly1305 or 0x1301 AES-128-GCM).
323bool parse_server_hello(std::string_view msg, std::string& server_pub_raw, unsigned& chosen_suite) {
324 if (msg.size() < 4 || msg[0] != 2) return false; // server_hello
325 std::string_view b = msg.substr(4);
326 if (b.size() < 2 + 32 + 1) return false;
327 std::size_t i = 2 + 32; // legacy_version + random
328 const unsigned sid_len = static_cast<unsigned char>(b[i]);
329 i += 1 + sid_len;
330 if (b.size() < i + 4) return false;
331 const unsigned suite = get16(b, i);
332 if (suite != 0x1303 && suite != 0x1301) return false; // must be one of the suites we offered
333 chosen_suite = suite;
334 i += 2 + 1; // suite + legacy_compression
335 if (b.size() < i + 2) return false;
336 const unsigned ext_len = get16(b, i);
337 i += 2;
338 const std::size_t ext_end = i + ext_len;
339 bool saw_13 = false;
340 while (i + 4 <= ext_end && ext_end <= b.size()) {
341 const unsigned etype = get16(b, i);
342 const unsigned elen = get16(b, i + 2);
343 i += 4;
344 if (i + elen > b.size()) return false;
345 if (etype == 43 && elen == 2 && get16(b, i) == 0x0304) saw_13 = true;
346 if (etype == 51 && elen >= 4 && get16(b, i) == 0x001d && get16(b, i + 2) == 32 &&
347 elen == 4 + 32) {
348 server_pub_raw = std::string(b.substr(i + 4, 32));
349 }
350 i += elen;
351 }
352 return saw_13 && server_pub_raw.size() == 32;
353}

355// Extract the Ed25519 public key from the leaf certificate's SubjectPublicKeyInfo: the DER
356// pattern 30 05 06 03 2B 65 70 (AlgorithmIdentifier { id-Ed25519 }) followed by
357// 03 21 00 <32-byte key> (BIT STRING). Returns "" when the cert key is not Ed25519.
358std::string ed25519_spki_key(std::string_view cert_der) {
359 static const unsigned char kPat[] = {0x30, 0x05, 0x06, 0x03, 0x2B, 0x65, 0x70,
360 0x03, 0x21, 0x00};
361 for (std::size_t i = 0; i + sizeof kPat + 32 <= cert_der.size(); ++i) {
362 if (std::memcmp(cert_der.data() + i, kPat, sizeof kPat) == 0) {
363 return std::string(cert_der.substr(i + sizeof kPat, 32));
364 }
365 }
366 return ""; // LCOV_EXCL_LINE: only when an Ed25519 CertificateVerify names a non-Ed25519 leaf — a malformed peer we don't mirror
367}

378long long handshake(long long fd, const std::string& server_name, bool insecure,
379 const std::string& ca_file) {
380 t_error.clear();
381
382 // Ephemeral X25519 key pair.
383 const std::string priv_raw = random_bytes(32);
384 if (priv_raw.size() != 32) {
385 fail("tls: system random unavailable");
386 return -1;
387 }
388 const std::string priv_hex = to_hex(priv_raw);
389 const std::string pub_hex = x25519::x25519_base(priv_hex);
390 const std::string pub_raw = from_hex(pub_hex);
391
392 std::string transcript; // concatenated handshake MESSAGES (no record headers)
393 const std::string client_hello = build_client_hello(server_name, pub_raw);
394 transcript += client_hello;
395 if (!write_record(fd, 22, client_hello)) {
396 fail("tls: cannot send ClientHello");
397 return -1;
398 }
399
400 // ServerHello (plaintext record, type 22; tolerate ChangeCipherSpec compat records).
401 std::string buffer, payload;
402 unsigned rtype = 0;
403 std::string server_pub_raw;
404 for (;;) {
405 if (!read_record(fd, buffer, rtype, payload)) {
406 fail("tls: connection closed before ServerHello");
407 return -1;
408 }
409 if (rtype == 20) continue; // ChangeCipherSpec (compat) — ignored
410 if (rtype == 21) {
411 fail("tls: server sent an alert instead of ServerHello — " + alert_text(payload));
412 return -1;
413 }
414 if (rtype != 22) {
415 fail("tls: unexpected record before ServerHello");
416 return -1;
417 }
418 break;
419 }
420 unsigned chosen_suite = 0;
421 if (!parse_server_hello(payload, server_pub_raw, chosen_suite)) {
422 fail("tls: malformed ServerHello (or the server refused TLS 1.3 + our cipher suites)");
423 return -1;
424 }
425 const bool aes_gcm = (chosen_suite == 0x1301); // else 0x1303 ChaCha20-Poly1305
426 transcript += payload;
427
428 // Key schedule through the handshake secrets.
429 const std::string shared_hex = x25519::x25519(priv_hex, to_hex(server_pub_raw));
430 if (shared_hex.empty()) {
431 fail("tls: invalid server key share");
432 return -1;
433 }
434 const std::string zeros(32, '\0');
435 const std::string early = hashlib::hkdf_extract(std::string(), zeros);
436 const std::string derived = derive_secret_impl(early, "derived", "");
437 const std::string hs_secret = hashlib::hkdf_extract(derived, from_hex(shared_hex));
438 const std::string c_hs = derive_secret_impl(hs_secret, "c hs traffic", transcript);
439 const std::string s_hs = derive_secret_impl(hs_secret, "s hs traffic", transcript);
440 Keys client_keys = traffic_keys(c_hs, aes_gcm);
441 Keys server_keys = traffic_keys(s_hs, aes_gcm);
442
443 // Encrypted handshake flight: EncryptedExtensions, Certificate, CertificateVerify, Finished.
444 std::string handshake_bytes; // decrypted, possibly spanning records
445 std::string cert_der;
446 std::vector<std::string> cert_chain; // the full certificate chain (leaf first) for validation
447 bool verified_cert = false, server_finished = false;
448 std::string transcript_at_cv, transcript_at_finished;
449 while (!server_finished) {
450 if (!read_record(fd, buffer, rtype, payload)) {
451 fail("tls: connection closed during the handshake");
452 return -1;
453 }
454 if (rtype == 20) continue; // compat ChangeCipherSpec
455 if (rtype == 21) {
456 fail("tls: server alert during the handshake — " + alert_text(payload));
457 return -1;
458 }
459 if (rtype != 23) {
460 fail("tls: unexpected plaintext record during the encrypted handshake");
461 return -1;
462 }
463 unsigned inner_type = 0;
464 std::string content;
465 if (!open_record(server_keys, payload, inner_type, content)) {
466 fail("tls: handshake record failed authentication");
467 return -1;
468 }
469 if (inner_type == 21) {
470 fail("tls: server alert during the handshake — " + alert_text(content));
471 return -1;
472 }
473 if (inner_type != 22) {
474 fail("tls: unexpected inner record type during the handshake");
475 return -1;
476 }
477 handshake_bytes += content;
478
479 // Drain complete handshake messages from the reassembly buffer.
480 while (handshake_bytes.size() >= 4) {
481 const unsigned mtype = static_cast<unsigned char>(handshake_bytes[0]);
482 const unsigned mlen = get24(handshake_bytes, 1);
483 if (handshake_bytes.size() < 4 + mlen) break;
484 const std::string msg = handshake_bytes.substr(0, 4 + mlen);
485 handshake_bytes.erase(0, 4 + mlen);
486
487 if (mtype == 11 && msg.size() > 4 + 4 + 3 + 3) { // Certificate
488 // certificate_request_context (1 byte, empty) + the cert list. Walk every
489 // CertificateEntry (3-byte length | cert DER | 2-byte extensions | extensions)
490 // so the FULL chain is available for path validation, not just the leaf.
491 std::size_t i = 4 + 1 + 3; // header + context length byte + cert-list length
492 while (i + 3 <= msg.size()) {
493 const unsigned clen = get24(msg, i);
494 i += 3;
495 if (i + clen > msg.size()) break;
496 cert_chain.push_back(msg.substr(i, clen));
497 i += clen;
498 if (i + 2 > msg.size()) break;
499 i += 2 + get16(msg, i); // skip the per-certificate extensions
500 }
501 if (!cert_chain.empty()) cert_der = cert_chain[0]; // the leaf signs CertificateVerify
502 transcript_at_cv = transcript + msg; // transcript THROUGH Certificate
503 }
504 if (mtype == 15) { // CertificateVerify
505 // cppcheck-suppress stlcstrConstructor // (ptr,len) subview of msg — not a c_str() copy
506 const std::string_view body(msg.data() + 4, msg.size() - 4);
507 if (body.size() < 4) {
508 fail("tls: malformed CertificateVerify");
509 return -1;
510 }
511 const unsigned alg = get16(body, 0);
512 const unsigned sig_len = get16(body, 2);
513 if (body.size() < 4 + sig_len) {
514 fail("tls: malformed CertificateVerify");
515 return -1;
516 }
517 // The signed content (same for every algorithm): 64 spaces, the
518 // context string, a NUL, then the handshake transcript hash.
519 std::string signed_content(64, ' ');
520 signed_content += "TLS 1.3, server CertificateVerify";
521 signed_content.push_back('\0');
522 signed_content += hashlib::sha256_digest(transcript_at_cv);
523 const std::string sig(body.substr(4, sig_len));
524 if (alg == 0x0807) { // ed25519 (signs the message directly)
525 const std::string spki = ed25519_spki_key(cert_der);
526 if (spki.size() != 32) {
527 fail("tls: certificate key is not Ed25519");
528 return -1;
529 }
530 if (!ed25519::verify(to_hex(spki), signed_content, to_hex(sig))) {
531 fail("tls: server CertificateVerify signature is INVALID");
532 return -1;
533 }
534 } else if (alg == 0x0403) { // ecdsa_secp256r1_sha256 (signs SHA-256(content))
535 const std::string point = p256::spki_ec_point(cert_der);
536 if (point.size() != 64) {
537 fail("tls: certificate key is not P-256 EC");
538 return -1;
539 }
540 if (!p256::verify_der(point, hashlib::sha256_digest(signed_content), sig)) {
541 fail("tls: server CertificateVerify (ECDSA P-256) is INVALID");
542 return -1;
543 }
544 } else if (alg == 0x0804) { // rsa_pss_rsae_sha256 (RSA leaf certificate)
545 // RSA-PSS verifies the signed_content directly (it hashes with SHA-256 internally),
546 // using the RSA public key extracted from the leaf cert's SubjectPublicKeyInfo.
547 if (!rsa::verify_pss_sha256(cert_der, signed_content, sig)) {
548 fail("tls: server CertificateVerify (RSA-PSS SHA-256) is INVALID");
549 return -1;
550 }
551 } else { // LCOV_EXCL_LINE: reached only if a server sends a CertificateVerify whose algorithm ignores our advertised signature_algorithms — a non-conformant peer we don't mirror
552 fail("tls: server certificate uses an algorithm cheatah cannot verify yet " // LCOV_EXCL_LINE
553 "(Ed25519, ECDSA P-256, and RSA-PSS SHA-256 are supported) — refusing an "
554 "unauthenticated connection");
555 return -1;
556 }
557 verified_cert = true;
558 }
559 if (mtype == 20) { // Finished
560 const std::string finished_key = expand_label_impl(s_hs, "finished", "", 32);
561 const std::string expect =
562 hashlib::hmac_sha256(finished_key, hashlib::sha256_digest(transcript));
563 // cppcheck-suppress stlcstrConstructor // (ptr,len) subview of msg — not a c_str() copy
564 const std::string_view got(msg.data() + 4, msg.size() - 4);
565 // Constant-time MAC compare (parity with the AEAD tag check): always scan all
566 // 32 bytes of the secret `expect`, never early-exiting on a mismatching byte, so
567 // timing cannot reveal a partial match of the handshake MAC.
568 unsigned char diff = (got.size() == expect.size()) ? 0 : 1;
569 for (std::size_t i = 0; i < expect.size(); ++i) {
570 const unsigned char g =
571 i < got.size() ? static_cast<unsigned char>(got[i]) : 0;
572 diff |= g ^ static_cast<unsigned char>(expect[i]);
573 }
574 if (diff != 0) {
575 fail("tls: server Finished MAC mismatch — handshake transcript tampered");
576 return -1;
577 }
578 transcript_at_finished = transcript + msg;
579 server_finished = true;
580 }
581 transcript += msg;
582 }
583 }
584 if (!verified_cert) {
585 fail("tls: server never proved possession of its certificate key");
586 return -1;
587 }
588
589 // AUTHENTICATE THE SERVER'S IDENTITY (unless the caller opted out of verification): build the
590 // presented chain to a trusted CA, match the hostname against the leaf's SAN, and check the
591 // validity dates. Key possession alone (above) does not prove identity — this is what stops an
592 // active man-in-the-middle presenting any certificate.
593 if (!insecure) {
594 x509::TrustStore custom;
595 const x509::TrustStore* store = &default_trust();
596 if (!ca_file.empty()) {
597 custom = x509::load_trust(ca_file);
598 store = &custom;
599 }
600 std::string verr;
601 if (!x509::validate(cert_chain, server_name, *store, static_cast<long long>(std::time(nullptr)),
602 verr)) {
603 fail("tls: certificate validation failed — " + verr);
604 return -1;
605 }
606 }
607
608 // Application traffic secrets (transcript through server Finished), then OUR Finished
609 // (sent under the handshake keys, with the transcript through the server's Finished).
610 const std::string derived2 = derive_secret_impl(hs_secret, "derived", "");
611 const std::string master = hashlib::hkdf_extract(derived2, zeros);
612 const std::string c_ap = derive_secret_impl(master, "c ap traffic", transcript_at_finished);
613 const std::string s_ap = derive_secret_impl(master, "s ap traffic", transcript_at_finished);
614
615 const std::string c_finished_key = expand_label_impl(c_hs, "finished", "", 32);
616 const std::string verify =
617 hashlib::hmac_sha256(c_finished_key, hashlib::sha256_digest(transcript));
618 std::string fin_msg;
619 fin_msg.push_back(20);
620 put24(fin_msg, static_cast<unsigned>(verify.size()));
621 fin_msg += verify;
622 if (!seal_record(fd, client_keys, 22, fin_msg)) {
623 fail("tls: cannot send client Finished");
624 return -1;
625 }
626
627 Session s;
628 s.fd = fd;
629 s.client_keys = traffic_keys(c_ap, aes_gcm);
630 s.server_keys = traffic_keys(s_ap, aes_gcm);
631 s.read_buffer = std::move(buffer); // bytes already pulled off the socket stay with us
632
633 const std::lock_guard<std::mutex> lock(g_mutex);
634 const long long handle = g_next_handle++;
635 g_sessions[handle] = std::move(s);
636 return handle;
637}

647long long send(long long session, const std::string& data) {
648 t_error.clear();
649 // Lock ONLY for the map lookup (see recv): the socket write below runs without
650 // the global lock so concurrent sessions don't serialize on each other.
651 Session* sp = nullptr;
652 {
653 const std::lock_guard<std::mutex> lock(g_mutex);
654 const auto it = g_sessions.find(session);
655 if (it == g_sessions.end() || it->second.closed) {
656 fail("tls: unknown or closed session");
657 return -1;
658 }
659 sp = &it->second;
660 }
661 Session& s = *sp;
662 // Respect the 16 KiB record plaintext bound.
663 std::string_view rest = data;
664 while (!rest.empty()) {
665 const std::size_t n = std::min<std::size_t>(rest.size(), 16384);
666 if (!seal_record(s.fd, s.client_keys, 23, rest.substr(0, n))) {
667 fail("tls: send failed");
668 return -1;
669 }
670 rest.remove_prefix(n);
671 }
672 return 0;
673}

675std::string recv(long long session, long long bufsize) {
676 t_error.clear();
677 if (bufsize <= 0) return "";
678 // Guard ONLY the map lookup. The per-session buffers (read_buffer/app_pending)
679 // and socket are owned by this session's single reader thread, so the blocking
680 // record I/O below runs WITHOUT the global lock — otherwise one session's
681 // blocking recv would serialize (and at shutdown, starve) every other session's
682 // recv/send. A std::map node address is stable until that node is erased, and a
683 // session is erased only by its own owner (after this loop), so the pointer is
684 // valid for this call. (g_mutex still serializes find/insert/erase on the map.)
685 Session* sp = nullptr;
686 {
687 const std::lock_guard<std::mutex> lock(g_mutex);
688 const auto it = g_sessions.find(session);
689 if (it == g_sessions.end()) {
690 fail("tls: unknown session");
691 return "";
692 }
693 sp = &it->second;
694 }
695 Session& s = *sp;
696 while (s.app_pending.empty() && !s.closed) {
697 unsigned rtype = 0;
698 std::string payload;
699 if (!read_record(s.fd, s.read_buffer, rtype, payload)) {
700 s.closed = true; // peer EOF (or socket timeout) — surfaced as ""
701 break;
702 }
703 if (rtype == 20) continue; // stray compat ChangeCipherSpec
704 if (rtype == 21) { // plaintext alert (illegal post-handshake, but final)
705 s.closed = true;
706 break;
707 }
708 if (rtype != 23) continue; // ignore anything else
709 unsigned inner_type = 0;
710 std::string content;
711 if (!open_record(s.server_keys, payload, inner_type, content)) {
712 fail("tls: record failed authentication");
713 s.closed = true;
714 break;
715 }
716 if (inner_type == 23) {
717 s.app_pending += content;
718 } else if (inner_type == 21) { // alert: close_notify (or fatal) ends the stream
719 s.closed = true;
720 } else if (inner_type == 22) {
721 // Post-handshake messages: NewSessionTicket(4) is ignored; a KeyUpdate(24)
722 // would change the peer's keys — unsupported, so end the stream rather than
723 // silently fail to decrypt what follows.
724 if (!content.empty() && static_cast<unsigned char>(content[0]) == 24) {
725 fail("tls: peer KeyUpdate is not supported");
726 s.closed = true;
727 }
728 }
729 }
730 const std::size_t n = std::min<std::size_t>(s.app_pending.size(),
731 static_cast<std::size_t>(bufsize));
732 const std::string out = s.app_pending.substr(0, n);
733 s.app_pending.erase(0, n);
734 return out;
735}
736
737long long close(long long session) {
738 t_error.clear();
739 const std::lock_guard<std::mutex> lock(g_mutex);
740 const auto it = g_sessions.find(session);
741 if (it == g_sessions.end()) return -1;
742 if (!it->second.closed) {
743 const std::string close_notify = {1, 0}; // warning, close_notify
744 seal_record(it->second.fd, it->second.client_keys, 21, close_notify);
745 }
746 g_sessions.erase(it);
747 return 0;
748}
749
750long long shutdown(long long session) {
751 t_error.clear();
752 const std::lock_guard<std::mutex> lock(g_mutex);
753 const auto it = g_sessions.find(session);
754 if (it == g_sessions.end()) return -1;
755 // Wake a reader blocked in recv() WITHOUT erasing the session (that stays the
756 // owner's job via close(), after it has joined the reader). Just half-close the
757 // socket so the blocking recv returns EOF.
758 return socket::shutdown(it->second.fd);
759}

764// ---- owning RAII session ----
765// Each method forwards to the handle-based free function above; the guard adds deterministic
766// close() (close_notify + session erase) on scope exit, so a `with` block cannot leak.
767
768Conn& Conn::operator=(Conn&& other) noexcept {
769 if (this != &other) {
770 if (session_ > 0) cheatah::tls::close(session_);
771 session_ = other.session_;
772 other.session_ = 0;
773 }
774 return *this;
775}
776Conn::~Conn() {
777 if (session_ > 0) cheatah::tls::close(session_);
778}
779long long Conn::send(const std::string& data) { return cheatah::tls::send(session_, data); }
780std::string Conn::recv(long long bufsize) { return cheatah::tls::recv(session_, bufsize); }
781long long Conn::shutdown() { return cheatah::tls::shutdown(session_); }
782long long Conn::close() {
783 if (session_ <= 0) return -1;
784 const long long rc = cheatah::tls::close(session_);
785 session_ = 0;
786 return rc;
787}
