36std::string to_hex(std::string_view raw) {
37 static constexpr char kHex[] = "0123456789abcdef";
38 std::string out;
39 out.reserve(raw.size() * 2);
40 for (const char ch : raw) {
41 out.push_back(kHex[static_cast<unsigned char>(ch) >> 4]);
42 out.push_back(kHex[static_cast<unsigned char>(ch) & 0xF]);
43 }
44 return out;
45}
46
47std::string from_hex(std::string_view hex) {
48 std::string out;
49 out.reserve(hex.size() / 2);
50 for (std::size_t i = 0; i + 1 < hex.size(); i += 2) {
51 unsigned v = 0;
52 for (int k = 0; k < 2; ++k) {
53 const char ch = hex[i + k];
54 v <<= 4;
55 if (ch >= '0' && ch <= '9') v |= static_cast<unsigned>(ch - '0');
56 else if (ch >= 'a' && ch <= 'f') v |= static_cast<unsigned>(ch - 'a' + 10);
57 else if (ch >= 'A' && ch <= 'F') v |= static_cast<unsigned>(ch - 'A' + 10); // LCOV_EXCL_LINE: from_hex is fed only lowercase x25519 hex; the uppercase branch is defensive parity
58 }
59 out.push_back(static_cast<char>(v));
60 }
61 return out;
62}

64// 16/24-bit big-endian helpers for the wire format.
65void put16(std::string& out, unsigned v) {
66 out.push_back(static_cast<char>(v >> 8));
67 out.push_back(static_cast<char>(v));
68}
69void put24(std::string& out, unsigned v) {
70 out.push_back(static_cast<char>(v >> 16));
71 out.push_back(static_cast<char>(v >> 8));
72 out.push_back(static_cast<char>(v));
73}
74unsigned get16(std::string_view s, std::size_t i) {
75 return (static_cast<unsigned char>(s[i]) << 8) | static_cast<unsigned char>(s[i + 1]);
76}
77unsigned get24(std::string_view s, std::size_t i) {
78 return (static_cast<unsigned char>(s[i]) << 16) | (static_cast<unsigned char>(s[i + 1]) << 8) |
79 static_cast<unsigned char>(s[i + 2]);
80}

86// HKDF-Expand-Label(secret, label, context, length) with the "tls13 " prefix.
87std::string expand_label_impl(std::string_view secret, std::string_view label,
88 std::string_view context, unsigned length) {
89 std::string info;
90 put16(info, length);
91 info.push_back(static_cast<char>(6 + label.size()));
92 info += "tls13 ";
93 info += label;
94 info.push_back(static_cast<char>(context.size()));
95 info += context;
96 return hashlib::hkdf_expand(secret, info, length);
97}

107// One traffic direction: AEAD key + iv + record sequence number.
108struct Keys {
109 std::string key_hex; // AEAD key (hex): 32 bytes for ChaCha20-Poly1305, 16 for AES-128-GCM
110 std::string iv; // 12-byte raw iv; per-record nonce = iv XOR seq
111 std::uint64_t seq = 0;
112 bool aes_gcm = false; // record cipher: false = ChaCha20-Poly1305 (0x1303), true = AES-128-GCM (0x1301)
113};
114
115// Both suites use the SHA-256 key schedule and a 12-byte iv; only the record-key length differs
116// (AES-128 = 16 bytes, ChaCha20 = 32). aes_gcm selects which record cipher seal/open_record use.
117Keys traffic_keys(std::string_view secret, bool aes_gcm) {
118 Keys k;
119 k.aes_gcm = aes_gcm;
120 k.key_hex = to_hex(expand_label_impl(secret, "key", "", aes_gcm ? 16 : 32));
121 k.iv = expand_label_impl(secret, "iv", "", 12);
122 return k;
123}
124
125// The per-record nonce: the 12-byte iv with the 8-byte big-endian sequence XORed into its tail.
126std::string nonce_hex(const Keys& k) {
127 std::string n = k.iv;
128 for (int i = 0; i < 8; ++i) {
129 n[4 + i] = static_cast<char>(static_cast<unsigned char>(n[4 + i]) ^
130 static_cast<unsigned char>(k.seq >> (8 * (7 - i))));
131 }
132 return to_hex(n);
133}

152// Read exactly one TLS record (header + payload) from the socket into (type, payload).
153// Blocking, bounded by the fd's socket timeout. False on EOF/short read.
154bool read_record(long long fd, std::string& buffer, unsigned& type, std::string& payload) {
155 while (buffer.size() < 5) {
156 const std::string chunk = sock::recv(fd, 16384);
157 if (chunk.empty()) return false;
158 buffer += chunk;
159 }
160 type = static_cast<unsigned char>(buffer[0]);
161 const unsigned len = get16(buffer, 3);
162 if (len > 16384 + 256) { // RFC bound + AEAD overhead: anything bigger is malformed
163 return false;
164 }
165 while (buffer.size() < 5 + len) {
166 const std::string chunk = sock::recv(fd, 16384);
167 if (chunk.empty()) return false;
168 buffer += chunk;
169 }
170 payload = buffer.substr(5, len);
171 buffer.erase(0, 5 + len);
172 return true;
173}

175bool write_record(long long fd, unsigned type, std::string_view payload) {
176 std::string rec;
177 rec.push_back(static_cast<char>(type));
178 put16(rec, 0x0303); // legacy_record_version
179 put16(rec, static_cast<unsigned>(payload.size()));
180 rec += payload;
181 return sock::sendall(fd, rec) == 0;
182}

184// Seal one application_data record (RFC 8446 §5.2): inner plaintext = content || content_type,
185// AAD = the record header, then ChaCha20-Poly1305.
186bool seal_record(long long fd, Keys& k, unsigned inner_type, std::string_view content) {
187 std::string inner(content);
188 inner.push_back(static_cast<char>(inner_type));
189 std::string aad;
190 aad.push_back(23);
191 put16(aad, 0x0303);
192 put16(aad, static_cast<unsigned>(inner.size() + 16));
193 const std::string ct = k.aes_gcm
194 ? aead::aes128gcm_encrypt(k.key_hex, nonce_hex(k), aad, inner)
195 : aead::chacha20poly1305_encrypt(k.key_hex, nonce_hex(k), aad, inner);
196 ++k.seq;
197 if (ct.empty()) return false;
198 return sock::sendall(fd, aad + ct) == 0;
199}

201// Open one encrypted record: returns the inner content and type, false on AEAD failure.
202bool open_record(Keys& k, std::string_view payload, unsigned& inner_type, std::string& content) {
203 std::string aad;
204 aad.push_back(23);
205 put16(aad, 0x0303);
206 put16(aad, static_cast<unsigned>(payload.size()));
207 std::string inner = k.aes_gcm
208 ? aead::aes128gcm_decrypt(k.key_hex, nonce_hex(k), aad, payload)
209 : aead::chacha20poly1305_decrypt(k.key_hex, nonce_hex(k), aad, payload);
210 ++k.seq;
211 if (inner.empty() && payload.size() > 16) return false; // tag mismatch (or empty record)
212 while (!inner.empty() && inner.back() == '\0') inner.pop_back(); // strip padding
213 if (inner.empty()) return false; // a record must carry a content type
214 inner_type = static_cast<unsigned char>(inner.back());
215 content = inner.substr(0, inner.size() - 1);
216 return true;
217}

221std::string random_bytes(std::size_t n) {
222 std::string out(n, '\0');
223 std::size_t got = 0;
224 while (got < n) {
225 const ssize_t r = getrandom(out.data() + got, n - got, 0);
226 if (r <= 0) return ""; // the kernel CSPRNG failing is fatal for key material
227 got += static_cast<std::size_t>(r);
228 }
229 return out;
230}

232// Build the ClientHello handshake MESSAGE (no record header). Fills `client_hello_random`.
233std::string build_client_hello(const std::string& server_name, std::string_view pub_raw) {
234 std::string body;
235 put16(body, 0x0303); // legacy_version
236 body += random_bytes(32); // random
237 body.push_back(32); // legacy_session_id (32 bytes, middlebox compatibility)
238 body += random_bytes(32);
239 put16(body, 4); // cipher_suites: two suites (4 bytes)
240 put16(body, 0x1303); // TLS_CHACHA20_POLY1305_SHA256 (preferred)
241 put16(body, 0x1301); // TLS_AES_128_GCM_SHA256 (some hosts only do AES-GCM)
242 body.push_back(1); // legacy_compression_methods
243 body.push_back(0); // null
244
245 std::string ext;
246 { // server_name (0)
247 std::string names;
248 names.push_back(0); // host_name
249 put16(names, static_cast<unsigned>(server_name.size()));
250 names += server_name;
251 std::string sni;
252 put16(sni, static_cast<unsigned>(names.size()));
253 sni += names;
254 put16(ext, 0);
255 put16(ext, static_cast<unsigned>(sni.size()));
256 ext += sni;
257 }
258 { // supported_groups (10): x25519 only
259 std::string g;
260 put16(g, 2);
261 put16(g, 0x001d);
262 put16(ext, 10);
263 put16(ext, static_cast<unsigned>(g.size()));
264 ext += g;
265 }
266 { // signature_algorithms (13): ed25519 (verifiable) + the common ones so real servers
267 // complete the handshake far enough for our explicit refusal to be diagnosable
268 std::string a;
269 put16(a, 6);
270 put16(a, 0x0807); // ed25519
271 put16(a, 0x0804); // rsa_pss_rsae_sha256 (verified — see rsa_verify.hpp)
272 put16(a, 0x0403); // ecdsa_secp256r1_sha256 (verified — see p256)
273 put16(ext, 13);
274 put16(ext, static_cast<unsigned>(a.size()));
275 ext += a;
276 }
277 { // supported_versions (43): TLS 1.3
278 std::string v;
279 v.push_back(2);
280 put16(v, 0x0304);
281 put16(ext, 43);
282 put16(ext, static_cast<unsigned>(v.size()));
283 ext += v;
284 }
285 { // key_share (51): our X25519 public key
286 std::string entry;
287 put16(entry, 0x001d);
288 put16(entry, 32);
289 entry += pub_raw;
290 std::string ks;
291 put16(ks, static_cast<unsigned>(entry.size()));
292 ks += entry;
293 put16(ext, 51);
294 put16(ext, static_cast<unsigned>(ks.size()));
295 ext += ks;
296 }
297 put16(body, static_cast<unsigned>(ext.size()));
298 body += ext;
299
300 std::string msg;
301 msg.push_back(1); // client_hello
302 put24(msg, static_cast<unsigned>(body.size()));
303 msg += body;
304 return msg;
305}

307// Human-readable TLS alert (RFC 8446 §6) from the 2 alert bytes — so a handshake refusal names its
308// cause (e.g. 40 handshake_failure = no common cipher/group; 70 protocol_version = no TLS 1.3).
309std::string alert_text(std::string_view p) {
310 if (p.size() < 2) return "(empty alert)";
311 const unsigned lvl = static_cast<unsigned char>(p[0]);
312 const unsigned d = static_cast<unsigned char>(p[1]);
313 const char* name = "unknown";
314 switch (d) {
315 case 0: name = "close_notify"; break;
316 case 10: name = "unexpected_message"; break;
317 case 20: name = "bad_record_mac"; break;
318 case 22: name = "record_overflow"; break;
319 case 40: name = "handshake_failure"; break;
320 case 42: name = "bad_certificate"; break;
321 case 43: name = "unsupported_certificate"; break;
322 case 47: name = "illegal_parameter"; break;
323 case 48: name = "unknown_ca"; break;
324 case 49: name = "access_denied"; break;
325 case 50: name = "decode_error"; break;
326 case 51: name = "decrypt_error"; break;
327 case 70: name = "protocol_version"; break;
328 case 71: name = "insufficient_security"; break;
329 case 80: name = "internal_error"; break;
330 case 109: name = "missing_extension"; break;
331 case 110: name = "unsupported_extension"; break;
332 case 112: name = "unrecognized_name"; break;
333 case 116: name = "certificate_required"; break;
334 case 120: name = "no_application_protocol"; break;
335 default: break;
336 }
337 return "alert level=" + std::to_string(lvl) + " description=" + std::to_string(d) + " (" + name + ")";
338}

340// Parse ServerHello: confirm TLS 1.3 + one of our suites, extract the server's X25519 key share and
341// the CHOSEN cipher suite (0x1303 ChaCha20-Poly1305 or 0x1301 AES-128-GCM).
342bool parse_server_hello(std::string_view msg, std::string& server_pub_raw, unsigned& chosen_suite) {
343 if (msg.size() < 4 || msg[0] != 2) return false; // server_hello
344 std::string_view b = msg.substr(4);
345 if (b.size() < 2 + 32 + 1) return false;
346 std::size_t i = 2 + 32; // legacy_version + random
347 const unsigned sid_len = static_cast<unsigned char>(b[i]);
348 i += 1 + sid_len;
349 if (b.size() < i + 4) return false;
350 const unsigned suite = get16(b, i);
351 if (suite != 0x1303 && suite != 0x1301) return false; // must be one of the suites we offered
352 chosen_suite = suite;
353 i += 2 + 1; // suite + legacy_compression
354 if (b.size() < i + 2) return false;
355 const unsigned ext_len = get16(b, i);
356 i += 2;
357 const std::size_t ext_end = i + ext_len;
358 bool saw_13 = false;
359 while (i + 4 <= ext_end && ext_end <= b.size()) {
360 const unsigned etype = get16(b, i);
361 const unsigned elen = get16(b, i + 2);
362 i += 4;
363 if (i + elen > b.size()) return false;
364 if (etype == 43 && elen == 2 && get16(b, i) == 0x0304) saw_13 = true;
365 if (etype == 51 && elen >= 4 && get16(b, i) == 0x001d && get16(b, i + 2) == 32 &&
366 elen == 4 + 32) {
367 server_pub_raw = std::string(b.substr(i + 4, 32));
368 }
369 i += elen;
370 }
371 return saw_13 && server_pub_raw.size() == 32;
372}

374// Extract the Ed25519 public key from the leaf certificate's SubjectPublicKeyInfo: the DER
375// pattern 30 05 06 03 2B 65 70 (AlgorithmIdentifier { id-Ed25519 }) followed by
376// 03 21 00 <32-byte key> (BIT STRING). Returns "" when the cert key is not Ed25519.
377std::string ed25519_spki_key(std::string_view cert_der) {
378 static const unsigned char kPat[] = {0x30, 0x05, 0x06, 0x03, 0x2B, 0x65, 0x70,
379 0x03, 0x21, 0x00};
380 for (std::size_t i = 0; i + sizeof kPat + 32 <= cert_der.size(); ++i) {
381 if (std::memcmp(cert_der.data() + i, kPat, sizeof kPat) == 0) {
382 return std::string(cert_der.substr(i + sizeof kPat, 32));
383 }
384 }
385 return ""; // LCOV_EXCL_LINE: only when an Ed25519 CertificateVerify names a non-Ed25519 leaf — a malformed peer we don't mirror
386}

388// ---- the handshake state machine -------------------------------------------------

390long long handshake(long long fd, const std::string& server_name) {
391 t_error.clear();
392
393 // Ephemeral X25519 key pair.
394 const std::string priv_raw = random_bytes(32);
395 if (priv_raw.size() != 32) {
396 fail("tls: system random unavailable");
397 return -1;
398 }
399 const std::string priv_hex = to_hex(priv_raw);
400 const std::string pub_hex = x25519::x25519_base(priv_hex);
401 const std::string pub_raw = from_hex(pub_hex);
402
403 std::string transcript; // concatenated handshake MESSAGES (no record headers)
404 const std::string client_hello = build_client_hello(server_name, pub_raw);
405 transcript += client_hello;
406 if (!write_record(fd, 22, client_hello)) {
407 fail("tls: cannot send ClientHello");
408 return -1;
409 }
410
411 // ServerHello (plaintext record, type 22; tolerate ChangeCipherSpec compat records).
412 std::string buffer, payload;
413 unsigned rtype = 0;
414 std::string server_pub_raw;
415 for (;;) {
416 if (!read_record(fd, buffer, rtype, payload)) {
417 fail("tls: connection closed before ServerHello");
418 return -1;
419 }
420 if (rtype == 20) continue; // ChangeCipherSpec (compat) — ignored
421 if (rtype == 21) {
422 fail("tls: server sent an alert instead of ServerHello — " + alert_text(payload));
423 return -1;
424 }
425 if (rtype != 22) {
426 fail("tls: unexpected record before ServerHello");
427 return -1;
428 }
429 break;
430 }
431 unsigned chosen_suite = 0;
432 if (!parse_server_hello(payload, server_pub_raw, chosen_suite)) {
433 fail("tls: malformed ServerHello (or the server refused TLS 1.3 + our cipher suites)");
434 return -1;
435 }
436 const bool aes_gcm = (chosen_suite == 0x1301); // else 0x1303 ChaCha20-Poly1305
437 transcript += payload;
438
439 // Key schedule through the handshake secrets.
440 const std::string shared_hex = x25519::x25519(priv_hex, to_hex(server_pub_raw));
441 if (shared_hex.empty()) {
442 fail("tls: invalid server key share");
443 return -1;
444 }
445 const std::string zeros(32, '\0');
446 const std::string early = hashlib::hkdf_extract(std::string(), zeros);
447 const std::string derived = derive_secret_impl(early, "derived", "");
448 const std::string hs_secret = hashlib::hkdf_extract(derived, from_hex(shared_hex));
449 const std::string c_hs = derive_secret_impl(hs_secret, "c hs traffic", transcript);
450 const std::string s_hs = derive_secret_impl(hs_secret, "s hs traffic", transcript);
451 Keys client_keys = traffic_keys(c_hs, aes_gcm);
452 Keys server_keys = traffic_keys(s_hs, aes_gcm);
453
454 // Encrypted handshake flight: EncryptedExtensions, Certificate, CertificateVerify, Finished.
455 std::string handshake_bytes; // decrypted, possibly spanning records
456 std::string cert_der;
457 bool verified_cert = false, server_finished = false;
458 std::string transcript_at_cv, transcript_at_finished;
459 while (!server_finished) {
460 if (!read_record(fd, buffer, rtype, payload)) {
461 fail("tls: connection closed during the handshake");
462 return -1;
463 }
464 if (rtype == 20) continue; // compat ChangeCipherSpec
465 if (rtype == 21) {
466 fail("tls: server alert during the handshake — " + alert_text(payload));
467 return -1;
468 }
469 if (rtype != 23) {
470 fail("tls: unexpected plaintext record during the encrypted handshake");
471 return -1;
472 }
473 unsigned inner_type = 0;
474 std::string content;
475 if (!open_record(server_keys, payload, inner_type, content)) {
476 fail("tls: handshake record failed authentication");
477 return -1;
478 }
479 if (inner_type == 21) {
480 fail("tls: server alert during the handshake — " + alert_text(content));
481 return -1;
482 }
483 if (inner_type != 22) {
484 fail("tls: unexpected inner record type during the handshake");
485 return -1;
486 }
487 handshake_bytes += content;
488
489 // Drain complete handshake messages from the reassembly buffer.
490 while (handshake_bytes.size() >= 4) {
491 const unsigned mtype = static_cast<unsigned char>(handshake_bytes[0]);
492 const unsigned mlen = get24(handshake_bytes, 1);
493 if (handshake_bytes.size() < 4 + mlen) break;
494 const std::string msg = handshake_bytes.substr(0, 4 + mlen);
495 handshake_bytes.erase(0, 4 + mlen);
496
497 if (mtype == 11 && msg.size() > 4 + 4 + 3 + 3) { // Certificate
498 // certificate_request_context (1 byte, empty) + cert list; take the leaf.
499 std::size_t i = 4 + 1 + 3; // header + context + list length
500 const unsigned leaf_len = get24(msg, i);
501 i += 3;
502 if (i + leaf_len <= msg.size()) cert_der = msg.substr(i, leaf_len);
503 transcript_at_cv = transcript + msg; // transcript THROUGH Certificate
504 }
505 if (mtype == 15) { // CertificateVerify
506 // cppcheck-suppress stlcstrConstructor // (ptr,len) subview of msg — not a c_str() copy
507 const std::string_view body(msg.data() + 4, msg.size() - 4);
508 if (body.size() < 4) {
509 fail("tls: malformed CertificateVerify");
510 return -1;
511 }
512 const unsigned alg = get16(body, 0);
513 const unsigned sig_len = get16(body, 2);
514 if (body.size() < 4 + sig_len) {
515 fail("tls: malformed CertificateVerify");
516 return -1;
517 }
518 // The signed content (same for every algorithm): 64 spaces, the
519 // context string, a NUL, then the handshake transcript hash.
520 std::string signed_content(64, ' ');
521 signed_content += "TLS 1.3, server CertificateVerify";
522 signed_content.push_back('\0');
523 signed_content += hashlib::sha256_digest(transcript_at_cv);
524 const std::string sig(body.substr(4, sig_len));
525 if (alg == 0x0807) { // ed25519 (signs the message directly)
526 const std::string spki = ed25519_spki_key(cert_der);
527 if (spki.size() != 32) {
528 fail("tls: certificate key is not Ed25519");
529 return -1;
530 }
531 if (!ed25519::verify(to_hex(spki), signed_content, to_hex(sig))) {
532 fail("tls: server CertificateVerify signature is INVALID");
533 return -1;
534 }
535 } else if (alg == 0x0403) { // ecdsa_secp256r1_sha256 (signs SHA-256(content))
536 const std::string point = p256::spki_ec_point(cert_der);
537 if (point.size() != 64) {
538 fail("tls: certificate key is not P-256 EC");
539 return -1;
540 }
541 if (!p256::verify_der(point, hashlib::sha256_digest(signed_content), sig)) {
542 fail("tls: server CertificateVerify (ECDSA P-256) is INVALID");
543 return -1;
544 }
545 } else if (alg == 0x0804) { // rsa_pss_rsae_sha256 (RSA leaf certificate)
546 // RSA-PSS verifies the signed_content directly (it hashes with SHA-256 internally),
547 // using the RSA public key extracted from the leaf cert's SubjectPublicKeyInfo.
548 if (!rsa::verify_pss_sha256(cert_der, signed_content, sig)) {
549 fail("tls: server CertificateVerify (RSA-PSS SHA-256) is INVALID");
550 return -1;
551 }
552 } else { // LCOV_EXCL_LINE: reached only if a server sends a CertificateVerify whose algorithm ignores our advertised signature_algorithms — a non-conformant peer we don't mirror
553 fail("tls: server certificate uses an algorithm cheatah cannot verify yet " // LCOV_EXCL_LINE
554 "(Ed25519, ECDSA P-256, and RSA-PSS SHA-256 are supported) — refusing an "
555 "unauthenticated connection");
556 return -1;
557 }
558 verified_cert = true;
559 }
560 if (mtype == 20) { // Finished
561 const std::string finished_key = expand_label_impl(s_hs, "finished", "", 32);
562 const std::string expect =
563 hashlib::hmac_sha256(finished_key, hashlib::sha256_digest(transcript));
564 // cppcheck-suppress stlcstrConstructor // (ptr,len) subview of msg — not a c_str() copy
565 const std::string_view got(msg.data() + 4, msg.size() - 4);
566 if (got.size() != expect.size() ||
567 std::memcmp(got.data(), expect.data(), expect.size()) != 0) {
568 fail("tls: server Finished MAC mismatch — handshake transcript tampered");
569 return -1;
570 }
571 transcript_at_finished = transcript + msg;
572 server_finished = true;
573 }
574 transcript += msg;
575 }
576 }
577 if (!verified_cert) {
578 fail("tls: server never proved possession of its certificate key");
579 return -1;
580 }
581
582 // Application traffic secrets (transcript through server Finished), then OUR Finished
583 // (sent under the handshake keys, with the transcript through the server's Finished).
584 const std::string derived2 = derive_secret_impl(hs_secret, "derived", "");
585 const std::string master = hashlib::hkdf_extract(derived2, zeros);
586 const std::string c_ap = derive_secret_impl(master, "c ap traffic", transcript_at_finished);
587 const std::string s_ap = derive_secret_impl(master, "s ap traffic", transcript_at_finished);
588
589 const std::string c_finished_key = expand_label_impl(c_hs, "finished", "", 32);
590 const std::string verify =
591 hashlib::hmac_sha256(c_finished_key, hashlib::sha256_digest(transcript));
592 std::string fin_msg;
593 fin_msg.push_back(20);
594 put24(fin_msg, static_cast<unsigned>(verify.size()));
595 fin_msg += verify;
596 if (!seal_record(fd, client_keys, 22, fin_msg)) {
597 fail("tls: cannot send client Finished");
598 return -1;
599 }
600
601 Session s;
602 s.fd = fd;
603 s.client_keys = traffic_keys(c_ap, aes_gcm);
604 s.server_keys = traffic_keys(s_ap, aes_gcm);
605 s.read_buffer = std::move(buffer); // bytes already pulled off the socket stay with us
606
607 const std::lock_guard<std::mutex> lock(g_mutex);
608 const long long handle = g_next_handle++;
609 g_sessions[handle] = std::move(s);
610 return handle;
611}

613} // namespace

619long long send(long long session, const std::string& data) {
620 t_error.clear();
621 // Lock ONLY for the map lookup (see recv): the socket write below runs without
622 // the global lock so concurrent sessions don't serialize on each other.
623 Session* sp = nullptr;
624 {
625 const std::lock_guard<std::mutex> lock(g_mutex);
626 const auto it = g_sessions.find(session);
627 if (it == g_sessions.end() || it->second.closed) {
628 fail("tls: unknown or closed session");
629 return -1;
630 }
631 sp = &it->second;
632 }
633 Session& s = *sp;
634 // Respect the 16 KiB record plaintext bound.
635 std::string_view rest = data;
636 while (!rest.empty()) {
637 const std::size_t n = std::min<std::size_t>(rest.size(), 16384);
638 if (!seal_record(s.fd, s.client_keys, 23, rest.substr(0, n))) {
639 fail("tls: send failed");
640 return -1;
641 }
642 rest.remove_prefix(n);
643 }
644 return 0;
645}
646
647std::string recv(long long session, long long bufsize) {
648 t_error.clear();
649 if (bufsize <= 0) return "";
650 // Guard ONLY the map lookup. The per-session buffers (read_buffer/app_pending)
651 // and socket are owned by this session's single reader thread, so the blocking
652 // record I/O below runs WITHOUT the global lock — otherwise one session's
653 // blocking recv would serialize (and at shutdown, starve) every other session's
654 // recv/send. A std::map node address is stable until that node is erased, and a
655 // session is erased only by its own owner (after this loop), so the pointer is
656 // valid for this call. (g_mutex still serializes find/insert/erase on the map.)
657 Session* sp = nullptr;
658 {
659 const std::lock_guard<std::mutex> lock(g_mutex);
660 const auto it = g_sessions.find(session);
661 if (it == g_sessions.end()) {
662 fail("tls: unknown session");
663 return "";
664 }
665 sp = &it->second;
666 }
667 Session& s = *sp;
668 while (s.app_pending.empty() && !s.closed) {
669 unsigned rtype = 0;
670 std::string payload;
671 if (!read_record(s.fd, s.read_buffer, rtype, payload)) {
672 s.closed = true; // peer EOF (or socket timeout) — surfaced as ""
673 break;
674 }
675 if (rtype == 20) continue; // stray compat ChangeCipherSpec
676 if (rtype == 21) { // plaintext alert (illegal post-handshake, but final)
677 s.closed = true;
678 break;
679 }
680 if (rtype != 23) continue; // ignore anything else
681 unsigned inner_type = 0;
682 std::string content;
683 if (!open_record(s.server_keys, payload, inner_type, content)) {
684 fail("tls: record failed authentication");
685 s.closed = true;
686 break;
687 }
688 if (inner_type == 23) {
689 s.app_pending += content;
690 } else if (inner_type == 21) { // alert: close_notify (or fatal) ends the stream
691 s.closed = true;
692 } else if (inner_type == 22) {
693 // Post-handshake messages: NewSessionTicket(4) is ignored; a KeyUpdate(24)
694 // would change the peer's keys — unsupported, so end the stream rather than
695 // silently fail to decrypt what follows.
696 if (!content.empty() && static_cast<unsigned char>(content[0]) == 24) {
697 fail("tls: peer KeyUpdate is not supported");
698 s.closed = true;
699 }
700 }
701 }
702 const std::size_t n = std::min<std::size_t>(s.app_pending.size(),
703 static_cast<std::size_t>(bufsize));
704 const std::string out = s.app_pending.substr(0, n);
705 s.app_pending.erase(0, n);
706 return out;
707}
708
709long long close(long long session) {
710 t_error.clear();
711 const std::lock_guard<std::mutex> lock(g_mutex);
712 const auto it = g_sessions.find(session);
713 if (it == g_sessions.end()) return -1;
714 if (!it->second.closed) {
715 const std::string close_notify = {1, 0}; // warning, close_notify
716 seal_record(it->second.fd, it->second.client_keys, 21, close_notify);
717 }
718 g_sessions.erase(it);
719 return 0;
720}
721
722long long shutdown(long long session) {
723 t_error.clear();
724 const std::lock_guard<std::mutex> lock(g_mutex);
725 const auto it = g_sessions.find(session);
726 if (it == g_sessions.end()) return -1;
727 // Wake a reader blocked in recv() WITHOUT erasing the session (that stays the
728 // owner's job via close(), after it has joined the reader). Just half-close the
729 // socket so the blocking recv returns EOF.
730 return socket::shutdown(it->second.fd);
731}

735// ---- owning RAII session ----
736// Each method forwards to the handle-based free function above; the guard adds deterministic
737// close() (close_notify + session erase) on scope exit, so a `with` block cannot leak.
738
739Conn& Conn::operator=(Conn&& other) noexcept {
740 if (this != &other) {
741 if (session_ > 0) cheatah::tls::close(session_);
742 session_ = other.session_;
743 other.session_ = 0;
744 }
745 return *this;
746}
747Conn::~Conn() {
748 if (session_ > 0) cheatah::tls::close(session_);
749}

751std::string Conn::recv(long long bufsize) { return cheatah::tls::recv(session_, bufsize); }
752long long Conn::shutdown() { return cheatah::tls::shutdown(session_); }
753long long Conn::close() {
754 if (session_ <= 0) return -1;
755 const long long rc = cheatah::tls::close(session_);
756 session_ = 0;
757 return rc;
758}
