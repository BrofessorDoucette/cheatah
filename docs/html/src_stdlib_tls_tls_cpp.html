28// The implementation walks the RFC top to bottom: record layer, transcript hash, the HKDF
29// key schedule, then the handshake state machine. Every secret derives through hashlib's
30// HKDF; every record seals/opens through the aead module; the ephemeral key is x25519.

41// ---- hex <-> bytes: the ONE canonical implementation lives in hashlib -----------
42// The crypto modules speak hex for keys; tls feeds from_hex only valid, even-length lowercase
43// x25519 hex, so the canonical from_hex's odd-length/non-hex throws are never reached here.
44using hashlib::to_hex; // bytes -> lowercase hex (string_view / (uint8_t*, n) overloads).
45using hashlib::from_hex; // hex -> bytes (throws on odd length / non-hex).

47// 16/24-bit big-endian helpers for the wire format.
48void put16(std::string& out, unsigned v) {
49 out.push_back(static_cast<char>(v >> 8));
50 out.push_back(static_cast<char>(v));
51}
52void put24(std::string& out, unsigned v) {
53 out.push_back(static_cast<char>(v >> 16));
54 out.push_back(static_cast<char>(v >> 8));
55 out.push_back(static_cast<char>(v));
56}
57unsigned get16(std::string_view s, std::size_t i) {
58 return (static_cast<unsigned char>(s[i]) << 8) | static_cast<unsigned char>(s[i + 1]);
59}
60unsigned get24(std::string_view s, std::size_t i) {
61 return (static_cast<unsigned char>(s[i]) << 16) | (static_cast<unsigned char>(s[i + 1]) << 8) |
62 static_cast<unsigned char>(s[i + 2]);
63}

69// HKDF-Expand-Label(secret, label, context, length) with the "tls13 " prefix. @p sha384 selects the
70// SHA-384 HKDF (for the TLS_AES_256_GCM_SHA384 key schedule); default is the SHA-256 schedule.
71std::string expand_label_impl(std::string_view secret, std::string_view label,
72 std::string_view context, unsigned length, bool sha384 = false) {
73 std::string info;
74 put16(info, length);
75 info.push_back(static_cast<char>(6 + label.size()));
76 info += "tls13 ";
77 info += label;
78 info.push_back(static_cast<char>(context.size()));
79 info += context;
80 return sha384 ? hashlib::hkdf_expand_sha384(secret, info, length)
81 : hashlib::hkdf_expand(secret, info, length);
82}
83
84// Derive-Secret(secret, label, transcript) = Expand-Label(secret, label, Hash(transcript), HashLen),
85// where Hash is the negotiated suite's hash (SHA-256, or SHA-384 when @p sha384).
86std::string derive_secret_impl(std::string_view secret, std::string_view label,
87 std::string_view transcript, bool sha384 = false) {
88 const std::string th =
89 sha384 ? hashlib::sha384_digest(transcript) : hashlib::sha256_digest(transcript);
90 return expand_label_impl(secret, label, th, sha384 ? 48 : 32, sha384);
91}

95// The negotiated record cipher: ChaCha20-Poly1305 (0x1303), AES-128-GCM (0x1301), or AES-256-GCM (0x1302).

98// Key-schedule hash dispatch: the SHA-256 schedule by default, the SHA-384 schedule for the
99// TLS_AES_256_GCM_SHA384 suite. (RFC 8446 §7.1: the schedule's Hash is the cipher suite's hash.)
100std::string ks_digest(bool sha384, std::string_view d) {
101 return sha384 ? hashlib::sha384_digest(d) : hashlib::sha256_digest(d);
102}
103std::string ks_extract(bool sha384, std::string_view salt, std::string_view ikm) {
104 return sha384 ? hashlib::hkdf_extract_sha384(salt, ikm) : hashlib::hkdf_extract(salt, ikm);
105}
106std::string ks_hmac(bool sha384, std::string_view key, std::string_view data) {
107 return sha384 ? hashlib::hmac_sha384(key, data) : hashlib::hmac_sha256(key, data);

110// One traffic direction: AEAD key + iv + record sequence number.
111struct Keys {
112 std::string key_hex; // AEAD key (hex): 32 bytes for ChaCha20 / AES-256-GCM, 16 for AES-128-GCM
113 std::string iv; // 12-byte raw iv; per-record nonce = iv XOR seq
114 std::uint64_t seq = 0;
115 Aead aead = Aead::Chacha20;
116};
117
118// Derive a direction's record keys from its traffic secret. Key length follows the AEAD (16 for AES-128,
119// 32 for AES-256 / ChaCha20); @p sha384 selects the SHA-384 key schedule (the 256 suite).
120Keys traffic_keys(std::string_view secret, Aead aead, bool sha384) {
121 Keys k;
122 k.aead = aead;
123 k.key_hex = to_hex(expand_label_impl(secret, "key", "", aead == Aead::Aes128 ? 16 : 32, sha384));
124 k.iv = expand_label_impl(secret, "iv", "", 12, sha384);
125 return k;
126}
127
128// The per-record nonce: the 12-byte iv with the 8-byte big-endian sequence XORed into its tail.
129std::string nonce_hex(const Keys& k) {
130 std::string n = k.iv;
131 for (int i = 0; i < 8; ++i) {
132 n[4 + i] = static_cast<char>(static_cast<unsigned char>(n[4 + i]) ^
133 static_cast<unsigned char>(k.seq >> (8 * (7 - i))));
134 }
135 return to_hex(n);
136}

155// Read exactly one TLS record (header + payload) from the socket into (type, payload).
156// Blocking, bounded by the fd's socket timeout. False on EOF/short read.
157bool read_record(long long fd, std::string& buffer, unsigned& type, std::string& payload) {
158 while (buffer.size() < 5) {
159 const std::string chunk = sock::recv(fd, 16384);
160 if (chunk.empty()) return false;
161 buffer += chunk;
162 }
163 type = static_cast<unsigned char>(buffer[0]);
164 const unsigned len = get16(buffer, 3);
165 if (len > 16384 + 256) { // RFC bound + AEAD overhead: anything bigger is malformed
166 return false;
167 }
168 while (buffer.size() < 5 + len) {
169 const std::string chunk = sock::recv(fd, 16384);
170 if (chunk.empty()) return false;
171 buffer += chunk;
172 }
173 payload = buffer.substr(5, len);
174 buffer.erase(0, 5 + len);
175 return true;
176}

178bool write_record(long long fd, unsigned type, std::string_view payload) {
179 std::string rec;
180 rec.push_back(static_cast<char>(type));
181 put16(rec, 0x0303); // legacy_record_version
182 put16(rec, static_cast<unsigned>(payload.size()));
183 rec += payload;
184 return sock::sendall(fd, rec) == 0;
185}

187// Seal one application_data record (RFC 8446 §5.2): inner plaintext = content || content_type,
188// AAD = the record header, then ChaCha20-Poly1305.
189bool seal_record(long long fd, Keys& k, unsigned inner_type, std::string_view content) {
190 std::string inner(content);
191 inner.push_back(static_cast<char>(inner_type));
192 std::string aad;
193 aad.push_back(23);
194 put16(aad, 0x0303);
195 put16(aad, static_cast<unsigned>(inner.size() + 16));
196 const std::string ct =
197 k.aead == Aead::Aes256 ? aead::aes256gcm_encrypt(k.key_hex, nonce_hex(k), aad, inner)
198 : k.aead == Aead::Aes128 ? aead::aes128gcm_encrypt(k.key_hex, nonce_hex(k), aad, inner)
199 : aead::chacha20poly1305_encrypt(k.key_hex, nonce_hex(k), aad, inner);
200 ++k.seq;
201 if (ct.empty()) return false;
202 return sock::sendall(fd, aad + ct) == 0;
203}

205// Open one encrypted record: returns the inner content and type, false on AEAD failure.
206bool open_record(Keys& k, std::string_view payload, unsigned& inner_type, std::string& content) {
207 std::string aad;
208 aad.push_back(23);
209 put16(aad, 0x0303);
210 put16(aad, static_cast<unsigned>(payload.size()));
211 std::string inner =
212 k.aead == Aead::Aes256 ? aead::aes256gcm_decrypt(k.key_hex, nonce_hex(k), aad, payload)
213 : k.aead == Aead::Aes128 ? aead::aes128gcm_decrypt(k.key_hex, nonce_hex(k), aad, payload)
214 : aead::chacha20poly1305_decrypt(k.key_hex, nonce_hex(k), aad, payload);
215 ++k.seq;
216 if (inner.empty() && payload.size() > 16) return false; // tag mismatch (or empty record)
217 while (!inner.empty() && inner.back() == '\0') inner.pop_back(); // strip padding
218 if (inner.empty()) return false; // a record must carry a content type
219 inner_type = static_cast<unsigned char>(inner.back());
220 content = inner.substr(0, inner.size() - 1);
221 return true;
222}

226std::string random_bytes(std::size_t n) {
227 std::string out(n, '\0');
228 // getentropy (Linux + macOS/BSD) is the portable CSPRNG read; getrandom is Linux-only.
229 // It is capped at 256 bytes per call, so loop for larger requests. A nonzero return means
230 // the OS could not supply randomness — fatal for key material, so bail with "".
231 std::size_t got = 0;
232 while (got < n) {
233 const std::size_t chunk = std::min<std::size_t>(n - got, 256);
234 if (::getentropy(out.data() + got, chunk) != 0) return "";
235 got += chunk;
236 }
237 return out;
238}

240// Build the ClientHello handshake MESSAGE (no record header). Fills `client_hello_random`.
241std::string build_client_hello(const std::string& server_name, std::string_view pub_raw) {
242 std::string body;
243 put16(body, 0x0303); // legacy_version
244 body += random_bytes(32); // random
245 body.push_back(32); // legacy_session_id (32 bytes, middlebox compatibility)
246 body += random_bytes(32);
247 put16(body, 6); // cipher_suites: three suites (6 bytes)
248 put16(body, 0x1303); // TLS_CHACHA20_POLY1305_SHA256 (preferred)
249 put16(body, 0x1301); // TLS_AES_128_GCM_SHA256 (some hosts only do AES-GCM)
250 put16(body, 0x1302); // TLS_AES_256_GCM_SHA384 (hosts that prefer the 256-bit suite)
251 body.push_back(1); // legacy_compression_methods
252 body.push_back(0); // null
253
254 std::string ext;
255 { // server_name (0)
256 std::string names;
257 names.push_back(0); // host_name
258 put16(names, static_cast<unsigned>(server_name.size()));
259 names += server_name;
260 std::string sni;
261 put16(sni, static_cast<unsigned>(names.size()));
262 sni += names;
263 put16(ext, 0);
264 put16(ext, static_cast<unsigned>(sni.size()));
265 ext += sni;
266 }
267 { // supported_groups (10): x25519 only
268 std::string g;
269 put16(g, 2);
270 put16(g, 0x001d);
271 put16(ext, 10);
272 put16(ext, static_cast<unsigned>(g.size()));
273 ext += g;
274 }
275 { // signature_algorithms (13): ed25519 (verifiable) + the common ones so real servers
276 // complete the handshake far enough for our explicit refusal to be diagnosable
277 std::string a;
278 put16(a, 8);
279 put16(a, 0x0807); // ed25519
280 put16(a, 0x0804); // rsa_pss_rsae_sha256 (verified — see rsa_verify.hpp)
281 put16(a, 0x0403); // ecdsa_secp256r1_sha256 (verified — see p256)
282 put16(a, 0x0503); // ecdsa_secp384r1_sha384 (verified — see p384)
283 put16(ext, 13);
284 put16(ext, static_cast<unsigned>(a.size()));
285 ext += a;
286 }
287 { // supported_versions (43): TLS 1.3
288 std::string v;
289 v.push_back(2);
290 put16(v, 0x0304);
291 put16(ext, 43);
292 put16(ext, static_cast<unsigned>(v.size()));
293 ext += v;
294 }
295 { // key_share (51): our X25519 public key
296 std::string entry;
297 put16(entry, 0x001d);
298 put16(entry, 32);
299 entry += pub_raw;
300 std::string ks;
301 put16(ks, static_cast<unsigned>(entry.size()));
302 ks += entry;
303 put16(ext, 51);
304 put16(ext, static_cast<unsigned>(ks.size()));
305 ext += ks;
306 }
307 put16(body, static_cast<unsigned>(ext.size()));
308 body += ext;
309
310 std::string msg;
311 msg.push_back(1); // client_hello
312 put24(msg, static_cast<unsigned>(body.size()));
313 msg += body;
314 return msg;
315}

317// Human-readable TLS alert (RFC 8446 §6) from the 2 alert bytes — so a handshake refusal names its
318// cause (e.g. 40 handshake_failure = no common cipher/group; 70 protocol_version = no TLS 1.3).
319std::string alert_text(std::string_view p) {
320 if (p.size() < 2) return "(empty alert)";
321 const unsigned lvl = static_cast<unsigned char>(p[0]);
322 const unsigned d = static_cast<unsigned char>(p[1]);
323 const char* name = "unknown";
324 switch (d) {
325 case 0: name = "close_notify"; break;
326 case 10: name = "unexpected_message"; break;
327 case 20: name = "bad_record_mac"; break;
328 case 22: name = "record_overflow"; break;
329 case 40: name = "handshake_failure"; break;
330 case 42: name = "bad_certificate"; break;
331 case 43: name = "unsupported_certificate"; break;
332 case 47: name = "illegal_parameter"; break;
333 case 48: name = "unknown_ca"; break;
334 case 49: name = "access_denied"; break;
335 case 50: name = "decode_error"; break;
336 case 51: name = "decrypt_error"; break;
337 case 70: name = "protocol_version"; break;
338 case 71: name = "insufficient_security"; break;
339 case 80: name = "internal_error"; break;
340 case 109: name = "missing_extension"; break;
341 case 110: name = "unsupported_extension"; break;
342 case 112: name = "unrecognized_name"; break;
343 case 116: name = "certificate_required"; break;
344 case 120: name = "no_application_protocol"; break;
345 default: break;
346 }
347 return "alert level=" + std::to_string(lvl) + " description=" + std::to_string(d) + " (" + name + ")";
348}

350// Parse ServerHello: confirm TLS 1.3 + one of our suites, extract the server's X25519 key share and
351// the CHOSEN cipher suite (0x1303 ChaCha20-Poly1305 or 0x1301 AES-128-GCM).
352bool parse_server_hello(std::string_view msg, std::string& server_pub_raw, unsigned& chosen_suite) {
353 if (msg.size() < 4 || msg[0] != 2) return false; // server_hello
354 std::string_view b = msg.substr(4);
355 if (b.size() < 2 + 32 + 1) return false;
356 std::size_t i = 2 + 32; // legacy_version + random
357 const unsigned sid_len = static_cast<unsigned char>(b[i]);
358 i += 1 + sid_len;
359 if (b.size() < i + 4) return false;
360 const unsigned suite = get16(b, i);
361 if (suite != 0x1303 && suite != 0x1301 && suite != 0x1302)
362 return false; // LCOV_EXCL_LINE: a server choosing a suite we did NOT offer is a malformed/hostile peer a conformant server never produces — the client-side mirror of the tested ParseClientHello "no suite in common" rejection
363 chosen_suite = suite;
364 i += 2 + 1; // suite + legacy_compression
365 if (b.size() < i + 2) return false;
366 const unsigned ext_len = get16(b, i);
367 i += 2;
368 const std::size_t ext_end = i + ext_len;
369 bool saw_13 = false;
370 while (i + 4 <= ext_end && ext_end <= b.size()) {
371 const unsigned etype = get16(b, i);
372 const unsigned elen = get16(b, i + 2);
373 i += 4;
374 if (i + elen > b.size()) return false;
375 if (etype == 43 && elen == 2 && get16(b, i) == 0x0304) saw_13 = true;
376 if (etype == 51 && elen >= 4 && get16(b, i) == 0x001d && get16(b, i + 2) == 32 &&
377 elen == 4 + 32) {
378 server_pub_raw = std::string(b.substr(i + 4, 32));
379 }
380 i += elen;
381 }
382 return saw_13 && server_pub_raw.size() == 32;
383}

385// Extract the Ed25519 public key from the leaf certificate's SubjectPublicKeyInfo: the DER
386// pattern 30 05 06 03 2B 65 70 (AlgorithmIdentifier { id-Ed25519 }) followed by
387// 03 21 00 <32-byte key> (BIT STRING). Returns "" when the cert key is not Ed25519.
388std::string ed25519_spki_key(std::string_view cert_der) {
389 static const unsigned char kPat[] = {0x30, 0x05, 0x06, 0x03, 0x2B, 0x65, 0x70,
390 0x03, 0x21, 0x00};
391 for (std::size_t i = 0; i + sizeof kPat + 32 <= cert_der.size(); ++i) {
392 if (std::memcmp(cert_der.data() + i, kPat, sizeof kPat) == 0) {
393 return std::string(cert_der.substr(i + sizeof kPat, 32));
394 }
395 }
396 return ""; // LCOV_EXCL_LINE: only when an Ed25519 CertificateVerify names a non-Ed25519 leaf — a malformed peer we don't mirror
397}

401// A PEM block's DER bytes (strict base64 — a non-alphabet byte rejects the block, like x509).
402// @p label is e.g. "CERTIFICATE" or "PRIVATE KEY". Returns "" if the block is absent/malformed.
403std::string pem_block(const std::string& pem, const std::string& label) {
404 const std::string begin = "-----BEGIN " + label + "-----";
405 const std::string end = "-----END " + label + "-----";
406 const std::size_t s = pem.find(begin);
407 if (s == std::string::npos) return "";
408 const std::size_t b = s + begin.size();
409 const std::size_t e = pem.find(end, b);
410 if (e == std::string::npos) return "";
411 return hashlib::base64_decode(pem.substr(b, e - b), /*strict=*/true);
412}

414// The 32-byte Ed25519 seed from a PKCS#8 private key DER: the id-Ed25519 AlgorithmIdentifier
415// (30 05 06 03 2B 65 70) followed by 04 22 04 20 (OCTET STRING { OCTET STRING[32] }) and the seed.
416std::string ed25519_seed_from_pkcs8(std::string_view der) {
417 static const unsigned char kPat[] = {0x30, 0x05, 0x06, 0x03, 0x2B, 0x65, 0x70,
418 0x04, 0x22, 0x04, 0x20};
419 for (std::size_t i = 0; i + sizeof kPat + 32 <= der.size(); ++i) {
420 if (std::memcmp(der.data() + i, kPat, sizeof kPat) == 0) {
421 return std::string(der.substr(i + sizeof kPat, 32));
422 }
423 }
424 return "";
425}

427// Parse a ClientHello: choose a cipher suite we support (ChaCha20 preferred), extract the client's
428// X25519 key share + its legacy_session_id (echoed in ServerHello), confirm it offered TLS 1.3.
429bool parse_client_hello(std::string_view msg, std::string& client_pub_raw, unsigned& chosen_suite,
430 std::string& session_id) {
431 client_pub_raw.clear(); // never leave stale out-params on a rejected/partial parse
432 session_id.clear();
433 if (msg.size() < 4 || static_cast<unsigned char>(msg[0]) != 1) return false; // client_hello
434 std::string_view b = msg.substr(4);
435 std::size_t i = 2 + 32; // legacy_version + random
436 if (b.size() < i + 1) return false;
437 const unsigned sid_len = static_cast<unsigned char>(b[i]);
438 i += 1;
439 if (b.size() < i + sid_len) return false;
440 session_id = std::string(b.substr(i, sid_len)); // must be echoed back verbatim
441 i += sid_len;
442 if (b.size() < i + 2) return false;
443 const unsigned cs_len = get16(b, i);
444 i += 2;
445 if (b.size() < i + cs_len) return false;
446 bool has_chacha = false, has_aes = false;
447 for (std::size_t j = 0; j + 2 <= cs_len; j += 2) {
448 const unsigned suite = get16(b, i + j);
449 if (suite == 0x1303) has_chacha = true;
450 if (suite == 0x1301) has_aes = true;
451 }
452 i += cs_len;
453 if (has_chacha) chosen_suite = 0x1303;
454 else if (has_aes) chosen_suite = 0x1301;
455 else return false; // no cipher suite in common
456 if (b.size() < i + 1) return false;
457 const unsigned comp_len = static_cast<unsigned char>(b[i]);
458 i += 1 + comp_len;
459 if (b.size() < i + 2) return false;
460 const unsigned ext_len = get16(b, i);
461 i += 2;
462 const std::size_t ext_end = i + ext_len;
463 bool saw_13 = false;
464 while (i + 4 <= ext_end && ext_end <= b.size()) {
465 const unsigned etype = get16(b, i);
466 const unsigned elen = get16(b, i + 2);
467 i += 4;
468 if (i + elen > b.size()) return false;
469 if (etype == 43 && elen >= 1) { // supported_versions (list): look for 0x0304
470 const unsigned n = static_cast<unsigned char>(b[i]);
471 for (std::size_t j = 0; j + 2 <= n && i + 1 + j + 2 <= b.size(); j += 2) {
472 if (get16(b, i + 1 + j) == 0x0304) saw_13 = true;
473 }
474 }
475 if (etype == 51 && elen >= 2) { // key_share (list): find our x25519 (0x001d)
476 const unsigned ks_len = get16(b, i);
477 std::size_t j = i + 2;
478 const std::size_t ks_end = i + 2 + ks_len;
479 while (j + 4 <= ks_end && ks_end <= b.size()) {
480 const unsigned group = get16(b, j);
481 const unsigned klen = get16(b, j + 2);
482 if (group == 0x001d && klen == 32 && j + 4 + 32 <= b.size()) {
483 client_pub_raw = std::string(b.substr(j + 4, 32));
484 }
485 j += 4 + klen;
486 }
487 }
488 i += elen;
489 }
490 return saw_13 && client_pub_raw.size() == 32;
491}

493// Build a ServerHello: echo the client's session id, our chosen suite, our X25519 key share.
494std::string build_server_hello(std::string_view session_id, unsigned suite,
495 std::string_view pub_raw) {
496 std::string body;
497 put16(body, 0x0303); // legacy_version
498 body += random_bytes(32); // random
499 body.push_back(static_cast<char>(session_id.size()));
500 body += session_id; // echo legacy_session_id (RFC 8446 §4.1.3)
501 put16(body, suite); // cipher_suite
502 body.push_back(0); // legacy_compression_method (null)
503
504 std::string ext;
505 put16(ext, 43); // supported_versions: TLS 1.3
506 put16(ext, 2);
507 put16(ext, 0x0304);
508 { // key_share (51): our X25519 KeyShareEntry
509 std::string entry;
510 put16(entry, 0x001d);
511 put16(entry, 32);
512 entry += pub_raw;
513 put16(ext, 51);
514 put16(ext, static_cast<unsigned>(entry.size()));
515 ext += entry;
516 }
517 put16(body, static_cast<unsigned>(ext.size()));
518 body += ext;
519
520 std::string msg;
521 msg.push_back(2); // server_hello
522 put24(msg, static_cast<unsigned>(body.size()));
523 msg += body;
524 return msg;
525}

527// The TLS 1.3 SERVER handshake over connected fd @p fd, presenting @p cert_pem (an Ed25519 leaf)
528// and proving possession with @p key_pem (its PKCS#8 Ed25519 private key). Mirror of handshake():
529// same key schedule and record I/O, roles reversed — we SIGN CertificateVerify instead of verifying
530// it, and send the certificate flight. Returns a session handle (>= 1) or -1 (see last_error()).
531long long server_handshake(long long fd, const std::string& cert_pem, const std::string& key_pem) {
532 t_error.clear();
533
534 const std::string leaf_der = pem_block(cert_pem, "CERTIFICATE");
535 if (leaf_der.empty()) {
536 fail("tls: server certificate PEM is missing or malformed");
537 return -1;
538 }
539 const std::string seed = ed25519_seed_from_pkcs8(pem_block(key_pem, "PRIVATE KEY"));
540 if (seed.size() != 32) {
541 fail("tls: server private key is not a PKCS#8 Ed25519 key");
542 return -1;
543 }
544 if (ed25519_spki_key(leaf_der).size() != 32) {
545 fail("tls: server certificate is not an Ed25519 leaf (only Ed25519 server certs are supported)");
546 return -1;
547 }
548
549 // ClientHello (plaintext; tolerate a leading ChangeCipherSpec compat record).
550 std::string buffer, payload;
551 unsigned rtype = 0;
552 for (;;) {
553 if (!read_record(fd, buffer, rtype, payload)) {
554 fail("tls: connection closed before ClientHello");
555 return -1;
556 }
557 if (rtype == 20) continue;
558 if (rtype != 22) {
559 fail("tls: expected a ClientHello");
560 return -1;
561 }
562 break;
563 }
564 std::string client_pub_raw, session_id;
565 unsigned suite = 0;
566 if (!parse_client_hello(payload, client_pub_raw, suite, session_id)) {
567 fail("tls: malformed ClientHello (or no TLS 1.3 / X25519 / shared cipher suite)");
568 return -1;
569 }
570 // The server offers only the SHA-256 suites (ChaCha20 / AES-128-GCM), so the key schedule is SHA-256.
571 const Aead aead = (suite == 0x1301) ? Aead::Aes128 : Aead::Chacha20;
572 std::string transcript = payload; // ClientHello
573
574 // Our ephemeral X25519 key pair, and ServerHello.
575 const std::string priv_raw = random_bytes(32);
576 if (priv_raw.size() != 32) {
577 fail("tls: system random unavailable"); // LCOV_EXCL_LINE: getentropy failure — unreachable on a working host
578 return -1; // LCOV_EXCL_LINE
579 }
580 const std::string priv_hex = to_hex(priv_raw);
581 const std::string pub_raw = from_hex(x25519::x25519_base(priv_hex));
582 const std::string server_hello = build_server_hello(session_id, suite, pub_raw);
583 transcript += server_hello;
584 if (!write_record(fd, 22, server_hello)) {
585 fail("tls: cannot send ServerHello"); // LCOV_EXCL_LINE: a mid-handshake socket write failure
586 return -1; // LCOV_EXCL_LINE
587 }
588 write_record(fd, 20, std::string(1, '\x01')); // middlebox-compat ChangeCipherSpec (not in transcript)
589
590 // Key schedule (identical to the client; the transcript now spans ClientHello + ServerHello).
591 const std::string shared_hex = x25519::x25519(priv_hex, to_hex(client_pub_raw));
592 if (shared_hex.empty()) {
593 fail("tls: invalid client key share");
594 return -1;
595 }
596 const std::string zeros(32, '\0');
597 const std::string early = hashlib::hkdf_extract(std::string(), zeros);
598 const std::string derived = derive_secret_impl(early, "derived", "");
599 const std::string hs_secret = hashlib::hkdf_extract(derived, from_hex(shared_hex));
600 const std::string c_hs = derive_secret_impl(hs_secret, "c hs traffic", transcript);
601 const std::string s_hs = derive_secret_impl(hs_secret, "s hs traffic", transcript);
602 Keys server_keys = traffic_keys(s_hs, aead, false); // we SEND under the server secret
603 Keys client_keys = traffic_keys(c_hs, aead, false); // we RECEIVE under the client secret
604
605 // Encrypted flight: EncryptedExtensions (empty), Certificate, CertificateVerify, Finished —
606 // each sealed as its own handshake record (inner type 22).
607 std::string ee;
608 ee.push_back(8); // encrypted_extensions
609 put24(ee, 2);
610 put16(ee, 0); // extensions: empty
611 transcript += ee;
612 if (!seal_record(fd, server_keys, 22, ee)) {
613 fail("tls: cannot send EncryptedExtensions"); // LCOV_EXCL_LINE: a mid-handshake socket write failure
614 return -1; // LCOV_EXCL_LINE
615 }
616
617 std::string cert_msg;
618 {
619 std::string entry; // one CertificateEntry
620 put24(entry, static_cast<unsigned>(leaf_der.size()));
621 entry += leaf_der;
622 put16(entry, 0); // per-cert extensions: none
623 std::string cbody;
624 cbody.push_back(0); // certificate_request_context: empty
625 put24(cbody, static_cast<unsigned>(entry.size()));
626 cbody += entry;
627 cert_msg.push_back(11); // certificate
628 put24(cert_msg, static_cast<unsigned>(cbody.size()));
629 cert_msg += cbody;
630 }
631 transcript += cert_msg;
632 if (!seal_record(fd, server_keys, 22, cert_msg)) {
633 fail("tls: cannot send Certificate"); // LCOV_EXCL_LINE: a mid-handshake socket write failure
634 return -1; // LCOV_EXCL_LINE
635 }
636
637 std::string cv_msg;
638 {
639 // Same signed content the client verifies: 64 spaces, the context string, a NUL, then the
640 // transcript hash through Certificate. Ed25519 signs the content directly.
641 std::string signed_content(64, ' ');
642 signed_content += "TLS 1.3, server CertificateVerify";
643 signed_content.push_back('\0');
644 signed_content += hashlib::sha256_digest(transcript);
645 const std::string sig = from_hex(ed25519::sign(to_hex(seed), signed_content));
646 std::string vbody;
647 put16(vbody, 0x0807); // ed25519
648 put16(vbody, static_cast<unsigned>(sig.size()));
649 vbody += sig;
650 cv_msg.push_back(15); // certificate_verify
651 put24(cv_msg, static_cast<unsigned>(vbody.size()));
652 cv_msg += vbody;
653 }
654 transcript += cv_msg;
655 if (!seal_record(fd, server_keys, 22, cv_msg)) {
656 fail("tls: cannot send CertificateVerify"); // LCOV_EXCL_LINE: a mid-handshake socket write failure
657 return -1; // LCOV_EXCL_LINE
658 }
659
660 std::string fin_msg;
661 {
662 const std::string finished_key = expand_label_impl(s_hs, "finished", "", 32);
663 const std::string verify =
664 hashlib::hmac_sha256(finished_key, hashlib::sha256_digest(transcript));
665 fin_msg.push_back(20); // finished
666 put24(fin_msg, static_cast<unsigned>(verify.size()));
667 fin_msg += verify;
668 }
669 if (!seal_record(fd, server_keys, 22, fin_msg)) {
670 fail("tls: cannot send server Finished"); // LCOV_EXCL_LINE: a mid-handshake socket write failure
671 return -1; // LCOV_EXCL_LINE
672 }
673 transcript += fin_msg;
674
675 // Application traffic secrets (transcript through the server's Finished).
676 const std::string derived2 = derive_secret_impl(hs_secret, "derived", "");
677 const std::string master = hashlib::hkdf_extract(derived2, zeros);
678 const std::string c_ap = derive_secret_impl(master, "c ap traffic", transcript);
679 const std::string s_ap = derive_secret_impl(master, "s ap traffic", transcript);
680
681 // Client Finished (encrypted under the client handshake keys); verify its MAC over the
682 // transcript through the server Finished.
683 const std::string c_finished_key = expand_label_impl(c_hs, "finished", "", 32);
684 const std::string expect =
685 hashlib::hmac_sha256(c_finished_key, hashlib::sha256_digest(transcript));
686 bool client_finished = false;
687 while (!client_finished) {
688 if (!read_record(fd, buffer, rtype, payload)) {
689 fail("tls: connection closed before the client Finished");
690 return -1;
691 }
692 if (rtype == 20) continue; // ChangeCipherSpec compat
693 if (rtype == 21) {
694 fail("tls: client alert during the handshake — " + alert_text(payload));
695 return -1;
696 }
697 if (rtype != 23) {
698 fail("tls: unexpected plaintext record awaiting the client Finished");
699 return -1;
700 }
701 unsigned inner_type = 0;
702 std::string content;
703 if (!open_record(client_keys, payload, inner_type, content)) {
704 fail("tls: client Finished failed authentication");
705 return -1;
706 }
707 if (inner_type != 22 || content.size() < 4 || static_cast<unsigned char>(content[0]) != 20) {
708 fail("tls: expected a client Finished"); // LCOV_EXCL_LINE: a key-scheduled malicious client we do not mirror
709 return -1; // LCOV_EXCL_LINE
710 }
711 // cppcheck-suppress stlcstrConstructor // (ptr,len) subview of content — not a c_str() copy
712 const std::string_view got(content.data() + 4, content.size() - 4);
713 unsigned char diff = (got.size() == expect.size()) ? 0 : 1; // constant-time MAC compare
714 for (std::size_t i = 0; i < expect.size(); ++i) {
715 const unsigned char g = i < got.size() ? static_cast<unsigned char>(got[i]) : 0;
716 diff |= g ^ static_cast<unsigned char>(expect[i]);
717 }
718 if (diff != 0) {
719 fail("tls: client Finished MAC mismatch — handshake transcript tampered"); // LCOV_EXCL_LINE: a key-scheduled malicious client we do not mirror
720 return -1; // LCOV_EXCL_LINE
721 }
722 client_finished = true;
723 }
724
725 Session s;
726 s.fd = fd;
727 s.client_keys = traffic_keys(s_ap, aead, false); // our sending direction (server app secret)
728 s.server_keys = traffic_keys(c_ap, aead, false); // the peer's direction (client app secret)
729 s.read_buffer = std::move(buffer);
730 const std::lock_guard<std::mutex> lock(g_mutex);
731 const long long handle = g_next_handle++;
732 g_sessions[handle] = std::move(s);
733 return handle;
734}

738// The system CA trust store, parsed once and reused (loading ~150 CA certs on every handshake
739// would be wasteful). Thread-safe initialization via the C++ function-local static.
740const x509::TrustStore& default_trust() {
741 static const x509::TrustStore store = x509::load_trust("");
742 return store;
743}
744
745long long handshake(long long fd, const std::string& server_name, bool insecure,
746 const std::string& ca_file) {
747 t_error.clear();
748
749 // Ephemeral X25519 key pair.
750 const std::string priv_raw = random_bytes(32);
751 if (priv_raw.size() != 32) {
752 fail("tls: system random unavailable");
753 return -1;
754 }
755 const std::string priv_hex = to_hex(priv_raw);
756 const std::string pub_hex = x25519::x25519_base(priv_hex);
757 const std::string pub_raw = from_hex(pub_hex);
758
759 std::string transcript; // concatenated handshake MESSAGES (no record headers)
760 const std::string client_hello = build_client_hello(server_name, pub_raw);
761 transcript += client_hello;
762 if (!write_record(fd, 22, client_hello)) {
763 fail("tls: cannot send ClientHello");
764 return -1;
765 }
766
767 // ServerHello (plaintext record, type 22; tolerate ChangeCipherSpec compat records).
768 std::string buffer, payload;
769 unsigned rtype = 0;
770 std::string server_pub_raw;
771 for (;;) {
772 if (!read_record(fd, buffer, rtype, payload)) {
773 fail("tls: connection closed before ServerHello");
774 return -1;
775 }
776 if (rtype == 20) continue; // ChangeCipherSpec (compat) — ignored
777 if (rtype == 21) {
778 fail("tls: server sent an alert instead of ServerHello — " + alert_text(payload));
779 return -1;
780 }
781 if (rtype != 22) {
782 fail("tls: unexpected record before ServerHello");
783 return -1;
784 }
785 break;
786 }
787 unsigned chosen_suite = 0;
788 if (!parse_server_hello(payload, server_pub_raw, chosen_suite)) {
789 fail("tls: malformed ServerHello (or the server refused TLS 1.3 + our cipher suites)");
790 return -1;
791 }
792 // The negotiated suite fixes both the record AEAD and the key-schedule hash.
793 const bool sha384 = (chosen_suite == 0x1302); // TLS_AES_256_GCM_SHA384 → SHA-384 schedule
794 const Aead aead = chosen_suite == 0x1302 ? Aead::Aes256
795 : chosen_suite == 0x1301 ? Aead::Aes128
796 : Aead::Chacha20;
797 transcript += payload;
798
799 // Key schedule through the handshake secrets.
800 const std::string shared_hex = x25519::x25519(priv_hex, to_hex(server_pub_raw));
801 if (shared_hex.empty()) {
802 fail("tls: invalid server key share");
803 return -1;
804 }
805 const std::string zeros(sha384 ? 48 : 32, '\0'); // HashLen zero bytes for Extract
806 const std::string early = ks_extract(sha384, std::string(), zeros);
807 const std::string derived = derive_secret_impl(early, "derived", "", sha384);
808 const std::string hs_secret = ks_extract(sha384, derived, from_hex(shared_hex));
809 const std::string c_hs = derive_secret_impl(hs_secret, "c hs traffic", transcript, sha384);
810 const std::string s_hs = derive_secret_impl(hs_secret, "s hs traffic", transcript, sha384);
811 Keys client_keys = traffic_keys(c_hs, aead, sha384);
812 Keys server_keys = traffic_keys(s_hs, aead, sha384);
813
814 // Encrypted handshake flight: EncryptedExtensions, Certificate, CertificateVerify, Finished.
815 std::string handshake_bytes; // decrypted, possibly spanning records
816 std::string cert_der;
817 std::vector<std::string> cert_chain; // the full certificate chain (leaf first) for validation
818 bool verified_cert = false, server_finished = false;
819 std::string transcript_at_cv, transcript_at_finished;
820 // Total-flight cap: the server's handshake messages accumulate into transcript / handshake_bytes /
821 // cert_chain BEFORE the certificate is validated, so a hostile (or MITM) server could otherwise
822 // stream unbounded records and exhaust memory pre-auth. A real TLS 1.3 flight — even a long cert
823 // chain of RSA-4096 leaves — is well under 256 KiB; cap there and fail closed beyond it.
824 constexpr std::size_t kMaxHandshakeFlight = 256 * 1024;
825 std::size_t flight_bytes = 0;
826 while (!server_finished) {
827 if (!read_record(fd, buffer, rtype, payload)) {
828 fail("tls: connection closed during the handshake");
829 return -1;
830 }
831 if (rtype == 20) continue; // compat ChangeCipherSpec
832 if (rtype == 21) {
833 fail("tls: server alert during the handshake — " + alert_text(payload));
834 return -1;
835 }
836 if (rtype != 23) {
837 fail("tls: unexpected plaintext record during the encrypted handshake");
838 return -1;
839 }
840 unsigned inner_type = 0;
841 std::string content;
842 if (!open_record(server_keys, payload, inner_type, content)) {
843 fail("tls: handshake record failed authentication");
844 return -1;
845 }
846 if (inner_type == 21) {
847 fail("tls: server alert during the handshake — " + alert_text(content));
848 return -1;
849 }
850 if (inner_type != 22) {
851 fail("tls: unexpected inner record type during the handshake");
852 return -1;
853 }
854 flight_bytes += content.size();
855 if (flight_bytes > kMaxHandshakeFlight) {
856 fail("tls: server handshake flight too large");
857 return -1;
858 }
859 handshake_bytes += content;
860
861 // Drain complete handshake messages from the reassembly buffer.
862 while (handshake_bytes.size() >= 4) {
863 const unsigned mtype = static_cast<unsigned char>(handshake_bytes[0]);
864 const unsigned mlen = get24(handshake_bytes, 1);
865 if (handshake_bytes.size() < 4 + mlen) break;
866 const std::string msg = handshake_bytes.substr(0, 4 + mlen);
867 handshake_bytes.erase(0, 4 + mlen);
868
869 if (mtype == 11 && msg.size() > 4 + 4 + 3 + 3) { // Certificate
870 // certificate_request_context (1 byte, empty) + the cert list. Walk every
871 // CertificateEntry (3-byte length | cert DER | 2-byte extensions | extensions)
872 // so the FULL chain is available for path validation, not just the leaf.
873 std::size_t i = 4 + 1 + 3; // header + context length byte + cert-list length
874 while (i + 3 <= msg.size()) {
875 const unsigned clen = get24(msg, i);
876 i += 3;
877 if (i + clen > msg.size()) break;
878 cert_chain.push_back(msg.substr(i, clen));
879 i += clen;
880 if (i + 2 > msg.size()) break;
881 i += 2 + get16(msg, i); // skip the per-certificate extensions
882 }
883 if (!cert_chain.empty()) cert_der = cert_chain[0]; // the leaf signs CertificateVerify
884 transcript_at_cv = transcript + msg; // transcript THROUGH Certificate
885 }
886 if (mtype == 15) { // CertificateVerify
887 // cppcheck-suppress stlcstrConstructor // (ptr,len) subview of msg — not a c_str() copy
888 const std::string_view body(msg.data() + 4, msg.size() - 4);
889 if (body.size() < 4) {
890 fail("tls: malformed CertificateVerify");
891 return -1;
892 }
893 const unsigned alg = get16(body, 0);
894 const unsigned sig_len = get16(body, 2);
895 if (body.size() < 4 + sig_len) {
896 fail("tls: malformed CertificateVerify");
897 return -1;
898 }
899 // The signed content (same for every algorithm): 64 spaces, the
900 // context string, a NUL, then the handshake transcript hash.
901 std::string signed_content(64, ' ');
902 signed_content += "TLS 1.3, server CertificateVerify";
903 signed_content.push_back('\0');
904 signed_content += ks_digest(sha384, transcript_at_cv); // transcript hash = suite hash
905 const std::string sig(body.substr(4, sig_len));
906 if (alg == 0x0807) { // ed25519 (signs the message directly)
907 const std::string spki = ed25519_spki_key(cert_der);
908 if (spki.size() != 32) {
909 fail("tls: certificate key is not Ed25519");
910 return -1;
911 }
912 if (!ed25519::verify(to_hex(spki), signed_content, to_hex(sig))) {
913 fail("tls: server CertificateVerify signature is INVALID");
914 return -1;
915 }
916 } else if (alg == 0x0403) { // ecdsa_secp256r1_sha256 (signs SHA-256(content))
917 const std::string point = p256::spki_ec_point(cert_der);
918 if (point.size() != 64) {
919 fail("tls: certificate key is not P-256 EC");
920 return -1;
921 }
922 if (!p256::verify_der(point, hashlib::sha256_digest(signed_content), sig)) {
923 fail("tls: server CertificateVerify (ECDSA P-256) is INVALID");
924 return -1;
925 }
926 } else if (alg == 0x0503) { // ecdsa_secp384r1_sha384 (signs SHA-384(content))
927 // The signature scheme's hash (SHA-384) is independent of the transcript hash
928 // inside signed_content (which is the negotiated suite's hash) — RFC 8446 §4.4.3.
929 const std::string point = p384::spki_ec_point(cert_der);
930 if (point.size() != 96) {
931 fail("tls: certificate key is not P-384 EC");
932 return -1;
933 }
934 if (!p384::verify_der(point, hashlib::sha384_digest(signed_content), sig)) {
935 fail("tls: server CertificateVerify (ECDSA P-384) is INVALID");
936 return -1;
937 }
938 } else if (alg == 0x0804) { // rsa_pss_rsae_sha256 (RSA leaf certificate)
939 // RSA-PSS verifies the signed_content directly (it hashes with SHA-256 internally),
940 // using the RSA public key extracted from the leaf cert's SubjectPublicKeyInfo.
941 if (!rsa::verify_pss_sha256(cert_der, signed_content, sig)) {
942 fail("tls: server CertificateVerify (RSA-PSS SHA-256) is INVALID");
943 return -1;
944 }
945 } else { // LCOV_EXCL_LINE: reached only if a server sends a CertificateVerify whose algorithm ignores our advertised signature_algorithms — a non-conformant peer we don't mirror
946 fail("tls: server certificate uses an algorithm cheatah cannot verify yet " // LCOV_EXCL_LINE
947 "(Ed25519, ECDSA P-256/P-384, and RSA-PSS SHA-256 are supported) — refusing an "
948 "unauthenticated connection");
949 return -1;
950 }
951 verified_cert = true;
952 }
953 if (mtype == 20) { // Finished
954 const std::string finished_key =
955 expand_label_impl(s_hs, "finished", "", sha384 ? 48 : 32, sha384);
956 const std::string expect =
957 ks_hmac(sha384, finished_key, ks_digest(sha384, transcript));
958 // cppcheck-suppress stlcstrConstructor // (ptr,len) subview of msg — not a c_str() copy
959 const std::string_view got(msg.data() + 4, msg.size() - 4);
960 // Constant-time MAC compare (parity with the AEAD tag check): always scan all
961 // HashLen bytes of the secret `expect` (32 for SHA-256, 48 for SHA-384), never
962 // early-exiting on a mismatching byte, so timing cannot reveal a partial match.
963 unsigned char diff = (got.size() == expect.size()) ? 0 : 1;
964 for (std::size_t i = 0; i < expect.size(); ++i) {
965 const unsigned char g =
966 i < got.size() ? static_cast<unsigned char>(got[i]) : 0;
967 diff |= g ^ static_cast<unsigned char>(expect[i]);
968 }
969 if (diff != 0) {
970 fail("tls: server Finished MAC mismatch — handshake transcript tampered");
971 return -1;
972 }
973 transcript_at_finished = transcript + msg;
974 server_finished = true;
975 }
976 transcript += msg;
977 }
978 }
979 if (!verified_cert) {
980 fail("tls: server never proved possession of its certificate key");
981 return -1;
982 }
983
984 // AUTHENTICATE THE SERVER'S IDENTITY (unless the caller opted out of verification): build the
985 // presented chain to a trusted CA, match the hostname against the leaf's SAN, and check the
986 // validity dates. Key possession alone (above) does not prove identity — this is what stops an
987 // active man-in-the-middle presenting any certificate.
988 if (!insecure) {
989 x509::TrustStore custom;
990 const x509::TrustStore* store = &default_trust();
991 if (!ca_file.empty()) {
992 custom = x509::load_trust(ca_file);
993 store = &custom;
994 }
995 std::string verr;
996 if (!x509::validate(cert_chain, server_name, *store, static_cast<long long>(std::time(nullptr)),
997 verr)) {
998 fail("tls: certificate validation failed — " + verr);
999 return -1;
1000 }
1001 }
1002
1003 // Application traffic secrets (transcript through server Finished), then OUR Finished
1004 // (sent under the handshake keys, with the transcript through the server's Finished).
1005 const std::string derived2 = derive_secret_impl(hs_secret, "derived", "", sha384);
1006 const std::string master = ks_extract(sha384, derived2, zeros);
1007 const std::string c_ap = derive_secret_impl(master, "c ap traffic", transcript_at_finished, sha384);
1008 const std::string s_ap = derive_secret_impl(master, "s ap traffic", transcript_at_finished, sha384);
1009
1010 const std::string c_finished_key =
1011 expand_label_impl(c_hs, "finished", "", sha384 ? 48 : 32, sha384);
1012 const std::string verify = ks_hmac(sha384, c_finished_key, ks_digest(sha384, transcript));
1013 std::string fin_msg;
1014 fin_msg.push_back(20);
1015 put24(fin_msg, static_cast<unsigned>(verify.size()));
1016 fin_msg += verify;
1017 if (!seal_record(fd, client_keys, 22, fin_msg)) {
1018 fail("tls: cannot send client Finished");
1019 return -1;
1020 }
1021
1022 Session s;
1023 s.fd = fd;
1024 s.client_keys = traffic_keys(c_ap, aead, sha384);
1025 s.server_keys = traffic_keys(s_ap, aead, sha384);
1026 s.read_buffer = std::move(buffer); // bytes already pulled off the socket stay with us
1027
1028 const std::lock_guard<std::mutex> lock(g_mutex);
1029 const long long handle = g_next_handle++;
1030 g_sessions[handle] = std::move(s);
1031 return handle;
1032}

1046long long send(long long session, const std::string& data) {
1047 t_error.clear();
1048 // Lock ONLY for the map lookup (see recv): the socket write below runs without
1049 // the global lock so concurrent sessions don't serialize on each other.
1050 Session* sp = nullptr;
1051 {
1052 const std::lock_guard<std::mutex> lock(g_mutex);
1053 const auto it = g_sessions.find(session);
1054 if (it == g_sessions.end() || it->second.closed) {
1055 fail("tls: unknown or closed session");
1056 return -1;
1057 }
1058 sp = &it->second;
1059 }
1060 Session& s = *sp;
1061 // Respect the 16 KiB record plaintext bound.
1062 std::string_view rest = data;
1063 while (!rest.empty()) {
1064 const std::size_t n = std::min<std::size_t>(rest.size(), 16384);
1065 if (!seal_record(s.fd, s.client_keys, 23, rest.substr(0, n))) {
1066 fail("tls: send failed");
1067 return -1;
1068 }
1069 rest.remove_prefix(n);
1070 }
1071 return 0;
1072}

1074std::string recv(long long session, long long bufsize) {
1075 t_error.clear();
1076 if (bufsize <= 0) return "";
1077 // Guard ONLY the map lookup. The per-session buffers (read_buffer/app_pending)
1078 // and socket are owned by this session's single reader thread, so the blocking
1079 // record I/O below runs WITHOUT the global lock — otherwise one session's
1080 // blocking recv would serialize (and at shutdown, starve) every other session's
1081 // recv/send. A std::map node address is stable until that node is erased, and a
1082 // session is erased only by its own owner (after this loop), so the pointer is
1083 // valid for this call. (g_mutex still serializes find/insert/erase on the map.)
1084 Session* sp = nullptr;
1085 {
1086 const std::lock_guard<std::mutex> lock(g_mutex);
1087 const auto it = g_sessions.find(session);
1088 if (it == g_sessions.end()) {
1089 fail("tls: unknown session");
1090 return "";
1091 }
1092 sp = &it->second;
1093 }
1094 Session& s = *sp;
1095 while (s.app_pending.empty() && !s.closed) {
1096 unsigned rtype = 0;
1097 std::string payload;
1098 if (!read_record(s.fd, s.read_buffer, rtype, payload)) {
1099 s.closed = true; // peer EOF (or socket timeout) — surfaced as ""
1100 break;
1101 }
1102 if (rtype == 20) continue; // stray compat ChangeCipherSpec
1103 if (rtype == 21) { // plaintext alert (illegal post-handshake, but final)
1104 s.closed = true;
1105 break;
1106 }
1107 if (rtype != 23) continue; // ignore anything else
1108 unsigned inner_type = 0;
1109 std::string content;
1110 if (!open_record(s.server_keys, payload, inner_type, content)) {
1111 fail("tls: record failed authentication");
1112 s.closed = true;
1113 break;
1114 }
1115 if (inner_type == 23) {
1116 s.app_pending += content;
1117 } else if (inner_type == 21) { // alert: close_notify (or fatal) ends the stream
1118 s.closed = true;
1119 } else if (inner_type == 22) {
1120 // Post-handshake messages: NewSessionTicket(4) is ignored; a KeyUpdate(24)
1121 // would change the peer's keys — unsupported, so end the stream rather than
1122 // silently fail to decrypt what follows.
1123 if (!content.empty() && static_cast<unsigned char>(content[0]) == 24) {
1124 fail("tls: peer KeyUpdate is not supported");
1125 s.closed = true;
1126 }
1127 }
1128 }
1129 const std::size_t n = std::min<std::size_t>(s.app_pending.size(),
1130 static_cast<std::size_t>(bufsize));
1131 const std::string out = s.app_pending.substr(0, n);
1132 s.app_pending.erase(0, n);
1133 return out;
1134}

1136long long close(long long session) {
1137 t_error.clear();
1138 const std::lock_guard<std::mutex> lock(g_mutex);
1139 const auto it = g_sessions.find(session);
1140 if (it == g_sessions.end()) return -1;
1141 if (!it->second.closed) {
1142 const std::string close_notify = {1, 0}; // warning, close_notify
1143 seal_record(it->second.fd, it->second.client_keys, 21, close_notify);
1144 }
1145 g_sessions.erase(it);
1146 return 0;
1147}
1148
1149long long shutdown(long long session) {
1150 t_error.clear();
1151 const std::lock_guard<std::mutex> lock(g_mutex);
1152 const auto it = g_sessions.find(session);
1153 if (it == g_sessions.end()) return -1;
1154 // Wake a reader blocked in recv() WITHOUT erasing the session (that stays the
1155 // owner's job via close(), after it has joined the reader). Just half-close the
1156 // socket so the blocking recv returns EOF.
1157 return socket::shutdown(it->second.fd);
1158}
