16std::string hex_of(std::string_view raw) {
17 static constexpr char kHex[] = "0123456789abcdef";
18 std::string out;
19 for (const char ch : raw) {
20 out.push_back(kHex[static_cast<unsigned char>(ch) >> 4]);
21 out.push_back(kHex[static_cast<unsigned char>(ch) & 0xF]);
22 }
23 return out;
24}

26void be16(std::string& o, unsigned v) {
27 o.push_back(static_cast<char>((v >> 8) & 0xFF));
28 o.push_back(static_cast<char>(v & 0xFF));
29}

31// A ClientHello handshake message with control over the fields parse_client_hello gates on:
32// the offered cipher suites, whether supported_versions advertises TLS 1.3, whether a valid
33// X25519 key_share is present, and the legacy_session_id length (echoed by the server).
34std::string make_client_hello(const std::vector<unsigned>& suites, bool tls13, bool x25519,
35 unsigned sid_len = 0) {
36 std::string body;
37 be16(body, 0x0303); // legacy_version
38 body.append(32, 'R'); // random
39 body.push_back(static_cast<char>(sid_len));
40 body.append(sid_len, 'S'); // legacy_session_id
41 be16(body, static_cast<unsigned>(suites.size() * 2));
42 for (unsigned s : suites) be16(body, s);
43 body.push_back(1); // legacy_compression_methods length
44 body.push_back(0); // null compression
45 std::string ext;
46 if (tls13) { // supported_versions: [TLS 1.3]
47 be16(ext, 43);
48 be16(ext, 3);
49 ext.push_back(2);
50 be16(ext, 0x0304);
51 }
52 if (x25519) { // key_share: one X25519 entry
53 std::string entry;
54 be16(entry, 0x001d);
55 be16(entry, 32);
56 entry.append(32, 'K');
57 std::string ks;
58 be16(ks, static_cast<unsigned>(entry.size()));
59 ks += entry;
60 be16(ext, 51);
61 be16(ext, static_cast<unsigned>(ks.size()));
62 ext += ks;
63 }
64 be16(body, static_cast<unsigned>(ext.size()));
65 body += ext;
66 std::string msg;
67 msg.push_back(1); // client_hello
68 msg.push_back(static_cast<char>((body.size() >> 16) & 0xFF));
69 msg.push_back(static_cast<char>((body.size() >> 8) & 0xFF));
70 msg.push_back(static_cast<char>(body.size() & 0xFF));
71 msg += body;
72 return msg;
73}
74} // namespace

76// RFC 8448 §3: the early secret HKDF-Extract(0, 0^32) and the "derived" secret from it.
77TEST(CheatahTls, KeySchedule) {
78 const std::string zeros(32, '\0');
79 const std::string early = cheatah::hashlib::hkdf_extract(std::string(), zeros);
80 EXPECT_EQ(hex_of(early), "33ad0a1c607ec03b09e6cd9893680ce210adf300aa1f2660e1b22e10f170f92a");
81 const std::string derived = cheatah::tls::detail::derive_secret(early, "derived", "");
82 EXPECT_EQ(hex_of(derived), "6f2615a108c702c5678f54fc9dbab69716c076189c48250cebeac3576c3611ba");
83}

85// HKDF-Expand-Label structure: deterministic, length-exact, label-sensitive.
86TEST(CheatahTls, ExpandLabel) {
87 const std::string secret(32, '\x42');
88 const std::string a = cheatah::tls::detail::expand_label(secret, "key", "", 32);
89 const std::string b = cheatah::tls::detail::expand_label(secret, "iv", "", 12);
90 EXPECT_EQ(a.size(), std::size_t{32});
91 EXPECT_EQ(b.size(), std::size_t{12});
92 EXPECT_NE(a.substr(0, 12), b); // different labels -> unrelated output
93 EXPECT_EQ(a, cheatah::tls::detail::expand_label(secret, "key", "", 32)); // deterministic
94}

96// The server's ClientHello parser: accept a well-formed hello (preferring ChaCha20), fall back to
97// AES-128-GCM when only that is offered, and reject every malformed / unsupported shape. Driving the
98// parser directly (a test seam) covers each refusal branch without a live network peer.
99TEST(CheatahTls, ParseClientHelloAcceptsAndNegotiates) {
100 namespace d = cheatah::tls::detail;
101 std::string pub, sid;
102 unsigned suite = 0;
103
104 // Both suites offered -> ChaCha20-Poly1305 preferred; the X25519 share + session id come back.
105 ASSERT_TRUE(d::parse_client_hello(make_client_hello({0x1303, 0x1301}, true, true, 4),
106 pub, suite, sid));
107 EXPECT_EQ(suite, 0x1303u);
108 EXPECT_EQ(pub, std::string(32, 'K'));
109 EXPECT_EQ(sid, std::string(4, 'S'));
110
111 // Only AES-128-GCM offered -> negotiate it.
112 ASSERT_TRUE(d::parse_client_hello(make_client_hello({0x1301}, true, true), pub, suite, sid));
113 EXPECT_EQ(suite, 0x1301u);
114}

116TEST(CheatahTls, ParseClientHelloRejectsMalformed) {
117 namespace d = cheatah::tls::detail;
118 std::string pub, sid;
119 unsigned suite = 0;
120
121 EXPECT_FALSE(d::parse_client_hello("", pub, suite, sid)); // too short
122 EXPECT_FALSE(d::parse_client_hello(std::string("\x02\x00\x00\x00", 4),
123 pub, suite, sid)); // not a client_hello
124 EXPECT_FALSE(d::parse_client_hello(make_client_hello({0x9999}, true, true),
125 pub, suite, sid)); // no cipher suite in common
126 EXPECT_FALSE(d::parse_client_hello(make_client_hello({0x1303}, false, true),
127 pub, suite, sid)); // no TLS 1.3 offered
128 EXPECT_FALSE(d::parse_client_hello(make_client_hello({0x1303}, true, false),
129 pub, suite, sid)); // no X25519 key share
130
131 // Truncations inside each length-prefixed field must be refused, never over-read. Cutting a
132 // well-formed hello at increasing offsets walks the bounds checks (session id, cipher-suite
133 // list, compression, extensions, key-share body) in turn.
134 const std::string full = make_client_hello({0x1303, 0x1301}, true, true, 4);
135 for (std::size_t cut = 4; cut < full.size(); ++cut) {
136 EXPECT_FALSE(d::parse_client_hello(full.substr(0, cut), pub, suite, sid))
137 << "truncation at " << cut << " must be rejected";
138 }
139}

141// PEM block extraction (strict base64) and the Ed25519 PKCS#8 seed parse, incl. the reject paths.
142TEST(CheatahTls, PemBlockExtractsAndRejects) {
143 namespace d = cheatah::tls::detail;
144 // "hi" base64 is "aGk=" -> a well-formed CERTIFICATE block round-trips to its bytes.
145 const std::string pem = "-----BEGIN CERTIFICATE-----\naGk=\n-----END CERTIFICATE-----\n";
146 EXPECT_EQ(d::pem_block(pem, "CERTIFICATE"), "hi");
147 EXPECT_EQ(d::pem_block(pem, "PRIVATE KEY"), ""); // label absent
148 EXPECT_EQ(d::pem_block("-----BEGIN CERTIFICATE-----\naGk=\n", "CERTIFICATE"), ""); // no END
149
150 // A real PKCS#8 Ed25519 key DER: 302e020100300506032b657004220420 || 32-byte seed.
151 std::string der = {static_cast<char>(0x30), 0x2e, 0x02, 0x01, 0x00, 0x30, 0x05, 0x06,
152 0x03, 0x2b, 0x65, 0x70, 0x04, 0x22, 0x04, 0x20};
153 der.append(32, static_cast<char>(0xAB));
154 EXPECT_EQ(d::ed25519_seed_from_pkcs8(der), std::string(32, static_cast<char>(0xAB)));
155 EXPECT_EQ(d::ed25519_seed_from_pkcs8("not a key"), ""); // pattern absent
156}

158// The ClientHello's cipher ORDER is a wire-format decision that depends on the host CPU: with AES-NI
159// we lead with AES-GCM, without it ChaCha20 leads. Read inline, whichever branch does not match the
160// build machine was dead code no test could reach — so the ordering went unpinned on every machine
161// except the one nobody was testing on. Taking the decision as a parameter makes both orders
162// checkable anywhere.
163TEST(CheatahTls, CipherPreferenceFollowsHardware) {
164 const auto suites = [](bool hw) {
165 std::string body;
166 cheatah::tls::detail::append_cipher_preference(body, hw);
167 EXPECT_EQ(body.size(), 6u) << "exactly three suites, two bytes each";
168 std::vector<unsigned> out;
169 for (std::size_t i = 0; i + 1 < body.size(); i += 2) {
170 out.push_back((static_cast<unsigned char>(body[i]) << 8) |
171 static_cast<unsigned char>(body[i + 1]));
172 }
173 return out;
174 };
175
176 // With hardware AES, AES-GCM runs multi-GB/s and must be offered first.
177 EXPECT_EQ(suites(true), (std::vector<unsigned>{0x1302, 0x1301, 0x1303}));
178 // Without it, scalar ChaCha20 is our faster path, so it leads.
179 EXPECT_EQ(suites(false), (std::vector<unsigned>{0x1303, 0x1301, 0x1302}));
180
181 // Both orders offer the SAME three suites — the preference changes, never the capability, so a
182 // server that honours client order can always still pick something we can actually speak.
183 auto a = suites(true), b = suites(false);
184 std::sort(a.begin(), a.end());
185 std::sort(b.begin(), b.end());
186 EXPECT_EQ(a, b);
187}
