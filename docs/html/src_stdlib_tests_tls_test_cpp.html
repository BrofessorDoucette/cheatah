1// Unit tests for the `tls` module's key schedule against the RFC 8448 trace constants
2// (the published TLS 1.3 test vectors, SHA-256 suite — suite-independent for the schedule).
3#include <gtest/gtest.h>
4
5#include <string>
6
7#include "hashlib.hpp"
8#include "tls.hpp"
9
10namespace {
11std::string hex_of(std::string_view raw) {
12 static constexpr char kHex[] = "0123456789abcdef";
13 std::string out;
14 for (const char ch : raw) {
15 out.push_back(kHex[static_cast<unsigned char>(ch) >> 4]);
16 out.push_back(kHex[static_cast<unsigned char>(ch) & 0xF]);
17 }
18 return out;
19}
20} // namespace
21
22// RFC 8448 §3: the early secret HKDF-Extract(0, 0^32) and the "derived" secret from it.
23TEST(CheatahTls, KeySchedule) {
24 const std::string zeros(32, '\0');
25 const std::string early = cheatah::hashlib::hkdf_extract(std::string(), zeros);
26 EXPECT_EQ(hex_of(early), "33ad0a1c607ec03b09e6cd9893680ce210adf300aa1f2660e1b22e10f170f92a");
27 const std::string derived = cheatah::tls::detail::derive_secret(early, "derived", "");
28 EXPECT_EQ(hex_of(derived), "6f2615a108c702c5678f54fc9dbab69716c076189c48250cebeac3576c3611ba");
29}
30
31// HKDF-Expand-Label structure: deterministic, length-exact, label-sensitive.
32TEST(CheatahTls, ExpandLabel) {
33 const std::string secret(32, '\x42');
34 const std::string a = cheatah::tls::detail::expand_label(secret, "key", "", 32);
35 const std::string b = cheatah::tls::detail::expand_label(secret, "iv", "", 12);
36 EXPECT_EQ(a.size(), std::size_t{32});
37 EXPECT_EQ(b.size(), std::size_t{12});
38 EXPECT_NE(a.substr(0, 12), b); // different labels -> unrelated output
39 EXPECT_EQ(a, cheatah::tls::detail::expand_label(secret, "key", "", 32)); // deterministic
40}
