2// per-function compile-run test (tests/purrc/hashlib_cr_test.cpp), this drives a
3// single cohesive program that hashes several inputs through the module and
4// asserts its exact stdout, exercising the function across multiple message
5// lengths in one run.
6//
7// All three are standard NIST/FIPS-180 SHA-256 test vectors:
8// "" -> the canonical empty-string digest,
9// "abc"-> the one-block vector,
10// the 56-byte message -> the two-block vector.
11//
12// Coverage — every function in stdlib/hashlib/hashlib.hpp:
13// sha256.
14#include "e2e_harness.hpp"
15
16TEST(StdlibE2E, Hashlib) {

18import hashlib
19

23io.print(hashlib.sha512(""))

25# The raw digests are the same bytes the hex form spells, so their lengths are fixed.
26io.print(len(hashlib.sha256_digest("abc")), len(hashlib.sha512_digest("abc")))
