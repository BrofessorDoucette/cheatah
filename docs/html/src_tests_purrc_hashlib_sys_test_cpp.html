3// System-level (whole-program) test for the `hashlib` stdlib module. Unlike the
4// per-function compile-run test (tests/purrc/hashlib_cr_test.cpp), this drives a
5// single cohesive program that hashes several inputs through the module and
6// asserts its exact stdout, exercising the function across multiple message
7// lengths in one run.
8//
9// All three are standard NIST/FIPS-180 SHA-256 test vectors:
10// "" -> the canonical empty-string digest,
11// "abc"-> the one-block vector,
12// the 56-byte message -> the two-block vector.
13//
14// Coverage — every function in stdlib/hashlib/hashlib.hpp:
15// sha256.
16#include "e2e_harness.hpp"
17
18TEST(StdlibE2E, Hashlib) {
19 e2e::expect_e2e("hashlib_sys", R"PURR(import io
20import hashlib
21
22io.print(hashlib.sha256(""))
23io.print(hashlib.sha256("abc"))
24io.print(hashlib.sha256("abcdbcdecdefdefgefghfghighijhijkijkljklmklmnlmnomnopnopq"))
25io.print(hashlib.sha512(""))
26io.print(hashlib.sha512("abc"))
27# The raw digests are the same bytes the hex form spells, so their lengths are fixed.
28io.print(len(hashlib.sha256_digest("abc")), len(hashlib.sha512_digest("abc")))
29)PURR",
