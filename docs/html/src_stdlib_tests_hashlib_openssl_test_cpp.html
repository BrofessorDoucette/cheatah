1// Byte-for-byte cross-check of the hashlib digests / HMAC / base64 against the system
2// `openssl` CLI over a range of inputs — empty, short, long, a sentence, and ALL 256 byte
3// values — so we validate against an independent reference implementation, not only the
4// fixed standard vectors in hashlib_test.cpp. The whole suite SKIPS when `openssl` is
5// unavailable, so it strengthens assurance where present without becoming a build/runtime
6// dependency. Input bytes go through a temp FILE (never the shell), so arbitrary bytes are
7// compared exactly; only ASCII, quote-free HMAC keys appear on the command line.

24bool has_openssl() { return std::system("openssl version >/dev/null 2>&1") == 0; }

26// Run @p cmd, capture its stdout.
27std::string capture(const std::string& cmd) {
28 std::array<char, 4096> buf{};
29 std::string out;
30 FILE* p = popen(cmd.c_str(), "r");
31 if (!p) return "";
32 while (std::fgets(buf.data(), static_cast<int>(buf.size()), p)) out += buf.data();
33 pclose(p);
34 return out;
35}

37// The first whitespace-delimited token — openssl `-r` prints "<hex> *<file>".
38std::string first_token(const std::string& s) {
39 std::string t;
40 for (char c : s) {
41 if (c == ' ' || c == '\t' || c == '\n' || c == '\r') break;
42 t += c;
43 }
44 return t;
45}
46
47// Strip trailing newlines (openssl base64 ends with one).
48std::string chomp(std::string s) {
49 while (!s.empty() && (s.back() == '\n' || s.back() == '\r')) s.pop_back();
50 return s;
51}
52
53// Lowercase hex of a raw byte string — hmac_sha256/512 return RAW bytes (by design),
54// while openssl `-r` prints hex, so encode hl's output before comparing.
55std::string to_hex(const std::string& raw) {
56 static constexpr char kHex[] = "0123456789abcdef";
57 std::string out;
58 out.reserve(raw.size() * 2);
59 for (unsigned char c : raw) {
60 out.push_back(kHex[c >> 4]);
61 out.push_back(kHex[c & 0xF]);
62 }
63 return out;
64}

66// Write @p data to a fresh temp file; returns its path (removed by the caller).
67std::string write_tmp(const std::string& data) {
68 char path[] = "/tmp/cheatah_ossl_XXXXXX";
69 const int fd = mkstemp(path);
70 if (fd >= 0) {
71 ssize_t off = 0;
72 while (off < static_cast<ssize_t>(data.size())) {
73 const ssize_t n = ::write(fd, data.data() + off, data.size() - off);
74 if (n <= 0) break;
75 off += n;
76 }
77 ::close(fd);
78 }
79 return std::string(path);
80}

82// The shared input corpus: empty, short, long-repeat, a sentence, and every byte 0x00..0xFF.
83const std::vector<std::string>& inputs() {
84 static const std::vector<std::string> v = [] {
85 std::string all;
86 for (int i = 0; i < 256; ++i) all.push_back(static_cast<char>(i));
87 return std::vector<std::string>{"", "abc", std::string(1000, 'a'),
88 "The quick brown fox jumps over the lazy dog", all};
89 }();
90 return v;
91}

95TEST(HashlibVsOpenssl, Sha256) {
96 if (!has_openssl()) GTEST_SKIP() << "openssl CLI not available";
97 for (const auto& in : inputs()) {
98 const std::string f = write_tmp(in);
99 const std::string ref = first_token(capture("openssl dgst -sha256 -r '" + f + "'"));
100 std::remove(f.c_str());
101 ASSERT_EQ(ref.size(), 64u) << "openssl output unexpected for input of size " << in.size();
102 EXPECT_EQ(hl::sha256(in), ref) << "sha256 mismatch for input of size " << in.size();
103 }
104}

117TEST(HashlibVsOpenssl, HmacSha256) {
118 if (!has_openssl()) GTEST_SKIP() << "openssl CLI not available";
119 for (const std::string key : {std::string("k"), std::string("secretkey"),
120 std::string(40, 'K')}) {
121 for (const auto& in : inputs()) {
122 const std::string f = write_tmp(in);
123 const std::string ref = first_token(
124 capture("openssl dgst -sha256 -hmac '" + key + "' -r '" + f + "'"));
125 std::remove(f.c_str());
126 ASSERT_EQ(ref.size(), 64u);
127 EXPECT_EQ(to_hex(hl::hmac_sha256(key, in)), ref)
128 << "hmac-sha256 mismatch (key '" << key << "', input size " << in.size() << ")";
129 }
130 }
