3/**
4 * @file routines.hpp
5 * @brief cheatah `linalg` — numpy's linear-algebra API on ndarray, with
6 * SIMD-friendly contiguous kernels (a .purr program writes `linalg.solve(A, b)`).
7 *
8 * `import linalg` to use it (auto-links ndarray). Unit tests:
9 * stdlib/tests/linalg_routines_test.cpp; SIMD reporting tested in
10 * tests/linalg/smoke_test.cpp. The suite runs under AddressSanitizer (the `asan`
11 * preset) and Valgrind (security/run-valgrind.sh) on every QA-gate run.
12 *
13 * Routines operate on `ndarray::NDArray` (2-D = matrix, 1-D = vector). They mirror
14 * https://numpy.org/doc/stable/reference/routines.linalg.html and are implemented
15 * in routines.cpp (LU w/ partial pivoting, Cholesky, Householder QR, one-sided
16 * Jacobi SVD, cyclic Jacobi symmetric eigen, Hessenberg + shifted-QR general
17 * eigen) at -O3 -march=native so the hot loops auto-vectorize.
18 *
19 * SIMD here is pure compiler auto-vectorization (no intrinsics). On a scalar build
20 * (no vector ISA) every routine still returns identical results, just slower — see
21 * simd.hpp's file comment for the full SIMD model, the no-SIMD behavior, and the
22 * compile-time-dispatch limitation.
23 *
24 * @note `n` below is the matrix dimension. The general eigensolvers `eig`/`eigvals`
25 * return a **complex** spectrum (@ref CNDArray) — a real matrix can have
26 * complex conjugate eigenvalue pairs — while the Hermitian solvers
27 * `eigh`/`eigvalsh` return a guaranteed-real spectrum. Routines that extract
28 * a working copy "allocate scratch O(n²) for the factorization" even when
29 * they return a scalar.
30 */

216/**
217 * Cholesky factor of a symmetric positive-definite matrix (throws otherwise).
218 *
219 * Requires a square matrix and computes L column by column reading only the
220 * lower triangle of @p a; if any pivot (the diagonal under the square root) is
221 * non-positive it throws "matrix is not positive-definite", which also catches
222 * non-SPD or non-symmetric input.
223 * @param a square SPD matrix.
224 * @return lower-triangular L with A = L·Lᵀ.
225 * @complexity O(n³).
226 * @alloc allocates a new NDArray result.
227 * @test LinalgRoutines.CholeskyAndQR
228 * @crtest LinalgCompileRun.Cholesky
229 * @systest StdlibE2E.Linalg
230 */
231NDArray cholesky(const NDArray& a); // lower-triangular L (A = L Lᵀ)
232/** Result of qr(): A = q·r with orthonormal q and upper-triangular r. */
233struct QR {
234 NDArray q; ///< Orthonormal columns, m×n (the Q in A = Q·R).
235 NDArray r; ///< Upper-triangular factor, n×n (the R in A = Q·R).
236};
237/**
238 * Reduced QR via Householder reflections (requires rows ≥ cols).
239 *
240 * Applies successive Householder reflectors to triangularize @p a, returning the
241 * thin/reduced factors; throws "qr requires rows >= cols" for wide matrices.
242 * Rank-deficient columns (zero pivot norm) are skipped, leaving the
243 * corresponding R entries zero.
244 * @param a m×n matrix.
245 * @return @ref QR with q (m×n, orthonormal cols) and r (n×n, upper-triangular).
246 * @complexity O(n³).
247 * @alloc allocates both members.
248 * @test LinalgRoutines.CholeskyAndQR
249 * @crtest LinalgCompileRun.Qr
250 * @systest StdlibE2E.Linalg
251 */
252QR qr(const NDArray& a);
253/** Result of svd(): A = u·diag(s)·vh. */
254struct SVD {
255 NDArray u; ///< Left singular vectors, m×n.
256 NDArray s; ///< Singular values in descending order (length n).
257 NDArray vh; ///< Right singular vectors transposed, n×n (the Vᵀ in A = u·diag(s)·Vᵀ).
258};
259/**
260 * Singular value decomposition (one-sided Jacobi; requires rows ≥ cols).
261 *
262 * Iterates one-sided Jacobi rotations (capped at 80 sweeps) to orthogonalize the
263 * columns, then normalizes them into u and sorts the singular values descending;
264 * throws "svd requires rows >= cols" for wide matrices (transpose first). Exact
265 * zero singular values yield a zero u column.
266 * @param a m×n matrix.
267 * @return @ref SVD with u (m×n), s (descending singular values), vh (n×n = Vᵀ).
268 * @complexity iterative O(n³).
269 * @alloc allocates all members.
270 * @test LinalgRoutines.SvdAndEigh
271 * @crtest LinalgCompileRun.Svd
272 * @systest StdlibE2E.Linalg
273 */
274SVD svd(const NDArray& a);

278struct Eig {

282/**
283 * Result of the general eig(): a **complex** spectrum, since a real matrix can have
284 * complex conjugate eigenvalue pairs. Column j of vectors is the eigenvector for values[j].
285 */
286struct EigC {
287 CNDArray values; ///< Eigenvalues (length n), complex.
288 CNDArray vectors; ///< Eigenvectors as columns: column j matches values[j].

290/**
291 * Eigen-decomposition of a general square matrix (**complex** spectrum and
292 * eigenvectors).
293 *
294 * For a symmetric @p a it delegates to @ref eigh (promoted to complex with zero
295 * imaginary part); otherwise it uses Hessenberg reduction + shifted QR for the
296 * eigenvalues, then **inverse iteration** for each eigenvector. A real matrix with a
297 * complex conjugate pair (e.g. a rotation) yields those complex eigenvalues and
298 * eigenvectors rather than throwing. Throws on a non-square matrix or if the QR
299 * iteration fails to converge.
300 * @param a square matrix.
301 * @return @ref EigC with complex values and matching complex eigenvector columns.
302 * @complexity iterative O(n³) via Hessenberg + shifted QR.
303 * @alloc allocates both members.
304 * @test LinalgRoutines.GeneralEig
305 * @crtest LinalgCompileRun.Eig
306 * @systest StdlibE2E.Linalg
307 */
308EigC eig(const NDArray& a); // general square matrix
309/**
310 * Eigenvalues of a general square matrix (**complex**), descending.
311 *
312 * Routes symmetric input through cyclic Jacobi and everything else through
313 * Hessenberg + shifted QR, then sorts the result descending (by real part, then by
314 * imaginary part). A real matrix with a complex conjugate pair yields those complex
315 * eigenvalues rather than throwing. Throws on a non-square matrix or non-convergence
316 * of the QR iteration.
317 * @param a square matrix.
318 * @return length-n complex vector of eigenvalues.
319 * @complexity iterative O(n³).
320 * @alloc allocates a new CNDArray result.
321 * @test LinalgRoutines.SvdAndEigh
322 * @crtest LinalgCompileRun.Eigvals
323 * @systest StdlibE2E.Linalg
324 */
325CNDArray eigvals(const NDArray& a);
326/**
327 * Eigen-decomposition of a symmetric matrix (cyclic Jacobi).
328 *
329 * Runs cyclic Jacobi (capped at 100 sweeps) to diagonalize @p a, returning real
330 * eigenvalues sorted descending with matching eigenvector columns; it reads the
331 * full matrix and assumes symmetry rather than checking it, so asymmetric input
332 * yields meaningless results. Throws on a non-square matrix.
333 * @param a square symmetric matrix.
334 * @return @ref Eig with descending values and matching eigenvectors.
335 * @complexity iterative O(n³).
336 * @alloc allocates both members.
337 * @test LinalgRoutines.SvdAndEigh
338 * @crtest LinalgCompileRun.Eigh
339 * @systest StdlibE2E.Linalg
340 */
341Eig eigh(const NDArray& a); // symmetric / Hermitian
342/**
343 * Eigenvalues of a symmetric matrix, descending (cyclic Jacobi).
344 *
345 * Same cyclic-Jacobi diagonalization as @ref eigh but discards the eigenvectors;
346 * assumes (does not verify) symmetry and throws on a non-square matrix.
347 * @param a square symmetric matrix.
348 * @return length-n vector of eigenvalues.
349 * @complexity iterative O(n³).
350 * @alloc allocates a new NDArray result.
351 * @test LinalgRoutines.EigvalshSymmetric
352 * @crtest LinalgCompileRun.Eigvalsh
353 * @systest StdlibE2E.Linalg
354 */
355NDArray eigvalsh(const NDArray& a);
356/**
357 * Result of the complex Hermitian eigh(): **real** eigenvalues (a Hermitian operator
358 * has a real spectrum) with **complex** eigenvectors. Column j of vectors is the
359 * (unit-norm) eigenvector for values[j].
360 */
361struct EighC {
362 NDArray values; ///< Eigenvalues (length n), real and descending.
363 CNDArray vectors; ///< Eigenvectors as columns: column j matches values[j].
364};
365/**
366 * Eigen-decomposition of a **complex Hermitian** matrix — real eigenvalues, complex
367 * eigenvectors.
368 *
369 * The quantum-mechanics workhorse: a Hermitian operator (`conj_transpose(a) == a`)
370 * has a guaranteed-real spectrum and orthonormal complex eigenvectors. Computed by
371 * the real symmetric Jacobi solver on the 2n×2n real embedding

373 * throws on a non-square matrix.
374 * @param a square complex Hermitian matrix.
375 * @return @ref EighC with descending real values and matching complex eigenvectors.
376 * @complexity iterative O(n³).
377 * @alloc allocates both members.
378 * @test LinalgRoutines.ComplexHermitianEigh
379 * @crtest LinalgCompileRun.EighComplex
380 * @systest StdlibE2E.LinalgComplex
381 */
382EighC eigh(const CNDArray& a);

497/**
498 * Solve A·x = b via LU with partial pivoting.

502 * A singular @p a does not throw but yields a garbage/overflowing solution
503 * (pivots are nudged off zero rather than detected).

505 * @param b right-hand-side vector.
506 * @return solution x.
507 * @complexity O(n³).
508 * @alloc allocates a new NDArray result.
509 * @test LinalgRoutines.SolveDetInv
510 * @crtest LinalgCompileRun.Solve
511 * @systest StdlibE2E.Linalg
512 */
513NDArray solve(const NDArray& a, const NDArray& b); // A x = b
514/**
515 * Least-squares solution min‖A·x − b‖ (computed as @ref pinv (a)·b).
516 *
517 * Forms the Moore–Penrose pseudo-inverse via SVD and multiplies it by @p b, so
518 * it handles over- and under-determined systems and returns the minimum-norm
519 * solution for rank-deficient @p a; @p b must be conformable for the
520 * @ref matmul step.
521 * @param a m×n matrix.
522 * @param b right-hand side.
523 * @return minimizing x.
524 * @complexity iterative O(n³) via SVD.
525 * @alloc allocates a new NDArray result.
526 * @test LinalgRoutines.Lstsq
527 * @crtest LinalgCompileRun.Lstsq
528 * @systest StdlibE2E.Linalg
529 */
530NDArray lstsq(const NDArray& a, const NDArray& b); // least-squares solution
531/**
532 * Matrix inverse via LU with partial pivoting.
533 *
534 * Factorizes @p a once and back-solves against each identity column; requires a
535 * square matrix (throws otherwise). A singular @p a does not throw but produces
536 * garbage/overflowing entries since zero pivots are nudged rather than detected.
537 * @param a square matrix.
538 * @return A⁻¹.
539 * @complexity O(n³).
540 * @alloc allocates a new NDArray result.
541 * @test LinalgRoutines.SolveDetInv
542 * @crtest LinalgCompileRun.Inv
