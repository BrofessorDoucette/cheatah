13// A real loopback round-trip: a listener accepts one client, reads its request,
14// and replies; a client thread connects, sends, and reads the reply. Exercises
15// tcp_listen / local_port / accept / recv / sendall / tcp_connect / send / close.
16TEST(CheatahSocket, Loopback) {
17 const long long lfd = sk::tcp_listen("127.0.0.1", 0, 4);
18 ASSERT_GE(lfd, 0) << sk::last_error();
19 const long long port = sk::local_port(lfd);
20 ASSERT_GT(port, 0);
21
22 std::string client_reply;
23 std::thread client([&] {
24 const long long cfd = sk::tcp_connect("127.0.0.1", port);
25 ASSERT_GE(cfd, 0);
26 ASSERT_EQ(sk::sendall(cfd, "ping"), 0);
27 client_reply = sk::recv(cfd, 64);
28 sk::close(cfd);
29 });
30
31 const long long conn = sk::accept(lfd);
32 ASSERT_GE(conn, 0) << sk::last_error();
33 EXPECT_EQ(sk::recv(conn, 64), "ping");
34 EXPECT_EQ(sk::sendall(conn, "pong"), 0);
35 sk::close(conn);
36
37 client.join();
38 EXPECT_EQ(client_reply, "pong");
39 sk::close(lfd);
40}

42// A larger, binary-ish payload to drive send()/sendall() through multiple writes
43// and confirm byte-exactness (including an embedded NUL).
44TEST(CheatahSocket, Sendall) {
45 const long long lfd = sk::tcp_listen("127.0.0.1", 0, 1);
46 ASSERT_GE(lfd, 0);
47 const long long port = sk::local_port(lfd);
48
49 std::string payload(8000, 'x');
50 payload[100] = '\0';
51 payload[7999] = 'Z';
52
53 std::thread client([&] {
54 const long long cfd = sk::tcp_connect("127.0.0.1", port);
55 ASSERT_GE(cfd, 0);
56 EXPECT_GT(sk::send(cfd, "hi"), 0); // single send()
57 EXPECT_EQ(sk::sendall(cfd, payload), 0);
58 sk::close(cfd);
59 });
60
61 const long long conn = sk::accept(lfd);
62 ASSERT_GE(conn, 0);
63 std::string got;
64 while (got.size() < payload.size() + 2) {
65 const std::string chunk = sk::recv(conn, 4096);
66 if (chunk.empty()) break;
67 got += chunk;
68 }
69 EXPECT_EQ(got, "hi" + payload);
70 sk::close(conn);
71 client.join();
72 sk::close(lfd);
73}

121// set_timeout (recv/send deadline) + shutdown (graceful SHUT_RDWR) on a live connection.
122// A loopback pair so both run on a real connected socket; shutdown wakes the peer's recv.
123TEST(CheatahSocket, TimeoutThenShutdown) {
124 const long long lfd = sk::tcp_listen("127.0.0.1", 0, 1);
125 ASSERT_GE(lfd, 0) << sk::last_error();
126 const long long port = sk::local_port(lfd);
127 ASSERT_GT(port, 0);
128 std::thread client([&] {
129 const long long c = sk::tcp_connect("127.0.0.1", port);
130 if (c >= 0) { sk::recv(c, 1); sk::close(c); }
131 });
132 const long long conn = sk::accept(lfd);
133 ASSERT_GE(conn, 0) << sk::last_error();
134 EXPECT_EQ(sk::set_timeout(conn, 500), 0); // positive timeout -> SO_RCVTIMEO/SO_SNDTIMEO
135 EXPECT_EQ(sk::shutdown(conn), 0); // graceful SHUT_RDWR
136 sk::close(conn);
137 client.join();
138 sk::close(lfd);
139}

153 sk::Conn bad = sk::open("127.0.0.1", dead);

165// Explicit close() releases the fd and is idempotent; the destructor then does nothing.
166TEST(CheatahSocket, ConnGuardClosesOnScopeExit) {
167 sk::Listener server = sk::serve("127.0.0.1", 0, 1);
168 ASSERT_TRUE(server.is_open()) << sk::last_error();
169 sk::Conn c = sk::open("127.0.0.1", server.local_port());
170 ASSERT_TRUE(c.is_open()) << sk::last_error();
171 EXPECT_GE(c.fd(), 0);
172 EXPECT_EQ(c.close(), 0); // explicit close
173 EXPECT_FALSE(c.is_open());
174 EXPECT_EQ(c.close(), -1); // idempotent
175 // `server` is closed by its destructor here — Valgrind confirms no fd leak.
176}

178// Move transfers fd ownership; the moved-from guard is left closed (never double-closed).
179TEST(CheatahSocket, ConnMoveTransfersOwnership) {
180 sk::Listener server = sk::serve("127.0.0.1", 0, 2);
181 ASSERT_TRUE(server.is_open());
182 sk::Conn a = sk::open("127.0.0.1", server.local_port());
183 ASSERT_TRUE(a.is_open());
184 const long long fd = a.fd();
185 sk::Conn b(std::move(a)); // move-construct
186 EXPECT_FALSE(a.is_open());
187 EXPECT_EQ(b.fd(), fd);
188 // Move-assign onto an already-open guard closes the overwritten fd first.
189 sk::Conn d = sk::open("127.0.0.1", server.local_port());
190 ASSERT_TRUE(d.is_open());
191 d = std::move(b);
192 EXPECT_FALSE(b.is_open());
193 EXPECT_EQ(d.fd(), fd);
194 EXPECT_EQ(d.close(), 0);
195}

197// Full loopback through the guards: serve()/open()/accept() + Conn send/sendall/recv/
198// set_timeout/local_port/shutdown/close — the whole owning API on a live connection.
199TEST(CheatahSocket, ConnLoopback) {
200 sk::Listener server = sk::serve("127.0.0.1", 0, 4);
201 ASSERT_TRUE(server.is_open()) << sk::last_error();
202 EXPECT_GE(server.fd(), 0);
203 const long long port = server.local_port();
204 ASSERT_GT(port, 0);
205
206 std::string client_reply;
207 std::thread client([&] {
208 sk::Conn c = sk::open("127.0.0.1", port);
209 ASSERT_TRUE(c.is_open());
210 EXPECT_EQ(c.set_timeout(1000), 0);
211 EXPECT_GT(c.local_port(), 0);
212 EXPECT_GT(c.send("pi"), 0); // single send()
213 EXPECT_EQ(c.sendall("ng"), 0); // looped sendall()
214 client_reply = c.recv(64);
215 // c is closed by its destructor at thread-scope exit.
216 });
217
218 sk::Conn conn = server.accept();
219 ASSERT_TRUE(conn.is_open()) << sk::last_error();
220 std::string got;
221 while (got.size() < 4) {
222 const std::string chunk = conn.recv(64);
223 if (chunk.empty()) break;
224 got += chunk;
225 }
226 EXPECT_EQ(got, "ping");
227 EXPECT_EQ(conn.sendall("pong"), 0);
228 EXPECT_EQ(conn.shutdown(), 0); // graceful half-close
229 EXPECT_EQ(conn.close(), 0);
230
231 client.join();
232 EXPECT_EQ(client_reply, "pong");
233 // `server` is closed by its destructor.
234}

236// Listener move semantics + idempotent close.
237TEST(CheatahSocket, ListenerLoopback) {
238 sk::Listener a = sk::serve("127.0.0.1", 0, 1);
239 ASSERT_TRUE(a.is_open()) << sk::last_error();
240 const long long port = a.local_port();
241 ASSERT_GT(port, 0);
242 sk::Listener b(std::move(a)); // move-construct
243 EXPECT_FALSE(a.is_open());
244 EXPECT_EQ(b.local_port(), port);
245 sk::Listener c;
246 c = std::move(b); // move-assign onto a closed listener
247 EXPECT_FALSE(b.is_open());
248 EXPECT_GE(c.fd(), 0);
249 EXPECT_EQ(c.close(), 0);
250 EXPECT_EQ(c.close(), -1); // idempotent
251}
