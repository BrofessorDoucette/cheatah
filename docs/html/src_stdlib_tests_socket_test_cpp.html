10// A real loopback round-trip: a listener accepts one client, reads its request,
11// and replies; a client thread connects, sends, and reads the reply. Exercises
12// tcp_listen / local_port / accept / recv / sendall / tcp_connect / send / close.
13TEST(CheatahSocket, Loopback) {
14 const long long lfd = sk::tcp_listen("127.0.0.1", 0, 4);
15 ASSERT_GE(lfd, 0) << sk::last_error();
16 const long long port = sk::local_port(lfd);
17 ASSERT_GT(port, 0);
18
19 std::string client_reply;
20 std::thread client([&] {
21 const long long cfd = sk::tcp_connect("127.0.0.1", port);
22 ASSERT_GE(cfd, 0);
23 ASSERT_EQ(sk::sendall(cfd, "ping"), 0);
24 client_reply = sk::recv(cfd, 64);
25 sk::close(cfd);
26 });
27
28 const long long conn = sk::accept(lfd);
29 ASSERT_GE(conn, 0) << sk::last_error();
30 EXPECT_EQ(sk::recv(conn, 64), "ping");
31 EXPECT_EQ(sk::sendall(conn, "pong"), 0);
32 sk::close(conn);
33
34 client.join();
35 EXPECT_EQ(client_reply, "pong");
36 sk::close(lfd);
37}

39// A larger, binary-ish payload to drive send()/sendall() through multiple writes
40// and confirm byte-exactness (including an embedded NUL).
41TEST(CheatahSocket, Sendall) {
42 const long long lfd = sk::tcp_listen("127.0.0.1", 0, 1);
43 ASSERT_GE(lfd, 0);
44 const long long port = sk::local_port(lfd);
45
46 std::string payload(8000, 'x');
47 payload[100] = '\0';
48 payload[7999] = 'Z';
49
50 std::thread client([&] {
51 const long long cfd = sk::tcp_connect("127.0.0.1", port);
52 ASSERT_GE(cfd, 0);
53 EXPECT_GT(sk::send(cfd, "hi"), 0); // single send()
54 EXPECT_EQ(sk::sendall(cfd, payload), 0);
55 sk::close(cfd);
56 });
57
58 const long long conn = sk::accept(lfd);
59 ASSERT_GE(conn, 0);
60 std::string got;
61 while (got.size() < payload.size() + 2) {
62 const std::string chunk = sk::recv(conn, 4096);
63 if (chunk.empty()) break;
64 got += chunk;
65 }
66 EXPECT_EQ(got, "hi" + payload);
67 sk::close(conn);
68 client.join();
69 sk::close(lfd);
70}

72// The low-level BSD path: socket -> set_reuseaddr -> bind -> listen -> local_port.
73TEST(CheatahSocket, ListenLowLevel) {
74 const long long fd = sk::socket();
75 ASSERT_GE(fd, 0);
76 EXPECT_EQ(sk::set_reuseaddr(fd), 0);
77 EXPECT_EQ(sk::bind(fd, "127.0.0.1", 0), 0);
78 EXPECT_EQ(sk::listen(fd, 1), 0);
79 EXPECT_GT(sk::local_port(fd), 0);
80 EXPECT_EQ(sk::close(fd), 0);
81}

83// Connecting to a port nobody is listening on fails, and last_error() is set.
84TEST(CheatahSocket, ConnectRefused) {
85 // Bind+listen to grab a free port, then close it so the port is free again.
86 const long long probe = sk::tcp_listen("127.0.0.1", 0, 1);
87 ASSERT_GE(probe, 0);
88 const long long dead_port = sk::local_port(probe);
89 sk::close(probe);
90
91 EXPECT_LT(sk::tcp_connect("127.0.0.1", dead_port), 0);
92 EXPECT_FALSE(sk::last_error().empty());
93}
