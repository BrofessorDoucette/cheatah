3// cheatah::parsers::json — PoolBuilder: the construction policy for the POOLED parse path.
4//
5// The (now iterative) grammar in json.cpp drives a Builder as a small STACK MACHINE: begin_array/
6// begin_object open a container, add_element/add_member append children to the innermost open one,
7// and finish_array/finish_object close it and return its Node. PoolBuilder accumulates children on
8// reused scratch stacks, then on finish commits them contiguously into a reused POOL and returns a
9// span VIEW (ArrayView/ObjectView) — zero per-container heap allocation. The pools persist across
10// parses (one PoolBuilder, reused by a Parser), so after warm-up allocations amortize to ~0
11// (the reusable-parser model). No runtime polymorphism — a plain policy type.
12
13#include <cstddef>
14#include <span>
15#include <utility>
16#include <vector>
17
18#include "document.hpp" // Node, Member, ArrayView, ObjectView (via node.hpp)
19
20namespace cheatah::parsers::json {
21
22/**
23 * @brief The construction policy for the POOLED JSON parse path: a stack machine
24 * (begin/add/finish) that accumulates children on reused scratch stacks and commits them
25 * contiguously into reused pools, returning span VIEWS (ArrayView/ObjectView) — zero
26 * per-container heap allocation, amortized to ~0 across parses. A plain policy type (no
27 * runtime polymorphism).
28 */
29class PoolBuilder {
30public:
31 /// False: unescaped strings are zero-copy String<std::string_view> into the SOURCE text, so a
32 /// Document from this pooled path is valid only while the source (and these pools) outlive it.
33 static constexpr bool owns_strings = false;
34
35 /**
36 * Prepare for a fresh parse: drop the previous result but keep capacity, then reserve a safe
37 * upper bound so the pools never reallocate mid-parse (every committed span stays valid). The
38 * bounds are structural: an array element costs >= 2 input bytes ("0," / "0]") and an object
39 * member >= 4 (`"":0`), so size/2 nodes and size/4 members can never be exceeded. NOTE this is
40 * still O(input) memory reserved UP FRONT (~24x the text size) — for untrusted input, bound the
41 * document size before parsing, or use the owning path, which allocates only what the document
42 * actually contains. Reused capacity makes this ~free once warm.
43 *
44 * @param text_size the byte length of the source about to be parsed (sizes the pool reserve).
45 * @complexity O(1) amortized (no realloc after warm-up)
46 * @alloc grows the pools once (~text_size/2 nodes + text_size/4 members); reused thereafter
47 * @test Json.ParseObject
48 */
49 void reset(std::size_t text_size) {
50 node_pool_.clear();
51 member_pool_.clear();
52 node_scratch_.clear();
53 member_scratch_.clear();
54 open_bases_.clear();
55 node_pool_.reserve(text_size / 2 + 1);
56 member_pool_.reserve(text_size / 4 + 1);
57 }
58
59 /// Open an array: remember where its elements start on the node scratch stack.
60 /// @complexity O(1) @alloc amortized open_bases_ growth @test Json.PooledForms
61 void begin_array() { open_bases_.push_back(node_scratch_.size()); }
62 /// Open an object: remember where its members start on the member scratch stack.
63 /// @complexity O(1) @alloc amortized open_bases_ growth @test Json.PooledForms
64 void begin_object() { open_bases_.push_back(member_scratch_.size()); }
65
66 /// Append an element to the innermost open array.
67 /// @param n the element node to append (moved in).
68 /// @complexity O(1) @alloc amortized scratch growth @test Json.PooledForms
69 void add_element(Node&& n) { node_scratch_.push_back(std::move(n)); }
70 /// Append a member to the innermost open object.
71 /// @param m the key/value member to append (moved in).
72 /// @complexity O(1) @alloc amortized scratch growth @test Json.PooledForms
73 void add_member(Member&& m) { member_scratch_.push_back(std::move(m)); }
74
75 /**
76 * Close the innermost open array: commit its children into the pool and wrap them in an
77 * ArrayView span.
78 * @return a Node holding the closed array as an ArrayView into the pool.
79 * @complexity O(children) @alloc none (pool is pre-reserved) @test Json.ParseObject
80 */
81 Node finish_array() {
82 const std::size_t base = open_bases_.back();
83 open_bases_.pop_back();
84 Node n;
85 n.variant().emplace<ArrayView>(commit_nodes(base));
86 return n;
87 }
88 /**
89 * Close the innermost open object: commit its members into the pool and wrap them in an
90 * ObjectView span.
91 * @return a Node holding the closed object as an ObjectView into the pool.
92 * @complexity O(members) @alloc none @test Json.ParseObject
93 */
94 Node finish_object() {
95 const std::size_t base = open_bases_.back();
96 open_bases_.pop_back();
97 Node n;
98 n.variant().emplace<ObjectView>(commit_members(base));
99 return n;
100 }
101
102private:
103 // Move the scratch-stack tail [base, size) contiguously into the pool, pop it off the stack,
104 // and return a stable span view of it. The pool is reserved, so push_back never reallocates.
105 // @complexity O(children) @alloc none (pool pre-reserved) @test Json.PooledForms
106 std::span<const Node> commit_nodes(std::size_t base) {
107 const std::size_t count = node_scratch_.size() - base;
108 const std::size_t offset = node_pool_.size();
109 for (std::size_t i = base; i < node_scratch_.size(); ++i) {
110 node_pool_.push_back(std::move(node_scratch_[i]));
111 }
112 node_scratch_.resize(base);
113 return std::span<const Node>(node_pool_.data() + offset, count);
114 }
115 std::span<const Member> commit_members(std::size_t base) {
116 const std::size_t count = member_scratch_.size() - base;
117 const std::size_t offset = member_pool_.size();
118 for (std::size_t i = base; i < member_scratch_.size(); ++i) {
119 member_pool_.push_back(std::move(member_scratch_[i]));
120 }
121 member_scratch_.resize(base);
122 return std::span<const Member>(member_pool_.data() + offset, count);
123 }
124
125 std::vector<Node> node_pool_; // array elements (stable; ArrayView spans point here)
126 std::vector<Member> member_pool_; // object members (stable; ObjectView spans point here)
127 std::vector<Node> node_scratch_; // build stack for in-progress array elements
128 std::vector<Member> member_scratch_; // build stack for in-progress object members
129 std::vector<std::size_t> open_bases_; // scratch base of each currently-open container (LIFO)
130};
131
132} // namespace cheatah::parsers::json
