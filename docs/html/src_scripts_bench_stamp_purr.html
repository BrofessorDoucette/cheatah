scripts/bench/stamp.purr

1# Copyright (c) 2026 BigBrain LLC. MIT-licensed (see LICENSE).
2# Original work; see ACKNOWLEDGMENTS.md for the open-source ideas we build upon.
3# stamp — the provenance block for tables produced by .purr benchmark harnesses.
4#
5# tests/benchmarks/bench_main.cpp emits this same block for the Google Benchmark suites. The
6# three cheatah-language harnesses (app_compare, integral_threads, gen_bench_compare) compute
7# their own numbers and so must emit their own stamp; this module is the one place its shape
8# is defined, so the two producers cannot drift into disagreeing about what a stamp looks like.
9#
10# The fields scripts/bench_table.purr actually reads are `suite:` (which region this belongs
11# to), `commit:` and `watch:` (together: has the measured code moved since?) and
12# `publishable:`. Everything else is for the human who has to decide whether to believe the
13# number.
14#
15# Used as a library module: purrc --import-root scripts . See docs/imports.md.
16import io
17import os
18import string
19
20# Shell out and read the result back. `os.system` returns only an exit status — there is no
21# popen in the stdlib — so capturing means redirecting to a file, the idiom every .purr tool
22# in this repo uses.
23fn capture(cmd : str) -> str {
24 let tmp = "/tmp/cheatah_stamp_capture.out"
25 let rc = os.system(cmd + " > " + tmp + " 2>/dev/null")
26 if rc != 0 { return "" }
27 let out = string.strip(io.read_file(tmp))
28 os.remove(tmp)
29 return out
30}
31
32# The commit the measurement describes. `(dirty)` is recorded rather than hidden: a number
33# taken from an uncommitted tree cannot be traced back to source, and bench_table.purr refuses
34# to publish one.
35# Dirtiness is judged against the suite's OWN watched sources, not the whole tree — see the
36# long note in scripts/bench_run.sh. Markdown is excluded outright: a document cannot change
37# what code measures, so editing one must never invalidate a measurement.
38fn commit(watch : str) -> str {
39 let c = capture("git rev-parse --short HEAD")
40 if c == "" { return "unknown" }
41 os.system("git update-index --refresh >/dev/null 2>&1")
42 if watch == "" { return c } # nothing named to judge against
43 let spec = ""
44 for p in string.split(watch, ",") {
45 let t = string.strip(p)
46 if t != "" { spec = spec + " '" + t + "'" }
47 }
48 if spec == "" { return c }
49 if os.system("git diff --quiet --" + spec + " ':!*.md' ':!docs/bench'") != 0 {
50 return c + " (dirty)"
51 }
52 return c
53}
54
55# A machine string someone could reproduce on. `uname -m` alone says "x86_64", which
56# identifies nothing; the model name and the frequency governor are what actually explain a
57# laptop part's numbers.
58fn host() -> str {
59 let model = capture("awk -F': ' '/^model name/{print $2; exit}' /proc/cpuinfo")
60 if model == "" { model = capture("uname -m") }
61 let cpus = capture("nproc")
62 let kernel = capture("uname -sr")
63 let gov = capture("cat /sys/devices/system/cpu/cpu0/cpufreq/scaling_governor")
64 let s = model + ", " + cpus + " CPUs, " + kernel
65 if gov != "" { s = s + " (governor=" + gov + ")" }
66 return s
67}
68
69fn today() -> str { return capture("date +%Y-%m-%d") }
70
71# Build the stamp. `rounds`/`statistic` describe how the number was arrived at; `watch` lists
72# the sources whose change makes this table stale; `produced_by` is the command that
73# regenerates it, so a published table carries its own reproduction recipe.
74fn build(suite : str, watch : str, competitors : str, rounds : int, statistic : str,
75 produced_by : str) -> str {
76 let s = "<!-- cheatah-bench-stamp v1\n"
77 s = s + " suite: " + suite + "\n"
78 s = s + " generated: " + today() + "\n"
79 s = s + " commit: " + commit(watch) + "\n"
80 s = s + " host: " + host() + "\n"
81 s = s + " cpu-scaling: enabled\n"
82 s = s + " build: purrc -> -O3 -march=native\n"
83 s = s + " competitors: " + competitors + "\n"
84 s = s + " harness: rounds=" + io.str(rounds) + ", striated (every configuration "
85 s = s + "runs once per round)\n"
86 s = s + " statistic: " + statistic + "\n"
87 s = s + " watch: " + watch + "\n"
88 s = s + " publishable: true\n"
89 s = s + "\n PRODUCED BY:\n " + produced_by + "\n"
90 s = s + "-->\n\n"
91 return s
92}

94# Write a complete region body (stamp + table) where scripts/bench_table.purr expects it.
95fn write_region(path : str, stamp : str, table : str) {
96 with io.open(path, "w") as f {
97 f.write(stamp + table)
98 }
99 io.print("wrote", path)
100}
