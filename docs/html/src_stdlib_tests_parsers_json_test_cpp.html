1// In-process unit tests for the HEADER-ONLY parts of the `parsers` module that cheatah_tests can
2// instantiate WITHOUT linking the compiled DOM parser (parsers/json/json.cpp): the URL parser
3// (parsers::url::Parser), the JSON schema factories (field/object), the typed struct reader
4// (read<T>() — read.hpp, which drives the shared scan.hpp scanners), the pooled construction policy
5// (PoolBuilder), the SIMD scan primitives (simd.hpp), and the JSON token classes (Boolean/Null/
6// Number/Node). Every symbol here is header-only, so json.cpp stays OUT of this in-process
7// denominator (parsers_static is deliberately NOT linked — see CMakeLists); we cover the header
8// surface directly rather than by running the separate SIMD DOM parser.

37// A struct to hang a runtime-built schema off of (mirrors how requests' Options/Response get a
38// synthesized schema<>). field()/object() are constexpr but also run at runtime.
39struct Point {
40 long long x;
41 std::string label;
42};
43
44// field() and object() are constexpr factories; requests uses them at compile time via schema<>.
45// Building a schema at RUNTIME here executes their bodies so they are covered.
46// cppcheck-suppress syntaxError // cppcheck mis-parses the member-pointer/decltype schema template
47TEST(CheatahParsers, SchemaFactoriesAtRuntime) {
48 volatile bool run = true; // defeat constant-folding so the factories execute at runtime
49 if (run) {
50 const auto f = json::field("x", &Point::x);
51 const auto sch = json::object(f, json::field("label", &Point::label));
52 EXPECT_EQ(std::tuple_size_v<decltype(sch.fields)>, 2u);
53 EXPECT_EQ(std::get<0>(sch.fields).name, "x");
54 EXPECT_EQ(std::get<1>(sch.fields).name, "label");
55 }

58// The URL parser: scheme://host[:port][/path][?query], the exact grammar requests speaks.
59TEST(CheatahParsers, UrlParserComponents) {
60 url::Parser p;
61 url::Url u;
62 // Explicit port + path + query.
63 ASSERT_TRUE(p.parse("http://example.com:8080/a/b?x=1", u));
64 EXPECT_EQ(u.scheme, "http");
65 EXPECT_EQ(u.host, "example.com");
66 EXPECT_EQ(u.port, 8080);
67 EXPECT_EQ(u.target, "/a/b?x=1");
68 // Default port (no colon), scheme lowercased.
69 ASSERT_TRUE(p.parse("HTTP://host/path", u));
70 EXPECT_EQ(u.scheme, "http");
71 EXPECT_EQ(u.port, 80);
72 EXPECT_EQ(u.target, "/path");
73 // https default port; absent path -> "/".
74 ASSERT_TRUE(p.parse("https://secure.example", u));
75 EXPECT_EQ(u.port, 443);
76 EXPECT_EQ(u.target, "/");
77 // Bare-query form: no path, query present -> "/?...".
78 ASSERT_TRUE(p.parse("http://h?q=2", u));
79 EXPECT_EQ(u.target, "/?q=2");
80 // Max valid port.
81 ASSERT_TRUE(p.parse("http://host:65535/", u));
82 EXPECT_EQ(u.port, 65535);
83}

103// The header-only JSON token classes (node.hpp / number.hpp / boolean.hpp / null.hpp): construct
104// each token and read it back through value(), and exercise Node's variant() (mutable + const).
105// ----------------------------------------------------------------------------
106TEST(CheatahParsersJson, TokenClassesAndNodeVariant) {
107 // Number / Boolean / Null: constructor + value() accessor.
108 EXPECT_DOUBLE_EQ(json::Number{3.5}.value(), 3.5);
109 EXPECT_TRUE(json::Boolean{true}.value());
110 EXPECT_FALSE(json::Boolean{false}.value());
111 EXPECT_EQ(json::Null{}.value(), nullptr);
112
113 // Node wraps a token in its variant; variant() has a mutable and a const overload.
114 json::Node node{json::Number{42.0}};
115 EXPECT_DOUBLE_EQ(std::get<json::Number>(node.variant()).value(), 42.0); // mutable variant()
116 node.variant().emplace<json::Boolean>(true); // mutate via mutable ref
117 const json::Node& cref = node;
118 EXPECT_TRUE(std::get<json::Boolean>(cref.variant()).value()); // const variant()
119}

121// ----------------------------------------------------------------------------
122// The Array / Object / String container tokens (array.hpp / object.hpp / string.hpp): each is a
123// move-only class over its backing storage. Construct, move-construct, move-assign, read via value(),
124// and let them destruct — over BOTH the owning and viewing backings.
125// ----------------------------------------------------------------------------
126TEST(CheatahParsersJson, ContainerTokenLifecycle) {
127 // String: owning (std::string) + viewing (std::string_view). Deduction guides pick the backing.
128 json::String owned{std::string("owned")}; // String<std::string>
129 EXPECT_EQ(owned.value(), "owned");
130 json::String view{std::string_view("viewed")}; // String<std::string_view>
131 EXPECT_EQ(view.value(), "viewed");
132
133 // OwnedArray: build from a vector<Node>, move-construct and move-assign it.
134 std::vector<json::Node> elems;
135 elems.emplace_back(json::Number{1.0});
136 elems.emplace_back(json::Boolean{false});
137 json::OwnedArray arr{std::move(elems)};
138 json::OwnedArray arr2{std::move(arr)}; // move-construct
139 EXPECT_EQ(arr2.value().size(), 2u);
140 json::OwnedArray arr3{std::vector<json::Node>{}};
141 arr3 = std::move(arr2); // move-assign
142 ASSERT_EQ(arr3.value().size(), 2u);
143 EXPECT_DOUBLE_EQ(std::get<json::Number>(arr3.value()[0].variant()).value(), 1.0);
144
145 // OwnedObject: build from a vector<Member>, move-construct and move-assign it.
146 std::vector<json::Member> mem;
147 mem.push_back(json::Member{json::Node{json::String{std::string("k")}},
148 json::Node{json::Number{7.0}}});
149 json::OwnedObject obj{std::move(mem)};
150 json::OwnedObject obj2{std::move(obj)}; // move-construct
151 EXPECT_EQ(obj2.value().size(), 1u);
152 json::OwnedObject obj3{std::vector<json::Member>{}};
153 obj3 = std::move(obj2); // move-assign
154 ASSERT_EQ(obj3.value().size(), 1u);
155 EXPECT_DOUBLE_EQ(std::get<json::Number>(obj3.value()[0].second.variant()).value(), 7.0);
156}

158// ----------------------------------------------------------------------------
159// The SIMD scan primitives (simd.hpp): whitespace-skip (fast test + long run) and quote/backslash
160// find, over inputs long enough to exercise the AVX2 32-byte block AND its scalar tail.
161// ----------------------------------------------------------------------------
162TEST(CheatahParsersJson, SimdScanPrimitives) {
163 EXPECT_TRUE(jsimd::is_whitespace(' '));
164 EXPECT_TRUE(jsimd::is_whitespace('\t'));
165 EXPECT_FALSE(jsimd::is_whitespace('x'));
166
167 // skip_whitespace: a >40-byte run of spaces then 'X' — past the fast path, through the block scan.
168 const std::string ws = std::string(40, ' ') + "X" + std::string(5, ' ');
169 const char* p = jsimd::skip_whitespace(ws.data(), ws.data() + ws.size());
170 ASSERT_LT(p, ws.data() + ws.size());
171 EXPECT_EQ(*p, 'X');
172 // fast path: first byte already non-whitespace -> returns `it` unchanged.
173 const std::string none = "abc";
174 EXPECT_EQ(jsimd::skip_whitespace(none.data(), none.data() + none.size()), none.data());
175
176 // find_quote_or_backslash: 40 ordinary bytes then a quote, then a backslash later.
177 const std::string body = std::string(40, 'a') + "\"more\\x";
178 const char* q = jsimd::find_quote_or_backslash(body.data(), body.data() + body.size());
179 ASSERT_LT(q, body.data() + body.size());
180 EXPECT_EQ(*q, '"');
181 const char* bs = jsimd::find_quote_or_backslash(q + 1, body.data() + body.size());
182 EXPECT_EQ(*bs, '\\');
183}

185// ----------------------------------------------------------------------------
186// The shared low-level scanners (scan.hpp detail::): number parsing (Clinger fast path + fallbacks),
187// string scanning, and escape decoding incl. \uXXXX and a surrogate pair (append_utf8 / hex4).
188// ----------------------------------------------------------------------------
189TEST(CheatahParsersJson, ScanNumbersAndEscapes) {
190 auto parse_d = [](std::string_view s) {
191 json::Cursor c{s.data(), s.data() + s.size()};
192 double out = 0.0;
193 EXPECT_TRUE(jdet::parse_double_fast(c, out)) << s;
194 return out;
195 };
196 EXPECT_DOUBLE_EQ(parse_d("0"), 0.0);
197 EXPECT_DOUBLE_EQ(parse_d("-3"), -3.0);
198 EXPECT_DOUBLE_EQ(parse_d("2.5"), 2.5);
199 EXPECT_DOUBLE_EQ(parse_d("-12.5e2"), -1250.0);
200 EXPECT_DOUBLE_EQ(parse_d("1e-3"), 0.001);
201 // Outside the exact fast window (25 digits) -> std::from_chars fallback path.
202 EXPECT_DOUBLE_EQ(parse_d("1234567890123456789012345"), 1234567890123456789012345.0);
203
204 // parse_arithmetic on an integral T: the base-10 loop, negative, and the overflow-refetch branch.
205 auto parse_ll = [](std::string_view s) {
206 json::Cursor c{s.data(), s.data() + s.size()};
207 long long out = 0;
208 EXPECT_TRUE(jdet::parse_arithmetic(c, out)) << s;
209 return out;
210 };
211 EXPECT_EQ(parse_ll("42"), 42);
212 EXPECT_EQ(parse_ll("-9223372036854775808"), -9223372036854775807LL - 1); // INT64_MIN, refetch path
213 { // unsigned rejects a negative literal
214 const std::string neg = "-1";
215 json::Cursor c{neg.data(), neg.data() + neg.size()};
216 unsigned long long u = 0;
217 EXPECT_FALSE(jdet::parse_arithmetic(c, u));
218 }

220 // decode_escapes drives append_utf8 for every \uXXXX form via the 1/2/3/4-byte arms:
221 // A -> 'A' (1 byte), é -> U+00E9 (2 bytes), € -> U+20AC (3 bytes),
222 // 😀 -> U+1F600 (surrogate pair -> 4 bytes). Plus the simple two-char
223 // escapes \t \b \f \r \/ \\ \". The input is the LITERAL backslash-escape text.
224 {
225 // Build the escape text explicitly (each backslash is a real byte, not a C escape).
226 const std::string raw =
227 "a\\tb\\nA\\u0041\\u00e9\\u20ac\\ud83d\\ude00\\b\\f\\r\\/\\\\\\\"z";
228 std::string decoded;
229 ASSERT_TRUE(jdet::decode_escapes(raw, decoded));
230 EXPECT_EQ(decoded,
231 std::string("a\tb\nAA\xC3\xA9\xE2\x82\xAC\xF0\x9F\x98\x80\b\f\r/\\\"z"));
232 }
233 { // scan_string over a quoted literal with an embedded escaped quote
234 const std::string src = R"("he\"llo")";
235 json::Cursor c{src.data(), src.data() + src.size()};
236 std::string_view inner;
237 bool esc = false;
238 ASSERT_TRUE(jdet::scan_string(c, inner, esc));
239 EXPECT_TRUE(esc);
240 EXPECT_EQ(inner, R"(he\"llo)");
241 }
242 { // scan_string on a plain (escape-free) string: esc stays false, bulk path.
243 const std::string src = R"("plain")";
244 json::Cursor c{src.data(), src.data() + src.size()};
245 std::string_view inner;
246 bool esc = true;
247 ASSERT_TRUE(jdet::scan_string(c, inner, esc));
248 EXPECT_FALSE(esc);
249 EXPECT_EQ(inner, "plain");
250 }
251 { // an UNTERMINATED string (no closing quote) -> scan_string returns false.
252 const std::string src = "\"no end";
253 json::Cursor c{src.data(), src.data() + src.size()};
254 std::string_view inner;
255 bool esc = false;
256 EXPECT_FALSE(jdet::scan_string(c, inner, esc));
257 }
258 { // a dangling escape at end of input -> scan_string returns false.
259 const std::string src = "\"x\\";
260 json::Cursor c{src.data(), src.data() + src.size()};
261 std::string_view inner;
262 bool esc = false;
263 EXPECT_FALSE(jdet::scan_string(c, inner, esc));
264 }
265 // match(): the literal matcher used by skip_value's t/f/n arms — hit, miss, and too-short.
266 {
267 const std::string t = "true", n = "nullish";
268 json::Cursor ct{t.data(), t.data() + t.size()};
269 EXPECT_TRUE(jdet::match(ct, "true"));
270 json::Cursor cn{n.data(), n.data() + n.size()};
271 EXPECT_TRUE(jdet::match(cn, "null"));
272 const std::string sh = "tr";
273 json::Cursor cs{sh.data(), sh.data() + sh.size()};
274 EXPECT_FALSE(jdet::match(cs, "true")); // too short to match
275 }
276 // Malformed escapes are rejected (bad \u hex, dangling backslash, lone high surrogate, and an
277 // unknown escape selector \x -> the switch default).
278 {
279 std::string out;
280 EXPECT_FALSE(jdet::decode_escapes(R"(\uZZZZ)", out));
281 EXPECT_FALSE(jdet::decode_escapes("\\", out));
282 EXPECT_FALSE(jdet::decode_escapes(R"(\uD83Dx)", out));
283 EXPECT_FALSE(jdet::decode_escapes("\\x", out)); // unknown selector -> default: return false
284 }
285 { // an input ENDING exactly at an escape's end takes the post-loop `return true` (no npos run).
286 std::string out;
287 EXPECT_TRUE(jdet::decode_escapes("\\t", out)); // just "\t" -> a single tab, loop exits at end
288 EXPECT_EQ(out, "\t");
289 }
290 // skip_value: discard a complete nested value in one call — the object holds an array, a string,
291 // and the three literal forms true/false/null (the t/f/n match arms) plus a number.
292 {
293 const std::string src = R"({"a":[1,2,{"b":true}],"c":"x","d":false,"e":null,"f":-3.5} tail)";
294 json::Cursor c{src.data(), src.data() + src.size()};
295 ASSERT_TRUE(jdet::skip_value(c));
296 EXPECT_EQ(std::string_view(c.it, static_cast<std::size_t>(c.end - c.it)), " tail");
297 }
298 // skip_value rejects malformed shapes: a lone closing brace, and stray punctuation.
299 {
300 const std::string bad = "}";
301 json::Cursor c{bad.data(), bad.data() + bad.size()};
302 EXPECT_FALSE(jdet::skip_value(c));
303 }
304 {
305 const std::string bad = ",";
306 json::Cursor c{bad.data(), bad.data() + bad.size()};
307 EXPECT_FALSE(jdet::skip_value(c));
308 }
309 { // skip_value over just a bare literal (false) leaves the cursor at end.
310 const std::string src = "false";
311 json::Cursor c{src.data(), src.data() + src.size()};
312 EXPECT_TRUE(jdet::skip_value(c));
313 EXPECT_EQ(c.it, c.end);
314 }
315}

318// PoolBuilder (pool_builder.hpp): the pooled construction stack machine. Drive it as the DOM parser
319// would — begin/add/finish for a nested array + object — and read the resulting ArrayView/ObjectView.
320// ----------------------------------------------------------------------------
321TEST(CheatahParsersJson, PoolBuilderStackMachine) {
322 json::PoolBuilder b;
323 b.reset(64); // reserve pools
324
325 // Build the array [1, "k": inner-object] -> actually: an array holding a Number then an object.
326 b.begin_array();
327 b.add_element(json::Node{json::Number{1.0}});
328 b.begin_object(); // an object nested inside the array
329 b.add_member(json::Member{json::Node{json::Number{0.0}}, json::Node{json::Boolean{true}}});
330 json::Node inner_obj = b.finish_object(); // commit_members -> ObjectView
331 b.add_element(std::move(inner_obj));
332 json::Node arr = b.finish_array(); // commit_nodes -> ArrayView
333
334 ASSERT_TRUE(std::holds_alternative<json::ArrayView>(arr.variant()));
335 const std::span<const json::Node> elems = std::get<json::ArrayView>(arr.variant()).value();
336 ASSERT_EQ(elems.size(), 2u);
337 EXPECT_DOUBLE_EQ(std::get<json::Number>(elems[0].variant()).value(), 1.0);
338 ASSERT_TRUE(std::holds_alternative<json::ObjectView>(elems[1].variant()));
339 const auto members = std::get<json::ObjectView>(elems[1].variant()).value();
340 ASSERT_EQ(members.size(), 1u);
341 EXPECT_TRUE(std::get<json::Boolean>(members[0].second.variant()).value());
342}

372// read<T>() dispatches each field on its STATIC type: integral (parse_arithmetic base-10 loop),
373// double, string, optional (null -> nullopt), vector. An UNKNOWN key exercises detail::skip_value
374// over a nested value (object containing an array/number), which must be discarded and skipped.
375TEST(CheatahParsersJson, TypedReadWithUnknownKeys) {
376 Trade t{};
377 const bool ok = json::read(
378 R"({"qty": -100, "extra": {"junk": [1, 2, {"deep": true}], "s": "skip\tme"},
379 "price": 3.25, "sym": "AAPL", "lot": null, "tags": [10, 20, 30]})",
380 t);
381 ASSERT_TRUE(ok);
382 EXPECT_EQ(t.qty, -100); // negative integral, base-10 loop
383 EXPECT_DOUBLE_EQ(t.price, 3.25);
384 EXPECT_EQ(t.sym, "AAPL");
385 EXPECT_FALSE(t.lot.has_value()); // JSON null -> nullopt
386 ASSERT_EQ(t.tags.size(), 3u);
387 EXPECT_EQ(t.tags[1], 20);
388}

401// Keys in NON-schema order (exercises the hint-wrap field search), an ESCAPED string VALUE
402// (read_string -> decode_escapes), and an ESCAPED KEY (q == 'q', decode_escapes on the key).
403TEST(CheatahParsersJson, TypedReadEscapesAndKeyOrder) {
404 Trade t{};
405 // "tags" first, then "sym" with a \t escape, then the escaped key "qty" (== "qty").
406 const bool ok = json::read(

408 ASSERT_TRUE(ok);
409 EXPECT_EQ(t.qty, 9); // matched via the escaped key
410 EXPECT_EQ(t.sym, std::string("a\tb")); // value decoded through decode_escapes
411 ASSERT_EQ(t.tags.size(), 2u);
412 EXPECT_EQ(t.tags[0], 1);
413 EXPECT_FALSE(t.lot.has_value());
414}
