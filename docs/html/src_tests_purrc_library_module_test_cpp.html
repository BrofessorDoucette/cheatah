1// Copyright (c) 2026 BigBrain LLC. MIT-licensed (see LICENSE).
2// Original work; see ACKNOWLEDGMENTS.md for the open-source ideas we build upon.
3// End-to-end tests for purrc's cheatah-library emitter (`--emit-library`) and the
4// consumer-side integrity verification:
5// - a TRANSPARENT module inlines its generated C++ source into a signed header (so the
6// true code is always visible);
7// - an OPAQUE module (the default) ships only the API in the header and hides the
8// implementation in a signed static archive;
9// - a program that `import`s a module verifies the module's SHA-512 checksum before
10// compiling against it, and REFUSES a tampered module (fail-closed).
11// See cmake/CheatahModule.cmake and the first-party `parsers` module.
12#include <sys/stat.h>
13
14#include <cstdio>
15#include <cstdlib>
16#include <fstream>
17#include <sstream>
18#include <string>
19
20#include <gtest/gtest.h>

22#ifndef PURRC_PATH
23#define PURRC_PATH ""
24#endif
25#ifndef CHEATAH_RUNTIME_PATH
26#define CHEATAH_RUNTIME_PATH ""
27#endif
28#ifndef PURR_TEST_TMP
29#define PURR_TEST_TMP "."
30#endif

32namespace {
33bool exists(const std::string& p) {
34 struct stat st;
35 return ::stat(p.c_str(), &st) == 0;
36}
37std::string read_file(const std::string& p) {
38 std::ifstream f(p);
39 std::ostringstream s;
40 s << f.rdbuf();
41 return s.str();
42}
43void write_file(const std::string& p, const std::string& c) {
44 std::ofstream f(p);
45 f << c;
46}
47int run(const std::string& cmd) { return std::system((cmd + " >/dev/null 2>&1").c_str()); }
48
49// A fresh per-test module dir <tmp>/<tag>/<name>/, with <name>.purr written into it.
50std::string make_module_dir(const std::string& tag, const std::string& name) {
51 const std::string root = std::string(PURR_TEST_TMP) + "/libmod_" + tag;
52 const std::string dir = root + "/" + name;
53 run("rm -rf " + root + " && mkdir -p " + dir);
54 write_file(dir + "/" + name + ".purr", "# " + name + " library module (mechanism test)\n");
55 return dir;
56}
57const std::string kPurrc = PURRC_PATH;
58} // namespace

60TEST(LibraryModule, TransparentEmitsSignedSourceInHeader) {
61 const std::string dir = make_module_dir("t", "foo");
62 ASSERT_EQ(run(kPurrc + " --emit-library --transparent " + dir + "/foo.purr -o " + dir + "/foo.hpp"), 0);
63 EXPECT_TRUE(exists(dir + "/foo.hpp"));
64 EXPECT_TRUE(exists(dir + "/foo.hpp.sha512")); // signed (checksum sidecar)
65 const std::string hdr = read_file(dir + "/foo.hpp");
66 EXPECT_NE(hdr.find("namespace cheatah::foo"), std::string::npos);
67 EXPECT_NE(hdr.find("inline const char* module_abi()"), std::string::npos); // source inline
68 EXPECT_FALSE(exists(dir + "/libcheatah_foo.a")); // transparent: no compiled archive
69}

71TEST(LibraryModule, OpaqueHidesSourceInSignedArchive) {
72 const std::string dir = make_module_dir("o", "foo");
73 ASSERT_EQ(run(kPurrc + " --emit-library " + dir + "/foo.purr -o " + dir + "/foo.hpp"), 0); // default = opaque
74 const std::string hdr = read_file(dir + "/foo.hpp");
75 EXPECT_NE(hdr.find("const char* module_abi() noexcept;"), std::string::npos); // API only
76 EXPECT_EQ(hdr.find("return \"foo\""), std::string::npos); // implementation NOT in the header
77 EXPECT_TRUE(exists(dir + "/libcheatah_foo.a")); // compiled, hidden, signed
78 EXPECT_TRUE(exists(dir + "/libcheatah_foo.a.sha512"));
79}

81TEST(LibraryModule, ImportVerifiesAndRuns) {
82 const std::string dir = make_module_dir("i", "foo");
83 const std::string root = dir.substr(0, dir.rfind('/')); // the search root holding foo/
84 ASSERT_EQ(run(kPurrc + " --emit-library --transparent " + dir + "/foo.purr -o " + dir + "/foo.hpp"), 0);
85 const std::string prog = root + "/prog.purr", mod = root + "/prog.so";
86 write_file(prog, "import io\nimport foo\nio.print(foo.module_abi())\n");
87 ASSERT_EQ(run("CHEATAH_MODULE_PATH=" + root + " " + kPurrc + " " + prog + " -o " + mod), 0);
88 FILE* pipe = popen((std::string(CHEATAH_RUNTIME_PATH) + " " + mod + " 2>/dev/null").c_str(), "r");
89 ASSERT_NE(pipe, nullptr);
90 std::string out;
91 char buf[128];
92 while (std::fgets(buf, sizeof buf, pipe)) out += buf;
93 pclose(pipe);
94 EXPECT_EQ(out, "foo\n");
95}

97TEST(LibraryModule, CheatahLinkMarkerForwardsExternalLibs) {
98 // A hand-written C++ module whose hidden implementation needs a system library
99 // declares it with a `// cheatah-link:` header marker; purrc must append that flag
100 // to the consumer's link command. Here the impl spawns a std::thread, so it needs
101 // -lpthread; without the marker forwarding it the program would fail to link on a
102 // toolchain that does not pull pthread in by default.
103 const std::string root = std::string(PURR_TEST_TMP) + "/libmod_link";
104 const std::string dir = root + "/thr";
105 run("rm -rf " + root + " && mkdir -p " + dir);
106 write_file(dir + "/thr.hpp",
107 "#pragma once\n// cheatah-link: -lpthread\n"
108 "namespace cheatah::thr {\nlong long on_thread();\n}\n");
109 write_file(root + "/impl.cpp",
110 "#include \"thr.hpp\"\n#include <thread>\n"
111 "namespace cheatah::thr {\nlong long on_thread() {\n"
112 " long long r = 0;\n std::thread t([&r]{ r = 7; });\n t.join();\n return r;\n}\n}\n");
113 ASSERT_EQ(run("g++ -std=c++20 -fPIC -I" + dir + " -c " + root + "/impl.cpp -o " + root + "/impl.o"), 0);
114 ASSERT_EQ(run("ar rcs " + dir + "/libcheatah_thr.a " + root + "/impl.o"), 0);
115 // Wrap the redirecting checksum commands in `sh -c '...'` so run()'s appended
116 // `>/dev/null` redirects the subshell's stdout, not the sidecar file writes.
117 ASSERT_EQ(run("sh -c 'cd " + dir + " && sha512sum thr.hpp > thr.hpp.sha512 && "
118 "sha512sum libcheatah_thr.a > libcheatah_thr.a.sha512'"), 0);
119 const std::string prog = root + "/prog.purr", mod = root + "/prog.so";
120 write_file(prog, "import io\nimport thr\nio.print(thr.on_thread())\n");
121 ASSERT_EQ(run("CHEATAH_MODULE_PATH=" + root + " " + kPurrc + " " + prog + " -o " + mod), 0);
122 FILE* pipe = popen((std::string(CHEATAH_RUNTIME_PATH) + " " + mod + " 2>/dev/null").c_str(), "r");
123 ASSERT_NE(pipe, nullptr);
124 std::string out;
125 char buf[64];
126 while (std::fgets(buf, sizeof buf, pipe)) out += buf;
127 pclose(pipe);
128 EXPECT_EQ(out, "7\n");
129}

131TEST(LibraryModule, DocCommentsBecomeJavadoc) {
132 const std::string dir = make_module_dir("d", "foo");
133 write_file(dir + "/foo.purr",
134 "# foo — a documented module. Exists to prove doc emission.\n"
135 "#\n"
136 "# Second paragraph of the module doc.\n"
137 "\n"
138 "# Twice the input.\n"
139 "#\n"
140 "# @param x the input.\n"
141 "# @return 2*x.\n"
142 "fn twice(x) {\n"
143 " # a body comment: NOT documentation\n"
144 " return x * 2 # trailing comment: NOT documentation\n"
145 "}\n"
146 "\n"
147 "# A documented struct.\n"
148 "struct P {\n"
149 " x: int\n"
150 "\n"
151 " # A documented method.\n"
152 " fn get(self) {\n"
153 " return self.x\n"
154 " }\n"
155 "}\n"
156 "\n"
157 "fn undocumented(x) {\n"
158 " return x\n"
159 "}\n"
160 "\n"
161 "# Detached comment (blank line below) — NOT documentation.\n"
162 "\n"
163 "fn detached(x) {\n"
164 " return x\n"
165 "}\n");
166 ASSERT_EQ(run(kPurrc + " --emit-library --transparent " + dir + "/foo.purr -o " + dir + "/foo.hpp"), 0);
167 const std::string hdr = read_file(dir + "/foo.hpp");
168
169 // Module doc -> @file Javadoc, above the namespace.
170 EXPECT_NE(hdr.find("/**\n * @file foo.hpp\n *\n"
171 " * foo — a documented module. Exists to prove doc emission.\n"
172 " *\n"
173 " * Second paragraph of the module doc.\n */"),
174 std::string::npos)
175 << hdr;
176 // fn doc -> Javadoc directly above the emitted function; @tags pass through verbatim.
177 // Library free fns are emitted `inline` — multi-TU consumers would hit ODR otherwise.
178 EXPECT_NE(hdr.find("/**\n * Twice the input.\n *\n"
179 " * @param x the input.\n * @return 2*x.\n */\ninline auto twice("),
180 std::string::npos)
181 << hdr;
182 // struct + method docs (method Javadoc is indented with its member).
183 EXPECT_NE(hdr.find("/**\n * A documented struct.\n */\nstruct P {"), std::string::npos) << hdr;
184 EXPECT_NE(hdr.find(" /**\n * A documented method.\n */\n"), std::string::npos) << hdr;
185 // Non-docs stay out of the header: body/trailing/detached comments, and an
186 // undocumented function gets no Javadoc block.
187 EXPECT_EQ(hdr.find("NOT documentation"), std::string::npos) << hdr;
188 EXPECT_EQ(hdr.find("*/\ninline auto undocumented("), std::string::npos) << hdr;
189}

191TEST(LibraryModule, FromImportBindsEnumAndStructBare) {
192 // `import Sym[, Sym…] from mod` binds each symbol at file scope, so a consumer names a
193 // module's enum member or struct WITHOUT the module prefix — `QuatCol.W`, `Vec({…})` — and it
194 // resolves unambiguously to `::cheatah::geo::QuatCol::W` / `::cheatah::geo::Vec`.
195 const std::string dir = make_module_dir("fi", "geo");
196 const std::string root = dir.substr(0, dir.rfind('/'));
197 write_file(dir + "/geo.purr",
198 "import ndarray\n"
199 "enum QuatCol { X, Y, Z, W }\n"
200 "struct Vec { x: int y: int }\n"
201 "fn width() -> int { return 4 }\n"); // a FUNCTION, from-imported alongside the types
202 ASSERT_EQ(run(kPurrc + " --emit-library --transparent " + dir + "/geo.purr -o " + dir + "/geo.hpp"), 0);
203 const std::string prog = root + "/prog.purr", mod = root + "/prog.so";
204 write_file(prog,
205 "import io\n"
206 "import ndarray\n"
207 "import QuatCol, Vec, width from geo\n" // type, struct, AND function — one statement
208 "let q = ndarray.zeros([4])\n"
209 "q[QuatCol.W] = 7.0\n" // enum member as a bare subscript
210 "let v: Vec = Vec({ .x = 3, .y = 4 })\n" // struct as a bare type AND constructor
211 "io.print(q[QuatCol.W], v.x, v.y, width())\n"); // bare function call
212 ASSERT_EQ(run("CHEATAH_MODULE_PATH=" + root + " " + kPurrc + " " + prog + " -o " + mod), 0);
213 FILE* pipe = popen((std::string(CHEATAH_RUNTIME_PATH) + " " + mod + " 2>/dev/null").c_str(), "r");
214 ASSERT_NE(pipe, nullptr);
215 std::string out;
216 char buf[64];
217 while (std::fgets(buf, sizeof buf, pipe)) out += buf;
218 pclose(pipe);
219 EXPECT_EQ(out, "7 3 4 4\n");
220}

222TEST(LibraryModule, TamperedModuleFailsClosed) {
223 const std::string dir = make_module_dir("x", "foo");
224 const std::string root = dir.substr(0, dir.rfind('/'));
225 ASSERT_EQ(run(kPurrc + " --emit-library --transparent " + dir + "/foo.purr -o " + dir + "/foo.hpp"), 0);
226 { // tamper the header — its committed checksum no longer matches
227 std::ofstream f(dir + "/foo.hpp", std::ios::app);
228 f << "\n// tampered\n";
229 }
230 const std::string prog = root + "/prog.purr", mod = root + "/prog.so";
231 write_file(prog, "import io\nimport foo\nio.print(foo.module_abi())\n");
232 EXPECT_NE(run("CHEATAH_MODULE_PATH=" + root + " " + kPurrc + " " + prog + " -o " + mod), 0)
233 << "a consumer must refuse a module whose checksum no longer matches";
234}
