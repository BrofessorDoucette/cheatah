1// Compile-run unit tests for the `ed25519` module: one test per function, each a tiny
2// .purr compiled by purrc and run under the cheatah runtime. Keys/signatures are hex.
3// The deterministic cases use the RFC 8032 §7.1 known-answer vectors; generate() is
4// non-deterministic, so it is asserted by PROPERTY (a fresh key still signs+verifies).
5// Complements the in-process unit tests (stdlib/tests/ed25519_test.cpp) and the
6// per-module system-level test (StdlibE2E.Ed25519).
7#include "e2e_harness.hpp"
8
9// RFC 8032 TEST 3: seed c5aa… -> public key fc51….
10TEST(Ed25519CompileRun, PublicKey) {
11 e2e::expect_e2e("ed25519_public_key", R"PURR(import io
12import ed25519
13io.print(ed25519.public_key("c5aa8df43f9f837bedb7442f31dcb7b166d38535076f094b85ce3a2e0b4458f7"))
14)PURR", "fc51cd8e6218a1a38da47ed00230f0580816ed13ba3303ac5deb911548908025\n");
15}
16
17// Sign a message and verify it — the round trip a recipient performs.
18TEST(Ed25519CompileRun, SignVerify) {
19 e2e::expect_e2e("ed25519_sign_verify", R"PURR(import io
20import ed25519
21let secret = "c5aa8df43f9f837bedb7442f31dcb7b166d38535076f094b85ce3a2e0b4458f7"
22let pub = ed25519.public_key(secret)
23let sig = ed25519.sign(secret, "hello cheatah")
24io.print(ed25519.verify(pub, "hello cheatah", sig))
25)PURR", "True\n");
26}
27
28// generate() makes a fresh secret seed; it must still produce a working keypair.
29TEST(Ed25519CompileRun, Generate) {
30 e2e::expect_e2e("ed25519_generate", R"PURR(import io
31import ed25519
32let secret = ed25519.generate()
33let pub = ed25519.public_key(secret)
34let sig = ed25519.sign(secret, "msg")
35io.print(len(secret), ed25519.verify(pub, "msg", sig))
36)PURR", "64 True\n");
37}
