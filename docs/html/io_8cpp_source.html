60std::ios::openmode File::translate_mode(std::string_view mode) {
61 // Python modes: r / w / a (+ optional '+' for read-update, 'b' for binary).
62 std::ios::openmode m{};
63 const bool plus = mode.find('+') != std::string_view::npos;
64 if (mode.find('b') != std::string_view::npos) m |= std::ios::binary;
65 if (mode.find('r') != std::string_view::npos) { m |= std::ios::in; if (plus) m |= std::ios::out; }
66 if (mode.find('w') != std::string_view::npos) { m |= std::ios::out | std::ios::trunc; if (plus) m |= std::ios::in; }
67 if (mode.find('a') != std::string_view::npos) { m |= std::ios::out | std::ios::app; if (plus) m |= std::ios::in; }
68 if (m == std::ios::openmode{}) m = std::ios::in; // default 'r'
69 return m;
70}
