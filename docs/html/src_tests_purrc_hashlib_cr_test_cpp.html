tests/purrc/hashlib_cr_test.cpp

1// Compile-run unit tests for the `hashlib` module: one test per function. Each
2// writes a tiny .purr that calls a single hashlib function on a known input,
3// compiles it with purrc, runs it under the cheatah runtime, and asserts the
4// exact stdout. Complements the in-process unit tests
5// (stdlib/tests/hashlib_test.cpp) and the per-module system-level test
6// (StdlibE2E.Hashlib).
7#include "e2e_harness.hpp"
8
9// Standard NIST/FIPS-180 SHA-256 test vector for "abc".
10TEST(HashlibCompileRun, Sha256) {
11 e2e::expect_e2e("hashlib_sha256", R"PURR(import io
12import hashlib
13io.print(hashlib.sha256("abc"))
14)PURR", "ba7816bf8f01cfea414140de5dae2223b00361a396177a9cb410ff61f20015ad\n");
15}
16
17// Standard NIST/FIPS-180 SHA-512 test vector for "abc".
18TEST(HashlibCompileRun, Sha512) {
19 e2e::expect_e2e("hashlib_sha512", R"PURR(import io
20import hashlib
21io.print(hashlib.sha512("abc"))
22)PURR", "ddaf35a193617abacc417349ae20413112e6fa4e89a97ea20a9eeee64b55d39a"
23 "2192992a274fc1a836ba3c23a3feebbd454d4423643ce80e2a9ac94fa54ca49f\n");
24}
25
26// The raw digest forms return the bytes (32 / 64), not hex — assert their length so the
27// program output stays deterministic.
28TEST(HashlibCompileRun, Sha256Digest) {
29 e2e::expect_e2e("hashlib_sha256_digest", R"PURR(import io
30import hashlib
31io.print(len(hashlib.sha256_digest("abc")))
32)PURR", "32\n");
33}
34
35TEST(HashlibCompileRun, Sha512Digest) {
36 e2e::expect_e2e("hashlib_sha512_digest", R"PURR(import io
37import hashlib
38io.print(len(hashlib.sha512_digest("abc")))
39)PURR", "64\n");
40}
