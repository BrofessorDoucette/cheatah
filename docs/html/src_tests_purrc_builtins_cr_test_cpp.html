1// Compile-run unit tests for the `builtins` module: one test per built-in. Each
2// writes a tiny .purr that calls a single built-in (built-ins need no `import`
3// and take no module prefix, e.g. `len("hi")`), compiles it with purrc, runs it
4// under the cheatah runtime, and asserts the exact stdout. Complements the
5// in-process unit tests (stdlib/tests/builtins_test.cpp).
6//
7// Note: the keyword-named conversions are spelled with their Python names in
8// cheatah source (`int`/`float`/`bool`), which the compiler maps to
9// builtins::to_int/to_float/to_bool. The `hash` built-in is intentionally not
10// covered here: its value is implementation-defined and not stable across
11// compilers/libstdc++ versions, so it has no portable expected stdout.
12#include "e2e_harness.hpp"
13
14TEST(BuiltinsCompileRun, Len) {
15 e2e::expect_e2e("builtins_len", R"PURR(import io
16io.print(len("meow"))
17)PURR", "4\n");
18}
