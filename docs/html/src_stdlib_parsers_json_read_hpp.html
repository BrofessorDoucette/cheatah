3// cheatah::parsers::json — read<T>(): parse JSON DIRECTLY into a typed struct, no Node/variant DOM.
4//
5// struct Series { std::string symbol; std::vector<Ohlc> data; }; // + schema<> specializations
6// Series s;
7// if (json::read(text, s)) use(s);
8//
9// Each field is dispatched on its STATIC type (the concept set below), so a number lands in its
10// typed field via std::from_chars with no intermediate double, a string is copied (std::string) or
11// viewed (std::string_view), nested structs recurse on their schema, std::vector<T> reads a
12// variable-length array, std::array<T, N> reads a FIXED N-element array inline (no allocation), and
13// std::optional<T> reads JSON null as nullopt. Unknown keys are skipped. Validate is the same
14// compile-time switch as the DOM parser: with Validate=false every bounds/structure check is
15// `if constexpr`-removed for trusted input (UB on malformed input). No runtime polymorphism.
16//
17// The data-structure recursion (nested structs / vectors) is bounded by the SCHEMA depth, which is
18// fixed at compile time — not attacker-controlled — so natural recursion here cannot be overflowed
19// by deep input. (Skipping an unknown value of attacker-controlled depth uses detail::skip_value,
20// which is iterative.)

56// Mutually-recursive workers (value -> array/object -> value). Declared first so they can call one
57// another regardless of definition order.
58//
59// read_value is FORCE-inlined: the per-instantiation call graph is acyclic (a type cannot contain
60// itself), so this is safe, and it matters — the optimizer's size heuristic otherwise outlines the
61// scalar instantiations (read_value<long>, read_value<string_view>), adding a function-call round
62// trip to every scalar member; inlining the whole hot path removes it.
63// @complexity O(value size) @alloc only what the field type owns @test JsonRead.Scalars
64template <bool Validate, class T>
65[[gnu::always_inline]] inline bool read_value(Cursor& c, T& out);
66template <bool Validate, class Vec>
67bool read_array(Cursor& c, Vec& out);
68template <bool Validate, class Arr>
69bool read_fixed_array(Cursor& c, Arr& out);
70template <bool Validate, class T, class... Fields>
71bool read_object(Cursor& c, T& out, const ObjectSchema<Fields...>& sch);

73// Read a JSON string body into an owned std::string (decoding escapes) or a zero-copy view (only
74// when unescaped — an escaped string has no contiguous source to view).
75// @complexity O(|string|) @alloc only for an owned std::string @test JsonRead.ViewVsOwned
76template <bool Validate, class Str>
77bool read_string(Cursor& c, Str& out) {
78 if constexpr (Validate) {
79 if (c.it == c.end || *c.it != '"') {
80 return false;
81 }
82 }
83 std::string_view raw;
84 bool esc = false;
85 if (!scan_string(c, raw, esc)) {
86 return false;
87 }
88 if constexpr (std::is_same_v<Str, std::string_view>) {
89 if (esc) {
90 return false; // cannot view an escaped string; declare the field std::string instead
91 }
92 out = raw;
93 return true;
94 } else { // std::string (owned)
95 if (!esc) {
96 out.assign(raw);
97 return true;
98 }
99 return decode_escapes(raw, out);
100 }
101}

103template <bool Validate, class T>
104inline bool read_value(Cursor& c, T& out) {
105 skip_ws(c);
106 if constexpr (Validate) {
107 if (c.it == c.end) {
108 return false;
109 }
110 }
111 if constexpr (std::is_same_v<T, bool>) {
112 if (match(c, "true")) {
113 out = true;
114 return true;
115 }
116 if (match(c, "false")) {
117 out = false;
118 return true;
119 }
120 return false;
121 } else if constexpr (std::is_arithmetic_v<T>) { // integral or floating (bool handled above)
122 return parse_arithmetic(c, out);
123 } else if constexpr (std::is_same_v<T, std::string> ||
124 std::is_same_v<T, std::string_view>) {
125 return read_string<Validate>(c, out);
126 } else if constexpr (is_optional<T>::value) {
127 if (match(c, "null")) {
128 out.reset();
129 return true;
130 }
131 return read_value<Validate>(c, out.emplace());
132 } else if constexpr (is_std_array<T>::value) {
133 return read_fixed_array<Validate>(c, out); // fixed N elements, inline (no allocation)
134 } else if constexpr (is_vector<T>::value) {
135 return read_array<Validate>(c, out);
136 } else if constexpr (HasSchema<T>) {
137 return read_object<Validate>(c, out, schema<T>);
138 } else {
139 static_assert(sizeof(T) == 0,
140 "json::read: type is not a supported scalar/container and has no schema<T>");
141 return false;
142 }
143}

145// Read a variable-length JSON array into a std::vector, element by element, in place.
146// @complexity O(array size) @alloc vector growth (capacity reused via clear) @test JsonRead.Vectors
147template <bool Validate, class Vec>
148bool read_array(Cursor& c, Vec& out) {
149 if constexpr (Validate) {
150 if (c.it == c.end || *c.it != '[') {
151 return false;
152 }
153 }
154 ++c.it; // skip '['
155 skip_ws(c);
156 if constexpr (Validate) {
157 if (c.it == c.end) {
158 return false;
159 }
160 }
161 out.clear();
162 if (*c.it == ']') {
163 ++c.it;
164 return true; // empty array
165 }
166 for (;;) {
167 // Construct the element in place and read straight into it — no temporary, no move.
168 if (!read_value<Validate>(c, out.emplace_back())) {
169 return false;
170 }
171 skip_ws(c);
172 if constexpr (Validate) {
173 if (c.it == c.end) {
174 return false;
175 }
176 }
177 const char sep = *c.it++;
178 if (sep == ']') {
179 return true;
180 }
181 if (sep != ',') {
182 return false; // expected ',' or ']'
183 }
184 }
185}

187// Read a JSON array of EXACTLY N elements into a std::array<E, N> — fixed size, stored inline, so
188// it never allocates. A wrong element count (too few / too many) is a parse error under Validate.
189// @complexity O(N) @alloc none @test JsonRead.FixedArrays
190template <bool Validate, class Arr>
191bool read_fixed_array(Cursor& c, Arr& out) {
192 constexpr std::size_t kSize = std::tuple_size_v<Arr>;
193 if constexpr (Validate) {
194 if (c.it == c.end || *c.it != '[') {
195 return false;
196 }
197 }
198 ++c.it; // skip '['
199 for (std::size_t i = 0; i < kSize; ++i) {
200 if (!read_value<Validate>(c, out[i])) {
201 return false;
202 }
203 skip_ws(c);
204 if constexpr (Validate) {
205 // every element but the last is followed by ',', the last by ']' — anything else is
206 // the wrong arity or a missing separator.
207 if (c.it == c.end || *c.it != (i + 1 == kSize ? ']' : ',')) {
208 return false;
209 }
210 }
211 ++c.it; // consume the ',' (between elements) or the closing ']' (after the last)
212 }
213 if constexpr (kSize == 0) { // std::array<E, 0>: no elements, so still consume the ']'
214 skip_ws(c);
215 if constexpr (Validate) {
216 if (c.it == c.end || *c.it != ']') {
217 return false;
218 }
219 }
220 ++c.it;
221 }
222 return true;
223}

225// Match `key` against the schema's fields; read the value into the matching member, or skip it.
226// Read one complete `"key": value` member into the matching field of `out`. On entry the cursor is
227// at the opening quote of the key (or the whitespace before it on the general path).
228//
229// FAST PATH — predicted-literal match: `hint` names the field we EXPECT next (the one after the
230// previous match), because real JSON nearly always lists keys in schema order. We compare the
231// input bytes directly against that field's `"name":` literal — one short memcmp replaces the
232// whole key pipeline (quote scan, key compare, colon handling). In-order compact input takes this
233// path for every member. Anything else — whitespace inside the member syntax, out-of-order,
234// unknown, or escaped keys — leaves the cursor untouched and falls back to the general scan.
235// @complexity O(|key| + value) @alloc only a temp std::string when the key itself is escaped (rare); otherwise none of its own @test JsonRead.OutOfOrderKeys
236template <bool Validate, class T, class... Fields>
237bool read_one_member(Cursor& c, T& out, const ObjectSchema<Fields...>& sch, std::size_t& hint) {
238 constexpr std::size_t kCount = sizeof...(Fields);
239 bool ok = true;
240 std::size_t matched = kCount;
241 const auto read_into = [&](const auto& f, std::size_t index) {
242 matched = index;
243 ok = read_value<Validate>(c, out.*(f.ptr));
244 };
245
246 // ---- fast path: does the input start with the predicted field's `"name":` bytes? ----
247 const auto try_predicted = [&](const auto& f, std::size_t index) {
248 const std::string_view name = f.name;
249 const char* const p = c.it;
250 if (static_cast<std::size_t>(c.end - p) < name.size() + 3) {
251 return false; // not enough bytes for `"name":` — let the general path decide
252 }
253 if (p[0] != '"' || p[name.size() + 1] != '"' || p[name.size() + 2] != ':' ||
254 std::memcmp(p + 1, name.data(), name.size()) != 0) {
255 return false; // not the predicted key (cursor untouched)
256 }
257 c.it = p + name.size() + 3; // step past `"name":` in one go
258 read_into(f, index);
259 return true;
260 };
261 const auto predicted = [&]<std::size_t... Is>(std::index_sequence<Is...>) {
262 return (((Is == hint) && try_predicted(std::get<Is>(sch.fields), Is)) || ...);
263 };
264
265 if (!predicted(std::make_index_sequence<kCount>{})) {
266 // ---- general path: scan the key, then match it against the fields ----
267 if constexpr (Validate) {
268 if (c.it == c.end || *c.it != '"') {
269 return false; // key must be a string
270 }
271 }
272 std::string_view key;
273 bool esc = false;
274 if (!scan_string(c, key, esc)) {
275 return false;
276 }
277 std::string key_decoded; // only used when the key itself contains escapes (rare)
278 if (esc) {
279 if (!decode_escapes(key, key_decoded)) {
280 return false;
281 }
282 key = key_decoded;
283 }
284 skip_ws(c);
285 if constexpr (Validate) {
286 if (c.it == c.end || *c.it != ':') {
287 return false;
288 }
289 }
290 ++c.it; // skip ':'
291 const auto try_field = [&](const auto& f, std::size_t index) {
292 if (f.name != key) {
293 return false; // not this field — keep looking
294 }
295 read_into(f, index);
296 return true; // matched — stop the fold
297 };
298 // Search [hint, N) first, then wrap to [0, hint) — the `Is >= hint` guards on skipped
299 // indices are just integer compares, and the `||` folds short-circuit at the first match.
300 const auto search = [&]<std::size_t... Is>(std::index_sequence<Is...>) {
301 return ((Is >= hint && try_field(std::get<Is>(sch.fields), Is)) || ...) ||
302 ((Is < hint && try_field(std::get<Is>(sch.fields), Is)) || ...);
303 };
304 if (!search(std::make_index_sequence<kCount>{})) {
305 return skip_value(c); // a key not in the schema: discard its value
306 }
307 }
308 if (!ok) {
309 return false;
310 }
311 hint = (matched + 1 == kCount) ? 0 : matched + 1;
312 return true;
313}

315// Read a JSON object member-by-member into the schema'd struct `out`.
316// @complexity O(object size) @alloc only what the field types own @test JsonRead.NestedStructs
317template <bool Validate, class T, class... Fields>
318bool read_object(Cursor& c, T& out, const ObjectSchema<Fields...>& sch) {
319 if constexpr (Validate) {
320 if (c.it == c.end || *c.it != '{') {
321 return false;
322 }
323 }
324 ++c.it; // skip '{'
325 skip_ws(c);
326 if constexpr (Validate) {
327 if (c.it == c.end) {
328 return false;
329 }
330 }
331 if (*c.it == '}') {
332 ++c.it;
333 return true; // empty object
334 }
335 std::size_t hint = 0; // the field we expect NEXT (keys usually arrive in schema order)
336 for (;;) {
337 skip_ws(c);
338 if (!read_one_member<Validate>(c, out, sch, hint)) { // `"key": value` (predicted or scanned)
339 return false;
340 }
341 skip_ws(c);
342 if constexpr (Validate) {
343 if (c.it == c.end) {
344 return false;
345 }
346 }
347 const char sep = *c.it++;
348 if (sep == '}') {
349 return true;
350 }
351 if (sep != ',') {
352 return false; // expected ',' or '}'
353 }
354 }
355}

359/**
360 * Parse `text` directly into `out` (a struct with a schema<>, a supported scalar, std::vector,
361 * std::array, std::optional, or std::string/std::string_view). Returns true on success. With
362 * Validate=true the input is fully checked (including no trailing junk); with Validate=false every
363 * bounds/structure check is compiled out for trusted, well-formed input (malformed input is then
364 * undefined behavior, and no success/trailing check is performed — it always returns true).
365 *
366 * std::string fields OWN their characters (safe to outlive `text`); std::string_view fields VIEW
367 * `text` (which must then outlive `out`) and reject escaped strings.
368 *
369 * @complexity O(n) in the input length
370 * @alloc only the owned std::string fields / vector growth in `out` — no DOM, no Node tree
371 * @test JsonRead.*
372 */
373template <bool Validate = true, class T>
374[[nodiscard]] bool read(std::string_view text, T& out) {
375 Cursor c{text.data(), text.data() + text.size()};
376 if constexpr (Validate) {
377 if (!detail::read_value<true>(c, out)) {
378 return false;
379 }
380 detail::skip_ws(c);
381 return c.it == c.end; // reject trailing junk
382 } else {
383 // Trusted input: read with all checks compiled out, no success test, no trailing-junk check.
384 detail::read_value<false>(c, out);
385 return true;
386 }
387}
