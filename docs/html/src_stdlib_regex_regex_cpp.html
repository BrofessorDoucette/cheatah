13// A from-scratch, linear-time regex engine: the pattern compiles to a Thompson NFA and runs as
14// a lazy DFA (RE2-style). Matching touches only the input bytes (via string_view) and integer
15// program-counters — it NEVER allocates an intermediate string. The compiled program and its
16// DFA cache live in one heap-owned `Dfa` that a `Pattern` shares, so reuse is fast.
17namespace cheatah::regex {

21struct Inst {
22 enum Op : std::uint8_t { Byte, Split, Jmp, Match } op = Match;
23 std::uint64_t cls[4] = {0, 0, 0, 0}; // 256-bit accept set for Byte
24 int x = 0, y = 0; // targets
25};
26
27// ---- 256-bit byte class helpers ------------------------------------------------------
28void set_bit(std::uint64_t cls[4], unsigned char b) { cls[b >> 6] |= (std::uint64_t{1} << (b & 63)); }
29bool has_bit(const std::uint64_t cls[4], unsigned char b) { return (cls[b >> 6] >> (b & 63)) & 1u; }
30void set_range(std::uint64_t cls[4], unsigned lo, unsigned hi) {
31 for (unsigned c = lo; c <= hi; ++c) set_bit(cls, static_cast<unsigned char>(c));
32}
33void negate(std::uint64_t cls[4]) { for (int i = 0; i < 4; ++i) cls[i] = ~cls[i]; }
34
35// ---- the compiler: regex source -> NFA program (Thompson construction) ----------------
36struct Hole { int inst; int field; }; // a dangling exit: (instruction, 0=x|1=y)
37struct Frag { int start; std::vector<Hole> holes; };

39struct Compiler {
40 std::string_view src;
41 std::size_t i = 0;
42 std::vector<Inst> prog;
43 std::string err;
44
45 bool eof() const { return i >= src.size(); }
46 char peek() const { return i < src.size() ? src[i] : '\0'; }
47 char get() { return src[i++]; }
48 int emit(const Inst& in) { prog.push_back(in); return static_cast<int>(prog.size()) - 1; }
49 void patch(const std::vector<Hole>& holes, int target) {
50 for (const Hole& h : holes) (h.field == 0 ? prog[h.inst].x : prog[h.inst].y) = target;
51 }
52
53 Frag byte_frag(const std::uint64_t cls[4]) {
54 Inst in; in.op = Inst::Byte;
55 for (int k = 0; k < 4; ++k) in.cls[k] = cls[k];
56 int s = emit(in);
57 return Frag{s, {{s, 0}}};
58 }
59
60 bool escape_class(char c, std::uint64_t cls[4]) {
61 switch (c) {
62 case 'd': set_range(cls, '0', '9'); return true;
63 case 'w': set_range(cls, 'a', 'z'); set_range(cls, 'A', 'Z');
64 set_range(cls, '0', '9'); set_bit(cls, '_'); return true;
65 case 's': set_bit(cls, ' '); set_bit(cls, '\t'); set_bit(cls, '\n');
66 set_bit(cls, '\r'); set_bit(cls, '\f'); set_bit(cls, '\v'); return true;
67 case 'D': case 'W': case 'S': {
68 std::uint64_t base[4] = {0, 0, 0, 0};
69 escape_class(static_cast<char>(c + 32), base);
70 negate(base);
71 for (int k = 0; k < 4; ++k) cls[k] |= base[k];
72 return true;
73 }
74 default: return false;
75 }
76 }
77
78 Frag parse_alt() {
79 Frag left = parse_concat();
80 while (err.empty() && peek() == '|') {
81 get();
82 Frag right = parse_concat();
83 if (!err.empty()) return left;
84 Inst sp; sp.op = Inst::Split; sp.x = left.start; sp.y = right.start;
85 int s = emit(sp);
86 Frag f; f.start = s; f.holes = left.holes;
87 f.holes.insert(f.holes.end(), right.holes.begin(), right.holes.end());
88 left = f;
89 }
90 return left;
91 }
92
93 Frag parse_concat() {
94 Frag acc; acc.start = -1;
95 while (err.empty() && !eof() && peek() != '|' && peek() != ')') {
96 Frag r = parse_repeat();
97 if (!err.empty()) return acc;
98 if (acc.start == -1) acc = r;
99 else { patch(acc.holes, r.start); acc.holes = r.holes; }
100 }
101 if (acc.start == -1) { // empty -> an epsilon Jmp pass-through
102 Inst j; j.op = Inst::Jmp; int s = emit(j);
103 acc.start = s; acc.holes = {{s, 0}};
104 }
105 return acc;
106 }
107
108 Frag parse_repeat() {
109 Frag a = parse_atom();
110 while (err.empty() && (peek() == '*' || peek() == '+' || peek() == '?')) {
111 char q = get();
112 Inst sp; sp.op = Inst::Split;
113 if (q == '*') {
114 sp.x = a.start; int s = emit(sp);
115 patch(a.holes, s);
116 a = Frag{s, {{s, 1}}};
117 } else if (q == '+') {
118 sp.x = a.start; int s = emit(sp);
119 patch(a.holes, s);
120 a = Frag{a.start, {{s, 1}}};
121 } else { // '?'
122 sp.x = a.start; int s = emit(sp);
123 Frag f; f.start = s; f.holes = a.holes; f.holes.push_back({s, 1});
124 a = f;
125 }
126 }
127 return a;
128 }
129
130 Frag parse_atom() {
131 char c = peek();
132 if (c == '(') {
133 get();
134 Frag inner = parse_alt();
135 if (peek() != ')') { err = "unbalanced '('"; return inner; }
136 get();
137 return inner;
138 }
139 if (c == '[') return parse_class();
140 if (c == '.') { // any byte except newline
141 get();
142 std::uint64_t cls[4] = {~std::uint64_t{0}, ~std::uint64_t{0}, ~std::uint64_t{0}, ~std::uint64_t{0}};
143 cls['\n' >> 6] &= ~(std::uint64_t{1} << ('\n' & 63));
144 return byte_frag(cls);
145 }
146 if (c == '\\') {
147 get();
148 if (eof()) { err = "trailing backslash"; return {}; }
149 char e = get();
150 std::uint64_t cls[4] = {0, 0, 0, 0};
151 if (escape_class(e, cls)) return byte_frag(cls);
152 set_bit(cls, static_cast<unsigned char>(e)); // escaped literal
153 return byte_frag(cls);
154 }
155 if (c == ')' || c == '|' || c == '*' || c == '+' || c == '?') { err = "unexpected metacharacter"; return {}; }
156 get();
157 std::uint64_t cls[4] = {0, 0, 0, 0};
158 set_bit(cls, static_cast<unsigned char>(c));
159 return byte_frag(cls);
160 }
161
162 Frag parse_class() {
163 get(); // '['
164 bool neg = false;
165 if (peek() == '^') { get(); neg = true; }
166 std::uint64_t cls[4] = {0, 0, 0, 0};
167 bool first = true;
168 while (!eof() && (peek() != ']' || first)) {
169 first = false;
170 char c = get();
171 if (c == '\\' && !eof()) {
172 char e = get();
173 std::uint64_t sub[4] = {0, 0, 0, 0};
174 if (escape_class(e, sub)) { for (int k = 0; k < 4; ++k) cls[k] |= sub[k]; continue; }
175 c = e;
176 }
177 if (peek() == '-' && i + 1 < src.size() && src[i + 1] != ']') {
178 get();
179 char hi = get();
180 set_range(cls, static_cast<unsigned char>(c), static_cast<unsigned char>(hi));
181 } else {
182 set_bit(cls, static_cast<unsigned char>(c));
183 }
184 }
185 if (peek() != ']') { err = "unbalanced '['"; return {}; }
186 get();
187 if (neg) { negate(cls); cls['\n' >> 6] &= ~(std::uint64_t{1} << ('\n' & 63)); } // negated class excludes newline
188 return byte_frag(cls);
189 }
190};

194/// The compiled Thompson-NFA program plus its lazy-DFA cache — an internal implementation type
195/// (@ref Pattern holds one by `shared_ptr`; not part of the public surface).
196struct Dfa {
197 std::vector<Inst> prog; ///< the compiled Thompson-NFA program.
198 int start_unanchored = 0; ///< program entry pc for an unanchored start.
199 int start_anchored = 0; ///< program entry pc for an anchored start.
200 bool anchored_start = false; ///< whether the pattern is anchored at the start (`^`).
201 bool anchored_end = false; ///< whether the pattern is anchored at the end (`$`).
202
203 std::unordered_map<std::string, int> intern; ///< canonical (sorted) pc-set key -> DFA state id.
204 std::vector<std::vector<int>> pcs; ///< per-state pc set.
205 std::vector<char> accepts; ///< per-state "accepting" flag.
206 std::vector<int> tflat; ///< flat transition table (`tflat[state*256+byte]`; -1 = uncomputed).
207
208 std::uint64_t first[4] = {0, 0, 0, 0}; ///< bitset of bytes an anchored match can START with.
209 int first_count = 0; ///< popcount of `first` (0 = unknown / matches empty).
210 unsigned char first_byte = 0; ///< the single required first byte when `first_count == 1`.
211 bool matches_empty = false; ///< the pattern can match the empty string at any position.
212
213 /// Add the epsilon-closure of @p pc to @p out. @param pc start program counter.
214 /// @param out accumulates the reachable Byte/Match pcs. @param seen per-pc visited flags.
215 void add_closure(int pc, std::vector<int>& out, std::vector<char>& seen) const {
216 if (pc < 0 || seen[pc]) return;
217 seen[pc] = 1;
218 const Inst& in = prog[pc];
219 if (in.op == Inst::Jmp) add_closure(in.x, out, seen);
220 else if (in.op == Inst::Split) { add_closure(in.x, out, seen); add_closure(in.y, out, seen); }
221 else out.push_back(pc); // Byte or Match — a real state
222 }
223
224 /// Intern a pc @p set into a canonical DFA state (creating it if new).
225 /// @param set the pc set for the state. @return the state id.
226 int intern_state(std::vector<int> set) {
227 std::sort(set.begin(), set.end());
228 set.erase(std::unique(set.begin(), set.end()), set.end());
229 std::string key(reinterpret_cast<const char*>(set.data()), set.size() * sizeof(int));
230 auto it = intern.find(key);
231 if (it != intern.end()) return it->second;
232 int id = static_cast<int>(pcs.size());
233 bool acc = false;
234 for (int pc : set) if (prog[pc].op == Inst::Match) acc = true;
235 intern.emplace(std::move(key), id);
236 pcs.push_back(std::move(set));
237 accepts.push_back(acc ? 1 : 0);
238 tflat.resize(tflat.size() + 256, -1);
239 return id;
240 }
241
242 /// The start DFA state for program entry @p entry. @param entry the program entry pc.
243 /// @return the interned start state id.
244 int start_state(int entry) {
245 std::vector<int> out;
246 std::vector<char> seen(prog.size(), 0);
247 add_closure(entry, out, seen);
248 return intern_state(std::move(out));
249 }
250
251 /// Transition from @p state on input byte @p b, lazily filling the cache. @param state the current
252 /// DFA state id. @param b the input byte. @return the next state id.
253 int step(int state, unsigned char b) {
254 int cached = tflat[static_cast<std::size_t>(state) * 256 + b];
255 if (cached != -1) return cached;
256 std::vector<int> out;
257 std::vector<char> seen(prog.size(), 0);
258 for (int pc : pcs[state]) {
259 const Inst& in = prog[pc];
260 if (in.op == Inst::Byte && has_bit(in.cls, b)) add_closure(in.x, out, seen);
261 }
262 int next = intern_state(std::move(out)); // may grow tflat -> index AFTER, not before
263 tflat[static_cast<std::size_t>(state) * 256 + b] = next;
264 return next;
265 }
266};

268Pattern compile(std::string_view pattern) {
269 Pattern out;
270 auto dfa = std::make_shared<Dfa>();
271 if (!pattern.empty() && pattern.front() == '^') { dfa->anchored_start = true; pattern.remove_prefix(1); }
272 if (!pattern.empty() && pattern.back() == '$') { dfa->anchored_end = true; pattern.remove_suffix(1); }
273
274 Compiler c;
275 // Slots 0..1 = the unanchored `.*?` prefix, so an unanchored search begins a match at ANY
276 // position with no per-position work: 0: Split(body, 1); 1: Byte(any) -> 0.
277 c.emit(Inst{});
278 Inst anybyte; anybyte.op = Inst::Byte;
279 for (int k = 0; k < 4; ++k) anybyte.cls[k] = ~std::uint64_t{0};
280 anybyte.x = 0;
281 c.emit(anybyte);
282 c.src = pattern;
283
284 Frag f = c.parse_alt();
285 if (c.err.empty() && c.i != c.src.size()) c.err = "unexpected trailing input";
286 if (!c.err.empty()) { out.error = c.err; return out; }
287
288 Inst m; m.op = Inst::Match;
289 int match_pc = c.emit(m);
290 c.patch(f.holes, match_pc);
291 c.prog[0].op = Inst::Split;
292 c.prog[0].x = f.start; // anchored entry: the pattern body
293 c.prog[0].y = 1;
294
295 dfa->prog = std::move(c.prog);
296 dfa->start_unanchored = 0;
297 dfa->start_anchored = f.start;
298
299 // The bytes an anchored match can begin with (its start-state closure's Byte classes) — the
300 // fast prefix scan uses this to skip over input that can't begin a match.
301 {
302 std::vector<int> sc;
303 std::vector<char> seen(dfa->prog.size(), 0);
304 dfa->add_closure(dfa->start_anchored, sc, seen);
305 for (int pc : sc) {
306 const Inst& in = dfa->prog[pc];
307 if (in.op == Inst::Byte) for (int k = 0; k < 4; ++k) dfa->first[k] |= in.cls[k];
308 if (in.op == Inst::Match) dfa->matches_empty = true; // accepts the empty string
309 }
310 int cnt = 0, single = -1;
311 for (int b = 0; b < 256; ++b) if (has_bit(dfa->first, static_cast<unsigned char>(b))) { ++cnt; single = b; }
312 dfa->first_count = cnt;
313 if (cnt == 1) dfa->first_byte = static_cast<unsigned char>(single);
314 }
315
316 out.impl = std::move(dfa);
317 out.ok = true;
318 return out;
319}

322// Run the anchored DFA from byte offset `from`; true if it reaches an accepting state (that
323// also satisfies the end-anchor, if any). Touches only `text`'s bytes.
324bool run_anchored(Dfa& d, std::string_view text, std::size_t from, bool need_end, int start_state) {
325 int state = start_state;
326 for (std::size_t p = from;; ++p) {
327 if (d.accepts[state] && (!need_end || p == text.size())) return true;
328 if (p == text.size()) return false;
329 state = d.step(state, static_cast<unsigned char>(text[p]));
330 if (d.pcs[state].empty() && !d.accepts[state]) return false; // dead — cannot extend
331 }
332}

334bool run(const Pattern& re, std::string_view text, bool whole) {
335 if (!re.ok || !re.impl) return false;
336 Dfa& d = *re.impl;
337 const bool need_end = whole || d.anchored_end;
338 const int astart = d.start_state(d.start_anchored);
339
340 if (whole || d.anchored_start) // pinned to the start
341 return run_anchored(d, text, 0, need_end, astart);
342 if (d.matches_empty && !need_end) return true; // an empty match sits at position 0
343 if (d.first_count == 0) // no first-byte info -> full DFA pass
344 return run_anchored(d, text, 0, need_end, d.start_state(d.start_unanchored));
345
346 // Unanchored: skip to each candidate start byte with a fast scan, run the anchored DFA there.
347 const char* base = text.data();
348 const std::size_t n = text.size();
349 for (std::size_t i = 0; i < n;) {
350 if (d.first_count == 1) {
351 const void* hit = std::memchr(base + i, d.first_byte, n - i);
352 if (!hit) break;
353 i = static_cast<std::size_t>(static_cast<const char*>(hit) - base);
354 } else {
355 while (i < n && !has_bit(d.first, static_cast<unsigned char>(base[i]))) ++i;
356 if (i >= n) break;
357 }
358 if (run_anchored(d, text, i, need_end, astart)) return true;
359 ++i;
360 }
361 return false;
362}

365bool search(const Pattern& re, std::string_view text) { return run(re, text, false); }
366bool full_match(const Pattern& re, std::string_view text) { return run(re, text, true); }

368Match find(const Pattern& re, std::string_view text) {
369 Match r;
370 if (!re.ok || !re.impl) return r;
371 Dfa& d = *re.impl;
372 const std::size_t last_start = d.anchored_start ? 0 : text.size();
373 for (std::size_t i = 0; i <= last_start; ++i) {
374 int state = d.start_state(d.start_anchored);
375 long best_end = -1;
376 for (std::size_t p = i;; ++p) {
377 if (d.accepts[state] && (!d.anchored_end || p == text.size())) best_end = static_cast<long>(p);
378 if (p == text.size()) break;
379 state = d.step(state, static_cast<unsigned char>(text[p]));
380 if (d.pcs[state].empty() && !d.accepts[state]) break; // dead — no match can extend
381 }
382 if (best_end >= 0) {
383 r.matched = true;
384 r.begin = i;
385 r.end = static_cast<std::size_t>(best_end);
386 r.text.assign(text.data() + i, r.end - i); // OWN a copy of the matched bytes
387 return r;
388 }
389 }
390 return r;
