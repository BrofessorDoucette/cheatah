27// Consume the exact literal `lit` (e.g. "true") if present, else leave the cursor put.
28// @complexity O(|lit|) @alloc none @test Json.ParseObject
29inline bool match(Cursor& c, std::string_view lit) noexcept {
30 if (static_cast<std::size_t>(c.end - c.it) < lit.size()) {
31 return false;
32 }
33 if (std::string_view(c.it, lit.size()) != lit) {
34 return false;
35 }
36 c.it += lit.size();
37 return true;
38}

42// Append the UTF-8 encoding of code point `cp` (1..4 bytes) to `out`.
43// @complexity O(1) @alloc amortized growth of `out` @test JsonRead.Strings
44inline void append_utf8(std::uint32_t cp, std::string& out) {
45 if (cp <= 0x7F) {
46 out.push_back(static_cast<char>(cp));
47 } else if (cp <= 0x7FF) {
48 out.push_back(static_cast<char>(0xC0 | (cp >> 6)));
49 out.push_back(static_cast<char>(0x80 | (cp & 0x3F)));
50 } else if (cp <= 0xFFFF) {
51 out.push_back(static_cast<char>(0xE0 | (cp >> 12)));
52 out.push_back(static_cast<char>(0x80 | ((cp >> 6) & 0x3F)));
53 out.push_back(static_cast<char>(0x80 | (cp & 0x3F)));
54 } else {
55 out.push_back(static_cast<char>(0xF0 | (cp >> 18)));
56 out.push_back(static_cast<char>(0x80 | ((cp >> 12) & 0x3F)));
57 out.push_back(static_cast<char>(0x80 | ((cp >> 6) & 0x3F)));
58 out.push_back(static_cast<char>(0x80 | (cp & 0x3F)));
59 }
60}

62// Read 4 hex digits at raw[i..i+4) into cp.
63// @complexity O(1) @alloc none @test JsonRead.Strings
64inline bool hex4(std::string_view raw, std::size_t i, std::uint32_t& cp) {
65 if (i + 4 > raw.size()) {
66 return false;
67 }
68 std::uint32_t v = 0;
69 for (std::size_t k = 0; k < 4; ++k) {
70 const char ch = raw[i + k];
71 v <<= 4;
72 if (ch >= '0' && ch <= '9') {
73 v |= static_cast<std::uint32_t>(ch - '0');
74 } else if (ch >= 'a' && ch <= 'f') {
75 v |= static_cast<std::uint32_t>(ch - 'a' + 10);
76 } else if (ch >= 'A' && ch <= 'F') {
77 v |= static_cast<std::uint32_t>(ch - 'A' + 10);
78 } else {
79 return false;
80 }
81 }
82 cp = v;
83 return true;
84}

86// Decode the raw (escaped) inner bytes of a JSON string into `out`. Ordinary bytes between escapes
87// are copied in BULK (find the next backslash, append the whole run) rather than one at a time —
88// most of a string is non-escape, so this is a few memcpy-sized appends instead of N push_backs.
89// @complexity O(|raw|) @alloc `out` growth (reserved once up front) @test JsonRead.Strings
90inline bool decode_escapes(std::string_view raw, std::string& out) {
91 out.clear();
92 out.reserve(raw.size());
93 std::size_t i = 0;
94 while (i < raw.size()) {
95 const std::size_t bs = raw.find('\\', i); // next escape, or npos
96 const std::size_t run_end = (bs == std::string_view::npos) ? raw.size() : bs;
97 out.append(raw.data() + i, run_end - i); // bulk-copy the ordinary run
98 if (bs == std::string_view::npos) {
99 return true;
100 }
101 i = bs + 1; // step onto the escape selector
102 if (i >= raw.size()) {
103 return false;
104 }
105 switch (raw[i++]) { // consume the selector; i now points past it
106 case '"': out.push_back('"'); break;
107 case '\\': out.push_back('\\'); break;
108 case '/': out.push_back('/'); break;
109 case 'b': out.push_back('\b'); break;
110 case 'f': out.push_back('\f'); break;
111 case 'n': out.push_back('\n'); break;
112 case 'r': out.push_back('\r'); break;
113 case 't': out.push_back('\t'); break;
114 case 'u': {
115 std::uint32_t cp = 0;
116 if (!hex4(raw, i, cp)) { // the 4 hex digits at [i, i+4)
117 return false;
118 }
119 i += 4;
120 if (cp >= 0xD800 && cp <= 0xDBFF) { // high surrogate; expect a low surrogate \uXXXX
121 std::uint32_t lo = 0;
122 if (i + 1 >= raw.size() || raw[i] != '\\' || raw[i + 1] != 'u' ||
123 !hex4(raw, i + 2, lo) || lo < 0xDC00 || lo > 0xDFFF) {
124 return false;
125 }
126 i += 6;
127 cp = 0x10000 + ((cp - 0xD800) << 10) + (lo - 0xDC00);
128 }
129 append_utf8(cp, out);
130 break;
131 }
132 default:
133 return false;
134 }
135 }
136 return true;
137}

139// Scan a "..." string: set `raw` to its inner bytes and `esc` to whether it had escapes;
140// on entry c.it is at the opening quote, on success c.it is past the closing quote.
141// @complexity O(|string|) (SIMD 32 bytes/step) @alloc none (raw is a view) @test Json.Strings
142inline bool scan_string(Cursor& c, std::string_view& raw, bool& esc) {
143 ++c.it; // skip opening quote
144 const char* const start = c.it;
145 esc = false;
146 while (c.it < c.end) {
147 // SIMD-jump over ordinary content straight to the next quote or backslash (see simd.hpp).
148 c.it = simd::find_quote_or_backslash(c.it, c.end);
149 if (c.it == c.end) {
150 break; // unterminated
151 }
152 if (*c.it == '"') {
153 raw = std::string_view(start, static_cast<std::size_t>(c.it - start));
154 ++c.it; // skip closing quote
155 return true;
156 }
157 esc = true; // a backslash: this string has escapes (decoded later)
158 if (c.end - c.it < 2) {
159 return false; // dangling escape at end of input (checked BEFORE advancing — a pointer
160 // more than one-past-the-end is undefined behavior, even unused)
161 }
162 c.it += 2; // skip escape + escaped char (so \" does not end the string)
163 }
164 return false; // unterminated
165}

169// Powers of ten 10^0 .. 10^22 — every one is EXACTLY representable as a double (5^22 < 2^53), which
170// is what makes the float fast path below correctly rounded.
171inline constexpr double kPow10[23] = {1e0, 1e1, 1e2, 1e3, 1e4, 1e5, 1e6, 1e7,
172 1e8, 1e9, 1e10, 1e11, 1e12, 1e13, 1e14, 1e15,
173 1e16, 1e17, 1e18, 1e19, 1e20, 1e21, 1e22};

175// Parse a JSON double the fast way (Clinger): accumulate the digits into an integer mantissa, then
176// scale by one power of ten. When the mantissa fits in 53 bits AND the scale is within ±22, both
177// are exact doubles, so the single multiply/divide rounds once — bit-identical to std::from_chars
178// but without its general-format machinery. Anything outside that window (20+ digits, huge
179// exponents) falls back to std::from_chars for full correctness. Typical JSON numbers — prices,
180// quantities, timestamps — live entirely in the fast window.
181// @complexity O(digits) @alloc none @test JsonRead.NumbersEdge
182inline bool parse_double_fast(Cursor& c, double& out) {
183 const char* p = c.it;
184 const char* const end = c.end;
185 const bool negative = (p != end && *p == '-');
186 if (negative) {
187 ++p;
188 }
189
190 // Integer digits, then optional ".fraction" — both accumulate into ONE integer mantissa;
191 // each fraction digit just shifts the decimal exponent down by one.
192 const char* const int_start = p;
193 std::uint64_t mantissa = 0;
194 while (p != end && static_cast<unsigned>(*p - '0') <= 9u) {
195 mantissa = mantissa * 10u + static_cast<unsigned>(*p - '0');
196 ++p;
197 }
198 if (p == int_start) {
199 return false; // JSON requires at least one integer digit
200 }
201 std::int64_t digit_count = p - int_start;
202 int exp10 = 0;
203 if (p != end && *p == '.') {
204 ++p;
205 const char* const frac_start = p;
206 while (p != end && static_cast<unsigned>(*p - '0') <= 9u) {
207 mantissa = mantissa * 10u + static_cast<unsigned>(*p - '0');
208 ++p;
209 }
210 if (p == frac_start) {
211 return false; // JSON requires a digit after the decimal point
212 }
213 exp10 -= static_cast<int>(p - frac_start);
214 digit_count += p - frac_start;
215 }
216
217 // Optional exponent ("e"/"E", optional sign, digits).
218 if (p != end && (*p == 'e' || *p == 'E')) {
219 ++p;
220 bool exp_negative = false;
221 if (p != end && (*p == '+' || *p == '-')) {
222 exp_negative = (*p == '-');
223 ++p;
224 }
225 const char* const exp_start = p;
226 int e = 0;
227 while (p != end && static_cast<unsigned>(*p - '0') <= 9u) {
228 if (e < 10000) { // clamp; anything this large leaves the fast window anyway
229 e = e * 10 + (*p - '0');
230 }
231 ++p;
232 }
233 if (p == exp_start) {
234 return false; // 'e' with no digits
235 }
236 exp10 += exp_negative ? -e : e;
237 }
238
239 // The exactness window: <=19 digits means the mantissa accumulated without u64 overflow, the
240 // 2^53 test means it is an exact double, and |exp10| <= 22 means the scale is an exact double.
241 if (digit_count <= 19 && mantissa < (1ull << 53) && -22 <= exp10 && exp10 <= 22) {
242 double value = static_cast<double>(mantissa);
243 value = (exp10 >= 0) ? value * kPow10[exp10] : value / kPow10[-exp10];
244 out = negative ? -value : value;
245 c.it = p;
246 return true;
247 }
248
249 // Outside the window: std::from_chars handles arbitrary precision/exponents correctly.
250 const std::from_chars_result r = std::from_chars(c.it, end, out);
251 if (r.ec != std::errc()) {
252 return false;
253 }
254 c.it = r.ptr;
255 return true;
256}

258// Parse a JSON number into an arithmetic `out`. INTEGRAL types use a tight base-10 loop; double
259// uses the Clinger fast path above; other floating types (float, long double) defer to
260// std::from_chars. The integral loop is overflow-safe: a literal with more digits than the type can
261// ever hold exactly is re-parsed by std::from_chars, which detects out-of-range exactly; a negative
262// literal is rejected for unsigned fields rather than wrapped.
263// @complexity O(digits) @alloc none @test JsonRead.Scalars
264template <class T>
265inline bool parse_arithmetic(Cursor& c, T& out) {
266 if constexpr (std::is_integral_v<T> && !std::is_same_v<T, bool>) {
267 const char* p = c.it;
268 const bool negative = (p != c.end && *p == '-');
269 if (negative) {
270 if constexpr (std::is_unsigned_v<T>) {
271 return false; // a negative literal cannot fit an unsigned field
272 }
273 ++p;
274 }
275 if (p == c.end || *p < '0' || *p > '9') {
276 return false; // a number must have at least one digit
277 }
278 const char* const digit_start = p;
279 std::make_unsigned_t<T> magnitude = 0;
280 do {
281 magnitude = magnitude * 10 + static_cast<unsigned>(*p - '0');
282 ++p;
283 } while (p != c.end && *p >= '0' && *p <= '9');
284 if (p - digit_start > std::numeric_limits<T>::digits10) {
285 // More digits than T holds exactly: the accumulation above may have wrapped. Rare —
286 // re-parse with std::from_chars, which detects overflow exactly (and rejects it).
287 const std::from_chars_result r = std::from_chars(c.it, c.end, out);
288 if (r.ec != std::errc()) {
289 return false;
290 }
291 c.it = r.ptr;
292 return true;
293 }
294 c.it = p;
295 out = negative ? static_cast<T>(0) - static_cast<T>(magnitude) : static_cast<T>(magnitude);
296 return true;
297 } else if constexpr (std::is_same_v<T, double>) {
298 return parse_double_fast(c, out);
299 } else {
300 const std::from_chars_result r = std::from_chars(c.it, c.end, out);
301 if (r.ec != std::errc()) {
302 return false;
303 }
304 c.it = r.ptr;
305 return true;
306 }
307}

309// ---- skip an unknown value (iterative; depth-counted, so it cannot be stack-overflowed) ---------
310//
311// Consume exactly one complete JSON value (any shape) without storing it — used by the struct
312// reader to discard keys not present in the schema. Strings are skipped whole (so braces inside
313// them never miscount); containers are balanced with a depth counter rather than recursion, so an
314// adversarially deep unknown value costs O(depth) iterations and O(1) stack.
315// @complexity O(skipped bytes) @alloc none @test JsonRead.UnknownKeys
316inline bool skip_value(Cursor& c) {
317 std::size_t depth = 0;
318 do {
319 skip_ws(c);
320 if (c.it == c.end) {
321 return false;
322 }
323 switch (*c.it) {
324 case '{':
325 case '[':
326 ++c.it;
327 ++depth;
328 break;
329 case '}':
330 case ']':
331 if (depth == 0) {
332 return false;
333 }
334 ++c.it;
335 --depth;
336 break;
337 case '"': {
338 std::string_view raw;
339 bool esc = false;
340 if (!scan_string(c, raw, esc)) {
341 return false;
342 }
343 break;
344 }
345 case ',':
346 case ':':
347 if (depth == 0) {
348 return false; // stray punctuation is not a value
349 }
350 ++c.it; // structural punctuation inside a container we're skipping
351 break;
352 case 't':
353 if (!match(c, "true")) return false;
354 break;
355 case 'f':
356 if (!match(c, "false")) return false;
357 break;
358 case 'n':
359 if (!match(c, "null")) return false;
360 break;
361 default: {
362 double scratch = 0.0; // value discarded; we only need to advance past the number
363 if (!parse_arithmetic(c, scratch)) {
364 return false;
365 }
366 break;
367 }
368 }
369 } while (depth > 0);
370 return true;
371}
