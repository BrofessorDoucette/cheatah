3/**
4 * @file routines.hpp
5 * @brief cheatah `linalg` — numpy's linear-algebra API on ndarray, with
6 * SIMD-friendly contiguous kernels (a .purr program writes `linalg.solve(A, b)`).
7 *
8 * `import linalg` to use it (auto-links ndarray). Unit tests:
9 * stdlib/tests/linalg_routines_test.cpp; SIMD reporting tested in
10 * tests/linalg/smoke_test.cpp. The suite runs under AddressSanitizer (the `asan`
11 * preset) and Valgrind (security/run-valgrind.sh) on every QA-gate run.
12 *
13 * Routines operate on `ndarray::NDArray` (2-D = matrix, 1-D = vector). They mirror
14 * https://numpy.org/doc/stable/reference/routines.linalg.html and are implemented
15 * in routines.cpp (LU w/ partial pivoting, Cholesky, Householder QR, one-sided
16 * Jacobi SVD, cyclic Jacobi symmetric eigen, Hessenberg + shifted-QR general
17 * eigen) at -O3 -march=native so the hot loops auto-vectorize.
18 *
19 * @note `n` below is the matrix dimension. NDArray is double-only, so `eig`/
20 * `eigvals` return REAL eigenvalues and throw on a complex pair. Routines
21 * that extract a working copy "allocate scratch O(n²) for the factorization"
22 * even when they return a scalar.
23 */

36double vdot(const NDArray& a, const NDArray& b);

38double inner(const NDArray& a, const NDArray& b);
39/** Outer product of two vectors. @param a length-n vector. @param b length-m vector. @return n×m matrix aᵢbⱼ. @note O(n·m); allocates a new NDArray result. @test LinalgRoutines.VdotInnerOuterKron */

42NDArray matmul(const NDArray& a, const NDArray& b);

44NDArray matrix_power(const NDArray& a, long long n);

48// ---- Decompositions ----
49/** Cholesky factor of a symmetric positive-definite matrix (throws otherwise). @param a square SPD matrix. @return lower-triangular L with A = L·Lᵀ. @note O(n³); allocates a new NDArray result. @test LinalgRoutines.CholeskyAndQR */
50NDArray cholesky(const NDArray& a); // lower-triangular L (A = L Lᵀ)
51/** Result of qr(): A = q·r with orthonormal q and upper-triangular r. */
52struct QR { NDArray q; NDArray r; };
53/** Reduced QR via Householder reflections (requires rows ≥ cols). @param a m×n matrix. @return @ref QR with q (m×n, orthonormal cols) and r (n×n, upper-triangular). @note O(n³); allocates both members. @test LinalgRoutines.CholeskyAndQR */
54QR qr(const NDArray& a);
55/** Result of svd(): A = u·diag(s)·vh. */
56struct SVD { NDArray u; NDArray s; NDArray vh; };
57/** Singular value decomposition (one-sided Jacobi; requires rows ≥ cols). @param a m×n matrix. @return @ref SVD with u (m×n), s (descending singular values), vh (n×n = Vᵀ). @note iterative O(n³); allocates all members. @test LinalgRoutines.SvdAndEigh */
58SVD svd(const NDArray& a);

64Eig eig(const NDArray& a); // general square matrix

66NDArray eigvals(const NDArray& a);

68Eig eigh(const NDArray& a); // symmetric / Hermitian
69/** Eigenvalues of a symmetric matrix, descending (cyclic Jacobi). @param a square symmetric matrix. @return length-n vector of eigenvalues. @note iterative O(n³); allocates a new NDArray result. @test LinalgRoutines.EigvalshSymmetric */
