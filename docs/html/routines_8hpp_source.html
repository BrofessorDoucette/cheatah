3/**
4 * @file routines.hpp
5 * @brief cheatah `linalg` — numpy's linear-algebra API on ndarray, with
6 * SIMD-friendly contiguous kernels (a .purr program writes `linalg.solve(A, b)`).
7 *
8 * `import linalg` to use it (auto-links ndarray). Unit tests:
9 * stdlib/tests/linalg_routines_test.cpp; SIMD reporting tested in
10 * tests/linalg/smoke_test.cpp. The suite runs under AddressSanitizer (the `asan`
11 * preset) and Valgrind (security/run-valgrind.sh) on every QA-gate run.
12 *
13 * Routines operate on `ndarray::NDArray` (2-D = matrix, 1-D = vector). They mirror
14 * https://numpy.org/doc/stable/reference/routines.linalg.html and are implemented
15 * in routines.cpp (LU w/ partial pivoting, Cholesky, Householder QR, one-sided
16 * Jacobi SVD, cyclic Jacobi symmetric eigen, Hessenberg + shifted-QR general
17 * eigen) at -O3 -march=native so the hot loops auto-vectorize.
18 *
19 * @note `n` below is the matrix dimension. NDArray is double-only, so `eig`/
20 * `eigvals` return REAL eigenvalues and throw on a complex pair. Routines
21 * that extract a working copy "allocate scratch O(n²) for the factorization"
22 * even when they return a scalar.
23 */

37// ---- Matrix and vector products ----
38/**
39 * Dot product: 1-D inner product (vectors flattened).
40 * @param a,b same-length vectors.
41 * @return Σ aᵢbᵢ.
42 * @note O(n); copies both operands into scratch O(n), returns a double.

45double dot(const NDArray& a, const NDArray& b); // 1-D dot / 2-D matmul
46/**
47 * Vector dot product (alias of @ref dot; flattens N×1/1×N).
48 * @param a,b same-length vectors.

51 * @test LinalgRoutines.VdotInnerOuterKron

53double vdot(const NDArray& a, const NDArray& b);
54/**
55 * Inner product of two vectors (alias of @ref dot).
56 * @param a,b same-length vectors.

61double inner(const NDArray& a, const NDArray& b);
62/**

88NDArray matrix_power(const NDArray& a, long long n);

99// ---- Decompositions ----
100/**
101 * Cholesky factor of a symmetric positive-definite matrix (throws otherwise).
102 * @param a square SPD matrix.
103 * @return lower-triangular L with A = L·Lᵀ.
104 * @note O(n³); allocates a new NDArray result.
105 * @test LinalgRoutines.CholeskyAndQR
106 */
107NDArray cholesky(const NDArray& a); // lower-triangular L (A = L Lᵀ)
108/** Result of qr(): A = q·r with orthonormal q and upper-triangular r. */
109struct QR {
110 NDArray q; ///< Orthonormal columns, m×n (the Q in A = Q·R).
111 NDArray r; ///< Upper-triangular factor, n×n (the R in A = Q·R).
112};
113/**
114 * Reduced QR via Householder reflections (requires rows ≥ cols).
115 * @param a m×n matrix.
116 * @return @ref QR with q (m×n, orthonormal cols) and r (n×n, upper-triangular).
117 * @note O(n³); allocates both members.
118 * @test LinalgRoutines.CholeskyAndQR
119 */
120QR qr(const NDArray& a);
121/** Result of svd(): A = u·diag(s)·vh. */
122struct SVD {
123 NDArray u; ///< Left singular vectors, m×n.
124 NDArray s; ///< Singular values in descending order (length n).
125 NDArray vh; ///< Right singular vectors transposed, n×n (the Vᵀ in A = u·diag(s)·Vᵀ).
126};
127/**
128 * Singular value decomposition (one-sided Jacobi; requires rows ≥ cols).
129 * @param a m×n matrix.
130 * @return @ref SVD with u (m×n), s (descending singular values), vh (n×n = Vᵀ).
131 * @note iterative O(n³); allocates all members.
132 * @test LinalgRoutines.SvdAndEigh
133 */
134SVD svd(const NDArray& a);

136// ---- Matrix eigenvalues ----
137/** Result of eig() / eigh(): column j of vectors is the eigenvector for values[j]. */
138struct Eig {
139 NDArray values; ///< Eigenvalues (length n).
140 NDArray vectors; ///< Eigenvectors as columns: column j matches values[j] (empty if not computed).
141};
142/**
143 * Eigen-decomposition of a general square matrix (real spectrum only; throws on complex pairs).
144 * @param a square matrix.
145 * @return @ref Eig (vectors empty unless @p a is symmetric).
146 * @note iterative O(n³) via Hessenberg + shifted QR; allocates both members.
147 * @test LinalgRoutines.GeneralEig
148 */
149Eig eig(const NDArray& a); // general square matrix
150/**
151 * Eigenvalues of a general square matrix, descending (real only; throws on complex pairs).
152 * @param a square matrix.
153 * @return length-n vector of eigenvalues.
154 * @note iterative O(n³); allocates a new NDArray result.
155 * @test LinalgRoutines.SvdAndEigh
156 */
157NDArray eigvals(const NDArray& a);
158/**
159 * Eigen-decomposition of a symmetric matrix (cyclic Jacobi).
160 * @param a square symmetric matrix.
161 * @return @ref Eig with descending values and matching eigenvectors.
162 * @note iterative O(n³); allocates both members.
163 * @test LinalgRoutines.SvdAndEigh
164 */
165Eig eigh(const NDArray& a); // symmetric / Hermitian
166/**
167 * Eigenvalues of a symmetric matrix, descending (cyclic Jacobi).
168 * @param a square symmetric matrix.
169 * @return length-n vector of eigenvalues.
170 * @note iterative O(n³); allocates a new NDArray result.
171 * @test LinalgRoutines.EigvalshSymmetric
172 */
173NDArray eigvalsh(const NDArray& a);

175// ---- Norms and other numbers ----
176/**
177 * Norm: L2 for vectors, Frobenius for matrices.
178 * @param a vector or matrix.
179 * @return √Σ xᵢ².
180 * @note O(n) for vectors / O(n²) for matrices; copies into scratch, returns a double.
181 * @test LinalgRoutines.NormAndRank

183double norm(const NDArray& a); // default: Frobenius / L2
184/**
185 * 2-norm condition number σ_max/σ_min (∞ if singular).

187 * @return condition number.

190 * @test LinalgRoutines.SlogdetAndCond

192double cond(const NDArray& a);
193/**
194 * Determinant via LU with partial pivoting.
195 * @param a square matrix.
196 * @return det(A).
197 * @note O(n³); allocates scratch O(n²) for the factorization, returns a double.
198 * @test LinalgRoutines.SolveDetInv
199 */
200double det(const NDArray& a);
201/**
202 * Numerical rank from SVD singular-value thresholding.
203 * @param a matrix.
204 * @return rank.
205 * @note iterative O(n³) via SVD; allocates scratch O(n²) for the factorization.
206 * @test LinalgRoutines.NormAndRank
207 */
208long long matrix_rank(const NDArray& a);
209/** Result of slogdet(): det(A) = sign·exp(logabsdet). */
210struct SLogDet {
211 double sign; ///< Sign of the determinant (−1, 0, or +1).
212 double logabsdet; ///< Natural log of |det(A)|, so det(A) = sign·exp(logabsdet).
213};
214/**
215 * Sign and log|det| via LU (overflow-safe determinant).
216 * @param a square matrix.
217 * @return @ref SLogDet.
218 * @note O(n³); allocates scratch O(n²) for the factorization (the struct members are plain
219 * doubles).
220 * @test LinalgRoutines.SlogdetAndCond
221 */
222SLogDet slogdet(const NDArray& a);
223/**
224 * Trace: sum of the main diagonal.
225 * @param a matrix.
226 * @return Σ aᵢᵢ.
227 * @note O(n) summation, but copies the matrix into scratch O(n²) first; returns a double.
228 * @test LinalgRoutines.ProductsAndTrace

230double trace(const NDArray& a);
