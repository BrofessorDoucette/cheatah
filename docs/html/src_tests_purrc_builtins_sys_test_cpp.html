tests/purrc/builtins_sys_test.cpp

1// System-level "real program" test for the `builtins` module: a single cohesive
2// program that exercises EVERY public, purr-callable built-in declared in
3// stdlib/builtins/builtins.hpp in one run, then asserts its stdout byte-for-byte.
4//
5// Unlike the per-function compile-run suite (builtins_cr_test.cpp, one built-in
6// each), this is one small but genuine program: it inspects a string with
7// len/ord/chr, renders a number in every base (hex/oct/bin), shows its printable
8// repr with ascii, and runs the int()/float()/bool() conversions (the Python
9// spellings map to builtins::to_int/to_float/to_bool). The program is fully
10// DETERMINISTIC, so its output is asserted exactly.
11//
12// Coverage of stdlib/builtins/builtins.hpp (every purr-callable built-in):
13// len, ord, chr, hex, oct, bin, ascii,
14// int() -> to_int (from string and from float),
15// float()-> to_float(from string and from int),
16// bool() -> to_bool (from string and from number),
17// hash -> value is implementation-defined, so we assert only the
18// deterministic property hash("a") == hash("a") (not the value).
19//
20// Note: purrc treats a newline as a statement terminator, so each call stays on
21// its own source line.
22#include "e2e_harness.hpp"
23
24TEST(StdlibE2E, Builtins) {
25 e2e::expect_e2e("builtins_sys", R"PURR(import io
26import builtins
27
28# Inspect a string byte-by-byte using len/ord/chr (+ ascii for its repr).
29let s = "Cat"
30io.print(io.format("len({}) = {}", ascii(s), len(s)))
31
32let i = 0
33while i < len(s) {
34 let ch = chr(ord(s) + i)
35 io.print(io.format("ord/chr step {}: {}", i, ch))
36 i = i + 1
37}
38
39# Base representations of a number: hex / oct / bin.
40let n = 255
41io.print(io.format("{} -> hex={} oct={} bin={}", n, hex(n), oct(n), bin(n)))
42
43# Conversions: int()/float()/bool() map to to_int/to_float/to_bool.
44io.print(io.format("int(\"42\")={} int(3.9)={}", int("42"), int(3.9)))
45io.print(io.format("float(\"2.5\")={} float(7)={}", float("2.5"), float(7)))
46io.print(io.format("bool(\"x\")={} bool(0)={} bool(7)={}", bool("x"), bool(0), bool(7)))
47
48# hash: value is implementation-defined; assert only a deterministic property.
49io.print(io.format("hash stable: {}", bool(hash("a") == hash("a"))))
50)PURR",
51 "len('Cat') = 3\n"
52 "ord/chr step 0: C\n"
53 "ord/chr step 1: D\n"
54 "ord/chr step 2: E\n"
55 "255 -> hex=0xff oct=0o377 bin=0b11111111\n"
56 "int(\"42\")=42 int(3.9)=3\n"
57 "float(\"2.5\")=2.5 float(7)=7\n"
58 "bool(\"x\")=1 bool(0)=0 bool(7)=1\n"
59 "hash stable: 1\n");
60}
