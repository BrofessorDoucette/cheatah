5/**
6 * @file os.hpp
7 * @brief cheatah `os` — Python-like operating-system interface over
8 * `std::filesystem`, plus the `os.path` submodule. `import os` to use it.
9 *
10 * `import os` includes this header AND links `libcheatah_os`. Templated entry
11 * points (e.g. `os.path.join`) live here; the rest is compiled into the library.
12 * Unit tests: `stdlib/tests/os_test.cpp`; the suite runs under AddressSanitizer
13 * (the `asan` preset) and Valgrind (`security/run-valgrind.sh`) on every QA-gate
14 * run.
15 *
16 * @note Most calls touch the filesystem/environment, so they perform a syscall
17 * in addition to the cost noted per function; `n` is the path length.
18 */
19#include <cstdint>
20#include <filesystem>
21#include <string>
22#include <string_view>
23#include <utility>
24#include <vector>

26namespace cheatah::os {

28/// StringLike<T>: a std::string can be constructed from T — exactly what
29/// os.path.join() does (std::string(part)). Naming it yields a clear "constraint
30/// StringLike not satisfied" message while still accepting everything it does today
31/// (const char*, char arrays, std::string, std::string_view, …).
32template <typename T>
33concept StringLike = requires(const T& value) { std::string(value); };

35/**
36 * Current working directory.
37 *
38 * Queries the process's current directory via `std::filesystem::current_path`
39 * and returns it as an absolute path string.
40 * @return the absolute cwd.
41 * @complexity O(n) + a syscall.
42 * @alloc allocates the result string.
43 * @test CheatahOs.CwdAndCpuCount
44 * @crtest OsCompileRun.Getcwd
45 * @systest StdlibE2E.Os
46 */
47std::string getcwd();
48/**
49 * Change the working directory.
50 *
51 * Sets the process's current directory; subsequent relative paths resolve
52 * against it. Throws if @p path does not exist or is not a directory.
53 * @param path the target directory.
54 * @complexity O(1) + a syscall.
55 * @alloc none.
56 * @test CheatahOs.MakedirsAndChdir
57 * @crtest OsCompileRun.Chdir
58 * @systest StdlibE2E.Os
59 */
60void chdir(const std::string& path);
61/**
62 * List a directory's entries (basenames only).
63 *
64 * Iterates @p path and returns each entry's filename component (not a full
65 * path), in unspecified order; `.` and `..` are not included. Throws if @p path
66 * does not exist or is not a directory.
67 * @param path the directory (default `.`).
68 * @return the entry names.
69 * @complexity O(entries) + syscalls.
70 * @alloc allocates a vector of strings.
71 * @test CheatahOs.ListdirAndRename
72 * @crtest OsCompileRun.Listdir
73 * @systest StdlibE2E.Os
74 */
75std::vector<std::string> listdir(const std::string& path = ".");
76/**
77 * Create a single directory.
78 *
79 * Creates the leaf directory only; the parent must already exist (use
80 * makedirs to create missing parents). Does nothing if @p path already exists
81 * as a directory.
82 * @param path the directory to create.
83 * @complexity O(1) + a syscall.
84 * @alloc none.
85 * @test CheatahOs.MakeDirExistsThenRemove
86 * @crtest OsCompileRun.Mkdir
87 * @systest StdlibE2E.Os
88 */
89void mkdir(const std::string& path);
90/**
91 * Create a directory and any missing parents.
92 *
93 * Creates @p path along with every intermediate directory that does not yet
94 * exist. Succeeds without error if the full path already exists as a directory.
95 * @param path the nested directory to create.
96 * @complexity O(depth) + syscalls.
97 * @alloc none.
98 * @test CheatahOs.MakedirsAndChdir
99 * @crtest OsCompileRun.Makedirs
100 * @systest StdlibE2E.Os
101 */
102void makedirs(const std::string& path);
103/**
104 * Remove an (empty) directory.
105 *
106 * Deletes a single, empty directory; throws if @p path is non-empty. A missing
107 * @p path is a no-op (no error). Note this is the same `fs::remove` used by
108 * remove(), so it will also delete a regular file at @p path.
109 * @param path the directory to remove.
110 * @complexity O(1) + a syscall.
111 * @alloc none.
112 * @test CheatahOs.MakeDirExistsThenRemove
113 * @crtest OsCompileRun.Rmdir
114 * @systest StdlibE2E.Os
115 */
116void rmdir(const std::string& path);
117/**
118 * Remove a file or empty directory.
119 *
120 * Deletes a single file or empty directory and returns whether anything was
121 * removed; a missing @p path returns false rather than throwing. Throws if
122 * @p path is a non-empty directory.
123 * @param path the entry to remove.
124 * @return true iff something was removed.
125 * @complexity O(1) + a syscall.
126 * @alloc none.
127 * @test CheatahOs.FileQueriesIsfileAndGetsize
128 * @crtest OsCompileRun.Remove
129 * @systest StdlibE2E.Os
130 */
131bool remove(const std::string& path); // true if a file was removed
132/**
133 * Rename/move @p src to @p dst.
134 *
135 * Moves or renames an entry; an existing @p dst is overwritten when permitted
136 * by the underlying `fs::rename`. Crossing filesystems or other failures throw.
137 * @param src source path.
138 * @param dst destination path.
139 * @complexity O(1) + a syscall.
140 * @alloc none.
141 * @test CheatahOs.ListdirAndRename
142 * @crtest OsCompileRun.Rename
143 * @systest StdlibE2E.Os
144 */
145void rename(const std::string& src, const std::string& dst);

147/**
148 * Read an environment variable.
149 *
150 * Returns @p fallback (default `""`) when the variable is unset; an empty
151 * string result therefore does not distinguish "unset" from "set to empty".
152 * @param name the variable name.
153 * @param fallback returned when unset.
154 * @return the value, or @p fallback.
155 * @complexity O(1) + a syscall.
156 * @alloc allocates the returned string.
157 * @test CheatahOs.GetenvFallback, CheatahOs.SetenvThenGetenv
158 * @crtest OsCompileRun.Getenv
159 * @systest StdlibE2E.Os
160 */
161std::string getenv(const std::string& name, const std::string& fallback = "");
162/**
163 * Set an environment variable.
164 *
165 * When @p overwrite is false and the variable already exists, the existing
166 * value is kept; otherwise it is created or replaced. The change affects only
167 * this process and its future children.
168 * @param name the variable name.
169 * @param value the value to set.
170 * @param overwrite replace an existing value when true.
171 * @complexity O(1) + a syscall.
172 * @alloc may allocate inside the C library's environment table.
173 * @test CheatahOs.SetenvThenGetenv
174 * @crtest OsCompileRun.Setenv
175 * @systest StdlibE2E.Os
176 */
177void setenv(const std::string& name, const std::string& value, bool overwrite = true);

179/**
180 * Process id.
181 * @return the current process's pid.
182 * @complexity O(1) + a syscall.
183 * @alloc none.
184 * @test CheatahOs.PidAndSystem
185 * @crtest OsCompileRun.Getpid
186 * @systest StdlibE2E.Os

188int getpid();
189/**
190 * Logical CPU count.
191 *
192 * Reports `std::thread::hardware_concurrency()`, the number of concurrent
193 * threads supported; the standard allows it to return 0 when the value cannot
194 * be determined, so callers should treat 0 as "unknown".
195 * @return the number of hardware threads (0 if undetermined).
196 * @complexity O(1).
197 * @alloc none.
198 * @test CheatahOs.CwdAndCpuCount
199 * @crtest OsCompileRun.CpuCount
200 * @systest StdlibE2E.Os
201 */
202unsigned cpu_count();
203/**
204 * Run a shell command.
205 *
206 * Passes @p command to the system shell via `std::system` and blocks until it
207 * finishes; the returned status is implementation-defined (on POSIX, a wait
208 * status, conventionally decoded so that 0 means success).
209 * @param command the command line.
210 * @return the command's exit status.
211 * @complexity O(1) here + the cost of the spawned process (fork/exec via the shell).
212 * @alloc none.
213 * @test CheatahOs.PidAndSystem
214 * @crtest OsCompileRun.System
215 * @systest StdlibE2E.Os
216 */
217int system(const std::string& command);
218/**
219 * Cryptographically secure random bytes (like Python's `os.urandom`).
220 *
221 * Reads @p n bytes from the operating system's CSPRNG — `getentropy`/`/dev/urandom`
222 * on POSIX, `BCryptGenRandom` on Windows — suitable for keys and signatures. Unlike
223 * the `random` module (a deterministic, seedable PRNG), this is NOT reproducible and
224 * must not be seeded. Throws `std::runtime_error` if the OS source cannot be read
225 * (so a key is never built from non-random bytes).
226 * @param n the number of bytes to return.
227 * @return a string of @p n random bytes (may contain embedded NULs).
228 * @complexity O(n) plus a syscall.
229 * @alloc allocates the n-byte result.
230 * @test CheatahOs.Urandom
231 * @crtest OsCompileRun.Urandom
232 * @systest StdlibE2E.Os
233 */
234std::string urandom(int n);

236/**
237 * The loadable-module file extension for this platform.
238 *
239 * A compiled cheatah program is a native loadable module run by the `cheatah`
240 * host; its file extension is `.so` on Linux/BSD, `.dylib` on macOS, and `.dll`
241 * on Windows. Tools that build or name modules (e.g. the `biome` package manager)
242 * use this instead of hardcoding `.so`, so the paths they print and generate are
243 * correct on every platform. The result includes the leading dot.
244 * @return the platform module extension (e.g. `".so"`, `".dylib"`, `".dll"`).
245 * @complexity O(1).
246 * @alloc allocates the returned string.
247 * @test CheatahOs.ModuleExt
248 * @crtest OsCompileRun.ModuleExt
249 * @systest StdlibE2E.Os
250 */
251std::string module_ext();

256/**
257 * Join path components with the platform separator.
258 *
259 * Appends each component with `path::operator/=`, inserting a separator as
260 * needed; following `std::filesystem` rules, an absolute component discards
261 * everything joined before it. Purely lexical — the filesystem is not touched.
262 * @param first the first component.
263 * @param rest any further string-constructible components.
264 * @return e.g. `join("a","b","c") -> "a/b/c"`.
265 * @complexity O(total length).
266 * @alloc allocates the result string and per-part path temporaries.
267 * @test CheatahOs.PathJoin
268 * @crtest OsCompileRun.PathJoin
269 * @systest StdlibE2E.Os
270 */
271template <StringLike... Parts>
272std::string join(const std::string& first, const Parts&... rest) {
273 std::filesystem::path p(first);
274 ((p /= std::filesystem::path(std::string(rest))), ...);
275 return p.string();
276}

278/**
279 * Path existence test.
280 *
281 * Follows symlinks and is true for any existing entry — file, directory, or
282 * other; returns false for a missing path.
283 * @param p the path.
284 * @return true iff @p p exists.
285 * @complexity O(n) + a syscall.
286 * @alloc none.
287 * @test CheatahOs.MakeDirExistsThenRemove
288 * @crtest OsCompileRun.PathExists
289 * @systest StdlibE2E.Os
290 */
291bool exists(const std::string& p);
292/**
293 * Regular-file test.
294 *
295 * Returns false (rather than throwing) when @p p is missing or is a non-regular
296 * entry such as a directory; symlinks are followed to their target.
297 * @param p the path.
298 * @return true iff @p p is a regular file.
299 * @complexity O(n) + a syscall.
300 * @alloc none.
301 * @test CheatahOs.FileQueriesIsfileAndGetsize
302 * @crtest OsCompileRun.PathIsfile
303 * @systest StdlibE2E.Os
304 */
305bool isfile(const std::string& p);
306/**
307 * Directory test.
308 *
309 * Returns false (rather than throwing) when @p p is missing or is not a
310 * directory; symlinks are followed to their target.
311 * @param p the path.
312 * @return true iff @p p is a directory.
313 * @complexity O(n) + a syscall.
314 * @alloc none.
315 * @test CheatahOs.MakeDirExistsThenRemove
316 * @crtest OsCompileRun.PathIsdir
317 * @systest StdlibE2E.Os
318 */
319bool isdir(const std::string& p);
320/**
321 * Final path component.
322 *
323 * Returns the trailing filename component lexically, without touching the
324 * filesystem; a path ending in a separator (e.g. `a/b/`) yields an empty
325 * string, matching `std::filesystem::path::filename`.
326 * @param p the path.
327 * @return the basename (filename).
328 * @complexity O(n).
329 * @alloc allocates a path temporary and the result string.
330 * @test CheatahOs.PathBasenameDirname
331 * @crtest OsCompileRun.PathBasename
332 * @systest StdlibE2E.Os
333 */
334std::string basename(const std::string& p);
335/**
336 * Parent path.
337 *
338 * Returns everything before the final component lexically, without touching the
339 * filesystem; a bare filename with no separator (e.g. `file.txt`) yields an
340 * empty string, matching `std::filesystem::path::parent_path`.
341 * @param p the path.
342 * @return the directory portion of @p p.
343 * @complexity O(n).
344 * @alloc allocates a path temporary and the result string.
345 * @test CheatahOs.PathBasenameDirname
346 * @crtest OsCompileRun.PathDirname
347 * @systest StdlibE2E.Os
348 */
349std::string dirname(const std::string& p);
350/**
351 * Absolute path.
352 *
353 * Prepends the current working directory to a relative @p p; it does not
354 * collapse `.`/`..` segments or resolve symlinks (combine with normpath for
355 * that), and @p p need not exist.
356 * @param p the path.
357 * @return @p p resolved against the cwd.
358 * @complexity O(n) + a syscall (reads the cwd).
359 * @alloc allocates the result string.
360 * @test CheatahOs.AbspathAndNormpath
361 * @crtest OsCompileRun.PathAbspath
362 * @systest StdlibE2E.Os
363 */
364std::string abspath(const std::string& p);
365/**
366 * Lexically normalized path (collapses current-dir and parent-dir segments).
367 * @param p the path.
368 * @return the normalized path.
369 * @complexity O(n) (purely lexical, no syscall).
370 * @alloc allocates a path temporary and the result string.
371 * @test CheatahOs.AbspathAndNormpath
372 * @crtest OsCompileRun.PathNormpath
373 * @systest StdlibE2E.Os
374 */
375std::string normpath(const std::string& p);
376/**
377 * File size in bytes.
378 *
379 * Defined only for regular files; querying a missing path, or a directory or
380 * other non-regular entry, throws rather than returning a sentinel.
381 * @param p the file path.
382 * @return @p p's size.
383 * @complexity O(1) + a syscall.
384 * @alloc none.
385 * @test CheatahOs.FileQueriesIsfileAndGetsize
386 * @crtest OsCompileRun.PathGetsize
387 * @systest StdlibE2E.Os
388 */
389std::uintmax_t getsize(const std::string& p);

391/**
392 * Split a path into root and extension.
393 *
394 * Splits at the last dot of the final component so that concatenating the two
395 * results reproduces @p p; when there is no extension the whole path is the
396 * root and the extension is empty. The extension includes its leading dot, and
397 * a leading-dot name (e.g. `.bashrc`) is treated as having no extension.
398 * @param p the path.
399 * @return e.g. `splitext("dir/file.purr") -> {"dir/file", ".purr"}` (empty extension when
400 * none).
401 * @complexity O(n).
402 * @alloc allocates the two result strings and a path temporary.
403 * @test CheatahOs.PathSplitext
404 * @crtest OsCompileRun.PathSplitext
405 * @systest StdlibE2E.Os
406 */
407std::pair<std::string, std::string> splitext(const std::string& p);
