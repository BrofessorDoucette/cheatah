6 * `std::filesystem`, plus the `os.path` submodule. `import os` to use it.

26/// StringLike<T>: a std::string can be constructed from T — exactly what
27/// os.path.join() does (std::string(part)). Naming it yields a clear "constraint
28/// StringLike not satisfied" message while still accepting everything it does today
29/// (const char*, char arrays, std::string, std::string_view, …).
30template <typename T>
31concept StringLike = requires(const T& value) { std::string(value); };

220/**
221 * Join path components with the platform separator.
222 *
223 * Appends each component with `path::operator/=`, inserting a separator as
224 * needed; following `std::filesystem` rules, an absolute component discards
225 * everything joined before it. Purely lexical — the filesystem is not touched.
226 * @param first the first component.
227 * @param rest any further string-constructible components.
228 * @return e.g. `join("a","b","c") -> "a/b/c"`.
229 * @complexity O(total length).
230 * @alloc allocates the result string and per-part path temporaries.
231 * @test CheatahOs.PathJoin
232 * @crtest OsCompileRun.PathJoin
233 * @systest StdlibE2E.Os
234 */
235template <StringLike... Parts>
236std::string join(const std::string& first, const Parts&... rest) {
237 std::filesystem::path p(first);
238 ((p /= std::filesystem::path(std::string(rest))), ...);
239 return p.string();
240}

355/**
356 * Split a path into root and extension.
357 *
358 * Splits at the last dot of the final component so that concatenating the two
