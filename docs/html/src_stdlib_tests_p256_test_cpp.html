17// hex (big-endian) -> raw bytes
18std::string unhex(const std::string& h) {
19 auto nib = [](char c) -> int {
20 if (c >= '0' && c <= '9') return c - '0';
21 if (c >= 'a' && c <= 'f') return c - 'a' + 10;
22 return c - 'A' + 10;
23 };
24 std::string out;
25 out.reserve(h.size() / 2);
26 for (std::size_t i = 0; i + 1 < h.size(); i += 2)
27 out.push_back(static_cast<char>((nib(h[i]) << 4) | nib(h[i + 1])));
28 return out;
29}
30std::string hex(const std::string& b) {
31 static const char* d = "0123456789abcdef";
32 std::string out;
33 for (unsigned char c : b) {
34 out.push_back(d[c >> 4]);
35 out.push_back(d[c & 15]);
36 }
37 return out;
38}

86// The P-256 group order n (big-endian). A "hash" whose leftmost 256 bits are >= n
87// exercises the FIPS 186-4 reduction in hash_to_scalar (e -= n once). Using exactly
88// n also drives geq's all-limbs-equal return path.
89const std::string kN = "FFFFFFFF00000000FFFFFFFFFFFFFFFFBCE6FAADA7179E84F3B9CAC2FC632551";
90
91TEST(CheatahP256, HashToScalarReducesWhenGreaterThanOrder) {
92 const std::string pub = unhex(kUx) + unhex(kUy);
93 const std::string sig = unhex(kR) + unhex(kS);
94 // A 32-byte "hash" equal to n reduces to 0 (geq true via full equality); a hash
95 // of all-0xFF (> n) reduces by one n. Neither matches the real message, so verify
96 // returns false — we only need the reduction lines to execute.
97 EXPECT_FALSE(p256::verify_raw(pub, unhex(kN), sig));
98 EXPECT_FALSE(p256::verify_raw(pub, std::string(32, '\xff'), sig));
99 // Signing with such a digest must still produce a well-formed 64-byte signature
100 // that verifies against the derived key (round-trips through the reduced scalar).
101 const std::string msg_hash = std::string(32, '\xff');
102 const std::string s2 = p256::sign_raw(unhex(kPriv), msg_hash);
103 ASSERT_EQ(s2.size(), 64u);
104 EXPECT_TRUE(p256::verify_raw(pub, msg_hash, s2));
105}

107TEST(CheatahP256, VerifyDerWithLeadingZeroIntegers) {
108 // DER-encode the RFC 6979 (r, s): both start with a high bit set (0xEF / 0xF7),
109 // so DER requires a 0x00 sign byte — exercising der_to_rs's leading-zero strip.
110 const std::string r = unhex(kR), s = unhex(kS);
111 std::string der;
112 der.push_back(0x30);
113 der.push_back(0x46); // SEQUENCE, length 70
114 der.push_back(0x02);
115 der.push_back(0x21); // INTEGER, length 33 (32 + sign byte)
116 der.push_back(0x00);
117 der += r;
118 der.push_back(0x02);
119 der.push_back(0x21);
120 der.push_back(0x00);
121 der += s;
122 const std::string pub = unhex(kUx) + unhex(kUy);
123 const std::string hash = cheatah::hashlib::sha256_digest("sample");
124 EXPECT_TRUE(p256::verify_der(pub, hash, der));
125
126 // Malformed DER (wrong sequence length) must be rejected.
127 std::string bad = der;
128 bad[1] = 0x40;
129 EXPECT_FALSE(p256::verify_der(pub, hash, bad));
130}

132TEST(CheatahP256, VerifyRejectsOutOfRangeAndInfinity) {
133 const std::string pub = unhex(kUx) + unhex(kUy);
134 const std::string hash = cheatah::hashlib::sha256_digest("sample");
135 // r or s == 0 -> reject.
136 EXPECT_FALSE(p256::verify_raw(pub, hash, std::string(32, '\0') + unhex(kS)));
137 EXPECT_FALSE(p256::verify_raw(pub, hash, unhex(kR) + std::string(32, '\0')));
138 // r or s >= n -> reject (use n and n+ff... both out of range).
139 EXPECT_FALSE(p256::verify_raw(pub, hash, unhex(kN) + unhex(kS)));
140 EXPECT_FALSE(p256::verify_raw(pub, hash, unhex(kR) + std::string(32, '\xff')));
141 // Public coordinate >= field prime p (all-0xFF x) -> reject.
142 EXPECT_FALSE(p256::verify_raw(std::string(32, '\xff') + unhex(kUy), hash, unhex(kR) + unhex(kS)));
143 // Wrong-size inputs -> reject.
144 EXPECT_FALSE(p256::verify_raw("short", hash, unhex(kR) + unhex(kS)));
145 EXPECT_FALSE(p256::verify_raw(pub, hash, "short"));
146 // sign_raw rejects a bad private key.
147 EXPECT_TRUE(p256::sign_raw("short", hash).empty());
148 EXPECT_TRUE(p256::sign_raw(std::string(32, '\0'), hash).empty());
149 EXPECT_TRUE(p256::sign_raw(unhex(kN), hash).empty()); // d == n, out of range
150 EXPECT_TRUE(p256::public_from_private("short").empty());
151 EXPECT_TRUE(p256::public_from_private(std::string(32, '\0')).empty());
152}

154// Subtract two 32-byte big-endian values (a - b), assuming a >= b.
155std::string be_sub(const std::string& a, const std::string& b) {
156 std::string r(32, '\0');
157 int borrow = 0;
158 for (int i = 31; i >= 0; --i) {
159 int av = static_cast<unsigned char>(a[i]);
160 int bv = static_cast<unsigned char>(b[i]) + borrow;
161 int d = av - bv;
162 if (d < 0) {
163 d += 256;
164 borrow = 1;
165 } else {
166 borrow = 0;
167 }
168 r[i] = static_cast<char>(d);
169 }
170 return r;
171}

173// Verifying against a pubkey equal to G, and to -G, forces the Strauss-Shamir
174// precompute table to add a point to ITSELF (G+G -> H==0, Rr==0, doubling branch)
175// and to its NEGATION (G+(-G) -> H==0, Rr!=0, point-at-infinity branch). These are
176// the two jac_add group-law special cases; the signatures need only be range-valid.
177TEST(CheatahP256, VerifyHitsGroupLawSpecialCases) {
178 const std::string p = unhex("FFFFFFFF00000001000000000000000000000000FFFFFFFFFFFFFFFFFFFFFFFF");
179 // G = 1*G, derived from the implementation itself (avoids hand-transcribed hex).
180 const std::string G = p256::public_from_private(unhex(std::string(63, '0') + "1"));
181 ASSERT_EQ(G.size(), 64u);
182 const std::string gx = G.substr(0, 32), gy = G.substr(32, 32);
183 const std::string sig = unhex(kR) + unhex(kS);
184 const std::string hash = cheatah::hashlib::sha256_digest("sample");
185
186 // pubkey == G : tbl[1][1] = G + G (doubling special case)
187 (void)p256::verify_raw(G, hash, sig); // result irrelevant; the table path runs
188
189 // pubkey == -G = (Gx, p - Gy) : tbl[1][1] = G + (-G) (infinity special case)
190 const std::string pubNegG = gx + be_sub(p, gy);
191 (void)p256::verify_raw(pubNegG, hash, sig);
192}

194// The mod-n conditional subtraction only fires for x-coordinates in [n, p) — a
195// ~2^-128 event on real curve points, so it is driven directly through the test seam
196// on the SAME reduce_mod_n the sign/verify paths call.
197TEST(CheatahP256, ReduceModNBoundary) {
198 const std::string P = "FFFFFFFF00000001000000000000000000000000FFFFFFFFFFFFFFFFFFFFFFFF";
199 // value == n -> reduces to 0.
200 EXPECT_EQ(hex(p256::testonly::reduce_mod_n_be(unhex(kN))), std::string(64, '0'));
201 // value == p-1 (in [n, p)) -> reduces by exactly n: (p-1) - n.
202 const std::string pm1 = be_sub(unhex(P), unhex(std::string(63, '0') + "1"));
203 const std::string reduced = p256::testonly::reduce_mod_n_be(pm1);
204 EXPECT_EQ(reduced, be_sub(pm1, unhex(kN)));
205 // value < n -> unchanged.
206 const std::string small = unhex(std::string(63, '0') + "7");
207 EXPECT_EQ(p256::testonly::reduce_mod_n_be(small), small);
208}

210// The RFC 6979 retry loop and its "no candidate" exhaustion return are effectively
211// unreachable with real inputs (each rejection is a ~2^-128 event). The seam forces
212// rejections so the retry tail runs on the real signing code.
213TEST(CheatahP256, SignRetryLoop) {
214 const std::string hash = cheatah::hashlib::sha256_digest("sample");
215 const std::string pub = unhex(kUx) + unhex(kUy);
216 // Forcing one rejection still yields a valid (different) signature via the next
217 // candidate — exercising the retry tail while proving the loop stays correct.
218 const std::string sig = p256::testonly::sign_raw_skip(unhex(kPriv), hash, 1);
219 ASSERT_EQ(sig.size(), 64u);
220 EXPECT_TRUE(p256::verify_raw(pub, hash, sig));
221 // Forcing more rejections than the attempt cap exhausts the loop -> "".
222 EXPECT_TRUE(p256::testonly::sign_raw_skip(unhex(kPriv), hash, 100).empty());
223 // force_retries == 0 matches the public sign_raw exactly.
224 EXPECT_EQ(p256::testonly::sign_raw_skip(unhex(kPriv), hash, 0),
225 p256::sign_raw(unhex(kPriv), hash));
226}

228TEST(CheatahP256, SpkiExtractsPoint) {
229 // A SubjectPublicKeyInfo carrying the RFC 6979 public point: id-ecPublicKey +
230 // prime256v1, then BIT STRING 00 04 X Y.
231 const std::string point = std::string("\x04", 1) + unhex(kUx) + unhex(kUy); // 65 bytes
232 // Minimal SPKI: SEQUENCE { SEQUENCE { OID ecPublicKey, OID prime256v1 }, BIT STRING }
233 std::string spki;
234 // Algorithm OIDs (id-ecPublicKey 1.2.840.10045.2.1 and prime256v1 1.2.840.10045.3.1.7)
235 const std::string alg = unhex("301306072A8648CE3D020106082A8648CE3D030107");
236 std::string bitstr;
237 bitstr.push_back(0x03);
238 bitstr.push_back(0x42); // 66 bytes: 00 unused + 65-byte point
239 bitstr.push_back(0x00);
240 bitstr += point;
241 std::string inner = alg + bitstr;
242 spki.push_back(0x30);
243 spki.push_back(static_cast<char>(inner.size()));
244 spki += inner;
245 const std::string got = p256::spki_ec_point(spki);
246 ASSERT_EQ(got.size(), 64u);
247 EXPECT_EQ(got, unhex(kUx) + unhex(kUy));
248
249 // A DER with no uncompressed EC point returns "".
250 EXPECT_TRUE(p256::spki_ec_point(std::string("\x30\x03\x02\x01\x00", 5)).empty());
251 // A BIT STRING whose length byte is not 66 (compressed-point sized) is skipped.
252 std::string wronglen = std::string("\x03\x21\x00\x04", 4) + std::string(64, '\0');
253 EXPECT_TRUE(p256::spki_ec_point(wronglen).empty());
