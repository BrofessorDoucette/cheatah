19// hex (big-endian) -> raw bytes
20std::string unhex(const std::string& h) {
21 auto nib = [](char c) -> int {
22 if (c >= '0' && c <= '9') return c - '0';
23 if (c >= 'a' && c <= 'f') return c - 'a' + 10;
24 return c - 'A' + 10;
25 };
26 std::string out;
27 out.reserve(h.size() / 2);
28 for (std::size_t i = 0; i + 1 < h.size(); i += 2)
29 out.push_back(static_cast<char>((nib(h[i]) << 4) | nib(h[i + 1])));
30 return out;
31}
32std::string hex(const std::string& b) {
33 static const char* d = "0123456789abcdef";
34 std::string out;
35 for (unsigned char c : b) {
36 out.push_back(d[c >> 4]);
37 out.push_back(d[c & 15]);
38 }
39 return out;
40}

73// SECURITY (invalid-curve point validation, SP 800-56A): a public key whose coordinates are in
74// range but does NOT satisfy y^2 = x^3 - 3x + b is rejected before it enters the group law.
75TEST(CheatahP256, RejectsOffCurvePublicKey) {
76 const std::string hash = cheatah::hashlib::sha256_digest("sample");
77 const std::string sig = unhex(kR) + unhex(kS);
78 const std::string good = unhex(kUx) + unhex(kUy);
79 ASSERT_TRUE(p256::verify_raw(good, hash, sig)); // the genuine (on-curve) key verifies
80
81 std::string off = good;
82 off[63] ^= 0x01; // flip the low bit of y: still < p, but no longer on the curve
83 EXPECT_FALSE(p256::verify_raw(off, hash, sig));
84
85 // A point with y = 0 (never on this curve) is also refused.
86 std::string y_zero = good;
87 for (int i = 32; i < 64; ++i) y_zero[i] = 0;
88 EXPECT_FALSE(p256::verify_raw(y_zero, hash, sig));
89}

106// The P-256 group order n (big-endian). A "hash" whose leftmost 256 bits are >= n
107// exercises the FIPS 186-4 reduction in hash_to_scalar (e -= n once). Using exactly
108// n also drives geq's all-limbs-equal return path.
109const std::string kN = "FFFFFFFF00000000FFFFFFFFFFFFFFFFBCE6FAADA7179E84F3B9CAC2FC632551";
110
111TEST(CheatahP256, HashToScalarReducesWhenGreaterThanOrder) {
112 const std::string pub = unhex(kUx) + unhex(kUy);
113 const std::string sig = unhex(kR) + unhex(kS);
114 // A 32-byte "hash" equal to n reduces to 0 (geq true via full equality); a hash
115 // of all-0xFF (> n) reduces by one n. Neither matches the real message, so verify
116 // returns false — we only need the reduction lines to execute.
117 EXPECT_FALSE(p256::verify_raw(pub, unhex(kN), sig));
118 EXPECT_FALSE(p256::verify_raw(pub, std::string(32, '\xff'), sig));
119 // Signing with such a digest must still produce a well-formed 64-byte signature
120 // that verifies against the derived key (round-trips through the reduced scalar).
121 const std::string msg_hash = std::string(32, '\xff');
122 const std::string s2 = p256::sign_raw(unhex(kPriv), msg_hash);
123 ASSERT_EQ(s2.size(), 64u);
124 EXPECT_TRUE(p256::verify_raw(pub, msg_hash, s2));
125}

127TEST(CheatahP256, VerifyDerWithLeadingZeroIntegers) {
128 // DER-encode the RFC 6979 (r, s): both start with a high bit set (0xEF / 0xF7),
129 // so DER requires a 0x00 sign byte — exercising der_to_rs's leading-zero strip.
130 const std::string r = unhex(kR), s = unhex(kS);
131 std::string der;
132 der.push_back(0x30);
133 der.push_back(0x46); // SEQUENCE, length 70
134 der.push_back(0x02);
135 der.push_back(0x21); // INTEGER, length 33 (32 + sign byte)
136 der.push_back(0x00);
137 der += r;
138 der.push_back(0x02);
139 der.push_back(0x21);
140 der.push_back(0x00);
141 der += s;
142 const std::string pub = unhex(kUx) + unhex(kUy);
143 const std::string hash = cheatah::hashlib::sha256_digest("sample");
144 EXPECT_TRUE(p256::verify_der(pub, hash, der));
145
146 // Malformed DER (wrong sequence length) must be rejected.
147 std::string bad = der;
148 bad[1] = 0x40;
149 EXPECT_FALSE(p256::verify_der(pub, hash, bad));
150}

152TEST(CheatahP256, VerifyRejectsOutOfRangeAndInfinity) {
153 const std::string pub = unhex(kUx) + unhex(kUy);
154 const std::string hash = cheatah::hashlib::sha256_digest("sample");
155 // r or s == 0 -> reject.
156 EXPECT_FALSE(p256::verify_raw(pub, hash, std::string(32, '\0') + unhex(kS)));
157 EXPECT_FALSE(p256::verify_raw(pub, hash, unhex(kR) + std::string(32, '\0')));
158 // r or s >= n -> reject (use n and n+ff... both out of range).
159 EXPECT_FALSE(p256::verify_raw(pub, hash, unhex(kN) + unhex(kS)));
160 EXPECT_FALSE(p256::verify_raw(pub, hash, unhex(kR) + std::string(32, '\xff')));
161 // Public coordinate >= field prime p (all-0xFF x) -> reject.
162 EXPECT_FALSE(p256::verify_raw(std::string(32, '\xff') + unhex(kUy), hash, unhex(kR) + unhex(kS)));
163 // Wrong-size inputs -> reject.
164 EXPECT_FALSE(p256::verify_raw("short", hash, unhex(kR) + unhex(kS)));
165 EXPECT_FALSE(p256::verify_raw(pub, hash, "short"));
166 // sign_raw rejects a bad private key.
167 EXPECT_TRUE(p256::sign_raw("short", hash).empty());
168 EXPECT_TRUE(p256::sign_raw(std::string(32, '\0'), hash).empty());
169 EXPECT_TRUE(p256::sign_raw(unhex(kN), hash).empty()); // d == n, out of range
170 EXPECT_TRUE(p256::public_from_private("short").empty());
171 EXPECT_TRUE(p256::public_from_private(std::string(32, '\0')).empty());
172}

174// Subtract two 32-byte big-endian values (a - b), assuming a >= b.
175std::string be_sub(const std::string& a, const std::string& b) {
176 std::string r(32, '\0');
177 int borrow = 0;
178 for (int i = 31; i >= 0; --i) {
179 int av = static_cast<unsigned char>(a[i]);
180 int bv = static_cast<unsigned char>(b[i]) + borrow;
181 int d = av - bv;
182 if (d < 0) {
183 d += 256;
184 borrow = 1;
185 } else {
186 borrow = 0;
187 }
188 r[i] = static_cast<char>(d);
189 }
190 return r;
191}

193// Verifying against a pubkey equal to G, and to -G, forces the Strauss-Shamir
194// precompute table to add a point to ITSELF (G+G -> H==0, Rr==0, doubling branch)
195// and to its NEGATION (G+(-G) -> H==0, Rr!=0, point-at-infinity branch). These are
196// the two jac_add group-law special cases; the signatures need only be range-valid.
197TEST(CheatahP256, VerifyHitsGroupLawSpecialCases) {
198 const std::string p = unhex("FFFFFFFF00000001000000000000000000000000FFFFFFFFFFFFFFFFFFFFFFFF");
199 // G = 1*G, derived from the implementation itself (avoids hand-transcribed hex).
200 const std::string G = p256::public_from_private(unhex(std::string(63, '0') + "1"));
201 ASSERT_EQ(G.size(), 64u);
202 const std::string gx = G.substr(0, 32), gy = G.substr(32, 32);
203 const std::string sig = unhex(kR) + unhex(kS);
204 const std::string hash = cheatah::hashlib::sha256_digest("sample");
205
206 // pubkey == G : tbl[1][1] = G + G (doubling special case)
207 (void)p256::verify_raw(G, hash, sig); // result irrelevant; the table path runs
208
209 // pubkey == -G = (Gx, p - Gy) : tbl[1][1] = G + (-G) (infinity special case)
210 const std::string pubNegG = gx + be_sub(p, gy);
211 (void)p256::verify_raw(pubNegG, hash, sig);
212}

214// The mod-n conditional subtraction only fires for x-coordinates in [n, p) — a
215// ~2^-128 event on real curve points, so it is driven directly through the test seam
216// on the SAME reduce_mod_n the sign/verify paths call.
217TEST(CheatahP256, ReduceModNBoundary) {
218 const std::string P = "FFFFFFFF00000001000000000000000000000000FFFFFFFFFFFFFFFFFFFFFFFF";
219 // value == n -> reduces to 0.
220 EXPECT_EQ(hex(p256::testonly::reduce_mod_n_be(unhex(kN))), std::string(64, '0'));
221 // value == p-1 (in [n, p)) -> reduces by exactly n: (p-1) - n.
222 const std::string pm1 = be_sub(unhex(P), unhex(std::string(63, '0') + "1"));
223 const std::string reduced = p256::testonly::reduce_mod_n_be(pm1);
224 EXPECT_EQ(reduced, be_sub(pm1, unhex(kN)));
225 // value < n -> unchanged.
226 const std::string small = unhex(std::string(63, '0') + "7");
227 EXPECT_EQ(p256::testonly::reduce_mod_n_be(small), small);
228}

238// The RFC 6979 retry loop and its "no candidate" exhaustion return are effectively
239// unreachable with real inputs (each rejection is a ~2^-128 event). The seam forces
240// rejections so the retry tail runs on the real signing code.
241TEST(CheatahP256, SignRetryLoop) {
242 const std::string hash = cheatah::hashlib::sha256_digest("sample");
243 const std::string pub = unhex(kUx) + unhex(kUy);
244 // Forcing one rejection still yields a valid (different) signature via the next
245 // candidate — exercising the retry tail while proving the loop stays correct.
246 const std::string sig = p256::testonly::sign_raw_skip(unhex(kPriv), hash, 1);
247 ASSERT_EQ(sig.size(), 64u);
248 EXPECT_TRUE(p256::verify_raw(pub, hash, sig));
249 // Forcing more rejections than the attempt cap exhausts the loop -> "".
250 EXPECT_TRUE(p256::testonly::sign_raw_skip(unhex(kPriv), hash, 100).empty());
251 // force_retries == 0 matches the public sign_raw exactly.
252 EXPECT_EQ(p256::testonly::sign_raw_skip(unhex(kPriv), hash, 0),
253 p256::sign_raw(unhex(kPriv), hash));
254}

256TEST(CheatahP256, SpkiExtractsPoint) {
257 // A SubjectPublicKeyInfo carrying the RFC 6979 public point: id-ecPublicKey +
258 // prime256v1, then BIT STRING 00 04 X Y.
259 const std::string point = std::string("\x04", 1) + unhex(kUx) + unhex(kUy); // 65 bytes
260 // Minimal SPKI: SEQUENCE { SEQUENCE { OID ecPublicKey, OID prime256v1 }, BIT STRING }
261 std::string spki;
262 // Algorithm OIDs (id-ecPublicKey 1.2.840.10045.2.1 and prime256v1 1.2.840.10045.3.1.7)
263 const std::string alg = unhex("301306072A8648CE3D020106082A8648CE3D030107");
264 std::string bitstr;
265 bitstr.push_back(0x03);
266 bitstr.push_back(0x42); // 66 bytes: 00 unused + 65-byte point
267 bitstr.push_back(0x00);
268 bitstr += point;
269 std::string inner = alg + bitstr;
270 spki.push_back(0x30);
271 spki.push_back(static_cast<char>(inner.size()));
272 spki += inner;
273 const std::string got = p256::spki_ec_point(spki);
274 ASSERT_EQ(got.size(), 64u);
275 EXPECT_EQ(got, unhex(kUx) + unhex(kUy));
276
277 // A DER with no uncompressed EC point returns "".
278 EXPECT_TRUE(p256::spki_ec_point(std::string("\x30\x03\x02\x01\x00", 5)).empty());
279 // A BIT STRING whose length byte is not 66 (compressed-point sized) is skipped.
280 std::string wronglen = std::string("\x03\x21\x00\x04", 4) + std::string(64, '\0');
281 EXPECT_TRUE(p256::spki_ec_point(wronglen).empty());
