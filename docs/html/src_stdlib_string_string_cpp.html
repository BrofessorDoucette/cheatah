91std::string replace(std::string_view s, std::string_view from, std::string_view to) {
92 if (from.empty()) return std::string(s);
93 std::string r;
94 std::size_t prev = 0, pos;
95 while ((pos = s.find(from, prev)) != std::string_view::npos) {
96 r.append(s.substr(prev, pos - prev));
97 r.append(to);
98 prev = pos + from.size();
99 }
100 r.append(s.substr(prev));
101 return r;
102}

104std::vector<std::string> split(std::string_view s, std::string_view sep) {
105 std::vector<std::string> out;
106 if (sep.empty()) {
107 out.emplace_back(s);

109 }
110 std::size_t prev = 0, pos;

112 out.emplace_back(s.substr(prev, pos - prev));
113 prev = pos + sep.size();
114 }
115 out.emplace_back(s.substr(prev));
116 return out;

118std::vector<std::string> split(std::string_view s) {
119 std::vector<std::string> out;
120 const std::size_t n = s.size();
121 std::size_t i = 0;
122 while (i < n) {
123 while (i < n && in_set(s[i], whitespace)) ++i;
124 if (i >= n) break;
125 const std::size_t start = i;
126 while (i < n && !in_set(s[i], whitespace)) ++i;
127 out.emplace_back(s.substr(start, i - start));
128 }
129 return out;
130}
131std::vector<std::string> splitlines(std::string_view s) {
132 std::vector<std::string> out;
133 const std::size_t n = s.size();
134 std::size_t i = 0, start = 0;
135 while (i < n) {
136 if (s[i] == '\n' || s[i] == '\r') {
137 out.emplace_back(s.substr(start, i - start));
138 if (s[i] == '\r' && i + 1 < n && s[i + 1] == '\n') ++i;
139 ++i;
140 start = i;
141 } else {
142 ++i;
143 }
144 }
145 if (start < n) out.emplace_back(s.substr(start));
146 return out;
147}

165std::string center(std::string_view s, std::size_t width, std::string_view fill) {
166 const char f = fill.empty() ? ' ' : fill[0];
167 if (s.size() >= width) return std::string(s);
168 const std::size_t total = width - s.size();
169 const std::size_t left = total / 2;
170 return std::string(left, f) + std::string(s) + std::string(total - left, f);
171}
