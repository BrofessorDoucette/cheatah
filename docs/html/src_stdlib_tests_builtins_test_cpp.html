19TEST(CheatahBuiltins, Str) {
20 // Streamable template: integers, floats (default 6-sig-digit form), and strings.
21 EXPECT_EQ(b::str(42LL), "42");
22 EXPECT_EQ(b::str(-7LL), "-7");
23 EXPECT_EQ(b::str(3.14), "3.14");
24 EXPECT_EQ(b::str(std::string("hi")), "hi");
25 // bool overload: Python's capitalized spelling, not 1/0.
26 EXPECT_EQ(b::str(true), "True");
27 EXPECT_EQ(b::str(false), "False");
28}

30TEST(CheatahBuiltins, BaseReprs) {

34 EXPECT_EQ(b::hex(-255), "-0xff");

102TEST(CheatahBuiltins, IndexBoolList) {
103 // std::vector<bool> is bit-packed (proxy references, no .data()), so it has
104 // its own index overload; semantics match every other sequence.
105 const std::vector<bool> xs{true, false, true};
106 EXPECT_TRUE(b::index(xs, 0));
107 EXPECT_FALSE(b::index(xs, 1));
108 EXPECT_TRUE(b::index(xs, -1)); // negative from the end
109 EXPECT_THROW(b::index(xs, 3), std::out_of_range);
110}

112TEST(CheatahBuiltins, IndexDict) {

115 EXPECT_THROW(b::index(m, std::string("z")), std::out_of_range);

118TEST(CheatahBuiltins, SliceString) {
119 const std::string s = "hello world";
120 EXPECT_EQ(b::slice(s, 0, 5), "hello");
121 EXPECT_EQ(b::slice(s, 6, b::slice_end), "world"); // s[6:]
122 EXPECT_EQ(b::slice(s, 0, b::slice_end), s); // s[:]
123 EXPECT_EQ(b::slice(s, -5, b::slice_end), "world"); // negative start
124 EXPECT_EQ(b::slice(s, 3, 1), ""); // empty when lo >= hi
125 EXPECT_EQ(b::slice(s, 0, 100), s); // hi clamped to len
126}

128TEST(CheatahBuiltins, SliceList) {
129 const std::vector<long long> xs{1, 2, 3, 4, 5};
130 EXPECT_EQ(b::slice(xs, 1, 4), (std::vector<long long>{2, 3, 4}));
131 EXPECT_EQ(b::slice(xs, -2, b::slice_end), (std::vector<long long>{4, 5}));
132 EXPECT_TRUE(b::slice(xs, 3, 1).empty());

135TEST(CheatahBuiltins, Division) {
136 // truediv (the `/` operator) is ALWAYS floating-point, like Python 3.
137 EXPECT_DOUBLE_EQ(b::truediv(6, 4), 1.5); // int / int -> float
138 EXPECT_DOUBLE_EQ(b::truediv(6, 2), 3.0); // exact, but still a double
139 EXPECT_DOUBLE_EQ(b::truediv(7.0, 2.0), 3.5);
140 // floordiv (the `//` operator) floors toward -inf, the way Python does.
141 EXPECT_EQ(b::floordiv(7, 2), 3); // a%b != 0, same sign -> no adjust
142 EXPECT_EQ(b::floordiv(-7, 2), -4); // different signs -> floor adjust
143 EXPECT_EQ(b::floordiv(6, 2), 3); // exact (a%b == 0) -> no adjust
144 EXPECT_DOUBLE_EQ(b::floordiv(7.0, 2.0), 3.0); // floating operands -> floored double
145 EXPECT_DOUBLE_EQ(b::floordiv(7.0, 2), 3.0); // mixed -> floored double
146}
147
148// Integer `//` and `%` by zero raise a CONTROLLED error (std::domain_error) instead of undefined
149// behavior — C++ integer divide/modulo by zero is UB (SIGFPE). Float `/`,`//`,`%` are IEEE-safe.
150TEST(CheatahBuiltins, IntegerDivideAndModuloByZeroThrow) {
151 EXPECT_THROW(b::floordiv(1, 0), std::domain_error);
152 EXPECT_THROW(b::floordiv(-5, 0), std::domain_error);
153 EXPECT_THROW(b::mod(1, 0), std::domain_error);
154 EXPECT_THROW(b::mod(-5, 0), std::domain_error);
155}
