1# Copyright (c) 2026 BigBrain LLC. MIT-licensed (see LICENSE).
2# Original work; see ACKNOWLEDGMENTS.md for the open-source ideas we build upon.
3import io
4import time
5import math
6let N = 256
7let steps = 200
8let px: list<float> = []
9let py: list<float> = []
10let vx: list<float> = []
11let vy: list<float> = []
12for i in range(0, N) {
13 let fi = 1.0 + i
14 px.append(math.sin(fi * 1.3))
15 py.append(math.cos(fi * 0.7))
16 vx.append(0.0)
17 vy.append(0.0)
18}
19let dt = 0.001
20let eps = 0.01
21let t0 = time.monotonic()
22for s in range(0, steps) {
23 for i in range(0, N) {
24 let ax = 0.0
25 let ay = 0.0
26 for j in range(0, N) {
27 let dx = px[j] - px[i]
28 let dy = py[j] - py[i]
29 let r2 = dx * dx + dy * dy + eps
30 let inv = 1.0 / (r2 * math.sqrt(r2))
31 ax = ax + dx * inv
32 ay = ay + dy * inv
33 }
34 vx[i] = vx[i] + dt * ax
35 vy[i] = vy[i] + dt * ay
36 }
37 for i in range(0, N) {
38 px[i] = px[i] + dt * vx[i]
39 py[i] = py[i] + dt * vy[i]
40 }
41}
42let chk = 0.0
43for i in range(0, N) { chk = chk + px[i] + py[i] }
44let t1 = time.monotonic()
45io.print(chk)
46io.print(t1 - t0)
