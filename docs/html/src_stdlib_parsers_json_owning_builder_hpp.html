3// cheatah::parsers::json — OwningBuilder: the construction policy for the OWNING parse path.
4//
5// Same STACK-MACHINE interface as PoolBuilder (begin/add/finish driven by the iterative grammar),
6// but self-contained: each open container is built in its own std::vector and, on finish, the WHOLE
7// vector is moved into an OwnedArray/OwnedObject (O(1) — no per-element copies). Strings are copied
8// too (owns_strings), so the resulting Document owns its entire tree and is independent of any
9// Parser, pool, or the source buffer — safe to return, cache, or outlive the input. No runtime
10// polymorphism — a plain policy type.
11
12#include <utility>
13#include <vector>
14
15#include "document.hpp" // Node, Member, OwnedArray, OwnedObject (via node.hpp)
16
17namespace cheatah::parsers::json {
18
19/**
20 * @brief The construction policy for the OWNING JSON parse path: the same stack-machine interface
21 * as PoolBuilder (begin/add/finish), but each container is built in its own std::vector and,
22 * on finish, the whole vector is moved into an OwnedArray/OwnedObject (O(1), no per-element
23 * copies). Strings are copied too, so the resulting Document owns its entire tree and is
24 * independent of any Parser, pool, or source buffer. A plain policy type (no polymorphism).
25 */
26class OwningBuilder {
27public:
28 /// True: strings are COPIED into owned String<std::string> (not views into the source), so a
29 /// Document from this path is fully self-contained (cf. PoolBuilder::owns_strings == false).
30 static constexpr bool owns_strings = true;
31
32 /// Open an array by pushing a fresh, empty backing vector onto the array stack.
33 /// @complexity O(1) @alloc the stack slot (amortized) @test Json.OwningForms
34 void begin_array() { array_stack_.emplace_back(); }
35 /// Open an object by pushing a fresh, empty backing vector onto the object stack.
36 /// @complexity O(1) @alloc the stack slot (amortized) @test Json.OwningForms
37 void begin_object() { object_stack_.emplace_back(); }
38
39 /// Append an element to the innermost open array.
40 /// @param n the element node to append (moved in).
41 /// @complexity O(1) amortized @alloc vector growth @test Json.OwningForms
42 void add_element(Node&& n) { array_stack_.back().push_back(std::move(n)); }
43 /// Append a member to the innermost open object.
44 /// @param m the key/value member to append (moved in).
45 /// @complexity O(1) amortized @alloc vector growth @test Json.OwningForms
46 void add_member(Member&& m) { object_stack_.back().push_back(std::move(m)); }
47
48 /// Close the innermost open array: move its whole backing vector into an owned Node.
49 /// @return a Node holding the closed array as an OwnedArray.
50 /// @complexity O(1) (whole-vector move) @alloc none @test Json.OwningForms
51 Node finish_array() {
52 Node n;
53 n.variant().emplace<OwnedArray>(std::move(array_stack_.back()));
54 array_stack_.pop_back();
55 return n;
56 }
57 /// Close the innermost open object: move its whole backing vector into an owned Node.
58 /// @return a Node holding the closed object as an OwnedObject.
59 /// @complexity O(1) (whole-vector move) @alloc none @test Json.OwningForms
60 Node finish_object() {
61 Node n;
62 n.variant().emplace<OwnedObject>(std::move(object_stack_.back()));
63 object_stack_.pop_back();
64 return n;
65 }
66
67private:
68 std::vector<std::vector<Node>> array_stack_; // in-progress arrays (LIFO by nesting depth)
69 std::vector<std::vector<Member>> object_stack_; // in-progress objects (LIFO by nesting depth)
70};
71
72} // namespace cheatah::parsers::json
