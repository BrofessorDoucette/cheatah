12std::vector<std::ptrdiff_t> contiguous_strides(const std::vector<std::size_t>& shape) {
13 std::vector<std::ptrdiff_t> s(shape.size());
14 std::ptrdiff_t step = 1;
15 for (std::size_t i = shape.size(); i-- > 0;) {
16 s[i] = step;
17 step *= static_cast<std::ptrdiff_t>(shape[i]);
18 }
19 return s;
20}

22// Overflow-checked product of the dimensions. A huge shape would otherwise wrap
23// std::size_t and under-allocate, turning later element access into out-of-bounds
24// writes — so we reject it up front.
25std::size_t product(const std::vector<std::size_t>& shape) {
26 std::size_t t = 1;
27 for (std::size_t d : shape) {
28 if (d != 0 && t > std::numeric_limits<std::size_t>::max() / d) {
29 throw std::runtime_error("ndarray: shape too large (size overflow)");
30 }
31 t *= d;
32 }
33 return t;
34}

36// Convert signed dimensions/indices to sizes, rejecting negatives — a negative
37// cast to std::size_t becomes an enormous value (another under-allocation / OOB
38// vector). Validate at the boundary where untrusted shapes/indices enter.
39std::vector<std::size_t> to_size(const std::vector<long long>& v) {
40 std::vector<std::size_t> out(v.size());
41 for (std::size_t i = 0; i < v.size(); ++i) {
42 if (v[i] < 0) throw std::runtime_error("ndarray: negative dimension or index");
43 out[i] = static_cast<std::size_t>(v[i]);
44 }
45 return out;
46}

48// Advance a C-order multi-index odometer; returns false when it wraps past the end.
49bool next_index(std::vector<std::size_t>& idx, const std::vector<std::size_t>& shape) {
50 for (std::size_t i = shape.size(); i-- > 0;) {
51 if (++idx[i] < shape[i]) return true;
52 idx[i] = 0;
53 }
54 return false;
55}

57template <typename Op>
58NDArray binary_op(const NDArray& a, const NDArray& b, Op op) {
59 const std::vector<std::size_t> rshape = broadcast_shapes(a.shape(), b.shape());
60 const NDArray av = broadcast_to(a, rshape);
61 const NDArray bv = broadcast_to(b, rshape);
62 NDArray out(rshape);
63 auto& buf = *out.buffer();
64 std::vector<std::size_t> idx(rshape.size(), 0);
65 std::size_t flat = 0;
66 do {
67 buf[flat++] = op(av.at(idx), bv.at(idx)); // out is contiguous -> flat order
68 } while (!rshape.empty() && next_index(idx, rshape));
69 return out;
70}

72void format_rec(const NDArray& a, std::vector<std::size_t>& idx, std::size_t dim, std::string& out) {

105double NDArray::at(const std::vector<std::size_t>& index) const {
106 // Bounds-check: a wrong-rank or out-of-range index would otherwise compute an
107 // offset outside the backing buffer (out-of-bounds read).
108 if (index.size() != shape_.size()) {
109 throw std::runtime_error("ndarray: index has the wrong number of dimensions");
110 }
111 std::ptrdiff_t off = static_cast<std::ptrdiff_t>(offset_);
112 for (std::size_t i = 0; i < index.size(); ++i) {
113 if (index[i] >= shape_[i]) throw std::runtime_error("ndarray: index out of range");
114 off += static_cast<std::ptrdiff_t>(index[i]) * strides_[i];
115 }
116 return (*data_)[static_cast<std::size_t>(off)];
117}

119std::vector<std::size_t> broadcast_shapes(const std::vector<std::size_t>& a,
120 const std::vector<std::size_t>& b) {
121 const std::size_t n = std::max(a.size(), b.size());
122 std::vector<std::size_t> r(n);
123 for (std::size_t i = 0; i < n; ++i) {
124 // Align from the right (trailing dimensions).
125 const std::size_t da = (i < n - a.size()) ? 1 : a[i - (n - a.size())];
126 const std::size_t db = (i < n - b.size()) ? 1 : b[i - (n - b.size())];
127 if (da == db || db == 1) {
128 r[i] = da;
129 } else if (da == 1) {
130 r[i] = db;
131 } else {
132 throw std::runtime_error("ndarray: operands could not be broadcast together");
133 }
134 }
135 return r;
136}

138NDArray broadcast_to(const NDArray& a, const std::vector<std::size_t>& target) {
139 const std::size_t n = target.size();
140 if (a.ndim() > n) {
141 throw std::runtime_error("ndarray: cannot broadcast to fewer dimensions");
142 }
143 std::vector<std::ptrdiff_t> ns(n, 0); // stretched / missing dims -> stride 0
144 const std::size_t pad = n - a.ndim();
145 for (std::size_t i = 0; i < a.ndim(); ++i) {
146 const std::size_t adim = a.shape()[i];
147 if (adim == target[pad + i]) {
148 ns[pad + i] = a.strides()[i];
149 } else if (adim != 1) {
150 throw std::runtime_error("ndarray: shape not broadcastable to target");
151 } // adim == 1 -> stride stays 0 (stretch)
152 }
153 return NDArray(a.buffer(), target, ns, a.offset());
154}

181NDArray reshape(const NDArray& a, const std::vector<long long>& shape) {
182 const std::vector<std::size_t> ns = to_size(shape);
183 if (product(ns) != a.size()) {
184 throw std::runtime_error("ndarray: cannot reshape, size mismatch");
185 }
186 // Flatten in C-order (handles views), then lay out contiguously.
187 NDArray out(ns);
188 auto& buf = *out.buffer();
189 std::vector<std::size_t> idx(a.ndim(), 0);
190 std::size_t flat = 0;
191 do {
192 buf[flat++] = a.at(idx);
193 } while (a.ndim() != 0 && next_index(idx, a.shape()));
194 return out;
195}
