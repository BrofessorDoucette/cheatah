1// System-level test for the cheatah `time` module: a single cohesive .purr
2// program that exercises EVERY public function in stdlib/time/time.hpp
3// (time, time_ns, monotonic, monotonic_ns, perf_counter, perf_counter_ns,
4// process_time, sleep), compiles it with purrc into a loadable module, runs it
5// under the cheatah runtime, and asserts the exact stdout byte-for-byte.
6//
7// Clock readings are non-deterministic, so the program prints DETERMINISTIC
8// boolean PROPERTIES instead of raw values:
9// - wall clock (time/time_ns) is strictly positive;
10// - the monotonic / perf-counter clocks are positive and, after a sleep,
11// read >= their earlier values (they never run backward);
12// - process_time (CPU time) is non-negative.
13// Complements the in-process unit tests (stdlib/tests/time_test.cpp) and the
14// per-function compile-run tests (tests/purrc/time_cr_test.cpp).
15#include "e2e_harness.hpp"
16
17TEST(StdlibE2E, Time) {
18 e2e::expect_e2e("time_sys", R"PURR(import io
19import time
20
21# Wall-clock readings are positive.
22let w = time.time()
23let wn = time.time_ns()
24
25# Monotonic / perf clocks are positive and never run backward across a sleep.
26let m0 = time.monotonic()
27let mn0 = time.monotonic_ns()
28let p0 = time.perf_counter()
29let pn0 = time.perf_counter_ns()
30
31time.sleep(0.01)
32
33let m1 = time.monotonic()
34let mn1 = time.monotonic_ns()
35let p1 = time.perf_counter()
36let pn1 = time.perf_counter_ns()
37
38# CPU time consumed by the process is non-negative.
39let cpu = time.process_time()
40
41io.print(w > 0.0, wn > 0)
42io.print(m0 > 0.0, mn0 > 0, p0 > 0.0, pn0 > 0)
43io.print(m1 >= m0, mn1 >= mn0, p1 >= p0, pn1 >= pn0)
44io.print(cpu >= 0.0)
45)PURR",
46 "True True\n"
47 "True True True True\n"
48 "True True True True\n"
49 "True\n");
50}
