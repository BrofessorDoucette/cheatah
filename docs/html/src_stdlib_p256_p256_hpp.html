5/**
6 * @file p256.hpp
7 * @brief cheatah `p256` — the NIST P-256 (secp256r1) elliptic curve: ECDSA
8 * signature verification and deterministic (RFC 6979) signing, plus the
9 * DER/SPKI parsing needed to use it with TLS certificates and JWTs.
10 * From scratch, no external libraries. `import p256` to use it.
11 *
12 * It is the curve most of the world's TLS certificates and OAuth/JWT (ES256)
13 * are signed with, so it lets cheatah's TLS client verify a real server's
14 * certificate (ecdsa_secp256r1_sha256) and sign an ES256 JWT.
15 *
16 * Field/scalar arithmetic is Montgomery form over the two P-256 moduli (the
17 * field prime p and the group order n); points use Jacobian coordinates — a
18 * branchy Strauss-Shamir double chain for verification (public data) and a
19 * constant-time fixed-base comb (branch-free point ops, masked table selection)
20 * for the secret-scalar signing/keygen path. Signing uses an RFC 6979
21 * deterministic nonce (HMAC-SHA256), so it needs no entropy source and never
22 * repeats a nonce.
23 *
24 * Byte conventions: scalars/coordinates are 32 big-endian bytes; an uncompressed
25 * public key point is the 64 bytes X||Y; a raw ECDSA signature is the 64 bytes
26 * r||s (the JWT ES256 form); a DER signature is SEQUENCE{ INTEGER r, INTEGER s }
27 * (the TLS/X.509 form).
28 */
29
30#include <string>
31#include <string_view>
32
33namespace cheatah::p256 {
34
35/**
36 * Verify an ECDSA/P-256 signature.
37 * @param pubkey_xy the public key as 64 bytes (X||Y), big-endian.
38 * @param msg_hash the message digest (e.g. 32 bytes of SHA-256); if longer than
39 * 32 bytes it is truncated to the leftmost 256 bits (FIPS 186-4).
40 * @param sig_der the signature, DER-encoded SEQUENCE{INTEGER r, INTEGER s}.
41 * @return true iff the signature is valid for @p pubkey_xy over @p msg_hash.
42 * @complexity O(1) — two scalar multiplications, computed as one Strauss-Shamir double chain.
43 * @alloc a temporary raw r||s signature string.
44 * @test CheatahP256.VerifyKnownVector
45 */
46[[nodiscard]] bool verify_der(const std::string& pubkey_xy, const std::string& msg_hash,
47 const std::string& sig_der);
48
49/**
50 * Verify an ECDSA/P-256 signature given the raw 64-byte r||s form (the JWT
51 * ES256 layout).
52 * @param pubkey_xy the public key (64 bytes X||Y).
53 * @param msg_hash the digest (truncated to 256 bits if longer).
54 * @param sig_raw the signature as 64 bytes r||s.
55 * @return true iff valid.
56 * @complexity O(1) — two scalar multiplications, computed as one Strauss-Shamir double chain.
57 * @alloc none.
58 * @test CheatahP256.VerifyKnownVector
59 */
60[[nodiscard]] bool verify_raw(const std::string& pubkey_xy, const std::string& msg_hash,
61 const std::string& sig_raw);
62
63/**
64 * Deterministic ECDSA/P-256 signing (RFC 6979 nonce, HMAC-SHA256). Returns the
65 * raw 64-byte r||s signature — the form an ES256 JWT carries.
66 * @param privkey the private scalar d as 32 big-endian bytes.
67 * @param msg_hash the digest to sign (truncated to 256 bits if longer).
68 * @return the signature as 64 bytes r||s, or "" if @p privkey is not a valid scalar
69 * (wrong length, zero, or >= the group order n).
70 * @complexity O(1) — one fixed-base scalar multiplication per RFC 6979 candidate (almost always one).
71 * @alloc the returned signature plus RFC 6979 HMAC scratch strings.
72 * @test CheatahP256.SignKnownVector
73 */
74[[nodiscard]] std::string sign_raw(const std::string& privkey, const std::string& msg_hash);
75
76/**
77 * Encode a raw 64-byte r||s signature as the DER `SEQUENCE{INTEGER r, INTEGER s}`
78 * that TLS CertificateVerify and X.509 carry — the inverse of the parse inside
79 * @ref verify_der, with minimal-form integers (stripped leading zeros, 0x00 sign
80 * byte when the top bit is set). Composes with @ref sign_raw to produce the wire
81 * form a TLS 1.3 server sends for `ecdsa_secp256r1_sha256`.
82 * @param sig_raw the 64-byte r||s signature.
83 * @return the DER bytes, or "" if @p sig_raw has the wrong length or a zero integer.
84 * @complexity O(1).
85 * @alloc the returned string.
86 * @test CheatahP256.RsToDerRoundTripsAndRejects
87 */
88[[nodiscard]] std::string rs_to_der(const std::string& sig_raw);
89
90/**
91 * Derive the public key point from a private scalar: Q = d·G.
92 * @param privkey the private scalar d as 32 big-endian bytes.
93 * @return the public key as 64 bytes (X||Y), or "" if d is out of range.
94 * @complexity O(1) — one fixed-base scalar multiplication.
95 * @alloc the returned point.
96 * @test CheatahP256.PublicFromPrivate
97 */
98[[nodiscard]] std::string public_from_private(const std::string& privkey);
99
100/**
101 * Extract the P-256 public key point from a certificate / SubjectPublicKeyInfo
102 * DER: finds the `id-ecPublicKey` + `prime256v1` AlgorithmIdentifier and returns
103 * the uncompressed point's 64 bytes (X||Y). Returns "" if the SPKI is not a
104 * P-256 uncompressed EC key.
105 * @param spki_or_cert_der the certificate or SPKI DER bytes.
106 * @return the 64-byte X||Y point, or "" if not a P-256 EC key.
107 * @complexity O(der length).
108 * @alloc the returned point.
109 * @test CheatahP256.SpkiExtractsPoint
110 */
111[[nodiscard]] std::string spki_ec_point(std::string_view spki_or_cert_der);

114namespace testonly {
115/**
116 * TEST-ONLY seam. Reduce a 32-byte big-endian scalar modulo the group order n
117 * using the exact `reduce_mod_n` the signing/verification paths use. Exists so the
118 * conditional-subtraction branch (a value in [n, p), a ~2^-128 event on real curve
119 * x-coordinates) can be exercised on the real implementation without weakening it.
120 * Not part of the public P-256 API.
121 */
122[[nodiscard]] std::string reduce_mod_n_be(const std::string& be32);
123
124/**
125 * TEST-ONLY seam. Deterministic sign that first rejects `force_retries` valid RFC
126 * 6979 nonce candidates, exercising the retry loop (and, when force_retries exceeds
127 * the attempt cap, the "exhausted" empty return). With force_retries == 0 it is
128 * identical to sign_raw. Not part of the public P-256 API.
129 */
130[[nodiscard]] std::string sign_raw_skip(const std::string& privkey,
131 const std::string& msg_hash, int force_retries);
132
133/**
134 * TEST-ONLY seam. Differentially validate the constant-time point arithmetic used by the secret-
135 * scalar signing/keygen path (jac_add_ct / jac_double_ct) against the branchy reference ops across
136 * the general case and every special case (a==b, a==-b, infinity operands). Returns true iff they
137 * agree everywhere. Not part of the public P-256 API.
138 */
139[[nodiscard]] bool ct_point_selfcheck();
140} // namespace testonly
141#endif // CHEATAH_P256_TESTING
