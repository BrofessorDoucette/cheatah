1// System-level (whole-program) test for the `ed25519` stdlib module, written as a
2// worked EXAMPLE of the signature workflow that backs cheatah's binary-integrity
3// feature: a publisher signs a payload with a secret key, anyone holding only the
4// PUBLIC key can verify it, and ANY tampering — to the payload, the signature, or the
5// key — is rejected. This is the same Ed25519 the runtime uses to refuse a tampered
6// `.so` (see integrity_e2e_test.cpp).
7//
8// Coverage — every function in stdlib/ed25519/ed25519.hpp:
9// generate, public_key, sign, verify.
10//
11// Deterministic: signing is deterministic (RFC 8032), and the generate() result is
12// only exercised by property (a fresh key round-trips), so stdout is fixed.
13#include "e2e_harness.hpp"
14
15TEST(StdlibE2E, Ed25519) {
16 e2e::expect_e2e("ed25519_sys", R"PURR(import io
17import ed25519
18
19# A publisher's keypair. In real use the secret stays offline (purrc --keygen); only
20# the public key is shipped to verifiers. Here we use a fixed RFC 8032 seed so the
21# signature is reproducible.
22let secret = "c5aa8df43f9f837bedb7442f31dcb7b166d38535076f094b85ce3a2e0b4458f7"
23let pub = ed25519.public_key(secret)
24io.print("public:", pub)
25
26# Sign a payload (think: the bytes of a compiled module).
27let payload = "the trusted payload"
28let sig = ed25519.sign(secret, payload)
29
30# A verifier with ONLY the public key accepts the genuine payload+signature.
31io.print("genuine accepted:", ed25519.verify(pub, payload, sig))
32
33# An attacker who injects different bytes but keeps the old signature is rejected:
34# the signature no longer matches the payload.
35io.print("tampered payload rejected:", ed25519.verify(pub, "the INJECTED payload", sig) == false)
36
37# Tampering with the signature itself is rejected.
38let forged = "00" + sig[2:]
39io.print("forged signature rejected:", ed25519.verify(pub, payload, forged) == false)
40
41# A signature from a DIFFERENT key does not verify under this public key — a verifier
42# only trusts payloads signed by the key it pins.
43let other_secret = ed25519.generate()
44let other_sig = ed25519.sign(other_secret, payload)
45io.print("untrusted signer rejected:", ed25519.verify(pub, payload, other_sig) == false)
46
47# A freshly generated key still produces a working, verifiable signature.
48let fresh = ed25519.generate()
49let fresh_pub = ed25519.public_key(fresh)
50io.print("fresh keypair verifies:", ed25519.verify(fresh_pub, payload, ed25519.sign(fresh, payload)))
51)PURR",
52 "public: fc51cd8e6218a1a38da47ed00230f0580816ed13ba3303ac5deb911548908025\n"
53 "genuine accepted: True\n"
54 "tampered payload rejected: True\n"
55 "forged signature rejected: True\n"
56 "untrusted signer rejected: True\n"
57 "fresh keypair verifies: True\n");
58}
