17// field/curve arithmetic follows the public-domain TweetNaCl reference algorithm
18// (Bernstein, van Gastel, Janssen, Lange, Schwabe, Smetsers), reimplemented in C++ and
19// validated against the RFC 8032 known-answer vectors in stdlib/tests/ed25519_test.cpp.

51void car25519(gf& o) {
52 for (int i = 0; i < 16; ++i) {
53 o[i] += (1LL << 16);
54 const i64 c = o[i] >> 16;
55 // i<15: carry into the next limb; i==15: carry wraps as 38*(c-1) (since 2^256≡38).
56 o[(i + 1) * (i < 15)] += c - 1 + 37 * (c - 1) * (i == 15);
57 o[i] -= c << 16;
58 }
59}

61void sel25519(gf& p, gf& q, int b) {
62 const i64 c = ~(b - 1);
63 for (int i = 0; i < 16; ++i) {
64 const i64 t = c & (p[i] ^ q[i]);
65 p[i] ^= t;
66 q[i] ^= t;
67 }
68}

70void pack25519(u8* o, const gf& n) {
71 gf m{}, t = n;
72 car25519(t);
73 car25519(t);
74 car25519(t);
75 for (int j = 0; j < 2; ++j) {
76 m[0] = t[0] - 0xffed;
77 for (int i = 1; i < 15; ++i) {
78 m[i] = t[i] - 0xffff - ((m[i - 1] >> 16) & 1);
79 m[i - 1] &= 0xffff;
80 }
81 m[15] = t[15] - 0x7fff - ((m[14] >> 16) & 1);
82 const int b = (m[15] >> 16) & 1;
83 m[14] &= 0xffff;
84 sel25519(t, m, 1 - b);
85 }
86 for (int i = 0; i < 16; ++i) {
87 o[2 * i] = static_cast<u8>(t[i] & 0xff);
88 o[2 * i + 1] = static_cast<u8>(t[i] >> 8);
89 }
90}

92int neq25519(const gf& a, const gf& b) {
93 u8 c[32], d[32];
94 pack25519(c, a);
95 pack25519(d, b);
96 // crypto_verify_32 (constant-time): 0 if equal.
97 unsigned diff = 0;
98 for (int i = 0; i < 32; ++i) diff |= static_cast<unsigned>(c[i] ^ d[i]);
99 return (1 & ((diff - 1) >> 8)) - 1; // 0 if equal, -1 otherwise
100}

108void unpack25519(gf& o, const u8* n) {
109 for (int i = 0; i < 16; ++i) o[i] = n[2 * i] + (static_cast<i64>(n[2 * i + 1]) << 8);
110 o[15] &= 0x7fff;
111}

113void A(gf& o, const gf& a, const gf& b) {

116void Z(gf& o, const gf& a, const gf& b) {

119void M(gf& o, const gf& a, const gf& b) {
120 i64 t[31] = {0};
121 for (int i = 0; i < 16; ++i)
122 for (int j = 0; j < 16; ++j) t[i + j] += a[i] * b[j];
123 for (int i = 0; i < 15; ++i) t[i] += 38 * t[i + 16];
124 for (int i = 0; i < 16; ++i) o[i] = t[i];
125 car25519(o);
126 car25519(o);
127}

149void add(gf p[4], gf q[4]) {
150 gf a, b, c, d, t, e, f, g, h;
151 Z(a, p[1], p[0]);
152 Z(t, q[1], q[0]);
153 M(a, a, t);
154 A(b, p[0], p[1]);
155 A(t, q[0], q[1]);
156 M(b, b, t);
157 M(c, p[3], q[3]);
158 M(c, c, D2);
159 M(d, p[2], q[2]);
160 A(d, d, d);
161 Z(e, b, a);
162 Z(f, d, c);
163 A(g, d, c);
164 A(h, b, a);
165 M(p[0], e, f);
166 M(p[1], h, g);
167 M(p[2], g, f);
168 M(p[3], e, h);
169}

175void pack(u8* r, gf p[4]) {
176 gf tx, ty, zi;
177 inv25519(zi, p[2]);
178 M(tx, p[0], zi);
179 M(ty, p[1], zi);
180 pack25519(r, ty);
181 r[31] ^= par25519(tx) << 7;
182}

184void scalarmult(gf p[4], gf q[4], const u8* s) {
185 set25519(p[0], gf0);
186 set25519(p[1], gf1);
187 set25519(p[2], gf1);
188 set25519(p[3], gf0);
189 for (int i = 255; i >= 0; --i) {
190 const u8 b = (s[i / 8] >> (i & 7)) & 1;
191 cswap(p, q, b);
192 add(q, p);
193 add(p, p);
194 cswap(p, q, b);
195 }
196}

198void scalarbase(gf p[4], const u8* s) {
199 gf q[4];
200 set25519(q[0], X);
201 set25519(q[1], Y);
202 set25519(q[2], gf1);
203 M(q[3], X, Y);

207void modL(u8* r, i64 x[64]) {
208 for (int i = 63; i >= 32; --i) {
209 i64 carry = 0;
210 int j = i - 32;
211 for (; j < i - 12; ++j) {
212 x[j] += carry - 16 * x[i] * LCONST[j - (i - 32)];
213 carry = (x[j] + 128) >> 8;
214 x[j] -= carry << 8;
215 }
216 x[j] += carry;
217 x[i] = 0;
218 }
219 i64 carry = 0;
220 for (int j = 0; j < 32; ++j) {
221 x[j] += carry - (x[31] >> 4) * LCONST[j];
222 carry = x[j] >> 8;
223 x[j] &= 255;
224 }
225 for (int j = 0; j < 32; ++j) x[j] -= carry * LCONST[j];
226 for (int i = 0; i < 32; ++i) {
227 x[i + 1] += x[i] >> 8;
228 r[i] = static_cast<u8>(x[i] & 255);
229 }
230}

232void reduce(u8* r) {
233 i64 x[64];
234 for (int i = 0; i < 64; ++i) x[i] = static_cast<u64>(r[i]);
235 for (int i = 0; i < 64; ++i) r[i] = 0;
236 modL(r, x);

239// Whether the 32-byte little-endian scalar @p s is canonical, i.e. s < L (the group
240// order). RFC 8032 strict verification rejects a non-canonical S, which closes the
241// signature-malleability door (a forger can't add a multiple of L to S).
242bool scalar_is_canonical(const u8* s) {
243 for (int i = 31; i >= 0; --i) {
244 const u8 li = static_cast<u8>(LCONST[i]);
245 if (s[i] < li) return true;
246 if (s[i] > li) return false;
247 }
248 return false; // s == L is NOT canonical
249}

251int unpackneg(gf r[4], const u8 p[32]) {
252 gf t, chk, num, den, den2, den4, den6;
253 set25519(r[2], gf1);
254 unpack25519(r[1], p);
255 S(num, r[1]);
256 M(den, num, D);
257 Z(num, num, r[2]);
258 A(den, r[2], den);
259 S(den2, den);
260 S(den4, den2);
261 M(den6, den4, den2);
262 M(t, den6, num);
263 M(t, t, den);
264 pow2523(t, t);
265 M(t, t, num);
266 M(t, t, den);
267 M(t, t, den);
268 M(r[0], t, den);
269 S(chk, r[0]);
270 M(chk, chk, den);
271 if (neq25519(chk, num)) M(r[0], r[0], I);
272 S(chk, r[0]);
273 M(chk, chk, den);
274 if (neq25519(chk, num)) return -1;
275 if (par25519(r[0]) == (p[31] >> 7)) Z(r[0], gf0, r[0]);
276 M(r[3], r[0], r[1]);
277 return 0;
278}

280// SHA-512 of n bytes -> 64-byte digest, via cheatah::hashlib.
281void sha512(u8* out, const u8* in, std::size_t n) {
282 const std::string d =
283 hashlib::sha512_digest(std::string_view(reinterpret_cast<const char*>(in), n));
284 for (int i = 0; i < 64; ++i) out[i] = static_cast<u8>(d[i]);
285}

288std::string from_hex(std::string_view hex) {
289 if (hex.size() % 2 != 0) throw std::invalid_argument("ed25519: odd-length hex");
290 auto nib = [](char c) -> int {
291 if (c >= '0' && c <= '9') return c - '0';
292 if (c >= 'a' && c <= 'f') return c - 'a' + 10;
293 if (c >= 'A' && c <= 'F') return c - 'A' + 10;
294 return -1;
295 };
296 std::string out;
297 out.reserve(hex.size() / 2);
298 for (std::size_t i = 0; i < hex.size(); i += 2) {
299 const int hi = nib(hex[i]), lo = nib(hex[i + 1]);
300 if (hi < 0 || lo < 0) throw std::invalid_argument("ed25519: non-hex character");
301 out.push_back(static_cast<char>((hi << 4) | lo));
302 }
303 return out;
304}

306std::string to_hex(const u8* p, std::size_t n) {
307 static const char hexd[] = "0123456789abcdef";
308 std::string out;
309 out.reserve(n * 2);
310 for (std::size_t i = 0; i < n; ++i) {
311 out.push_back(hexd[p[i] >> 4]);
312 out.push_back(hexd[p[i] & 0xF]);
313 }
314 return out;
315}

317void secure_random(u8* out, std::size_t n) {
318#if defined(_WIN32)
319 if (::BCryptGenRandom(nullptr, out, static_cast<unsigned long>(n),
320 BCRYPT_USE_SYSTEM_PREFERRED_RNG) != 0)
321 throw std::runtime_error("ed25519: BCryptGenRandom failed");
322#else
323 std::size_t off = 0;
324 while (off < n) {
325 const std::size_t chunk = (n - off < 256) ? (n - off) : 256;
326 if (::getentropy(out + off, chunk) != 0) throw std::runtime_error("ed25519: getentropy failed");
327 off += chunk;
328 }
329#endif
330}

360std::string sign(std::string_view secret_hex, std::string_view message) {
361 const std::string seed = from_hex(secret_hex);
362 if (seed.size() != 32) throw std::invalid_argument("ed25519: secret seed must be 32 bytes (64 hex)");
363 const u8* sd = reinterpret_cast<const u8*>(seed.data());
364 const std::size_t n = message.size();
365
366 u8 d[64];
367 sha512(d, sd, 32);
368 d[0] &= 248;
369 d[31] &= 127;
370 d[31] |= 64; // a = d[0..31] (clamped scalar)
371
372 u8 pub[32];
373 {
374 gf p[4];
375 scalarbase(p, d);
376 pack(pub, p);
377 }
378
379 // r = SHA512(prefix || M), prefix = d[32..63]
380 std::string rbuf_in(reinterpret_cast<const char*>(d + 32), 32);
381 rbuf_in.append(message.data(), n);
382 u8 r[64];
383 sha512(r, reinterpret_cast<const u8*>(rbuf_in.data()), rbuf_in.size());
384 reduce(r);
385
386 // R = r * B
387 u8 R[32];
388 {
389 gf p[4];
390 scalarbase(p, r);
391 pack(R, p);
392 }
393
394 // k = SHA512(R || A || M)
395 std::string kbuf_in(reinterpret_cast<const char*>(R), 32);
396 kbuf_in.append(reinterpret_cast<const char*>(pub), 32);
397 kbuf_in.append(message.data(), n);
398 u8 h[64];
399 sha512(h, reinterpret_cast<const u8*>(kbuf_in.data()), kbuf_in.size());
400 reduce(h);
401
402 // S = (r + k*a) mod L
403 i64 x[64] = {0};
404 for (int i = 0; i < 32; ++i) x[i] = static_cast<u64>(r[i]);
405 for (int i = 0; i < 32; ++i)
406 for (int j = 0; j < 32; ++j) x[i + j] += static_cast<i64>(h[i]) * static_cast<i64>(d[j]);
407 u8 Sout[32];
408 modL(Sout, x);
409
410 u8 sig[64];
411 for (int i = 0; i < 32; ++i) sig[i] = R[i];
412 for (int i = 0; i < 32; ++i) sig[32 + i] = Sout[i];
413 return to_hex(sig, 64);
414}

416bool verify(std::string_view public_hex, std::string_view message, std::string_view signature_hex) {
417 std::string pub, sig;
418 try {
419 pub = from_hex(public_hex);
420 sig = from_hex(signature_hex);
421 } catch (const std::exception&) {
422 return false; // malformed hex -> reject
423 }
424 if (pub.size() != 32 || sig.size() != 64) return false;
425 const u8* A_ = reinterpret_cast<const u8*>(pub.data());
426 const u8* sg = reinterpret_cast<const u8*>(sig.data());
427
428 if (!scalar_is_canonical(sg + 32)) return false; // reject non-canonical S (S >= L)
429
430 gf q[4];
431 if (unpackneg(q, A_) != 0) return false; // not a valid public key point
432
433 // h = SHA512(R || A || M)
434 std::string hin(reinterpret_cast<const char*>(sg), 32); // R
435 hin.append(reinterpret_cast<const char*>(A_), 32); // A
436 hin.append(message.data(), message.size()); // M
437 u8 h[64];
438 sha512(h, reinterpret_cast<const u8*>(hin.data()), hin.size());
439 reduce(h);
440
441 // p = S*B - h*A (q already holds -A)
442 gf p[4];
443 scalarmult(p, q, h); // p = h * (-A)
444 gf g[4];
445 scalarbase(g, sg + 32); // S * B
446 add(p, g);
447
448 u8 t[32];
449 pack(t, p);
450 // Accept iff the recomputed R equals the signature's R, constant-time.
451 unsigned diff = 0;
452 for (int i = 0; i < 32; ++i) diff |= static_cast<unsigned>(sg[i] ^ t[i]);
453 return diff == 0;
454}
