1// Copyright (c) 2026 BigBrain LLC. MIT-licensed (see LICENSE).
2// Original work; see ACKNOWLEDGMENTS.md for the open-source ideas we build upon.
3//
4// bench_pairs — the ONE place that knows how a cheatah benchmark is paired with its
5// competitor twin.
6//
7// The suffix convention differs per file for historical reasons, and renaming is not an
8// option: scripts/bench_gate.sh greps `_(fixed|glm)$` to build its shards, and
9// tests/benchmarks/compiler_bench_baseline.csv is keyed by benchmark name. So instead of
10// one convention enforced by renaming, this header holds the whole table:
11//
12// eigen_compare_bench.cpp BM_<op>_cheatah/<n> vs BM_<op>_eigen/<n>
13// glm_compare_bench.cpp BM_<op><n>_cheatah vs BM_<op><n>_glm
14// fixed_glm_bench.cpp BM_<op>_<type>_fixed vs BM_<op>_<type>_glm
15// crypto_openssl_bench.cpp BM_Crypto<P>_Cheatah vs BM_Crypto<P>_OpenSSL
16//
17// A name splits into <stem>/<args>; the last '_'-delimited token of the stem names the
18// side. Comparison is case-insensitive, because the crypto file capitalizes and the
19// numeric files do not.
20//
21// The landmine this header exists to defuse: BM_CryptoChaCha20Poly1305_Cheatah_Into ends
22// in `Into`, which is not a side token at all. Its side is the token BEFORE that. Treated
23// naively it would either be dropped or — worse — read as a rival of the plain
24// `_Cheatah` row and reported as cheatah losing to itself. Here it classifies as OURS
25// with variant `Into`, and the variant stays in the pair key so it can never pair with
26// anything but an identically-named rival (there is none, so it simply reports alone).
27#ifndef CHEATAH_TESTS_BENCHMARKS_BENCH_PAIRS_HPP
28#define CHEATAH_TESTS_BENCHMARKS_BENCH_PAIRS_HPP
29
30#include <array>
31#include <cctype>
32#include <string>
33#include <string_view>
34
35namespace cheatah::bench {
36
37// Which library a benchmark row measures.
38enum class Side { Ours, Rival, Unknown };
39
40struct Pairing {
41 Side side = Side::Unknown;
42 std::string key; // name with the side token removed; two sides of a pair share it
43 std::string label; // the rival's display name ("glm", "eigen", "openssl", …); ours: ""
44};
45
46namespace detail {
47
48// Side tokens, matched case-insensitively against the stem's last '_'-delimited token.
49inline constexpr std::array<std::string_view, 2> kOurs{"cheatah", "fixed"};
50inline constexpr std::array<std::string_view, 6> kRivals{"eigen", "glm", "openssl",
51 "std", "boost", "re2"};
52
53inline bool iequals(std::string_view a, std::string_view b) {
54 if (a.size() != b.size()) return false;
55 for (std::size_t i = 0; i < a.size(); ++i) {
56 const auto ca = static_cast<unsigned char>(a[i]);
57 const auto cb = static_cast<unsigned char>(b[i]);
58 if (std::tolower(ca) != std::tolower(cb)) return false;
59 }
60 return true;
61}
62
63// The canonical lowercase label for a token, or "" when the token names no side.
64inline std::string_view classify(std::string_view token, Side& side) {
65 for (std::string_view o : kOurs) {
66 if (iequals(token, o)) { side = Side::Ours; return o; }
67 }
68 for (std::string_view r : kRivals) {
69 if (iequals(token, r)) { side = Side::Rival; return r; }
70 }
71 side = Side::Unknown;
72 return {};
73}
74
75} // namespace detail
76
77// How a rival's name is written in a published table. `classify_benchmark` returns the
78// lowercase token it matched on (that is what pairing needs); a reader wants the library's
79// own capitalisation. Kept beside kRivals so the vocabulary has exactly one home.
80inline std::string display_label(std::string_view label) {
81 if (detail::iequals(label, "glm")) return "GLM";
82 if (detail::iequals(label, "openssl")) return "OpenSSL";
83 if (detail::iequals(label, "eigen")) return "Eigen";
84 if (detail::iequals(label, "re2")) return "RE2";
85 if (detail::iequals(label, "std")) return "std::regex";
86 if (detail::iequals(label, "boost")) return "Boost";
87 return std::string(label);
88}
89
90// Classify one benchmark name. `name` is Run::run_name.str() — it carries any `/args`
91// suffix but NOT the `_median`/`_mean` aggregate suffix (that lives on Run::aggregate_name).
92inline Pairing classify_benchmark(std::string_view name) {
93 Pairing p;
94
95 // Split <stem>/<args>. Args stay attached to the key so BM_dot_cheatah/64 pairs with
96 // BM_dot_eigen/64 and never with BM_dot_eigen/4096.
97 const std::size_t slash = name.find('/');
98 const std::string_view stem = name.substr(0, slash);
99 const std::string_view args = (slash == std::string_view::npos) ? std::string_view{}
100 : name.substr(slash);
101
102 const std::size_t under = stem.rfind('_');
103 if (under == std::string_view::npos) return p;
104
105 Side side = Side::Unknown;
106 std::string_view label = detail::classify(stem.substr(under + 1), side);
107
108 if (side == Side::Unknown) {
109 // Not a side token — this may be a VARIANT of ours, e.g. `..._Cheatah_Into`. Look
110 // one token further left; if that names a side, the trailing token is the variant.
111 const std::size_t prev = stem.substr(0, under).rfind('_');
112 if (prev == std::string_view::npos) return p;
113 const std::string_view variant = stem.substr(under + 1);
114 label = detail::classify(stem.substr(prev + 1, under - prev - 1), side);
115 if (side == Side::Unknown) return p;
116 // Keep the variant IN the key: a variant must never pair with the plain row.
117 p.side = side;
118 p.key = std::string(stem.substr(0, prev)) + "_" + std::string(variant) + std::string(args);
119 p.label = (side == Side::Rival) ? std::string(label) : std::string{};
120 return p;
121 }
122
123 p.side = side;
124 p.key = std::string(stem.substr(0, under)) + std::string(args);
125 p.label = (side == Side::Rival) ? std::string(label) : std::string{};
126 return p;
127}
128
129} // namespace cheatah::bench
