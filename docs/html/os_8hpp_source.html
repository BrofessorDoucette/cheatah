26/// StringLike<T>: a std::string can be constructed from T — exactly what
27/// os.path.join() does (std::string(part)). Naming it yields a clear "constraint
28/// StringLike not satisfied" message while still accepting everything it does today
29/// (const char*, char arrays, std::string, std::string_view, …).
30template <typename T>
31concept StringLike = requires(const T& value) { std::string(value); };

139/**
140 * Join path components with the platform separator.
141 * @param first the first component.
142 * @param rest any further string-constructible components.
143 * @return e.g. `join("a","b","c") -> "a/b/c"`.
144 * @note O(total length); allocates the result string and per-part path temporaries.
145 * @test CheatahOs.PathJoin
146 */
147template <StringLike... Parts>
148std::string join(const std::string& first, const Parts&... rest) {
149 std::filesystem::path p(first);
150 ((p /= std::filesystem::path(std::string(rest))), ...);
151 return p.string();
152}
