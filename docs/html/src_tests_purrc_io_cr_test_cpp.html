1// Compile-run unit tests for the `io` module: one test per purr-callable function.
2// Each writes a tiny .purr that exercises a single io entry point, compiles it with
3// purrc, runs it under the cheatah runtime, and asserts the exact stdout. Complements
4// the in-process unit tests (stdlib/tests/io_test.cpp) and the per-module
5// system-level test (StdlibE2E.Io).
6//
7// File I/O tests write to a unique path under /tmp from inside the .purr program, then
8// read it back and io.print the content, so each test is self-contained and
9// deterministic. io.input is intentionally NOT covered here: it reads from stdin, which
10// the e2e harness does not feed (it is covered by the in-process test
11// CheatahIo.InputReadsALine instead).
12#include "e2e_harness.hpp"
13
14TEST(IoCompileRun, Str) {
15 e2e::expect_e2e("io_str", R"PURR(import io
16io.print(io.str(42))
17)PURR", "42\n");
18}

32// io.print renders a struct PRETTY (nice + readable by default): on multiple indented lines.
33TEST(IoCompileRun, PrintPrettyStruct) {
34 e2e::expect_e2e("io_print_pretty", R"PURR(import io
35struct Point {
36 x: float
37 y: float
38}
39let p = Point({.x = 1, .y = 2})
40io.print(p)
41)PURR",
42 "Point(\n x = 1,\n y = 2\n)\n");
43}

45// io.rprint prints a struct RAW — its compact `Name(field=value, …)` form, exactly as stored.
46TEST(IoCompileRun, Rprint) {
47 e2e::expect_e2e("io_rprint", R"PURR(import io
48struct Point {
49 x: float
50 y: float
51}
52let p = Point({.x = 1, .y = 2})
53io.rprint(p)
54)PURR",

58// Nested structs pretty-print recursively, each level indented under its parent.
59TEST(IoCompileRun, PrintPrettyNestedStruct) {
60 e2e::expect_e2e("io_print_pretty_nested", R"PURR(import io
61struct Inner { a: float }
62struct Outer {
63 inner: Inner
64 b: float
65}
66let o = Outer({.inner = Inner({.a = 5}), .b = 7})
67io.print(o)
68)PURR",
69 "Outer(\n inner = Inner(\n a = 5\n ),\n b = 7\n)\n");
70}
