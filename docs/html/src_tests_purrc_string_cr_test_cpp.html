1// Copyright (c) 2026 BigBrain LLC. MIT-licensed (see LICENSE).
2// Original work; see ACKNOWLEDGMENTS.md for the open-source ideas we build upon.
3// Compile-run unit tests for the `string` module: one test per function. Each
4// writes a tiny .purr that calls a single string function, compiles it with
5// purrc, runs it under the cheatah runtime, and asserts the exact stdout.
6// Complements the in-process unit tests (stdlib/tests/string_test.cpp) and the
7// per-module system-level test (StdlibE2E.String).
8//
9// List-returning functions (split/splitlines) are joined back into a single
10// deterministic string before printing. Booleans print as True/False.
11#include "e2e_harness.hpp"
12
13TEST(StringCompileRun, Upper) {
14 e2e::expect_e2e("string_upper", R"PURR(import io
15import string
16io.print(string.upper("meow"))
17)PURR", "MEOW\n");
18}
