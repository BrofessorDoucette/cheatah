1// Compile-run unit tests for the `string` module: one test per function. Each
2// writes a tiny .purr that calls a single string function, compiles it with
3// purrc, runs it under the cheatah runtime, and asserts the exact stdout.
4// Complements the in-process unit tests (stdlib/tests/string_test.cpp) and the
5// per-module system-level test (StdlibE2E.String).
6//
7// List-returning functions (split/splitlines) are joined back into a single
8// deterministic string before printing. Booleans print as True/False.
9#include "e2e_harness.hpp"
10

16}
