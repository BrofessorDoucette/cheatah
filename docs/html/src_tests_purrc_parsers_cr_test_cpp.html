1// Copyright (c) 2026 BigBrain LLC. MIT-licensed (see LICENSE).
2// Original work; see ACKNOWLEDGMENTS.md for the open-source ideas we build upon.
3// Compile-run system tests for the `parsers` module (C++-authored: parsers.url + parsers.json)
4// and the LANGUAGE features it exercises:
5// * dotted TYPE imports — `import parsers.url.Parser as Parser` aliases a class, and the
6// alias constructs (`Parser()`) and method-calls (`p.parse(...)`) like a local struct;
7// * compiler-synthesized JSON schemas — `import parsers` makes purrc emit a
8// cheatah::parsers::json::schema<> specialization for every struct the program defines,
9// so `parsers.json.read(text, value)` parses JSON STRAIGHT into user structs with no
10// user-written schema (the typed reader).
11// Each test writes a .purr, compiles it with purrc, runs it, and asserts exact stdout.
12#include "e2e_harness.hpp"
13
14// Dotted type imports: both parsers.url types, construction, method call, field access.
15TEST(ParsersCompileRun, UrlParserImport) {
16 e2e::expect_e2e("parsers_url_import", R"PURR(import parsers.url.Parser as Parser
17import parsers.url.Url as Url
18import io
19
20let p = Parser()
21let u = Url()
22if p.parse("http://example.com:8080/data?x=1", u) {
23 io.print(u.host)
24 io.print(u.port)
25 io.print(u.target)
26}
27)PURR", "example.com\n8080\n/data?x=1\n");
28}
29
30// The URL parser rejects malformed input through the same .purr surface.
31TEST(ParsersCompileRun, UrlParserRejects) {
32 e2e::expect_e2e("parsers_url_rejects", R"PURR(import parsers.url.Parser as Parser
33import parsers.url.Url as Url
34import io
35
36let p = Parser()
37let u = Url()
38if p.parse("ftp://example.com/x", u) {
39 io.print("accepted")
40} else {
41 io.print("rejected")
42}
43)PURR", "rejected\n");
44}

57// THE TYPED READER: parsers.json.read parses JSON straight into a .purr struct — the schema
58// is synthesized by purrc from the struct's typed fields (str/float/bool here).
59TEST(ParsersCompileRun, TypedReader) {
60 e2e::expect_e2e("parsers_typed_reader", R"PURR(import parsers
61import io
62
63struct Quote {
64 symbol: str
65 price: float
66 live: bool
67}
68
69let q = Quote("", 0.0, false)
70if parsers.json.read("{\"symbol\":\"SPX\",\"price\":7386.65,\"live\":true}", q) {
71 io.print(q.symbol)
72 io.print(q.price)
73 io.print(q.live)
74}
75)PURR", "SPX\n7386.65\nTrue\n");
76}
77
78// Schema synthesis composes: nested structs and list<T> fields (vector reads).
79TEST(ParsersCompileRun, TypedReaderNested) {
80 e2e::expect_e2e("parsers_typed_nested", R"PURR(import parsers
81import io
82
83struct Row {
84 v: float
85}
86struct Series {
87 name: str
88 rows: list<Row>
89}
90
91let rows: list<Row> = []
92let s = Series("", rows)
93if parsers.json.read("{\"name\":\"av\",\"rows\":[{\"v\":1.5},{\"v\":2.5}]}", s) {
94 io.print(s.name)
95 io.print(len(s.rows))
96 io.print(s.rows[1].v)
97}
98)PURR", "av\n2\n2.5\n");
99}

101// The validating reader REJECTS malformed input (and unknown keys are skipped, not errors).
102TEST(ParsersCompileRun, TypedReaderRejectsMalformed) {
103 e2e::expect_e2e("parsers_typed_rejects", R"PURR(import parsers
104import io
105
106struct Quote {
107 price: float
108}
109
110let q = Quote(0.0)
111if parsers.json.read("{\"price\":}", q) {
112 io.print("accepted")
113} else {
114 io.print("rejected")
115}
116if parsers.json.read("{\"unknown\":[1,2],\"price\":3.5}", q) {
117 io.print(q.price)
118}
119)PURR", "rejected\n3.5\n");
120}
