tests/purrc/linalg_vs_python_test.cpp

1// System-level tests for the (newly optimized) linalg library. Each runs a checked-in
2// .purr program through the real purrc → cheatah pipeline and cross-checks its output,
3// number for number, against an equivalent NumPy program. The .purr/.py sources live in
4// tests/purrc/linalg_programs/ so they can be opened and edited in the editor — open a
5// .purr there to see exactly what each test runs.
6//
7// Same fixed, well-conditioned symmetric 4×4 matrix feeds every op. Skipped (not
8// failed) when python3 + numpy is unavailable, so the suite stays green without numpy.
9#include <string>
10
11#include "e2e_harness.hpp"
12
13#ifndef LINALG_PROGRAMS_DIR
14#define LINALG_PROGRAMS_DIR "."
15#endif
16
17namespace {
18std::string prog(const char* base, const char* ext) {
19 return std::string(LINALG_PROGRAMS_DIR) + "/" + base + ext;
20}
21// Cross-check linalg_programs/<base>.purr against linalg_programs/<base>.py.
22void check(const char* base, bool sorted = false) {
23 e2e::expect_purr_matches_python(base, prog(base, ".purr"), prog(base, ".py"),
24 /*tol=*/1e-4, sorted);
25}
26} // namespace
27
28TEST(LinalgVsPython, Solve) { check("solve"); }
29TEST(LinalgVsPython, Det) { check("det"); }
30TEST(LinalgVsPython, Inv) { check("inv"); }
31TEST(LinalgVsPython, Matmul) { check("matmul"); }
32// Spectra: numpy and cheatah may order the values differently, so compare unordered.
33TEST(LinalgVsPython, Eigvalsh) { check("eigvalsh", /*sorted=*/true); }
34TEST(LinalgVsPython, Svdvals) { check("svdvals", /*sorted=*/true); }
