tests/purrc/linalg_vs_python_test.cpp

1// Copyright (c) 2026 BigBrain LLC. MIT-licensed (see LICENSE).
2// Original work; see ACKNOWLEDGMENTS.md for the open-source ideas we build upon.
3// System-level tests for the (newly optimized) linalg library. Each runs a checked-in
4// .purr program through the real purrc → cheatah pipeline and cross-checks its output,
5// number for number, against an equivalent NumPy program. The .purr/.py sources live in
6// tests/purrc/linalg_programs/ so they can be opened and edited in the editor — open a
7// .purr there to see exactly what each test runs.
8//
9// Same fixed, well-conditioned symmetric 4×4 matrix feeds every op. Skipped (not
10// failed) when python3 + numpy is unavailable, so the suite stays green without numpy.
11#include <string>
12
13#include "e2e_harness.hpp"
14
15#ifndef LINALG_PROGRAMS_DIR
16#define LINALG_PROGRAMS_DIR "."
17#endif
18
19namespace {
20std::string prog(const char* base, const char* ext) {
21 return std::string(LINALG_PROGRAMS_DIR) + "/" + base + ext;
22}
23// Cross-check linalg_programs/<base>.purr against linalg_programs/<base>.py.
24void check(const char* base, bool sorted = false) {
25 e2e::expect_purr_matches_python(base, prog(base, ".purr"), prog(base, ".py"),
26 /*tol=*/1e-4, sorted);
27}
28} // namespace
29
30TEST(LinalgVsPython, Solve) { check("solve"); }
31TEST(LinalgVsPython, Det) { check("det"); }
32TEST(LinalgVsPython, Inv) { check("inv"); }
33TEST(LinalgVsPython, Matmul) { check("matmul"); }
34// Spectra: numpy and cheatah may order the values differently, so compare unordered.
35TEST(LinalgVsPython, Eigvalsh) { check("eigvalsh", /*sorted=*/true); }
36TEST(LinalgVsPython, Svdvals) { check("svdvals", /*sorted=*/true); }
