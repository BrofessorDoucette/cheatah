2// Original work; see ACKNOWLEDGMENTS.md for the open-source ideas we build upon.
3// Micro-benchmarks for cheatah::ndarray — the generic numeric core
4// (`basic_ndarray<T>`). These exercise the element-wise ops (vectorized via the
5// `std::execution::unseq` policy on the contiguous fast path) and the full
6// reduction, across element TYPES (double and integer — the type is part of the
7// array) and SIZES, plus the strided/broadcast fallback for comparison.
8//
9// Build with the `release` preset (`-O3 -march=native`) for meaningful numbers:
10// ./build/release/bin/cheatah_benchmarks --benchmark_filter=ndarray|Elementwise|Sum
11#include "ndarray.hpp"
12
13#include <cstddef>
14#include <cstdint>
15#include <vector>
16
17#include <benchmark/benchmark.h>
18
19namespace nd = cheatah::ndarray;
20
21namespace {
22
23// A contiguous 1-D array of `n` ascending values of element type T (so the array's
24// element type — and thus the kernel that gets instantiated/vectorized — is T).
25template <typename T>
26nd::basic_ndarray<T> iota_array(std::size_t n) {
27 std::vector<T> v(n);
28 for (std::size_t i = 0; i < n; ++i) v[i] = static_cast<T>(i % 1024);
29 return nd::array(v);
30}
31
32// Element-wise add of two same-shape contiguous arrays — the `std::transform(unseq)`
33// SIMD fast path. Templated so the SAME kernel is measured for double and integer.
34template <typename T>
35void BM_ElementwiseAdd(benchmark::State& state) {
36 const std::size_t n = static_cast<std::size_t>(state.range(0));
37 const auto a = iota_array<T>(n);
38 const auto b = iota_array<T>(n);
39 for (auto _ : state) {
40 benchmark::DoNotOptimize(nd::add(a, b));
41 }
42 state.SetItemsProcessed(state.iterations() * static_cast<std::int64_t>(n));
43 state.SetBytesProcessed(state.iterations() * static_cast<std::int64_t>(n) * 2 *
44 static_cast<std::int64_t>(sizeof(T)));
45}
46BENCHMARK_TEMPLATE(BM_ElementwiseAdd, double)->Arg(1024)->Arg(65536)->Arg(1 << 20);
47BENCHMARK_TEMPLATE(BM_ElementwiseAdd, long long)->Arg(1024)->Arg(65536)->Arg(1 << 20);
48
49// Full reduction (sum across all elements) — the `std::reduce(unseq)` path.
50template <typename T>
51void BM_Sum(benchmark::State& state) {
52 const std::size_t n = static_cast<std::size_t>(state.range(0));
53 const auto a = iota_array<T>(n);
54 for (auto _ : state) {
55 benchmark::DoNotOptimize(nd::sum(a));
56 }
57 state.SetItemsProcessed(state.iterations() * static_cast<std::int64_t>(n));
58 state.SetBytesProcessed(state.iterations() * static_cast<std::int64_t>(n) *
59 static_cast<std::int64_t>(sizeof(T)));
60}
61BENCHMARK_TEMPLATE(BM_Sum, double)->Arg(1024)->Arg(65536)->Arg(1 << 20);
62BENCHMARK_TEMPLATE(BM_Sum, long long)->Arg(1024)->Arg(65536)->Arg(1 << 20);
63
64// Broadcast add (a 0-d scalar stretched over a vector) — the strided, stride-0
65// odometer FALLBACK. Compare against BM_ElementwiseAdd<double> at the same size to
66// see what the contiguous SIMD fast path buys.
67void BM_BroadcastAddFallback(benchmark::State& state) {
68 const std::size_t n = static_cast<std::size_t>(state.range(0));
69 const auto a = iota_array<double>(n);
70 const nd::NDArray s = nd::scalar(2.0);
71 for (auto _ : state) {
72 benchmark::DoNotOptimize(nd::add(a, s));
73 }
74 state.SetItemsProcessed(state.iterations() * static_cast<std::int64_t>(n));
75}
76BENCHMARK(BM_BroadcastAddFallback)->Arg(1024)->Arg(65536);
