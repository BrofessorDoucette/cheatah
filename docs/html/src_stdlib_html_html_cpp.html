17void append_utf8(std::string& out, std::uint32_t cp) {
18 if (cp <= 0x7F) {
19 out.push_back(static_cast<char>(cp));
20 } else if (cp <= 0x7FF) {
21 out.push_back(static_cast<char>(0xC0 | (cp >> 6)));
22 out.push_back(static_cast<char>(0x80 | (cp & 0x3F)));
23 } else if (cp <= 0xFFFF) {
24 out.push_back(static_cast<char>(0xE0 | (cp >> 12)));
25 out.push_back(static_cast<char>(0x80 | ((cp >> 6) & 0x3F)));
26 out.push_back(static_cast<char>(0x80 | (cp & 0x3F)));
27 } else {
28 out.push_back(static_cast<char>(0xF0 | (cp >> 18)));
29 out.push_back(static_cast<char>(0x80 | ((cp >> 12) & 0x3F)));
30 out.push_back(static_cast<char>(0x80 | ((cp >> 6) & 0x3F)));
31 out.push_back(static_cast<char>(0x80 | (cp & 0x3F)));
32 }
33}

55// Decode the reference whose body (between '&' and the optional ';') is `body`.
56// Returns true and appends to `out` on success; false leaves it to the caller.
57bool decode_reference(std::string_view body, std::string& out) {
58 if (body.empty()) return false;
59 if (body.front() == '#') { // numeric: &#169; or &#xA9;
60 std::uint32_t cp = 0;
61 bool hex = body.size() > 1 && (body[1] == 'x' || body[1] == 'X');
62 std::size_t i = hex ? 2 : 1;
63 if (i >= body.size()) return false;
64 for (; i < body.size(); ++i) {
65 const char c = body[i];
66 std::uint32_t digit;
67 if (c >= '0' && c <= '9') {
68 digit = static_cast<std::uint32_t>(c - '0');
69 } else if (hex && c >= 'a' && c <= 'f') {
70 digit = static_cast<std::uint32_t>(c - 'a' + 10);
71 } else if (hex && c >= 'A' && c <= 'F') {
72 digit = static_cast<std::uint32_t>(c - 'A' + 10);
73 } else {
74 return false;
75 }
76 cp = cp * (hex ? 16 : 10) + digit;
77 if (cp > 0x10FFFF) return false; // beyond Unicode range
78 }
79 if (cp == 0) return false;
80 append_utf8(out, cp);
81 return true;
82 }
83 const auto& table = named_entities();
84 const auto it = table.find(body);
85 if (it == table.end()) return false;
86 append_utf8(out, it->second);
87 return true;
88}

92std::string escape(std::string_view s, bool quote) {
93 std::string out;
94 out.reserve(s.size());
95 for (const char c : s) {
96 switch (c) {
97 case '&': out += "&amp;"; break;
98 case '<': out += "&lt;"; break;
99 case '>': out += "&gt;"; break;
100 case '"': out += quote ? "&quot;" : "\""; break;
101 case '\'': out += quote ? "&#x27;" : "'"; break;
102 default: out.push_back(c);
103 }
104 }
105 return out;
106}

108std::string unescape(std::string_view s) {
109 std::string out;
110 out.reserve(s.size());
111 for (std::size_t i = 0; i < s.size();) {
112 if (s[i] != '&') {
113 out.push_back(s[i++]);
114 continue;
115 }
116 // Find the terminating ';' within a sane window (entity names are short).
117 const std::size_t semi = s.find(';', i + 1);
118 const std::size_t limit = i + 1 + 32;
119 if (semi != std::string_view::npos && semi <= limit && semi > i + 1 &&
120 decode_reference(s.substr(i + 1, semi - i - 1), out)) {
121 i = semi + 1;
122 } else {
123 out.push_back(s[i++]); // a bare '&' or unknown reference: keep verbatim
124 }
125 }
126 return out;
127}

133bool is_name_start(char c) {
134 return (c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z') || c == '_' || c == ':';
135}
136bool is_name_char(char c) {
137 return is_name_start(c) || (c >= '0' && c <= '9') || c == '-' || c == '.';
138}
139bool is_space(char c) { return c == ' ' || c == '\t' || c == '\n' || c == '\r' || c == '\f'; }

141std::string lower_str(std::string_view s) {
142 std::string out;
143 out.reserve(s.size());
144 for (const char c : s) out.push_back(lower(c));
145 return out;
146}

148// Case-insensitive check that s[at..] begins with `lit` (lit is lowercase).
149bool matches_ci(std::string_view s, std::size_t at, std::string_view lit) {
150 if (at + lit.size() > s.size()) return false;
151 for (std::size_t k = 0; k < lit.size(); ++k) {
152 if (lower(s[at + k]) != lit[k]) return false;
153 }
154 return true;
155}

159std::vector<Token> parse(std::string_view html) {
160 std::vector<Token> tokens;
161 const std::size_t n = html.size();
162 std::size_t i = 0;
163 std::string text; // accumulates a run of character data
164
165 auto flush_text = [&] {
166 if (text.empty()) return;
167 tokens.push_back({"data", "", unescape(text), {}});
168 text.clear();
169 };
170
171 while (i < n) {
172 if (html[i] != '<') {
173 text.push_back(html[i++]);
174 continue;
175 }
176 // Something starting with '<'. Decide what it is.
177 if (matches_ci(html, i, "<!--")) { // comment
178 flush_text();
179 const std::size_t start = i + 4;
180 std::size_t end = html.find("-->", start);
181 const std::size_t stop = (end == std::string_view::npos) ? n : end;
182 tokens.push_back({"comment", "", std::string(html.substr(start, stop - start)), {}});
183 i = (end == std::string_view::npos) ? n : end + 3;
184 continue;
185 }
186 if (i + 1 < n && html[i + 1] == '!') { // declaration <!DOCTYPE ...>
187 flush_text();
188 const std::size_t start = i + 2;
189 std::size_t end = html.find('>', start);
190 const std::size_t stop = (end == std::string_view::npos) ? n : end;
191 tokens.push_back({"decl", "", std::string(html.substr(start, stop - start)), {}});
192 i = (end == std::string_view::npos) ? n : end + 1;
193 continue;
194 }
195 if (i + 1 < n && html[i + 1] == '?') { // processing instruction <? ... >
196 flush_text();
197 const std::size_t start = i + 2;
198 std::size_t end = html.find('>', start);
199 const std::size_t stop = (end == std::string_view::npos) ? n : end;
200 tokens.push_back({"pi", "", std::string(html.substr(start, stop - start)), {}});
201 i = (end == std::string_view::npos) ? n : end + 1;
202 continue;
203 }
204 if (i + 1 < n && html[i + 1] == '/') { // end tag </name>
205 const std::size_t name_start = i + 2;
206 std::size_t j = name_start;
207 while (j < n && is_name_char(html[j])) ++j;
208 if (j == name_start && !(j < n && is_name_start(html[j]))) {
209 // not a real name (e.g. "</ "): only a tag if name present
210 }
211 if (j > name_start) {
212 flush_text();
213 std::size_t end = html.find('>', j);
214 tokens.push_back({"endtag", lower_str(html.substr(name_start, j - name_start)), "", {}});
215 i = (end == std::string_view::npos) ? n : end + 1;
216 continue;
217 }
218 // malformed: treat '<' as data
219 text.push_back(html[i++]);
220 continue;
221 }
222 if (i + 1 < n && is_name_start(html[i + 1])) { // start tag <name ...>
223 flush_text();
224 std::size_t j = i + 1;
225 while (j < n && is_name_char(html[j])) ++j;
226 const std::string tag = lower_str(html.substr(i + 1, j - (i + 1)));
227
228 std::vector<Attr> attrs;
229 // Parse attributes until '>' or '/>' or end-of-input.
230 while (j < n && html[j] != '>') {
231 while (j < n && is_space(html[j])) ++j;
232 if (j < n && html[j] == '/') { ++j; continue; } // self-close marker
233 if (j >= n || html[j] == '>') break;
234 // attribute name
235 const std::size_t an = j;
236 while (j < n && !is_space(html[j]) && html[j] != '=' && html[j] != '>' &&
237 html[j] != '/') {
238 ++j;
239 }
240 if (j == an) { ++j; continue; } // stray char, skip
241 std::string name = lower_str(html.substr(an, j - an));
242 std::string value;
243 while (j < n && is_space(html[j])) ++j;
244 if (j < n && html[j] == '=') {
245 ++j;
246 while (j < n && is_space(html[j])) ++j;
247 if (j < n && (html[j] == '"' || html[j] == '\'')) {
248 const char q = html[j++];
249 const std::size_t vs = j;
250 while (j < n && html[j] != q) ++j;
251 value = unescape(html.substr(vs, j - vs));
252 if (j < n) ++j; // closing quote
253 } else {
254 const std::size_t vs = j;
255 while (j < n && !is_space(html[j]) && html[j] != '>') ++j;
256 value = unescape(html.substr(vs, j - vs));
257 }
258 }
259 attrs.push_back({std::move(name), std::move(value)});
260 }
261 // Detect self-closing: last non-space before '>' was '/'.
262 bool self_close = false;
263 if (j < n && html[j] == '>') {
264 std::size_t k = j;
265 while (k > i && is_space(html[k - 1])) --k;
266 if (k > i && html[k - 1] == '/') self_close = true;
267 }
268 const std::size_t after = (j < n) ? j + 1 : n;
269
270 tokens.push_back({self_close ? "startendtag" : "starttag", tag, "", std::move(attrs)});
271
272 // Raw-text elements: emit their body verbatim, then the end tag.
273 if (!self_close && (tag == "script" || tag == "style")) {
274 const std::string close = "</" + tag;
275 std::size_t k = after;
276 while (k < n) {
277 if (html[k] == '<' && matches_ci(html, k, close)) break;
278 ++k;
279 }
280 if (k > after) {
281 tokens.push_back({"data", "", std::string(html.substr(after, k - after)), {}});
282 }
283 if (k < n) { // consume the matching close tag
284 std::size_t end = html.find('>', k);
285 tokens.push_back({"endtag", tag, "", {}});
286 i = (end == std::string_view::npos) ? n : end + 1;
287 } else {
288 i = n;
289 }
290 continue;
291 }
292 i = after;
293 continue;
294 }
295 // A '<' that starts nothing recognizable -> literal data.
296 text.push_back(html[i++]);
297 }
298 flush_text();
299 return tokens;
300}

302std::string get_attr(const Token& t, std::string_view name) {
303 const std::string key = lower_str(name);
304 for (const Attr& a : t.attrs) {
305 if (a.name == key) return a.value;
306 }
307 return "";
308}

310bool has_attr(const Token& t, std::string_view name) {
311 const std::string key = lower_str(name);
312 for (const Attr& a : t.attrs) {
313 if (a.name == key) return true;
314 }
315 return false;
