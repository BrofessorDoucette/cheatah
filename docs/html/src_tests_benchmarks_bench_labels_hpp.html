tests/benchmarks/bench_labels.hpp

1// Copyright (c) 2026 BigBrain LLC. MIT-licensed (see LICENSE).
2// Original work; see ACKNOWLEDGMENTS.md for the open-source ideas we build upon.
3//
4// bench_labels — human-readable display names for benchmark keys.
5//
6// Some published tables want a name a reader recognises ("AES-128-GCM (AES-NI + PCLMULQDQ)")
7// rather than the benchmark identifier ("BM_CryptoAes128Gcm"). The mapping has to live
8// SOMEWHERE, and where it lives decides whether it drifts.
9//
10// It lives next to the BENCHMARK() registrations, via CHEATAH_BENCH_LABEL. That placement is
11// the whole point:
12//
13// * deleting a benchmark deletes its label in the same edit, so a label can never outlive
14// the thing it names;
15// * adding a benchmark without a label makes the raw key appear in the published table —
16// visibly wrong, rather than a silently missing row.
17//
18// The alternative — a table inside bench_main.cpp keyed by strings from a file it does not
19// include — is a second list that nothing keeps in step with the first. That is exactly the
20// class of drift this whole benchmarking effort exists to remove, so it is not used here.
21//
22// Keys are PAIR keys: the side suffix (_Cheatah / _OpenSSL) is already stripped by
23// bench_pairs.hpp's classify_benchmark, so one label covers both sides of a comparison.
24#ifndef CHEATAH_TESTS_BENCHMARKS_BENCH_LABELS_HPP
25#define CHEATAH_TESTS_BENCHMARKS_BENCH_LABELS_HPP
26
27#include <map>
28#include <string>
29
30namespace cheatah::bench {
31
32// Function-local static: the registrars below run during static initialisation, and this
33// guarantees the map exists by the time the first one calls it, whatever link order gives us.
34inline std::map<std::string, std::string>& label_registry() {
35 static std::map<std::string, std::string> m;
36 return m;
37}
38
39struct LabelRegistrar {
40 LabelRegistrar(const char* key, const char* display) { label_registry()[key] = display; }
41};
42
43// The display name, or the key itself when none was registered — never an empty cell.
44inline std::string label_for(const std::string& key) {
45 const auto it = label_registry().find(key);
46 return it == label_registry().end() ? key : it->second;
47}
48
49} // namespace cheatah::bench
50
51// Two levels so __LINE__ expands before pasting.
52#define CHEATAH_BENCH_LABEL_CAT_(a, b) a##b
53#define CHEATAH_BENCH_LABEL_CAT(a, b) CHEATAH_BENCH_LABEL_CAT_(a, b)
54
55#define CHEATAH_BENCH_LABEL(key, display) \
56 static const ::cheatah::bench::LabelRegistrar CHEATAH_BENCH_LABEL_CAT( \
57 cheatah_bench_label_, __LINE__){key, display}
58
59#endif // CHEATAH_TESTS_BENCHMARKS_BENCH_LABELS_HPP
