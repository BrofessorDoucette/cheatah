3/**
4 * @file ndarray.hpp
5 * @brief cheatah `ndarray` — our own numpy-flavored N-dimensional array (of
6 * doubles) with NumPy broadcasting, surfaced as a `NDArray` class plus free
7 * functions (a .purr program writes `ndarray.zeros([2, 3])`).
8 * See https://numpy.org/doc/stable/user/basics.broadcasting.html.
9 *
10 * `import ndarray` includes this header and links `libcheatah_ndarray`. Unit tests:
11 * `stdlib/tests/ndarray_test.cpp`; the suite runs under AddressSanitizer (the `asan`
12 * preset) and Valgrind (`security/run-valgrind.sh`) on every QA-gate run.
13 *
14 * @note Design (the "pointers + a bit of thinking"): the elements live in a shared
15 * buffer (`std::shared_ptr<std::vector<double>>`) and an array is a VIEW into
16 * it — {shape, strides, offset}. That makes reshape and **broadcast**
17 * zero-copy: to stretch a dimension of size 1 we give it a stride of 0, so
18 * every index along it reads the same element. Shared ownership = memory-safe,
19 * no manual frees. `size` below is the element count (product of dims).
20 */

53/// Numeric<T>: an arithmetic element type an ndarray can store (int or float
54/// family). Storage, construction, and elementwise +-* require only this.
55template <typename T>
56concept Numeric = std::is_arithmetic_v<T>;
57/// FloatingPoint<T>: a real floating type. The linalg decompositions (solve, inv,
58/// det, svd, eig) need division/√, so they constrain to this — calling them on an
59/// integer array fails with a clear "FloatingPoint not satisfied", not template spam.
60template <typename T>
61concept FloatingPoint = std::floating_point<T>;

63/// \cond INTERNAL
64template <typename T>
65struct is_complex : std::false_type {};
66template <typename U>
67struct is_complex<std::complex<U>> : std::bool_constant<std::is_floating_point_v<U>> {};
68/// \endcond

70/// Whether `T` is a `std::complex` of a floating type — the trait behind @ref Field.
71template <typename T>
72inline constexpr bool is_complex_v = is_complex<T>::value;

74/// Field<T>: a scalar an ndarray can store — a real arithmetic type OR a
75/// `std::complex` of a floating type. This is what makes **complex** matrices and
76/// vectors first-class (Hermitian operators, complex wavefunctions), and lets a
77/// REAL matrix yield the COMPLEX eigenvalues it mathematically has.
78template <typename T>
79concept Field = std::is_arithmetic_v<T> || is_complex_v<T>;

103namespace detail {
104/// C-order (row-major) strides for a shape.
105inline std::vector<std::ptrdiff_t> contiguous_strides(const std::vector<std::size_t>& shape) {
106 std::vector<std::ptrdiff_t> s(shape.size());
107 std::ptrdiff_t step = 1;
108 for (std::size_t i = shape.size(); i-- > 0;) {
109 s[i] = step;
110 step *= static_cast<std::ptrdiff_t>(shape[i]);
111 }
112 return s;
113}
114/// Overflow-checked product of the dimensions (a wrapped size_t would under-allocate,
115/// turning later element access into out-of-bounds writes — reject it up front).
116inline std::size_t product(const std::vector<std::size_t>& shape) {
117 std::size_t t = 1;
118 for (std::size_t d : shape) {
119 if (d != 0 && t > std::numeric_limits<std::size_t>::max() / d) {
120 throw std::runtime_error("ndarray: shape too large (size overflow)");
121 }
122 t *= d;
123 }
124 return t;
125}
126/// Convert signed dims/indices to sizes, rejecting negatives (a negative cast to
127/// size_t becomes huge -> under-allocation / OOB). Validate at the boundary.
128inline std::vector<std::size_t> to_size(const std::vector<long long>& v) {
129 std::vector<std::size_t> out(v.size());
130 for (std::size_t i = 0; i < v.size(); ++i) {
131 if (v[i] < 0) throw std::runtime_error("ndarray: negative dimension or index");
132 out[i] = static_cast<std::size_t>(v[i]);
133 }
134 return out;
135}
136/// Advance a C-order multi-index odometer; false when it wraps past the end.
137inline bool next_index(std::vector<std::size_t>& idx, const std::vector<std::size_t>& shape) {
138 for (std::size_t i = shape.size(); i-- > 0;) {
139 if (++idx[i] < shape[i]) return true;
140 idx[i] = 0;
141 }
142 return false;
143}
144} // namespace detail

146/**
147 * @brief An N-dimensional array of `T` (a @ref Field element type — real or complex):
148 * a view ({shape, strides, offset}) over a shared element buffer.
149 *
150 * Copies are cheap and share the buffer; reshape/broadcast produce new views without
151 * copying elements. Index math goes through @ref at, which bounds-checks. The element
152 * type is deduced from the data (e.g. `array([1,2,3])` is integer, `array([1.0,…])`
153 * is double); `NDArray` is the default `basic_ndarray<double>`. Complex element types
154 * (`std::complex<double>`) make complex matrices/vectors — and the complex eigenvalues
155 * a real matrix can have — first-class.
156 */
157template <Field T>
158class basic_ndarray {
159public:
160 using value_type = T; ///< The stored element type (the @ref Field `T`).
161 /**
162 * Construct an empty 0-d array with a fresh empty buffer.
163 *
164 * Leaves shape and strides empty and the buffer holding no elements; note this
165 * is distinct from a 0-d scalar (see @ref scalar), whose buffer holds one element.
166 * @complexity O(1).
167 * @alloc allocates the empty shared buffer.
168 * @test CheatahNDArray.ToStringScalar
169 * @systest StdlibE2E.Ndarray
170 */
171 basic_ndarray() : data_(std::make_shared<std::vector<T>>()) {}
172 /**
173 * Construct a contiguous array of @p shape filled with @p fill.
174 *
175 * Allocates a fresh buffer of `product(shape)` doubles all set to @p fill and
176 * computes C-order (row-major) strides; the dimension product is overflow-checked.
177 * @param shape the dimensions.
178 * @param fill value for every element.
179 * @complexity O(size).
180 * @alloc allocates a new NDArray buffer (shared_ptr<vector<double>>) of
181 * `product(shape)` elements; throws if the shape overflows size_t.
182 * @test CheatahNDArray.ShapeFactoriesAndReductions
183 * @systest StdlibE2E.Ndarray
184 */
185 explicit basic_ndarray(std::vector<std::size_t> shape, T fill = T{}) // contiguous
186 : data_(std::make_shared<std::vector<T>>()), shape_(std::move(shape)) {
187 data_->assign(detail::product(shape_), fill);
188 strides_ = detail::contiguous_strides(shape_);
189 }
190 /**
191 * Construct a view from explicit buffer/shape/strides/offset (used by views).
192 *
193 * Stores the supplied members verbatim with no validation or copy, so the new
194 * array shares ownership of @p data; callers (e.g. @ref broadcast_to) are
195 * responsible for passing strides/offset that stay within the buffer.
196 * @param data the shared element buffer.
197 * @param shape the dimensions.
198 * @param strides element strides per dimension.
199 * @param offset starting flat offset into @p data.
200 * @complexity O(1).
201 * @alloc shares @p data, no element copy.
202 * @test CheatahNDArray.BroadcastTo
203 * @systest StdlibE2E.Ndarray
204 */
205 basic_ndarray(std::shared_ptr<std::vector<T>> data, std::vector<std::size_t> shape,
206 std::vector<std::ptrdiff_t> strides, std::size_t offset)
207 : data_(std::move(data)), shape_(std::move(shape)), strides_(std::move(strides)),
208 offset_(offset) {}
209
210 /**
211 * The shape (dimensions).
212 * @return reference to the shape vector.
213 * @complexity O(1).
214 * @alloc none.
215 * @test CheatahNDArray.ShapeFactoriesAndReductions
216 * @systest StdlibE2E.Ndarray
217 */
218 const std::vector<std::size_t>& shape() const { return shape_; }
219 /**
220 * The element strides.
221 * @return reference to the strides vector.
222 * @complexity O(1).
223 * @alloc none.
224 * @test CheatahNDArray.BroadcastTo
225 * @systest StdlibE2E.Ndarray
226 */
227 const std::vector<std::ptrdiff_t>& strides() const { return strides_; }
228 /**
229 * The number of dimensions (rank).
230 * @return `shape().size()`.
231 * @complexity O(1).
232 * @alloc none.
233 * @test CheatahNDArray.BroadcastingAdd
234 * @systest StdlibE2E.Ndarray
235 */
236 std::size_t ndim() const { return shape_.size(); }
237 /**
238 * The element count (product of dims; 1 for a 0-d scalar).
239 *
240 * Recomputes the overflow-checked product of the shape on each call rather than
241 * caching it; an empty (no-dimension) shape yields 1.
242 * @return the number of elements.
243 * @complexity O(ndim).
244 * @alloc none; throws on size overflow.
245 * @test CheatahNDArray.ShapeFactoriesAndReductions
246 * @systest StdlibE2E.Ndarray
247 */
248 std::size_t size() const { return detail::product(shape_); } // 1 for a 0-d scalar
249
250 /**
251 * Read one element by multi-index (via strides).
252 *
253 * Computes the flat buffer position as `offset + sum(index[i] * strides[i])`, so
254 * it correctly resolves views (including broadcast dims with stride 0); throws if
255 * @p index has the wrong rank or any coordinate is out of range for its dimension.
256 * @param index one coordinate per dimension.
257 * @return the element value.
258 * @complexity O(ndim).
259 * @alloc none; bounds-checks rank and range, throwing on a bad index.
260 * @test CheatahNDArray.ShapeFactoriesAndReductions,
261 * CheatahNDArray.RejectsMaliciousShapesAndIndices
262 * @systest StdlibE2E.Ndarray
263 */
264 T at(const std::vector<std::size_t>& index) const { // element via strides
265 // Bounds-check: a wrong-rank or out-of-range index would otherwise compute an
266 // offset outside the backing buffer (out-of-bounds read).
267 if (index.size() != shape_.size()) {
268 throw std::runtime_error("ndarray: index has the wrong number of dimensions");
269 }
270 std::ptrdiff_t off = static_cast<std::ptrdiff_t>(offset_);
271 for (std::size_t i = 0; i < index.size(); ++i) {
272 if (index[i] >= shape_[i]) throw std::runtime_error("ndarray: index out of range");
273 off += static_cast<std::ptrdiff_t>(index[i]) * strides_[i];
274 }
275 return (*data_)[static_cast<std::size_t>(off)];
276 }
277 /**
278 * The shared backing buffer.
279 * @return reference to the element buffer shared_ptr.
280 * @complexity O(1).
281 * @alloc none.
282 * @test CheatahNDArray.BroadcastingAdd
283 * @systest StdlibE2E.Ndarray
284 */
285 const std::shared_ptr<std::vector<T>>& buffer() const { return data_; }
286 /**
287 * The flat offset into the buffer where this view starts.
288 * @return the offset.
289 * @complexity O(1).
290 * @alloc none.
291 * @test CheatahNDArray.BroadcastTo
292 * @systest StdlibE2E.Ndarray
293 */
294 std::size_t offset() const { return offset_; }
295
296 /**
297 * Python-style text rendering, e.g. `"[[1, 2], [3, 4]]"` — the `str()` hook that
298 * makes an NDArray printable via `io.print` (io's `HasStr` protocol). Defers to
299 * the free `to_string`; defined out-of-line below, where `to_string` is declared.
300 * @return the array formatted as nested brackets.
301 * @complexity O(n) in the element count.
302 * @alloc allocates the result string.
303 * @test CheatahNDArray.ToString
304 * @systest StdlibE2E.Ndarray
305 */
306 std::string str() const;
307
308private:
309 std::shared_ptr<std::vector<T>> data_;
310 std::vector<std::size_t> shape_;
311 std::vector<std::ptrdiff_t> strides_; // element strides
312 std::size_t offset_ = 0;
313};

337// ==========================================================================
338// Templated free functions. The element type T is deduced from the data
339// (array([1,2,3]) -> long long, array([1.0,…]) -> double); every op is
340// constrained by Numeric. Elementwise ops vectorize via the std::execution
341// policies (declarative SIMD); broadcasting/strided views fall back to a
342// C-order scalar walk. NDArray (= basic_ndarray<double>) is the default.
343// ==========================================================================

345/**
346 * Whether @p a is a contiguous C-order block (no broadcast/stride-0/permuted view),
347 * so its elements live consecutively from `offset()` and can be walked flatly.
348 * @param a the array (or view) to test.
349 * @return true if @p a's strides are the C-order strides for its shape.
350 * @complexity O(ndim).
351 * @alloc none.
352 * @test CheatahNDArray.BroadcastingAdd
353 * @systest StdlibE2E.Ndarray
354 */
355template <Field T>
356inline bool is_contiguous(const basic_ndarray<T>& a) {
357 return a.strides() == detail::contiguous_strides(a.shape());
358}

360/**
361 * A zero-copy view of @p a stretched to @p target (size-1 / missing dims get stride 0).
362 * @param a source array.
363 * @param target the shape to stretch to.
364 * @return a VIEW sharing @p a's buffer (no element copy).
365 * @test CheatahNDArray.BroadcastTo
366 * @systest StdlibE2E.Ndarray
367 */
368template <Field T>
369basic_ndarray<T> broadcast_to(const basic_ndarray<T>& a, const std::vector<std::size_t>& target) {
370 const std::size_t n = target.size();
371 if (a.ndim() > n) throw std::runtime_error("ndarray: cannot broadcast to fewer dimensions");
372 std::vector<std::ptrdiff_t> ns(n, 0); // stretched / missing dims -> stride 0
373 const std::size_t pad = n - a.ndim();
374 for (std::size_t i = 0; i < a.ndim(); ++i) {
375 const std::size_t adim = a.shape()[i];
376 if (adim == target[pad + i]) {
377 ns[pad + i] = a.strides()[i];
378 } else if (adim != 1) {
379 throw std::runtime_error("ndarray: shape not broadcastable to target");
380 } // adim == 1 -> stride stays 0 (stretch)
381 }
382 return basic_ndarray<T>(a.buffer(), target, ns, a.offset());
383}

474/**
475 * 1-D range `[start, stop)` stepping by @p step; element type deduced from the args.
476 * @param start first value.
477 * @param stop exclusive bound.
478 * @param step increment (throws if zero); a step pointing away from @p stop yields empty.
479 * @return a 1-D `basic_ndarray<T>` of the generated values.
480 * @complexity O(count).
481 * @alloc allocates a new buffer; throws if @p step is zero.
482 * @test CheatahNDArray.Arange
483 * @crtest NdarrayCompileRun.Arange
484 * @systest StdlibE2E.Ndarray
485 */
486template <Numeric T>
487basic_ndarray<T> arange(T start, T stop, T step) {
488 if (step == T{}) throw std::runtime_error("ndarray: arange step must be non-zero");
489 std::vector<T> v;
490 for (T x = start; (step > T{}) ? (x < stop) : (x > stop); x += step) v.push_back(x);
491 return array(v);
492}
493/**
494 * Reshape @p a to @p shape (same element count); reads in C-order so views/broadcasts
495 * are flattened into a fresh contiguous buffer (a copy, not an alias).
496 * @param a source array.
497 * @param shape the new dimensions (signed; throws on a negative).
498 * @return a new contiguous `basic_ndarray<T>` with the data in C-order.
499 * @complexity O(size).
500 * @alloc allocates a new buffer; throws on size mismatch or negative dims.
501 * @test CheatahNDArray.BroadcastingAdd, CheatahNDArray.ReshapeSizeMismatchThrows
502 * @crtest NdarrayCompileRun.Reshape
503 * @systest StdlibE2E.Ndarray
504 */
505template <Field T>
506basic_ndarray<T> reshape(const basic_ndarray<T>& a, const std::vector<long long>& shape) {
507 const std::vector<std::size_t> ns = detail::to_size(shape);
508 if (detail::product(ns) != a.size()) {
509 throw std::runtime_error("ndarray: cannot reshape, size mismatch");
510 }
511 basic_ndarray<T> out(ns);
512 auto& buf = *out.buffer();
513 std::vector<std::size_t> idx(a.ndim(), 0);
514 std::size_t flat = 0;
515 do {
516 buf[flat++] = a.at(idx);
517 } while (a.ndim() != 0 && detail::next_index(idx, a.shape()));
518 return out;
519}

521// ---- element-wise ops (broadcasting, vectorized) ----
522/**
523 * Broadcast @p a and @p b to their common shape and apply @p op elementwise into a
524 * fresh contiguous result. Fast path: when both operands are contiguous, a flat
525 * `std::transform` under the `unseq` policy (SIMD); otherwise a C-order scalar walk.
526 * @param a first operand.
527 * @param b second operand.
528 * @param op the binary operation applied to corresponding elements.
529 * @return `op(a, b)` broadcast to the common shape; throws if shapes don't broadcast.
530 * @complexity O(size of result).
531 * @alloc allocates the result buffer.
532 * @test CheatahNDArray.BroadcastingAdd
533 * @systest StdlibE2E.Ndarray
534 */
535template <Field T, typename Op>
536basic_ndarray<T> binary_op(const basic_ndarray<T>& a, const basic_ndarray<T>& b, Op op) {
537 const std::vector<std::size_t> rshape = broadcast_shapes(a.shape(), b.shape());
538 basic_ndarray<T> out(rshape);
539 auto& obuf = *out.buffer();
540 // Scalar fast paths: `array ⊕ scalar` (or the reverse) is by far the most common
541 // broadcast, and the general strided walk below does a bounds-checked at() per
542 // element (no SIMD). When the other operand is a single value over a contiguous
543 // full-shape array, it's just a flat vectorizable loop.
544 if (b.size() == 1 && a.shape() == rshape && is_contiguous(a)) {
545 const T s = (*b.buffer())[b.offset()];
546 const T* ad = a.buffer()->data() + a.offset();
547 for (std::size_t i = 0; i < obuf.size(); ++i) obuf[i] = op(ad[i], s);
548 return out;
549 }
550 if (a.size() == 1 && b.shape() == rshape && is_contiguous(b)) {
551 const T s = (*a.buffer())[a.offset()];
552 const T* bd = b.buffer()->data() + b.offset();
553 for (std::size_t i = 0; i < obuf.size(); ++i) obuf[i] = op(s, bd[i]);
554 return out;
555 }
556 const basic_ndarray<T> av = broadcast_to(a, rshape);
557 const basic_ndarray<T> bv = broadcast_to(b, rshape);
558 if (is_contiguous(av) && is_contiguous(bv)) {
559 const auto& abuf = *av.buffer();
560 const auto& bbuf = *bv.buffer();
561 std::transform(CHEATAH_UNSEQ abuf.begin() + av.offset(),
562 abuf.begin() + av.offset() + obuf.size(), bbuf.begin() + bv.offset(),
563 obuf.begin(), op);
564 return out;
565 }
566 std::vector<std::size_t> idx(rshape.size(), 0);
567 std::size_t flat = 0;
568 do {
569 obuf[flat++] = op(av.at(idx), bv.at(idx));
570 } while (!rshape.empty() && detail::next_index(idx, rshape));
571 return out;
572}

632/// Map @p a element-wise through @p f into a fresh contiguous array of element type
633/// `U` (which may differ from `T` — e.g. complex→real for @ref real). Contiguous
634/// fast path via `std::transform(unseq)`; otherwise a C-order walk.
635template <typename U, Field T, typename F>
636basic_ndarray<U> map_array(const basic_ndarray<T>& a, F f) {
637 basic_ndarray<U> out(a.shape());
638 auto& obuf = *out.buffer();
639 if (is_contiguous(a)) {
640 const auto& abuf = *a.buffer();
641 std::transform(CHEATAH_UNSEQ abuf.begin() + a.offset(),
642 abuf.begin() + a.offset() + a.size(), obuf.begin(), f);
643 return out;
644 }
645 std::vector<std::size_t> idx(a.ndim(), 0);
646 std::size_t flat = 0;
647 do {
648 obuf[flat++] = f(a.at(idx));
649 } while (a.ndim() != 0 && next_index(idx, a.shape()));
650 return out;
651}

653// Out-of-line, separately-compiled (-ffast-math) double-precision SIMD kernels for the
654// element-wise ufuncs — see ufunc_simd.cpp. They vectorize the transcendentals through
655// libmvec, which the default flags cannot; isolating -ffast-math to that file keeps the
656// rest of cheatah's arithmetic strict.
657void simd_sqrt_f64(const double*, double*, std::size_t);
658void simd_cbrt_f64(const double*, double*, std::size_t);
659void simd_exp_f64(const double*, double*, std::size_t);
660void simd_log_f64(const double*, double*, std::size_t);
661void simd_sin_f64(const double*, double*, std::size_t);
662void simd_cos_f64(const double*, double*, std::size_t);
663void simd_tan_f64(const double*, double*, std::size_t);

665/// Map a ufunc over @p a: a *contiguous double* array goes through the precompiled SIMD
666/// @p kernel; everything else (float, or a strided/broadcast view) uses the generic
667/// scalar @p fallback. Same result either way — the kernel just vectorizes the hot case.
668template <FloatingPoint T, class Kernel, class Fallback>
669basic_ndarray<T> map_ufunc(const basic_ndarray<T>& a, Kernel kernel, Fallback fallback) {
670 if constexpr (std::is_same_v<T, double>) {
671 if (is_contiguous(a)) {
672 basic_ndarray<T> out(a.shape());
673 kernel(a.buffer()->data() + a.offset(), out.buffer()->data(), a.size());
674 return out;
675 }
676 }
677 return map_array<T>(a, fallback);
678}

681/**
682 * Build a complex array from real and imaginary parts (element-wise `re + im·j`),
683 * broadcasting the two together — the way to construct a complex matrix/vector
684 * (a wavefunction, a Hermitian operator) since cheatah literals are real.
685 * @param re the real parts (a real floating array).
686 * @param im the imaginary parts (a real floating array, broadcast against @p re).
687 * @return a `basic_ndarray<std::complex<T>>` of `re + im·j`; throws if the shapes don't broadcast.
688 * @complexity O(size of result).
689 * @alloc allocates the result buffer.
690 * @test CheatahNDArray.ComplexConstructAndParts
691 * @crtest NdarrayCompileRun.Complex
692 * @systest StdlibE2E.NdarrayComplex
693 */
694template <FloatingPoint T>
695basic_ndarray<std::complex<T>> complex(const basic_ndarray<T>& re, const basic_ndarray<T>& im) {
696 using C = std::complex<T>;
697 const std::vector<std::size_t> rshape = broadcast_shapes(re.shape(), im.shape());
698 const basic_ndarray<T> rv = broadcast_to(re, rshape);
699 const basic_ndarray<T> iv = broadcast_to(im, rshape);
700 basic_ndarray<C> out(rshape);
701 auto& obuf = *out.buffer();
702 std::vector<std::size_t> idx(rshape.size(), 0);
703 std::size_t flat = 0;
704 do {
705 obuf[flat++] = C(rv.at(idx), iv.at(idx));
706 } while (!rshape.empty() && detail::next_index(idx, rshape));
707 return out;
708}
709/**
710 * Element-wise complex conjugate (`a − b·j` for each `a + b·j`); on a real array it
711 * is the identity (a copy). Type-preserving. Used to form Hermitian adjoints and
712 * conjugate-linear inner products.
713 * @param a the array.
714 * @return a fresh array of the same element type with each element conjugated.
715 * @complexity O(size).
716 * @alloc allocates the result buffer.
717 * @test CheatahNDArray.ComplexConstructAndParts
718 * @crtest NdarrayCompileRun.Conj
719 * @systest StdlibE2E.NdarrayComplex
720 */
721template <Field T>
722basic_ndarray<T> conj(const basic_ndarray<T>& a) {
723 return detail::map_array<T>(a, [](T x) -> T {
724 if constexpr (is_complex_v<T>) {
725 return std::conj(x);
726 } else {
727 return x;
728 }
729 });
730}
731/**
732 * The real parts as a real array (the identity on a real array). For `a + b·j` it
733 * returns `a`.
734 * @param a the array.
735 * @return a `basic_ndarray<real_base_t<T>>` of the real parts.
736 * @complexity O(size).
737 * @alloc allocates the result buffer.
738 * @test CheatahNDArray.ComplexConstructAndParts
739 * @crtest NdarrayCompileRun.Real
740 * @systest StdlibE2E.NdarrayComplex
741 */
742template <Field T>
743basic_ndarray<real_base_t<T>> real(const basic_ndarray<T>& a) {
744 using R = real_base_t<T>;
745 return detail::map_array<R>(a, [](T x) -> R {
746 if constexpr (is_complex_v<T>) {
747 return x.real();
748 } else {
749 return x;
750 }
751 });
752}
753/**
754 * The imaginary parts as a real array (all zeros for a real array). For `a + b·j` it
755 * returns `b`.
756 * @param a the array.
757 * @return a `basic_ndarray<real_base_t<T>>` of the imaginary parts.
758 * @complexity O(size).
759 * @alloc allocates the result buffer.
760 * @test CheatahNDArray.ComplexConstructAndParts
761 * @crtest NdarrayCompileRun.Imag
762 * @systest StdlibE2E.NdarrayComplex
763 */
764template <Field T>
765basic_ndarray<real_base_t<T>> imag(const basic_ndarray<T>& a) {

884/**
885 * Sum of all elements — a full reduction across every axis (vectorized via
886 * `std::reduce(unseq)` when contiguous, else a C-order walk); empty sums to 0.
887 * @param a the array.
888 * @return the total, as the element type @p T.
889 * @complexity O(size).
890 * @alloc none.
891 * @test CheatahNDArray.ShapeFactoriesAndReductions
892 * @crtest NdarrayCompileRun.Sum
893 * @systest StdlibE2E.Ndarray
894 */
895template <Field T>
896T sum(const basic_ndarray<T>& a) {
897 if (is_contiguous(a)) {
898 const auto& buf = *a.buffer();
899 return std::reduce(CHEATAH_UNSEQ buf.begin() + a.offset(),
900 buf.begin() + a.offset() + a.size(), T{});
901 }
902 T s{};
903 std::vector<std::size_t> idx(a.ndim(), 0);
904 do {
905 s += a.at(idx);
906 } while (a.ndim() != 0 && detail::next_index(idx, a.shape()));
907 return s;
908}
909/**
910 * Mean of all elements, always as a `double` (0.0 for an empty array — no divide-by-zero).
911 * @param a the array.
912 * @return the average as a double.
913 * @complexity O(size).
914 * @alloc none.
915 * @test CheatahNDArray.ShapeFactoriesAndReductions
916 * @crtest NdarrayCompileRun.Mean
917 * @systest StdlibE2E.Ndarray
918 */
919template <Numeric T>
920double mean(const basic_ndarray<T>& a) {
921 const std::size_t n = a.size();
922 return n == 0 ? 0.0 : static_cast<double>(sum(a)) / static_cast<double>(n);
923}

971namespace detail {
972/// Format one element. Real types go through `operator<<`; a complex element is
973/// rendered Python-style as `a+bj` / `a-bj` (not the `std::complex` default
974/// `(a,b)`), so a complex spectrum prints the way a cheatah user expects.
975template <typename T>
976std::string format_scalar(const T& v) {
977 std::ostringstream os;
978 if constexpr (is_complex_v<T>) {
979 using R = real_base_t<T>;
980 // Flush negative zero to +0 so a conjugate prints "1+0j", not "1+-0j".
981 const auto nz = [](R x) -> R { return x == R{0} ? R{0} : x; };
982 os << nz(v.real());
983 if (v.imag() < R{0}) {
984 os << "-" << nz(-v.imag()) << "j";
985 } else {
986 os << "+" << nz(v.imag()) << "j";
987 }
988 } else {
989 os << v;
990 }
991 return os.str();
992}
993
994/// Recursively format @p a into nested brackets (each element via `format_scalar`).
995template <Field T>
996void format_rec(const basic_ndarray<T>& a, std::vector<std::size_t>& idx, std::size_t dim,
997 std::string& out) {
998 if (dim == a.ndim()) {
999 out += format_scalar(a.at(idx));
1000 return;
1001 }
1002 out += "[";
1003 for (std::size_t i = 0; i < a.shape()[dim]; ++i) {
1004 if (i != 0) out += ", ";
1005 idx[dim] = i;
1006 format_rec(a, idx, dim + 1, out);
1007 }
1008 out += "]";
1009}
1010} // namespace detail

1012/**
1013 * Render as a nested-bracket string, e.g. `"[[1, 2], [3, 4]]"` (a 0-d scalar renders
1014 * as the bare number). Each element is formatted with the default `ostream` precision.
1015 * @param a the array.
1016 * @return the textual representation.
1017 * @complexity O(size).
1018 * @alloc allocates the result string.
1019 * @test CheatahNDArray.ToStringScalar, CheatahNDArray.BroadcastingAdd
1020 * @crtest NdarrayCompileRun.ToString
1021 * @systest StdlibE2E.Ndarray
1022 */
1023template <Field T>
1024std::string to_string(const basic_ndarray<T>& a) {
1025 if (a.ndim() == 0) {
1026 return detail::format_scalar(a.at({}));
1027 }
1028 std::vector<std::size_t> idx(a.ndim(), 0);
1029 std::string out;
1030 detail::format_rec(a, idx, 0, out);
1031 return out;
1032}
