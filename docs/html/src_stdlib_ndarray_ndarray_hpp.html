3/**
4 * @file ndarray.hpp
5 * @brief cheatah `ndarray` — our own numpy-flavored N-dimensional array (of
6 * doubles) with NumPy broadcasting, surfaced as a `NDArray` class plus free
7 * functions (a .purr program writes `ndarray.zeros([2, 3])`).
8 * See https://numpy.org/doc/stable/user/basics.broadcasting.html.
9 *
10 * `import ndarray` includes this header and links `libcheatah_ndarray`. Unit tests:
11 * `stdlib/tests/ndarray_test.cpp`; the suite runs under AddressSanitizer (the `asan`
12 * preset) and Valgrind (`security/run-valgrind.sh`) on every QA-gate run.
13 *
14 * @note Design (the "pointers + a bit of thinking"): the elements live in a shared
15 * buffer (`std::shared_ptr<std::vector<double>>`) and an array is a VIEW into
16 * it — {shape, strides, offset}. That makes reshape and **broadcast**
17 * zero-copy: to stretch a dimension of size 1 we give it a stride of 0, so
18 * every index along it reads the same element. Shared ownership = memory-safe,
19 * no manual frees. `size` below is the element count (product of dims).
20 */

55/// Numeric<T>: an arithmetic element type an ndarray can store (int or float

105namespace detail {
106/// \cond INTERNAL
107/// An allocator identical to `std::allocator<T>` in every respect EXCEPT that
108/// DEFAULT (no-value) construction — what `vector(n)` / `resize(n)` perform — leaves a
109/// trivially-constructible element UNINITIALIZED instead of value-initializing it to 0.
110///
111/// Every ndarray op that allocates a result buffer it then overwrites in full (binary
112/// ops, ufuncs, reshape, array()) would otherwise pay a throwaway zero-fill of the whole
113/// buffer first. That wasted write pass is hidden on compute-heavy ops but DOMINATES
114/// bandwidth-bound ones — `add` was ≈1.5× of NumPy purely from the extra memset. With
115/// this allocator the sizing path skips it; the value-filling forms (`assign(n, v)`,
116/// `vector(n, v)` used by zeros/full/scalar) are untouched and still initialize.
117template <typename T>
118struct default_init_allocator : std::allocator<T> {
119 using std::allocator<T>::allocator;
120 template <typename U>
121 struct rebind {
122 using other = default_init_allocator<U>;
123 };
124 /// Default construction: a trivially-constructible element is left uninitialized.
125 template <typename U>
126 void construct(U* p) noexcept(std::is_nothrow_default_constructible_v<U>) {
127 ::new (static_cast<void*>(p)) U; // default-init (no `()`): no zeroing for trivial U
128 }
129 /// Every other construction (value-fill, copy, emplace) behaves exactly as normal.
130 template <typename U, typename... Args>
131 void construct(U* p, Args&&... args) {
132 ::new (static_cast<void*>(p)) U(std::forward<Args>(args)...);
133 }
134};
135/// \endcond
136
137/// C-order (row-major) strides for a shape.
138inline std::vector<std::ptrdiff_t> contiguous_strides(const std::vector<std::size_t>& shape) {
139 std::vector<std::ptrdiff_t> s(shape.size());
140 std::ptrdiff_t step = 1;
141 for (std::size_t i = shape.size(); i-- > 0;) {
142 s[i] = step;
143 step *= static_cast<std::ptrdiff_t>(shape[i]);
144 }
145 return s;
146}
147/// Overflow-checked product of the dimensions (a wrapped size_t would under-allocate,
148/// turning later element access into out-of-bounds writes — reject it up front).
149inline std::size_t product(const std::vector<std::size_t>& shape) {
150 std::size_t t = 1;
151 for (std::size_t d : shape) {
152 if (d != 0 && t > std::numeric_limits<std::size_t>::max() / d) {
153 throw std::runtime_error("ndarray: shape too large (size overflow)");
154 }
155 t *= d;
156 }
157 return t;
158}
159/// Convert signed dims/indices to sizes, rejecting negatives (a negative cast to
160/// size_t becomes huge -> under-allocation / OOB). Validate at the boundary.
161inline std::vector<std::size_t> to_size(const std::vector<long long>& v) {
162 std::vector<std::size_t> out(v.size());
163 for (std::size_t i = 0; i < v.size(); ++i) {
164 if (v[i] < 0) throw std::runtime_error("ndarray: negative dimension or index");
165 out[i] = static_cast<std::size_t>(v[i]);
166 }
167 return out;
168}
169/// Advance a C-order multi-index odometer; false when it wraps past the end.
170inline bool next_index(std::vector<std::size_t>& idx, const std::vector<std::size_t>& shape) {
171 for (std::size_t i = shape.size(); i-- > 0;) {
172 if (++idx[i] < shape[i]) return true;
173 idx[i] = 0;
174 }
175 return false;
176}

178/// Peel `std::vector<>` layers off a (possibly deeply nested) list type to reach the
179/// leaf scalar — `nested_scalar_t<std::vector<std::vector<double>>>` is `double`.
180template <typename V> struct nested_scalar { using type = V; };
181template <typename U> struct nested_scalar<std::vector<U>> {
182 using type = typename nested_scalar<U>::type;
183};
184template <typename V> using nested_scalar_t = typename nested_scalar<V>::type;
185
186/// Whether `V` is a `std::vector<…>` (used to tell a nested list from a scalar leaf).
187template <typename V> inline constexpr bool is_std_vector_v = false;
188template <typename U> inline constexpr bool is_std_vector_v<std::vector<U>> = true;
189
190/// Flatten a scalar leaf into the C-order buffer (recursion base case).
191template <Field T>
192void nested_collect(const T& x, std::vector<T>& flat, std::vector<std::size_t>&, std::size_t) {
193 flat.push_back(x);
194}
195/// Walk a nested list: record each axis length the first time it is seen, reject a
196/// ragged list (a row whose length differs from its siblings — numpy does too), and
197/// flatten the leaves in C-order. The leaf scalar must be a @ref Field.
198template <typename U>
199 requires Field<nested_scalar_t<U>>
200void nested_collect(const std::vector<U>& v, std::vector<nested_scalar_t<U>>& flat,
201 std::vector<std::size_t>& shape, std::size_t depth) {
202 if (depth == shape.size()) shape.push_back(v.size());
203 else if (shape[depth] != v.size())
204 throw std::runtime_error("ndarray: array(...) ragged nested list (a row's length "
205 "differs from its siblings)");
206 for (const U& e : v) nested_collect(e, flat, shape, depth + 1);
207}
208} // namespace detail

210/// The backing store of an ndarray: a flat, contiguous, shared element buffer. It uses
211/// @ref detail::default_init_allocator so a freshly-sized result buffer that an op is
212/// about to overwrite in full is not needlessly zero-filled first. zeros/full/scalar,
213/// which value-fill, are unaffected — only the no-value sizing path skips initialization.
214template <Field T>
215using buffer_t = std::vector<T, detail::default_init_allocator<T>>;
216
217/**
218 * @brief An N-dimensional array of `T` (a @ref Field element type — real or complex):
219 * a view ({shape, strides, offset}) over a shared element buffer.
220 *
221 * Copies are cheap and share the buffer; reshape/broadcast produce new views without
222 * copying elements. Index math goes through @ref at, which bounds-checks. The element
223 * type is deduced from the data (e.g. `array([1,2,3])` is integer, `array([1.0,…])`
224 * is double); `NDArray` is the default `basic_ndarray<double>`. Complex element types
225 * (`std::complex<double>`) make complex matrices/vectors — and the complex eigenvalues
226 * a real matrix can have — first-class.
227 */
228template <Field T>
229class basic_ndarray {
230public:
231 using value_type = T; ///< The stored element type (the @ref Field `T`).
232 /**
233 * Construct an empty 0-d array with a fresh empty buffer.
234 *
235 * Leaves shape and strides empty and the buffer holding no elements; note this
236 * is distinct from a 0-d scalar (see @ref scalar), whose buffer holds one element.
237 * @complexity O(1).
238 * @alloc allocates the empty shared buffer.
239 * @test CheatahNDArray.ToStringScalar
240 * @systest StdlibE2E.Ndarray
241 */
242 basic_ndarray() : data_(std::make_shared<buffer_t<T>>()) {}
243 /**
244 * Construct a contiguous array of @p shape filled with @p fill.
245 *
246 * Allocates a fresh buffer of `product(shape)` doubles all set to @p fill and
247 * computes C-order (row-major) strides; the dimension product is overflow-checked.
248 * @param shape the dimensions.
249 * @param fill value for every element.
250 * @complexity O(size).
251 * @alloc allocates a new NDArray buffer (shared_ptr<vector<double>>) of
252 * `product(shape)` elements; throws if the shape overflows size_t.
253 * @test CheatahNDArray.ShapeFactoriesAndReductions
254 * @systest StdlibE2E.Ndarray
255 */
256 explicit basic_ndarray(std::vector<std::size_t> shape, T fill = T{}) // contiguous
257 : data_(std::make_shared<buffer_t<T>>()), shape_(std::move(shape)) {
258 // resize (default-init: no zero pass) then std::fill — the fill goes through
259 // operator= on built elements, which keeps libstdc++'s memset/SIMD fast path.
260 // (vector::assign(n, v) would route through the allocator's construct, which a
261 // default-init allocator forces element-by-element — measurably slower.)
262 data_->resize(detail::product(shape_));
263 std::fill(data_->begin(), data_->end(), fill);
264 strides_ = detail::contiguous_strides(shape_);
265 }
266 /**
267 * Build a contiguous array of @p shape whose buffer is sized but left
268 * UNINITIALIZED — for internal ops (binary ops, ufuncs, reshape, array) that
269 * immediately write every element, so paying for a zero-fill first is pure waste.
270 * @param shape the dimensions.
271 * @return an array of @p shape with an uninitialized buffer.
272 * @complexity O(1) beyond the allocation (no element initialization).
273 * @alloc allocates an uninitialized `product(shape)`-element buffer; overflow-checked.
274 * @test CheatahNDArray.BroadcastingAdd
275 * @systest StdlibE2E.Ndarray
276 */
277 static basic_ndarray uninitialized(std::vector<std::size_t> shape) {
278 basic_ndarray out;
279 out.shape_ = std::move(shape);
280 out.data_->resize(detail::product(out.shape_)); // default-init alloc -> no zero-fill
281 out.strides_ = detail::contiguous_strides(out.shape_);
282 return out;
283 }
284 /**
285 * Construct a view from explicit buffer/shape/strides/offset (used by views).
286 *
287 * Stores the supplied members verbatim with no validation or copy, so the new
288 * array shares ownership of @p data; callers (e.g. @ref broadcast_to) are
289 * responsible for passing strides/offset that stay within the buffer.
290 * @param data the shared element buffer.
291 * @param shape the dimensions.
292 * @param strides element strides per dimension.
293 * @param offset starting flat offset into @p data.
294 * @complexity O(1).
295 * @alloc shares @p data, no element copy.
296 * @test CheatahNDArray.BroadcastTo
297 * @systest StdlibE2E.Ndarray
298 */
299 basic_ndarray(std::shared_ptr<buffer_t<T>> data, std::vector<std::size_t> shape,
300 std::vector<std::ptrdiff_t> strides, std::size_t offset)
301 : data_(std::move(data)), shape_(std::move(shape)), strides_(std::move(strides)),
302 offset_(offset) {}
303
304 /**
305 * The shape (dimensions).
306 * @return reference to the shape vector.
307 * @complexity O(1).
308 * @alloc none.
309 * @test CheatahNDArray.ShapeFactoriesAndReductions
310 * @systest StdlibE2E.Ndarray
311 */
312 const std::vector<std::size_t>& shape() const { return shape_; }
313 /**
314 * The element strides.
315 * @return reference to the strides vector.
316 * @complexity O(1).
317 * @alloc none.
318 * @test CheatahNDArray.BroadcastTo
319 * @systest StdlibE2E.Ndarray
320 */
321 const std::vector<std::ptrdiff_t>& strides() const { return strides_; }
322 /**
323 * The number of dimensions (rank).
324 * @return `shape().size()`.
325 * @complexity O(1).
326 * @alloc none.
327 * @test CheatahNDArray.BroadcastingAdd
328 * @systest StdlibE2E.Ndarray
329 */
330 std::size_t ndim() const { return shape_.size(); }
331 /**
332 * The element count (product of dims; 1 for a 0-d scalar).
333 *
334 * Recomputes the overflow-checked product of the shape on each call rather than
335 * caching it; an empty (no-dimension) shape yields 1.
336 * @return the number of elements.
337 * @complexity O(ndim).
338 * @alloc none; throws on size overflow.
339 * @test CheatahNDArray.ShapeFactoriesAndReductions
340 * @systest StdlibE2E.Ndarray
341 */
342 std::size_t size() const { return detail::product(shape_); } // 1 for a 0-d scalar
343
344 /**
345 * Read one element by multi-index (via strides).
346 *
347 * Computes the flat buffer position as `offset + sum(index[i] * strides[i])`, so
348 * it correctly resolves views (including broadcast dims with stride 0); throws if
349 * @p index has the wrong rank or any coordinate is out of range for its dimension.
350 * @param index one coordinate per dimension.
351 * @return the element value.
352 * @complexity O(ndim).
353 * @alloc none; bounds-checks rank and range, throwing on a bad index.
354 * @test CheatahNDArray.ShapeFactoriesAndReductions,
355 * CheatahNDArray.RejectsMaliciousShapesAndIndices
356 * @systest StdlibE2E.Ndarray
357 */
358 T at(const std::vector<std::size_t>& index) const { // element via strides
359 // Bounds-check: a wrong-rank or out-of-range index would otherwise compute an
360 // offset outside the backing buffer (out-of-bounds read).
361 if (index.size() != shape_.size()) {
362 throw std::runtime_error("ndarray: index has the wrong number of dimensions");
363 }
364 std::ptrdiff_t off = static_cast<std::ptrdiff_t>(offset_);
365 for (std::size_t i = 0; i < index.size(); ++i) {
366 if (index[i] >= shape_[i]) throw std::runtime_error("ndarray: index out of range");
367 off += static_cast<std::ptrdiff_t>(index[i]) * strides_[i];
368 }
369 return (*data_)[static_cast<std::size_t>(off)];
370 }
371 /**
372 * The shared backing buffer.
373 * @return reference to the element buffer shared_ptr.
374 * @complexity O(1).
375 * @alloc none.
376 * @test CheatahNDArray.BroadcastingAdd
377 * @systest StdlibE2E.Ndarray
378 */
379 const std::shared_ptr<buffer_t<T>>& buffer() const { return data_; }
380 /**
381 * The flat offset into the buffer where this view starts.
382 * @return the offset.
383 * @complexity O(1).
384 * @alloc none.
385 * @test CheatahNDArray.BroadcastTo
386 * @systest StdlibE2E.Ndarray
387 */
388 std::size_t offset() const { return offset_; }

392 * makes an NDArray printable via `io.print` (io's `HasStr` protocol). Defers to

396 * @alloc allocates the result string.
397 * @test CheatahNDArray.ToString
398 * @systest StdlibE2E.Ndarray

400 std::string str() const;

415 void cheatah_pretty_print(std::ostream& os, long long indent) const;

417private:
418 std::shared_ptr<buffer_t<T>> data_;
419 std::vector<std::size_t> shape_;
420 std::vector<std::ptrdiff_t> strides_; // element strides
421 std::size_t offset_ = 0;
422};

446// ==========================================================================
447// Templated free functions. The element type T is deduced from the data
448// (array([1,2,3]) -> long long, array([1.0,…]) -> double); every op is
449// constrained by Numeric. Elementwise ops vectorize via the std::execution
450// policies (declarative SIMD); broadcasting/strided views fall back to a
451// C-order scalar walk. NDArray (= basic_ndarray<double>) is the default.
452// ==========================================================================
453
454/**
455 * Whether @p a is a contiguous C-order block (no broadcast/stride-0/permuted view),
456 * so its elements live consecutively from `offset()` and can be walked flatly.
457 * @param a the array (or view) to test.
458 * @return true if @p a's strides are the C-order strides for its shape.
459 * @complexity O(ndim).
460 * @alloc none.
461 * @test CheatahNDArray.BroadcastingAdd
462 * @systest StdlibE2E.Ndarray
463 */
464template <Field T>
465inline bool is_contiguous(const basic_ndarray<T>& a) {
466 // C-order contiguity WITHOUT materializing the reference strides: walk the dims
467 // back-to-front and check each stride equals the running size product. The old
468 // `strides() == contiguous_strides(shape())` heap-allocated a vector on every call
469 // — a fixed cost that dominated small-n reductions (e.g. dot, where it ran twice
470 // per call). Same result as the comparison, zero allocation. O(ndim).
471 const std::vector<std::size_t>& shape = a.shape();
472 const std::vector<std::ptrdiff_t>& strides = a.strides();
473 std::ptrdiff_t expect = 1;
474 for (std::size_t i = shape.size(); i-- > 0;) {
475 if (strides[i] != expect) return false;
476 expect *= static_cast<std::ptrdiff_t>(shape[i]);
477 }
478 return true;
479}

481/**
482 * A zero-copy view of @p a stretched to @p target (size-1 / missing dims get stride 0).
483 * @param a source array.
484 * @param target the shape to stretch to.
485 * @return a VIEW sharing @p a's buffer (no element copy).
486 * @test CheatahNDArray.BroadcastTo
487 * @systest StdlibE2E.Ndarray
488 */
489template <Field T>
490basic_ndarray<T> broadcast_to(const basic_ndarray<T>& a, const std::vector<std::size_t>& target) {
491 const std::size_t n = target.size();
492 if (a.ndim() > n) throw std::runtime_error("ndarray: cannot broadcast to fewer dimensions");
493 std::vector<std::ptrdiff_t> ns(n, 0); // stretched / missing dims -> stride 0
494 const std::size_t pad = n - a.ndim();
495 for (std::size_t i = 0; i < a.ndim(); ++i) {
496 const std::size_t adim = a.shape()[i];
497 if (adim == target[pad + i]) {
498 ns[pad + i] = a.strides()[i];
499 } else if (adim != 1) {
500 throw std::runtime_error("ndarray: shape not broadcastable to target");
501 } // adim == 1 -> stride stays 0 (stretch)
502 }
503 return basic_ndarray<T>(a.buffer(), target, ns, a.offset());
504}

507/**
508 * 1-D array from a list of values; the element type is the list's element type
509 * (`array([1,2,3])` is integer, `array([1.0,…])` is double).
510 * @param values the elements, copied into a fresh contiguous buffer.
511 * @return a contiguous 1-D `basic_ndarray<T>`.
512 * @complexity O(n).
513 * @alloc allocates a new buffer of `values.size()` elements.
514 * @test CheatahNDArray.ShapeFactoriesAndReductions
515 * @crtest NdarrayCompileRun.Array
516 * @systest StdlibE2E.Ndarray

519basic_ndarray<T> array(const std::vector<T>& values) {

521 std::copy(values.begin(), values.end(), a.buffer()->begin());

524/**
525 * N-dimensional array from a **nested** list — `array([[1, 2], [3, 4]])` is 2-D,
526 * `array([[[1],[2]],[[3],[4]]])` is 3-D, and so on to any depth. The shape is read off
527 * the nesting (outer list = axis 0, …) and the leaf scalar type is deduced; the list
528 * must be **rectangular** (every sibling row the same length) or it throws, exactly as
529 * numpy rejects a ragged array. Selected only when the argument is itself a list of
530 * lists, so it never competes with the 1-D @ref array overload above.
531 * @tparam V the element type of the outer list — itself a `std::vector<…>` whose leaf
532 * is a @ref Field.
533 * @param values the nested list (rows, planes, …), copied into a fresh C-order buffer.
534 * @return a contiguous `basic_ndarray<T>` of the inferred shape.
535 * @complexity O(size).
536 * @alloc allocates one buffer of `product(shape)` elements; throws on a ragged list.
537 * @test CheatahNDArray.NestedArrayConstruction
538 * @crtest NdarrayCompileRun.NestedArray
539 * @systest StdlibE2E.Ndarray
540 */
541template <typename V>
542 requires detail::is_std_vector_v<V> && Field<detail::nested_scalar_t<V>>
543basic_ndarray<detail::nested_scalar_t<V>> array(const std::vector<V>& values) {
544 using T = detail::nested_scalar_t<V>;
545 std::vector<std::size_t> shape;
546 std::vector<T> flat;
547 detail::nested_collect(values, flat, shape, 0);
548 basic_ndarray<T> a = basic_ndarray<T>::uninitialized(shape);
549 std::move(flat.begin(), flat.end(), a.buffer()->begin());
550 return a;
551}

554 * which the `std::vector<T>` overload can't do directly).

623/**
624 * 1-D range `[start, stop)` stepping by @p step; element type deduced from the args.
625 * @param start first value.
626 * @param stop exclusive bound.
627 * @param step increment (throws if zero); a step pointing away from @p stop yields empty.
628 * @return a 1-D `basic_ndarray<T>` of the generated values.
629 * @complexity O(count).
630 * @alloc allocates a new buffer; throws if @p step is zero.
631 * @test CheatahNDArray.Arange
632 * @crtest NdarrayCompileRun.Arange
633 * @systest StdlibE2E.Ndarray
634 */
635template <Numeric T>
636basic_ndarray<T> arange(T start, T stop, T step) {
637 if (step == T{}) throw std::runtime_error("ndarray: arange step must be non-zero");
638 std::vector<T> v;

640 return array(v);

642/**
643 * Reshape @p a to @p shape (same element count); reads in C-order so views/broadcasts
644 * are flattened into a fresh contiguous buffer (a copy, not an alias).
645 * @param a source array.
646 * @param shape the new dimensions (signed; throws on a negative).
647 * @return a new contiguous `basic_ndarray<T>` with the data in C-order.
648 * @complexity O(size).
649 * @alloc allocates a new buffer; throws on size mismatch or negative dims.
650 * @test CheatahNDArray.BroadcastingAdd, CheatahNDArray.ReshapeSizeMismatchThrows
651 * @crtest NdarrayCompileRun.Reshape
652 * @systest StdlibE2E.Ndarray
653 */
654template <Field T>
655basic_ndarray<T> reshape(const basic_ndarray<T>& a, const std::vector<long long>& shape) {
656 const std::vector<std::size_t> ns = detail::to_size(shape);
657 if (detail::product(ns) != a.size()) {
658 throw std::runtime_error("ndarray: cannot reshape, size mismatch");
659 }
660 basic_ndarray<T> out = basic_ndarray<T>::uninitialized(ns); // every element is written below
661 auto& buf = *out.buffer();
662 // Contiguous source (the common case — e.g. reshaping a freshly built array): copy
663 // the flat block in one shot instead of walking a per-element bounds-checked
664 // odometer.
665 if (is_contiguous(a)) {
666 const T* src = a.buffer()->data() + a.offset();
667 std::copy(src, src + a.size(), buf.begin());
668 return out;
669 }
670 std::vector<std::size_t> idx(a.ndim(), 0);
671 std::size_t flat = 0;
672 do {
673 buf[flat++] = a.at(idx);
674 } while (a.ndim() != 0 && detail::next_index(idx, a.shape()));
675 return out;
676}

678// ---- element-wise ops (broadcasting, vectorized) ----
679/**
680 * Broadcast @p a and @p b to their common shape and apply @p op elementwise into a
681 * fresh contiguous result. Fast path: when both operands are contiguous, a flat
682 * `std::transform` under the `unseq` policy (SIMD); otherwise a C-order scalar walk.
683 * @param a first operand.
684 * @param b second operand.
685 * @param op the binary operation applied to corresponding elements.
686 * @return `op(a, b)` broadcast to the common shape; throws if shapes don't broadcast.
687 * @complexity O(size of result).
688 * @alloc allocates the result buffer.
689 * @test CheatahNDArray.BroadcastingAdd
690 * @systest StdlibE2E.Ndarray
691 */
692template <Field T, typename Op>
693basic_ndarray<T> binary_op(const basic_ndarray<T>& a, const basic_ndarray<T>& b, Op op) {
694 const std::vector<std::size_t> rshape = broadcast_shapes(a.shape(), b.shape());
695 basic_ndarray<T> out = basic_ndarray<T>::uninitialized(rshape); // every element written below
696 auto& obuf = *out.buffer();
697 // Scalar fast paths: `array ⊕ scalar` (or the reverse) is by far the most common
698 // broadcast, and the general strided walk below does a bounds-checked at() per
699 // element (no SIMD). When the other operand is a single value over a contiguous
700 // full-shape array, it's a flat loop we hand to the unseq transform so it vectorizes
701 // the same way the array⊕array path does.
702 if (b.size() == 1 && a.shape() == rshape && is_contiguous(a)) {
703 const T s = (*b.buffer())[b.offset()];
704 const auto first = a.buffer()->begin() + a.offset();
705 std::transform(CHEATAH_UNSEQ first, first + obuf.size(), obuf.begin(),
706 [s, op](T x) { return op(x, s); });
707 return out;
708 }
709 if (a.size() == 1 && b.shape() == rshape && is_contiguous(b)) {
710 const T s = (*a.buffer())[a.offset()];
711 const auto first = b.buffer()->begin() + b.offset();
712 std::transform(CHEATAH_UNSEQ first, first + obuf.size(), obuf.begin(),
713 [s, op](T x) { return op(s, x); });
714 return out;
715 }
716 const basic_ndarray<T> av = broadcast_to(a, rshape);
717 const basic_ndarray<T> bv = broadcast_to(b, rshape);
718 if (is_contiguous(av) && is_contiguous(bv)) {
719 const auto& abuf = *av.buffer();
720 const auto& bbuf = *bv.buffer();
721 std::transform(CHEATAH_UNSEQ abuf.begin() + av.offset(),
722 abuf.begin() + av.offset() + obuf.size(), bbuf.begin() + bv.offset(),
723 obuf.begin(), op);
724 return out;
725 }
726 std::vector<std::size_t> idx(rshape.size(), 0);
727 std::size_t flat = 0;
728 do {
729 obuf[flat++] = op(av.at(idx), bv.at(idx));
730 } while (!rshape.empty() && detail::next_index(idx, rshape));
731 return out;
732}
733/**
734 * Element-wise `a + b` with broadcasting.
735 * @param a first operand.
736 * @param b second operand.
737 * @return `a + b` broadcast to the common shape.
738 * @complexity O(size of result). @alloc allocates the result.
739 * @test CheatahNDArray.BroadcastingAdd
740 * @crtest NdarrayCompileRun.Add
741 * @systest StdlibE2E.Ndarray
742 */
743template <Field T>
744basic_ndarray<T> add(const basic_ndarray<T>& a, const basic_ndarray<T>& b) {

792/// Map @p a element-wise through @p f into a fresh contiguous array of element type
793/// `U` (which may differ from `T` — e.g. complex→real for @ref real). Contiguous
794/// fast path via `std::transform(unseq)`; otherwise a C-order walk.
795template <typename U, Field T, typename F>
796basic_ndarray<U> map_array(const basic_ndarray<T>& a, F f) {
797 basic_ndarray<U> out = basic_ndarray<U>::uninitialized(a.shape()); // fully written below
798 auto& obuf = *out.buffer();
799 if (is_contiguous(a)) {
800 const auto& abuf = *a.buffer();
801 std::transform(CHEATAH_UNSEQ abuf.begin() + a.offset(),
802 abuf.begin() + a.offset() + a.size(), obuf.begin(), f);
803 return out;
804 }
805 std::vector<std::size_t> idx(a.ndim(), 0);
806 std::size_t flat = 0;
807 do {
808 obuf[flat++] = f(a.at(idx));
809 } while (a.ndim() != 0 && next_index(idx, a.shape()));
810 return out;
811}

813// Out-of-line, separately-compiled (-ffast-math) double-precision SIMD kernels for the
814// element-wise ufuncs — see ufunc_simd.cpp. They vectorize the transcendentals through
815// libmvec, which the default flags cannot; isolating -ffast-math to that file keeps the
816// rest of cheatah's arithmetic strict.
817void simd_sqrt_f64(const double*, double*, std::size_t);
818void simd_cbrt_f64(const double*, double*, std::size_t);
819void simd_exp_f64(const double*, double*, std::size_t);
820void simd_log_f64(const double*, double*, std::size_t);
821void simd_sin_f64(const double*, double*, std::size_t);
822void simd_cos_f64(const double*, double*, std::size_t);
823void simd_tan_f64(const double*, double*, std::size_t);

825/// Map a ufunc over @p a: a *contiguous double* array goes through the precompiled SIMD
826/// @p kernel; everything else (float, or a strided/broadcast view) uses the generic
827/// scalar @p fallback. Same result either way — the kernel just vectorizes the hot case.
828template <FloatingPoint T, class Kernel, class Fallback>
829basic_ndarray<T> map_ufunc(const basic_ndarray<T>& a, Kernel kernel, Fallback fallback) {
830 if constexpr (std::is_same_v<T, double>) {
831 if (is_contiguous(a)) {
832 basic_ndarray<T> out = basic_ndarray<T>::uninitialized(a.shape()); // kernel fills it
833 kernel(a.buffer()->data() + a.offset(), out.buffer()->data(), a.size());
834 return out;
835 }
836 }
837 return map_array<T>(a, fallback);
838}

841/**
842 * Build a complex array from real and imaginary parts (element-wise `re + im·j`),
843 * broadcasting the two together — the way to construct a complex matrix/vector
844 * (a wavefunction, a Hermitian operator) since cheatah literals are real.
845 * @param re the real parts (a real floating array).
846 * @param im the imaginary parts (a real floating array, broadcast against @p re).
847 * @return a `basic_ndarray<std::complex<T>>` of `re + im·j`; throws if the shapes don't broadcast.
848 * @complexity O(size of result).
849 * @alloc allocates the result buffer.
850 * @test CheatahNDArray.ComplexConstructAndParts
851 * @crtest NdarrayCompileRun.Complex
852 * @systest StdlibE2E.NdarrayComplex
853 */
854template <FloatingPoint T>
855basic_ndarray<std::complex<T>> complex(const basic_ndarray<T>& re, const basic_ndarray<T>& im) {
856 using C = std::complex<T>;
857 const std::vector<std::size_t> rshape = broadcast_shapes(re.shape(), im.shape());
858 const basic_ndarray<T> rv = broadcast_to(re, rshape);
859 const basic_ndarray<T> iv = broadcast_to(im, rshape);
860 basic_ndarray<C> out(rshape);
861 auto& obuf = *out.buffer();
862 std::vector<std::size_t> idx(rshape.size(), 0);
863 std::size_t flat = 0;
864 do {
865 obuf[flat++] = C(rv.at(idx), iv.at(idx));
866 } while (!rshape.empty() && detail::next_index(idx, rshape));
867 return out;
868}
869/**
870 * Element-wise complex conjugate (`a − b·j` for each `a + b·j`); on a real array it
871 * is the identity (a copy). Type-preserving. Used to form Hermitian adjoints and
872 * conjugate-linear inner products.
873 * @param a the array.
874 * @return a fresh array of the same element type with each element conjugated.
875 * @complexity O(size).
876 * @alloc allocates the result buffer.
877 * @test CheatahNDArray.ComplexConstructAndParts
878 * @crtest NdarrayCompileRun.Conj
879 * @systest StdlibE2E.NdarrayComplex
880 */
881template <Field T>
882basic_ndarray<T> conj(const basic_ndarray<T>& a) {
883 return detail::map_array<T>(a, [](T x) -> T {
884 if constexpr (is_complex_v<T>) {
885 return std::conj(x);
886 } else {
887 return x;
888 }
889 });
890}

1030/**
1031 * Element-wise absolute value (the array form of `math.abs`; ≈ `numpy.abs`).
1032 * @param a a floating-point array.
1033 * @return a fresh same-shape array with `|x|` for each element.
1034 * @complexity O(size). @alloc allocates the result buffer.
1035 * @test CheatahNDArray.ElementwiseMath
1036 * @systest StdlibE2E.NdarrayMath
1037 */
1038template <FloatingPoint T>
1039basic_ndarray<T> abs(const basic_ndarray<T>& a) {
1040 return detail::map_array<T>(a, [](T x) { return std::fabs(x); });
1041}

1044/**
1045 * Sum of all elements — a full reduction across every axis (vectorized via
1046 * `std::reduce(unseq)` when contiguous, else a C-order walk); empty sums to 0.
1047 * @param a the array.
1048 * @return the total, as the element type @p T.
1049 * @complexity O(size).
1050 * @alloc none.
1051 * @test CheatahNDArray.ShapeFactoriesAndReductions
1052 * @crtest NdarrayCompileRun.Sum
1053 * @systest StdlibE2E.Ndarray
1054 */
1055template <Field T>
1056T sum(const basic_ndarray<T>& a) {
1057 if (is_contiguous(a)) {
1058 // Eight independent accumulators so the reduction vectorizes (a single running
1059 // sum — or a plain std::reduce, which libstdc++ left-folds for FP without
1060 // -ffast-math — serializes on FP-add latency, the dot/norm mistake).
1061 const T* p = a.buffer()->data() + a.offset();
1062 const std::size_t n = a.size();
1063 T s0{}, s1{}, s2{}, s3{}, s4{}, s5{}, s6{}, s7{};
1064 std::size_t i = 0;
1065 for (; i + 8 <= n; i += 8) {
1066 s0 += p[i + 0]; s1 += p[i + 1]; s2 += p[i + 2]; s3 += p[i + 3];
1067 s4 += p[i + 4]; s5 += p[i + 5]; s6 += p[i + 6]; s7 += p[i + 7];
1068 }
1069 T s = ((s0 + s1) + (s2 + s3)) + ((s4 + s5) + (s6 + s7));
1070 for (; i < n; ++i) s += p[i];
1071 return s;
1072 }
1073 T s{};
1074 std::vector<std::size_t> idx(a.ndim(), 0);
1075 do {
1076 s += a.at(idx);
1077 } while (a.ndim() != 0 && detail::next_index(idx, a.shape()));
1078 return s;
1079}
1080/**
1081 * Mean of all elements, always as a `double` (0.0 for an empty array — no divide-by-zero).
1082 * @param a the array.
1083 * @return the average as a double.
1084 * @complexity O(size).
1085 * @alloc none.
1086 * @test CheatahNDArray.ShapeFactoriesAndReductions
1087 * @crtest NdarrayCompileRun.Mean
1088 * @systest StdlibE2E.Ndarray
1089 */
1090template <Numeric T>
1091double mean(const basic_ndarray<T>& a) {
1092 const std::size_t n = a.size();
1093 return n == 0 ? 0.0 : static_cast<double>(sum(a)) / static_cast<double>(n);
1094}

1097 * basic_ndarray::at; rejects negative coordinates).
1098 * @param a the array.
1099 * @param index one coordinate per dimension (signed; throws on a negative).
1100 * @return the element value (type @p T); throws on a wrong-rank/out-of-range index.
1101 * @complexity O(ndim).
1102 * @alloc none.
1103 * @test CheatahNDArray.ShapeFactoriesAndReductions

1142namespace detail {
1143/// Format one element. Real types go through `operator<<`; a complex element is
1144/// rendered Python-style as `a+bj` / `a-bj` (not the `std::complex` default
1145/// `(a,b)`), so a complex spectrum prints the way a cheatah user expects.
1146template <typename T>
1147std::string format_scalar(const T& v) {
1148 std::ostringstream os;
1149 if constexpr (is_complex_v<T>) {
1150 using R = real_base_t<T>;
1151 // Flush negative zero to +0 so a conjugate prints "1+0j", not "1+-0j".
1152 const auto nz = [](R x) -> R { return x == R{0} ? R{0} : x; };
1153 os << nz(v.real());
1154 if (v.imag() < R{0}) {
1155 os << "-" << nz(-v.imag()) << "j";
1156 } else {
1157 os << "+" << nz(v.imag()) << "j";
1158 }
1159 } else {
1160 os << v;
1161 }
1162 return os.str();
1163}
1164
1165/// Recursively format @p a into nested brackets (each element via `format_scalar`).
1166template <Field T>
1167void format_rec(const basic_ndarray<T>& a, std::vector<std::size_t>& idx, std::size_t dim,
1168 std::string& out) {
1169 if (dim == a.ndim()) {
1170 out += format_scalar(a.at(idx));
1171 return;
1172 }
1173 out += "[";
1174 for (std::size_t i = 0; i < a.shape()[dim]; ++i) {
1175 if (i != 0) out += ", ";
1176 idx[dim] = i;
1177 format_rec(a, idx, dim + 1, out);
1178 }
1179 out += "]";
1180}
1181
1182/// Like format_rec but ABBREVIATES a large array: an axis longer than `2*edge` shows its
1183/// first and last `edge` items with `...` between, recursively. Summarization is enabled by
1184/// @p summarize (the caller turns it on only past a total-size threshold), so small arrays
1185/// print in full.
1186template <Field T>
1187void format_rec_trunc(const basic_ndarray<T>& a, std::vector<std::size_t>& idx, std::size_t dim,
1188 std::string& out, std::size_t edge, bool summarize) {
1189 if (dim == a.ndim()) {
1190 out += format_scalar(a.at(idx));
1191 return;
1192 }
1193 const std::size_t n = a.shape()[dim];
1194 const bool trunc = summarize && n > 2 * edge;
1195 out += "[";
1196 bool first = true;
1197 for (std::size_t i = 0; i < n; ++i) {
1198 if (trunc && i >= edge && i < n - edge) {
1199 if (i == edge) {
1200 if (!first) out += ", ";
1201 out += "...";
1202 first = false;
1203 }
1204 continue; // skip the abbreviated middle
1205 }
1206 if (!first) out += ", ";
1207 first = false;
1208 idx[dim] = i;
1209 format_rec_trunc(a, idx, dim + 1, out, edge, summarize);
1210 }
1211 out += "]";
1212}
1213
1214/// to_string, but ABBREVIATED with `...` when the array is large (total size beyond a
1215/// numpy-style threshold) — the readable default for `io.print`. `io.rprint`/`to_string`
1216/// keep the full form.
1217template <Field T>
1218std::string to_string_pretty(const basic_ndarray<T>& a) {
1219 if (a.ndim() == 0) return format_scalar(a.at({}));
1220 constexpr std::size_t kEdge = 3; // items kept at each end of an abbreviated axis
1221 constexpr std::size_t kThreshold = 1000; // only summarize past this many elements (numpy)
1222 std::vector<std::size_t> idx(a.ndim(), 0);
1223 std::string out;
1224 format_rec_trunc(a, idx, 0, out, kEdge, a.size() > kThreshold);
1225 return out;
1226}
1227} // namespace detail

1229/**
1230 * Render as a nested-bracket string, e.g. `"[[1, 2], [3, 4]]"` (a 0-d scalar renders
1231 * as the bare number). Each element is formatted with the default `ostream` precision.
1232 * @param a the array.
1233 * @return the textual representation.
1234 * @complexity O(size).
1235 * @alloc allocates the result string.
1236 * @test CheatahNDArray.ToStringScalar, CheatahNDArray.BroadcastingAdd
1237 * @crtest NdarrayCompileRun.ToString
1238 * @systest StdlibE2E.Ndarray
1239 */
1240template <Field T>
1241std::string to_string(const basic_ndarray<T>& a) {
1242 if (a.ndim() == 0) {
1243 return detail::format_scalar(a.at({}));
1244 }
1245 std::vector<std::size_t> idx(a.ndim(), 0);
1246 std::string out;
1247 detail::format_rec(a, idx, 0, out);
1248 return out;
1249}

1251template <Field T>
1252inline std::string basic_ndarray<T>::str() const {
1253 return to_string(*this);
1254}

1260 * this way; `io.print` instead uses @ref cheatah_pretty_print to abbreviate large arrays.
