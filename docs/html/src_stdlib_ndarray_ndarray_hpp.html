3/**
4 * @file ndarray.hpp
5 * @brief cheatah `ndarray` — our own numpy-flavored N-dimensional array (of
6 * doubles) with NumPy broadcasting, surfaced as a `NDArray` class plus free
7 * functions (a .purr program writes `ndarray.zeros([2, 3])`).
8 * See https://numpy.org/doc/stable/user/basics.broadcasting.html.
9 *
10 * `import ndarray` includes this header and links `libcheatah_ndarray`. Unit tests:
11 * `stdlib/tests/ndarray_test.cpp`; the suite runs under AddressSanitizer (the `asan`
12 * preset) and Valgrind (`security/run-valgrind.sh`) on every QA-gate run.
13 *
14 * @note Design (the "pointers + a bit of thinking"): the elements live in a shared
15 * buffer (`std::shared_ptr<std::vector<double>>`) and an array is a VIEW into
16 * it — {shape, strides, offset}. That makes reshape and **broadcast**
17 * zero-copy: to stretch a dimension of size 1 we give it a stride of 0, so
18 * every index along it reads the same element. Shared ownership = memory-safe,
19 * no manual frees. `size` below is the element count (product of dims).
20 */
21#include <cstddef>
22#include <memory>
23#include <string>
24#include <vector>

28/**
29 * @brief An N-dimensional array of doubles: a view ({shape, strides, offset}) over
30 * a shared element buffer.
31 *
32 * Copies are cheap and share the buffer; reshape/broadcast produce new views without
33 * copying elements. Index math goes through @ref at, which bounds-checks.
34 */
35class NDArray {
36public:
37 /**
38 * Construct an empty 0-d array with a fresh empty buffer.
39 *
40 * Leaves shape and strides empty and the buffer holding no elements; note this
41 * is distinct from a 0-d scalar (see @ref scalar), whose buffer holds one element.
42 * @complexity O(1).
43 * @alloc allocates the empty shared buffer.
44 * @test CheatahNDArray.ToStringScalar
45 * @systest StdlibE2E.Ndarray
46 */
47 NDArray();
48 /**
49 * Construct a contiguous array of @p shape filled with @p fill.
50 *
51 * Allocates a fresh buffer of `product(shape)` doubles all set to @p fill and
52 * computes C-order (row-major) strides; the dimension product is overflow-checked.
53 * @param shape the dimensions.
54 * @param fill value for every element.
55 * @complexity O(size).
56 * @alloc allocates a new NDArray buffer (shared_ptr<vector<double>>) of
57 * `product(shape)` elements; throws if the shape overflows size_t.
58 * @test CheatahNDArray.ShapeFactoriesAndReductions
59 * @systest StdlibE2E.Ndarray
60 */
61 explicit NDArray(std::vector<std::size_t> shape, double fill = 0.0); // contiguous
62 /**
63 * Construct a view from explicit buffer/shape/strides/offset (used by views).
64 *
65 * Stores the supplied members verbatim with no validation or copy, so the new
66 * array shares ownership of @p data; callers (e.g. @ref broadcast_to) are
67 * responsible for passing strides/offset that stay within the buffer.
68 * @param data the shared element buffer.
69 * @param shape the dimensions.
70 * @param strides element strides per dimension.
71 * @param offset starting flat offset into @p data.
72 * @complexity O(1).
73 * @alloc shares @p data, no element copy.
74 * @test CheatahNDArray.BroadcastTo
75 * @systest StdlibE2E.Ndarray
76 */
77 NDArray(std::shared_ptr<std::vector<double>> data, std::vector<std::size_t> shape,
78 std::vector<std::ptrdiff_t> strides, std::size_t offset);
79
80 /**
81 * The shape (dimensions).
82 * @return reference to the shape vector.
83 * @complexity O(1).
84 * @alloc none.
85 * @test CheatahNDArray.ShapeFactoriesAndReductions
86 * @systest StdlibE2E.Ndarray
87 */
88 const std::vector<std::size_t>& shape() const { return shape_; }
89 /**
90 * The element strides.
91 * @return reference to the strides vector.
92 * @complexity O(1).
93 * @alloc none.
94 * @test CheatahNDArray.BroadcastTo
95 * @systest StdlibE2E.Ndarray
96 */
97 const std::vector<std::ptrdiff_t>& strides() const { return strides_; }
98 /**
99 * The number of dimensions (rank).
100 * @return `shape().size()`.
101 * @complexity O(1).
102 * @alloc none.
103 * @test CheatahNDArray.BroadcastingAdd
104 * @systest StdlibE2E.Ndarray
105 */
106 std::size_t ndim() const { return shape_.size(); }
107 /**
108 * The element count (product of dims; 1 for a 0-d scalar).
109 *
110 * Recomputes the overflow-checked product of the shape on each call rather than
111 * caching it; an empty (no-dimension) shape yields 1.
112 * @return the number of elements.
113 * @complexity O(ndim).
114 * @alloc none; throws on size overflow.
115 * @test CheatahNDArray.ShapeFactoriesAndReductions
116 * @systest StdlibE2E.Ndarray
117 */
118 std::size_t size() const; // product of dims (1 for a 0-d scalar)
119
120 /**
121 * Read one element by multi-index (via strides).
122 *
123 * Computes the flat buffer position as `offset + sum(index[i] * strides[i])`, so
124 * it correctly resolves views (including broadcast dims with stride 0); throws if
125 * @p index has the wrong rank or any coordinate is out of range for its dimension.
126 * @param index one coordinate per dimension.
127 * @return the element value.
128 * @complexity O(ndim).
129 * @alloc none; bounds-checks rank and range, throwing on a bad index.
130 * @test CheatahNDArray.ShapeFactoriesAndReductions,
131 * CheatahNDArray.RejectsMaliciousShapesAndIndices
132 * @systest StdlibE2E.Ndarray
133 */
134 double at(const std::vector<std::size_t>& index) const; // element via strides
135 /**
136 * The shared backing buffer.
137 * @return reference to the element buffer shared_ptr.
138 * @complexity O(1).
139 * @alloc none.
140 * @test CheatahNDArray.BroadcastingAdd
141 * @systest StdlibE2E.Ndarray
142 */
143 const std::shared_ptr<std::vector<double>>& buffer() const { return data_; }
144 /**
145 * The flat offset into the buffer where this view starts.
146 * @return the offset.
147 * @complexity O(1).
148 * @alloc none.
149 * @test CheatahNDArray.BroadcastTo
150 * @systest StdlibE2E.Ndarray
151 */
152 std::size_t offset() const { return offset_; }
153
154private:
155 std::shared_ptr<std::vector<double>> data_;
156 std::vector<std::size_t> shape_;
157 std::vector<std::ptrdiff_t> strides_; // element strides
158 std::size_t offset_ = 0;
159};

161/**
162 * The broadcast result shape of two shapes (NumPy rules).
163 *
164 * Aligns the shapes from the trailing (rightmost) dimension, treating missing
165 * leading dims as 1; each output dim is the non-1 input dim, and two unequal dims
166 * that are both not 1 are incompatible and throw.
167 * @param a first shape.
168 * @param b second shape.
169 * @return the broadcast shape (trailing-aligned).
170 * @complexity O(max(ndim)).
171 * @alloc allocates the small result vector; throws if the shapes are incompatible.
172 * @test CheatahNDArray.BroadcastShapeRules
173 * @systest StdlibE2E.Ndarray
174 */
175std::vector<std::size_t> broadcast_shapes(const std::vector<std::size_t>& a,
176 const std::vector<std::size_t>& b);
177/**
178 * A zero-copy view of @p a stretched to @p target (size-1 / missing dims get stride 0).
179 *
180 * Right-aligns @p a's dims under @p target, keeping the original stride where dims
181 * match and using stride 0 where a size-1 (or missing leading) dim is stretched;
182 * throws if @p a has more dims than @p target or a non-1 dim doesn't match the target.
183 * @param a source array.
184 * @param target the shape to stretch to.
185 * @return a VIEW sharing @p a's buffer (no element copy).
186 * @complexity O(target ndim).
187 * @alloc allocates only the small strides vector for the view, not the data;
188 * throws if @p a is not broadcastable to @p target.
189 * @test CheatahNDArray.BroadcastTo
190 * @systest StdlibE2E.Ndarray
191 */
192NDArray broadcast_to(const NDArray& a, const std::vector<std::size_t>& target);

270/**
271 * 1-D range `[start, stop)` stepping by @p step (Python `range`-like).
272 *
273 * Accumulates values from @p start, advancing by @p step while below @p stop (or
274 * above it when @p step is negative), so a step pointing away from @p stop yields an
275 * empty array; throws when @p step is zero. Uses floating-point accumulation, so
276 * rounding may affect the final count.
277 * @param start first value.
278 * @param stop exclusive bound.
279 * @param step increment (non-zero).
280 * @return a 1-D NDArray of the generated values.
281 * @complexity O(count) where count = number of steps.
282 * @alloc allocates a new NDArray buffer
283 * (shared_ptr<vector<double>>); throws if @p step is zero.
284 * @test CheatahNDArray.Arange
285 * @crtest NdarrayCompileRun.Arange
286 * @systest StdlibE2E.Ndarray
287 */
288NDArray arange(double start, double stop, double step);
289/**
290 * Reshape @p a to @p shape (same element count).
291 *
292 * Reads @p a's elements in C-order (so it works through arbitrary views/broadcasts)
293 * into a brand-new contiguous buffer with the requested shape, i.e. it copies rather
294 * than aliasing; throws if the new size differs from @p a's or any dim is negative.
295 * @param a source array.
296 * @param shape the new dimensions (rejects negatives).
297 * @return a new contiguous NDArray with the data laid out in C-order.
298 * @complexity O(size).
299 * @alloc flattens through views and allocates a new NDArray buffer
300 * (shared_ptr<vector<double>>); throws on size mismatch or negative dims.
301 * @test CheatahNDArray.BroadcastingAdd, CheatahNDArray.ReshapeSizeMismatchThrows
302 * @crtest NdarrayCompileRun.Reshape
303 * @systest StdlibE2E.Ndarray
304 */
305NDArray reshape(const NDArray& a, const std::vector<long long>& shape);
