35// Parse a JSON string straight into `out`. OwnsStrings is the builder's string policy: when false
36// (pooled/view path) an unescaped string is a zero-copy String<std::string_view> into the SOURCE
37// text; when true (owning path) it is COPIED into an owned String<std::string> so the Document is
38// self-contained and may outlive the source (e.g. the cache parses a temporary buffer). An ESCAPED
39// string is always decoded into owned storage regardless, since the decoded bytes have no source.
40// @complexity O(|string|) @alloc only for owned/escaped strings @test Json.Strings
41template <bool OwnsStrings>
42bool parse_string(Cursor& c, Node& out) {
43 std::string_view raw;
44 bool esc = false;
45 if (!scan_string(c, raw, esc)) {
46 return false;
47 }
48 if (!esc) {
49 if constexpr (OwnsStrings) {
50 out.variant().emplace<String<std::string>>(std::string(raw)); // owned copy
51 } else {
52 out.variant().emplace<String<std::string_view>>(raw); // zero-copy view into source
53 }
54 return true;
55 }
56 std::string decoded;
57 if (!decode_escapes(raw, decoded)) {
58 return false;
59 }
60 out.variant().emplace<String<std::string>>(std::move(decoded)); // owned
61 return true;
62}

64// Parse a JSON number into `out` (emplaced in place).

67 double d = 0.0;
68 if (!detail::parse_arithmetic(c, d)) {
69 return false;
70 }
71 out.variant().emplace<Number>(d);
72 return true;

77// Append `s` to `out` as a quoted JSON string, re-escaping specials and control bytes (\u00XX).
78// @complexity O(|s|) @alloc amortized `out` growth @test Json.DumpRoundTrip
79void dump_string(std::string_view s, std::string& out) {
80 static constexpr char kHex[] = "0123456789abcdef";
81 out.push_back('"');
82 for (const char ch : s) {
83 switch (ch) {
84 case '"': out.append("\\\""); break;
85 case '\\': out.append("\\\\"); break;
86 case '\b': out.append("\\b"); break;
87 case '\f': out.append("\\f"); break;
88 case '\n': out.append("\\n"); break;
89 case '\r': out.append("\\r"); break;
90 case '\t': out.append("\\t"); break;
91 default:
92 if (static_cast<unsigned char>(ch) < 0x20) {
93 out.append("\\u00");
94 out.push_back(kHex[(static_cast<unsigned char>(ch) >> 4) & 0xF]);
95 out.push_back(kHex[static_cast<unsigned char>(ch) & 0xF]);
96 } else {
97 out.push_back(ch);
98 }
99 }
100 }
101 out.push_back('"');
102}

104// Serialize ITERATIVELY with an explicit stack of open containers — symmetric with the iterative
105// parser, so a document of any nesting depth dumps without exhausting the C++ call stack (the
106// parser accepts adversarially deep input; the serializer must survive it too).
107// @complexity O(nodes) @alloc the frame stack, O(depth) @test Json.DeepDumpRoundTrip
108void dump_to(const Node& root, std::string& out) {
109 // One open container being emitted: which kind, how many children are already written, and a
110 // span of its remaining children (exactly one of the spans is used, chosen by is_object).
111 struct Frame {
112 bool is_object;
113 std::size_t next;
114 std::span<const Node> elements;
115 std::span<const Member> members;
116 };
117 std::vector<Frame> stack;
118 const Node* value = &root;
119
120 for (;;) {
121 // (A) Emit one VALUE. Scalars are appended whole; a container appends its opener and pushes
122 // a frame, so its children are emitted by the loop below.
123 std::visit(
124 [&](const auto& tok) {
125 using T = std::decay_t<decltype(tok)>;
126 if constexpr (std::is_same_v<T, Null>) {
127 out.append("null");
128 } else if constexpr (std::is_same_v<T, Boolean>) {
129 out.append(tok.value() ? "true" : "false");
130 } else if constexpr (std::is_same_v<T, Number>) {
131 char buf[32];
132 const std::to_chars_result r = std::to_chars(buf, buf + sizeof buf, tok.value());
133 out.append(buf, r.ptr);
134 } else if constexpr (std::is_same_v<T, String<std::string_view>> ||
135 std::is_same_v<T, String<std::string>>) {
136 dump_string(tok.value(), out);
137 } else if constexpr (std::is_same_v<T, OwnedArray> || std::is_same_v<T, ArrayView>) {
138 out.push_back('[');
139 stack.push_back(Frame{.is_object = false,
140 .next = 0,
141 .elements = tok.value(),
142 .members = {}});
143 } else { // Object
144 out.push_back('{');
145 stack.push_back(Frame{.is_object = true,
146 .next = 0,
147 .elements = {},
148 .members = tok.value()});
149 }
150 },
151 value->variant());
152
153 // (B) Find the next value to emit: write separators/keys for the innermost open container,
154 // closing every container that has run out of children along the way.
155 value = nullptr;
156 while (!stack.empty() && value == nullptr) {
157 Frame& top = stack.back();
158 if (top.is_object) {
159 if (top.next == top.members.size()) {
160 out.push_back('}');
161 stack.pop_back();
162 continue; // this object is finished — keep ascending
163 }
164 if (top.next > 0) {
165 out.push_back(',');
166 }
167 const Member& m = top.members[top.next++];
168 dump_string(to_view(m.first), out); // the key (a string Node)
169 out.push_back(':');
170 value = &m.second;
171 } else {
172 if (top.next == top.elements.size()) {
173 out.push_back(']');
174 stack.pop_back();
175 continue; // this array is finished — keep ascending
176 }
177 if (top.next > 0) {
178 out.push_back(',');
179 }
180 value = &top.elements[top.next++];
181 }
182 }
183 if (value == nullptr) {
184 return; // stack empty: the whole document has been emitted
185 }
186 }
187}

191std::string_view to_view(const Node& value) noexcept {
192 const Node::variant_type& v = value.variant();
193 if (const String<std::string_view>* view = std::get_if<String<std::string_view>>(&v)) {
194 return view->value();
195 }
196 if (const String<std::string>* owned = std::get_if<String<std::string>>(&v)) {
197 return owned->value();
198 }
199 return {};
200}

202template <bool Validate>
203Document parse(std::string_view text, bool* ok) {
204 // A fresh Parser yields a self-contained OWNING Document (no pool references), so it is safe to
205 // return even though the Parser is destroyed here. Reuse a Parser (Parser::parse for the pooled
206 // view form, or Parser::parse_owning) to amortize allocation across many parses. Validate
207 // forwards straight through to the grammar (default true; <false> compiles the checks out).
208 Parser p;
209 return p.parse_owning<Validate>(text, ok);
210}

224// ---- reusable Parser: ONE iterative grammar, two Builders -------------------
225//
226// The recursion is unrolled into a loop over an explicit frame_stack_: each open container is a
227// Frame, the Builder (driven as a stack machine: begin/add/finish) holds the partial container, and
228// completed values "bubble up" to their parent. Nesting depth therefore costs heap, not C++ call
229// frames — no stack overflow on adversarially deep input. The grammar is parameterized on (1) a
230// compile-time `bool Validate` and (2) a Builder policy (PoolBuilder -> span VIEWS into a reused
231// pool, or OwningBuilder -> self-contained owned vectors). Every bounds/structure check sits inside
232// `if constexpr (Validate)`, so with Validate=false the standard DISCARDS those statements at
233// compile time (validation gone from the binary, not merely optimized away); trusted input is
234// assumed well-formed, so the elided checks never fire and the result is identical (malformed input
235// under Validate=false is UB). Instantiated for both modes and both builders below.
236
237template <bool Validate, class Builder>
238bool Parser::parse_value(Cursor& c, Node& out, Builder& b) {
239 frame_stack_.clear();
240 Node value; // the most-recently-completed value, bubbling up toward its parent / the root
241
242 for (;;) {
243 // (A) Inside an object, the next child is a member: read its "key" and the ':' first.
244 if (!frame_stack_.empty() && frame_stack_.back().is_object) {
245 skip_ws(c);
246 if constexpr (Validate) {
247 if (c.it == c.end || *c.it != '"') {
248 return false; // key must be a string
249 }
250 }
251 if (!parse_string<Builder::owns_strings>(c, frame_stack_.back().key)) {
252 return false;
253 }
254 skip_ws(c);
255 if constexpr (Validate) {
256 if (c.it == c.end || *c.it != ':') {
257 return false;
258 }
259 }
260 ++c.it; // skip ':'
261 }
262
263 // (B) Read one value. A scalar fills `value`; an opening '['/'{' begins a container and (if
264 // non-empty) pushes a frame and loops back to read its first child. An empty container
265 // is finished immediately and falls through to (C).
266 skip_ws(c);
267 if constexpr (Validate) {
268 if (c.it == c.end) {
269 return false;
270 }
271 }
272 bool opened = false;
273 switch (*c.it) {
274 case '[':
275 ++c.it;
276 b.begin_array();
277 skip_ws(c);
278 if constexpr (Validate) {
279 if (c.it == c.end) {
280 return false; // unterminated right after '['
281 }
282 }
283 if (*c.it == ']') {
284 ++c.it;
285 value = b.finish_array(); // empty array
286 } else {
287 frame_stack_.push_back(Frame{false, Node{}});
288 opened = true;
289 }
290 break;
291 case '{':
292 ++c.it;
293 b.begin_object();
294 skip_ws(c);
295 if constexpr (Validate) {
296 if (c.it == c.end) {
297 return false; // unterminated right after '{'
298 }
299 }
300 if (*c.it == '}') {
301 ++c.it;
302 value = b.finish_object(); // empty object
303 } else {
304 frame_stack_.push_back(Frame{true, Node{}});
305 opened = true;
306 }
307 break;
308 case '"':
309 if (!parse_string<Builder::owns_strings>(c, value)) {
310 return false;
311 }
312 break;
313 case 't':
314 if (!match(c, "true")) {
315 return false;
316 }
317 value.variant().emplace<Boolean>(true);
318 break;
319 case 'f':
320 if (!match(c, "false")) {
321 return false;
322 }
323 value.variant().emplace<Boolean>(false);
324 break;
325 case 'n':
326 if (!match(c, "null")) {
327 return false;
328 }
329 value.variant().emplace<Null>();
330 break;
331 default:
332 if (!parse_number(c, value)) { // digit or '-'
333 return false;
334 }
335 break;
336 }
337 if (opened) {
338 continue; // a non-empty container was opened: loop back to read its first child/key
339 }
340
341 // (C) `value` is complete. Attach it to the enclosing container, then ascend through every
342 // container that closes here (each closed container becomes the next `value`).
343 for (;;) {
344 if (frame_stack_.empty()) {
345 out = std::move(value); // `value` was the whole document
346 return true;
347 }
348 Frame& top = frame_stack_.back();
349 if (top.is_object) {
350 b.add_member(Member{std::move(top.key), std::move(value)});
351 } else {
352 b.add_element(std::move(value));
353 }
354 skip_ws(c);
355 if constexpr (Validate) {
356 if (c.it == c.end) {
357 return false;
358 }
359 }
360 const char sep = *c.it++;
361 const char closer = top.is_object ? '}' : ']';
362 if (sep == ',') {
363 break; // more children: loop back to (A)/(B)
364 }
365 if (sep == closer) {
366 value = top.is_object ? b.finish_object() : b.finish_array();
367 frame_stack_.pop_back();
368 continue; // ascend: the closed container is now `value` for ITS parent
369 }
370 return false; // expected ',' or the closing bracket/brace
371 }
372 }
373}

375// In the Validate=false instantiations `ok` is never referenced (the whole else branch below omits
376// it), so it is [[maybe_unused]]; the unchecked binary contains no *ok store at all.
377template <bool Validate>
378Document Parser::parse(std::string_view text, [[maybe_unused]] bool* ok) {
379 pool_.reset(text.size()); // reuse capacity; reserve a safe upper bound so spans never dangle
380
381 Cursor c{text.data(), text.data() + text.size()};
382 Node root; // default-constructed to null
383 if constexpr (Validate) {
384 bool good = parse_value<true>(c, root, pool_); // ArrayView/ObjectView into the reused pool
385 if (good) {
386 skip_ws(c);
387 good = (c.it == c.end); // reject trailing junk
388 }
389 if (!good) {
390 root.variant().emplace<Null>();
391 }
392 if (ok != nullptr) {
393 *ok = good;
394 }
395 } else {
396 // Trusted input: parse with NO validation — no bounds/structure checks, no trailing-junk
397 // check, and `ok` is not touched (an unchecked parse has no validity to report; malformed
398 // input is undefined behavior, not a reported error).
399 parse_value<false>(c, root, pool_);
400 }
401 return root;
402}

404template <bool Validate>
405Document Parser::parse_owning(std::string_view text, [[maybe_unused]] bool* ok) {
406 Cursor c{text.data(), text.data() + text.size()};
407 Node root; // default-constructed to null
408 OwningBuilder owning; // local vectors -> OwnedArray/OwnedObject (no references into this Parser)
409 if constexpr (Validate) {
410 bool good = parse_value<true>(c, root, owning);
411 if (good) {
412 skip_ws(c);
413 good = (c.it == c.end); // reject trailing junk
414 }
415 if (!good) {
416 root.variant().emplace<Null>();
417 }
418 if (ok != nullptr) {
419 *ok = good;
420 }
421 } else {
422 parse_value<false>(c, root, owning); // trusted input: no validation, `ok` left untouched
423 }
424 return root;
425}
