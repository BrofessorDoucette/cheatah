28/// Streamable<T>: T can be written to a `std::ostream` — the ONLY thing str(),
29/// print(), format() and File::write() actually need. Naming the requirement turns
30/// a deep `operator<<` instantiation error into a clear "constraint Streamable not
31/// satisfied" message, without narrowing what those templates already accept.
32template <typename T>
33concept Streamable = requires(std::ostream& os, const T& value) {
34 { os << value } -> std::convertible_to<std::ostream&>;
35};

37/**
38 * Python `str()`: stringify any streamable value.
39 * @param value the value to render.
40 * @return @p value formatted as text.
41 * @note O(n) in the output length; allocates the result string (via an ostringstream).
42 * @test CheatahIo.StrFormatsPythonStyle
43 */
44template <Streamable T>
45std::string str(const T& value) {
46 std::ostringstream os;
47 os << value;

67/**
68 * Python `print(*args)`: space-separated, newline-terminated, to stdout (sep=' ', end='\n').
69 * @param args zero or more streamable values.
70 * @note O(total output length); each arg is routed through str() so values format the Python
71 * way (e.g. bool → True/False), allocating temporary strings.
72 * @test CheatahIo.PrintWritesSpaceSeparatedLine, CheatahIo.PrintNoArgsIsJustNewline
73 */
74template <Streamable... Args>
75void print(const Args&... args) {
76 std::size_t i = 0;
77 ((std::cout << (i++ ? " " : "") << str(args)), ...);
78 std::cout << '\n';
79}

107namespace detail {
108/**
109 * Base case of the recursive formatter: emit the remainder of @p fmt verbatim.
110 * @param os destination stream.
111 * @param fmt remaining format text (no placeholders left to fill).
112 */
113void format_into(std::ostringstream& os, std::string_view fmt);
114/**
115 * Recursive step: write text up to the next `{}`, substitute @p arg, recurse on the rest.
116 * @param os destination stream.
117 * @param fmt remaining format text.
118 * @param arg value substituted at the next `{}` placeholder.
119 * @param rest values for the remaining placeholders.
120 */
121template <Streamable T, typename... Rest>
122void format_into(std::ostringstream& os, std::string_view fmt, const T& arg, const Rest&... rest) {
123 const std::size_t brace = fmt.find("{}");
124 if (brace == std::string_view::npos) {
125 os << fmt; // more args than placeholders — drop the extras
126 return;
127 }
128 os << fmt.substr(0, brace) << arg;
129 format_into(os, fmt.substr(brace + 2), rest...);
130}
131} // namespace detail

133/**
134 * Sequential `{}` substitution — the common case of Python's str.format() / f-strings.
135 * @param fmt format string with `{}` placeholders.
136 * @param args values substituted left-to-right (extras dropped, missing placeholders left
137 * as-is).
138 * @return the formatted string.
139 * @note O(len(fmt) + total arg output); allocates the result string (via an ostringstream).
140 * @test CheatahIo.FormatSubstitutesBraces, CheatahIo.FormatMultiArgAndExtraArgs
141 */
142template <Streamable... Args>
143std::string format(std::string_view fmt, const Args&... args) {
144 std::ostringstream os;
145 detail::format_into(os, fmt, args...);
146 return os.str();
147}
