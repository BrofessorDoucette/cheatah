25// Generate a throwaway self-signed cert (@p newkey selects the key algorithm — "ed25519",
26// "rsa:2048", "ec" …) and start `openssl s_server` on @p port restricted to @p ciphersuites.
27// Returns true when the server is accepting. Killed via pkill in stop(). This is what lets the
28// system tests exercise each leaf-cert algorithm (Ed25519 / RSA-PSS / ECDSA P-256) and each record
29// cipher (ChaCha20-Poly1305 / AES-128-GCM) against a real TLS 1.3 peer.
30// @complexity O(1) (two subprocesses) @alloc the command strings @test TlsSys (helper)
31class OpensslServer {
32public:
33 explicit OpensslServer(long long port, const std::string& newkey = "ed25519",
34 const std::string& ciphersuites = "TLS_CHACHA20_POLY1305_SHA256")
35 : port_(port),
36 cert_(std::string(PURR_TEST_TMP) + "/tls_test_cert.pem"),
37 key_(std::string(PURR_TEST_TMP) + "/tls_test_key.pem") {
38 const std::string gen = "openssl req -x509 -newkey " + newkey + " -keyout '" + key_ +
39 "' -out '" + cert_ +
40 "' -days 2 -nodes -subj /CN=localhost 2>/dev/null";
41 ok_ = std::system(gen.c_str()) == 0;
42 if (!ok_) return;
43 const std::string serve = "openssl s_server -accept " + std::to_string(port_) +
44 " -cert '" + cert_ + "' -key '" + key_ +
45 "' -tls1_3 -ciphersuites " + ciphersuites + " -www "
46 ">/dev/null 2>&1 &";
47 ok_ = std::system(serve.c_str()) == 0;
48 std::this_thread::sleep_for(std::chrono::milliseconds(700)); // let it bind
49 }
50 ~OpensslServer() {
51 const std::string kill = "pkill -f 's_server -accept " + std::to_string(port_) + "'";
52 std::system(kill.c_str());
53 }
54 [[nodiscard]] bool ok() const { return ok_; }
55 [[nodiscard]] long long port() const { return port_; }
56
57private:
58 long long port_;
59 std::string cert_, key_;
60 bool ok_ = false;
61};

65// The full handshake: X25519 exchange, transcript-verified Finished, Ed25519 CertificateVerify,
66// then an HTTP exchange over the encrypted channel (s_server -www answers GET with 200).
67TEST(TlsSys, HandshakeAgainstOpenssl) {
68 OpensslServer server(47931);
69 ASSERT_TRUE(server.ok()) << "could not start openssl s_server (test infrastructure)";
70 const long long fd = sock::tcp_connect("127.0.0.1", server.port());
71 ASSERT_GE(fd, 0);
72 sock::set_timeout(fd, 5000);
73 const long long s = tls::client_connect(fd, "localhost");
74 ASSERT_GE(s, 0) << tls::last_error();
75
76 ASSERT_EQ(tls::send(s, "GET / HTTP/1.0\r\n\r\n"), 0) << tls::last_error();
77 std::string all;
78 for (;;) {
79 const std::string chunk = tls::recv(s, 65536);
80 if (chunk.empty()) break;
81 all += chunk;
82 }
83 EXPECT_EQ(all.compare(0, 15, "HTTP/1.0 200 ok"), 0) << all.substr(0, 60);
84 tls::close(s);
85 sock::close(fd);
86}

88// The owning-guard round trip: socket::Conn + tls::Conn (the `with`-friendly RAII API) run the
89// same X25519 handshake + encrypted GET, but the fd and TLS session are released deterministically
90// by the guards' destructors. Exercises tls::open/Conn::send/recv/shutdown/close/is_open/id and
91// both move operations end-to-end against a real peer.
92TEST(TlsSys, ConnGuardRoundTrip) {
93 OpensslServer server(47960);
94 ASSERT_TRUE(server.ok()) << "could not start openssl s_server (test infrastructure)";
95 sock::Conn tcp = sock::open("127.0.0.1", server.port());
96 ASSERT_TRUE(tcp.is_open()) << sock::last_error();
97 tcp.set_timeout(5000);
98 tls::Conn conn = tls::open(tcp.fd(), "localhost");
99 ASSERT_TRUE(conn.is_open()) << tls::last_error();
100 EXPECT_GT(conn.id(), 0);
101
102 tls::Conn active(std::move(conn)); // move-construct: transfer ownership
103 EXPECT_FALSE(conn.is_open());
104 ASSERT_EQ(active.send("GET / HTTP/1.0\r\n\r\n"), 0) << tls::last_error();
105 std::string all;
106 for (;;) {
107 const std::string chunk = active.recv(65536);
108 if (chunk.empty()) break;
109 all += chunk;
110 }
111 EXPECT_EQ(all.compare(0, 15, "HTTP/1.0 200 ok"), 0) << all.substr(0, 60);
112
113 tls::Conn sink;
114 sink = std::move(active); // move-assign onto a closed guard
115 EXPECT_FALSE(active.is_open());
116 EXPECT_EQ(sink.shutdown(), 0);
117 EXPECT_EQ(sink.close(), 0);
118 EXPECT_EQ(sink.close(), -1); // idempotent
119 // `tcp` (socket::Conn) closes the fd via its destructor here — no leak.
120}

123// Connect to a local TLS 1.3 server, GET /, and return true iff it answered 200 over the encrypted
124// channel — i.e. the FULL cheatah handshake (key exchange + leaf-cert verify + record cipher) and an
125// encrypted round trip all succeeded. The per-cert/per-cipher system tests below assert on this.
126bool handshake_gets_200(long long port) {
127 const long long fd = sock::tcp_connect("127.0.0.1", port);
128 if (fd < 0) return false;
129 sock::set_timeout(fd, 5000);
130 const long long s = tls::client_connect(fd, "localhost");
131 if (s < 0) {
132 sock::close(fd);
133 return false;
134 }
135 tls::send(s, "GET / HTTP/1.0\r\n\r\n");
136 std::string all;
137 for (;;) {
138 const std::string c = tls::recv(s, 65536);
139 if (c.empty()) break;
140 all += c;
141 if (all.size() > 40) break;
142 }
143 tls::close(s);
144 sock::close(fd);
145 return all.compare(0, 12, "HTTP/1.0 200") == 0;
146}
147} // namespace

217// A server that, right after the client's ClientHello, sends a plaintext TLS alert record instead of
218// a ServerHello must fail the handshake with a NAMED alert. This is adversarial crafted-byte input
219// (a 7-byte record: content_type 21 || version 0303 || length 0002 || level || description) — NOT a
220// mirrored handshake — and it drives alert_text()'s human-readable name table over every RFC 8446
221// alert description the client recognizes, so a refusal reports a specific cause, not a bare code.
222TEST(TlsSys, NamesEveryHandshakeAlert) {
223 // description code -> the exact substring alert_text() must place in last_error().
224 const std::vector<std::pair<int, std::string>> alerts = {
225 {0, "close_notify"}, {10, "unexpected_message"},
226 {20, "bad_record_mac"}, {22, "record_overflow"},
227 {42, "bad_certificate"}, {43, "unsupported_certificate"},
228 {47, "illegal_parameter"}, {48, "unknown_ca"},
229 {49, "access_denied"}, {50, "decode_error"},
230 {51, "decrypt_error"}, {70, "protocol_version"},
231 {71, "insufficient_security"},{80, "internal_error"},
232 {109, "missing_extension"}, {110, "unsupported_extension"},
233 {112, "unrecognized_name"}, {116, "certificate_required"},
234 {120, "no_application_protocol"},
235 {200, "unknown"}, // an unlisted code -> the default "unknown" name
236 };
237 for (const auto& [code, name] : alerts) {
238 const long long listen_fd = sock::tcp_listen("127.0.0.1", 0, 4);
239 ASSERT_GE(listen_fd, 0);
240 const long long port = sock::local_port(listen_fd);
241 std::thread peer([listen_fd, code]() {
242 const long long client = sock::accept(listen_fd);
243 if (client < 0) return;
244 sock::recv(client, 16384); // drain the ClientHello, then answer with a fatal alert
245 // Build the 7-byte alert record explicitly — a "\x00" in a string literal would
246 // terminate it early, so append each byte (type 21 || ver 0303 || len 0002 || lvl || desc).
247 std::string alert;
248 alert.push_back(static_cast<char>(21)); // content_type = alert
249 alert.push_back(static_cast<char>(0x03)); // legacy record version 0x0303
250 alert.push_back(static_cast<char>(0x03));
251 alert.push_back(static_cast<char>(0x00)); // length = 2
252 alert.push_back(static_cast<char>(0x02));
253 alert.push_back(static_cast<char>(2)); // level = fatal
254 alert.push_back(static_cast<char>(code & 0xFF)); // description
255 sock::sendall(client, alert);
256 sock::close(client);
257 });
258 const long long fd = sock::tcp_connect("127.0.0.1", port);
259 ASSERT_GE(fd, 0);
260 sock::set_timeout(fd, 2000);
261 const long long s = tls::client_connect(fd, "localhost");
262 EXPECT_LT(s, 0) << "an alert instead of ServerHello must fail the handshake (code " << code << ")";
263 EXPECT_NE(tls::last_error().find(name), std::string::npos)
264 << "alert " << code << " should be named '" << name << "': " << tls::last_error();
265 sock::close(fd);
266 peer.join();
267 sock::close(listen_fd);
268 }
269}

271// A short (1-byte) alert record must not misparse: alert_text reports "(empty alert)" rather than
272// reading a description byte that is not there — the p.size() < 2 guard.
273TEST(TlsSys, ShortAlertRecordIsHandled) {
274 const long long listen_fd = sock::tcp_listen("127.0.0.1", 0, 4);
275 ASSERT_GE(listen_fd, 0);
276 const long long port = sock::local_port(listen_fd);
277 std::thread peer([listen_fd]() {
278 const long long client = sock::accept(listen_fd);
279 if (client < 0) return;
280 sock::recv(client, 16384);
281 std::string alert; // an alert record with length=1 (truncated body: level only, no desc)
282 alert.push_back(static_cast<char>(21)); // content_type = alert
283 alert.push_back(static_cast<char>(0x03)); // legacy record version 0x0303
284 alert.push_back(static_cast<char>(0x03));
285 alert.push_back(static_cast<char>(0x00)); // length = 1
286 alert.push_back(static_cast<char>(0x01));
287 alert.push_back(static_cast<char>(2)); // one byte only (level, no description)
288 sock::sendall(client, alert);
289 sock::close(client);
290 });
291 const long long fd = sock::tcp_connect("127.0.0.1", port);
292 ASSERT_GE(fd, 0);
293 sock::set_timeout(fd, 2000);
294 const long long s = tls::client_connect(fd, "localhost");
295 EXPECT_LT(s, 0);
296 EXPECT_NE(tls::last_error().find("(empty alert)"), std::string::npos) << tls::last_error();
297 sock::close(fd);
298 peer.join();
299 sock::close(listen_fd);
300}
