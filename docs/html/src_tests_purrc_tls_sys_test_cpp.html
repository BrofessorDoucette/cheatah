27// Generate a throwaway self-signed cert (@p newkey selects the key algorithm — "ed25519",
28// "rsa:2048", "ec" …) and start `openssl s_server` on @p port restricted to @p ciphersuites.
29// Returns true when the server is accepting. Killed via pkill in stop(). This is what lets the
30// system tests exercise each leaf-cert algorithm (Ed25519 / RSA-PSS / ECDSA P-256) and each record
31// cipher (ChaCha20-Poly1305 / AES-128-GCM) against a real TLS 1.3 peer.
32// @complexity O(1) (two subprocesses) @alloc the command strings @test TlsSys (helper)
33class OpensslServer {
34public:
35 explicit OpensslServer(long long port, const std::string& newkey = "ed25519",
36 const std::string& ciphersuites = "TLS_CHACHA20_POLY1305_SHA256")
37 // Per-port cert/key paths so concurrent TlsSys tests (ctest -j) don't clobber each other's
38 // files — the client now VALIDATES the cert, so a shared path would race into failures.
39 : port_(port),
40 cert_(std::string(PURR_TEST_TMP) + "/tls_test_cert_" + std::to_string(port) + ".pem"),
41 key_(std::string(PURR_TEST_TMP) + "/tls_test_key_" + std::to_string(port) + ".pem") {
42 // Self-signed, with a subjectAltName so it is a valid trust anchor for "localhost": passed
43 // to the client as a ca_file (or via $SSL_CERT_FILE) it authenticates itself.
44 const std::string gen = "openssl req -x509 -newkey " + newkey + " -keyout '" + key_ +
45 "' -out '" + cert_ +
46 "' -days 2 -nodes -subj /CN=localhost "
47 "-addext subjectAltName=DNS:localhost 2>/dev/null";
48 ok_ = std::system(gen.c_str()) == 0;
49 if (!ok_) return;
50 const std::string serve = "openssl s_server -accept " + std::to_string(port_) +
51 " -cert '" + cert_ + "' -key '" + key_ +
52 "' -tls1_3 -ciphersuites " + ciphersuites + " -www "
53 ">/dev/null 2>&1 &";
54 ok_ = std::system(serve.c_str()) == 0;
55 std::this_thread::sleep_for(std::chrono::milliseconds(700)); // let it bind
56 }
57 ~OpensslServer() {
58 const std::string kill = "pkill -f 's_server -accept " + std::to_string(port_) + "'";
59 std::system(kill.c_str());
60 }
61 [[nodiscard]] bool ok() const { return ok_; }
62 [[nodiscard]] long long port() const { return port_; }
63 [[nodiscard]] const std::string& cert_path() const { return cert_; }
64
65private:
66 long long port_;
67 std::string cert_, key_;
68 bool ok_ = false;
69};

73// The full handshake: X25519 exchange, transcript-verified Finished, Ed25519 CertificateVerify,
74// then an HTTP exchange over the encrypted channel (s_server -www answers GET with 200).
75TEST(TlsSys, HandshakeAgainstOpenssl) {
76 OpensslServer server(47931);
77 ASSERT_TRUE(server.ok()) << "could not start openssl s_server (test infrastructure)";
78 const long long fd = sock::tcp_connect("127.0.0.1", server.port());
79 ASSERT_GE(fd, 0);
80 sock::set_timeout(fd, 5000);
81 const long long s = tls::client_connect(fd, "localhost", false, server.cert_path());
82 ASSERT_GE(s, 0) << tls::last_error();
83
84 ASSERT_EQ(tls::send(s, "GET / HTTP/1.0\r\n\r\n"), 0) << tls::last_error();
85 std::string all;
86 for (;;) {
87 const std::string chunk = tls::recv(s, 65536);
88 if (chunk.empty()) break;
89 all += chunk;
90 }
91 EXPECT_EQ(all.compare(0, 15, "HTTP/1.0 200 ok"), 0) << all.substr(0, 60);
92 tls::close(s);
93 sock::close(fd);
94}

96// The owning-guard round trip: socket::Conn + tls::Conn (the `with`-friendly RAII API) run the
97// same X25519 handshake + encrypted GET, but the fd and TLS session are released deterministically
98// by the guards' destructors. Exercises tls::open/Conn::send/recv/shutdown/close/is_open/id and
99// both move operations end-to-end against a real peer.
100TEST(TlsSys, ConnGuardRoundTrip) {
101 OpensslServer server(47960);
102 ASSERT_TRUE(server.ok()) << "could not start openssl s_server (test infrastructure)";
103 sock::Conn tcp = sock::open("127.0.0.1", server.port());
104 ASSERT_TRUE(tcp.is_open()) << sock::last_error();
105 tcp.set_timeout(5000);
106 tls::Conn conn = tls::open(tcp.fd(), "localhost", false, server.cert_path());
107 ASSERT_TRUE(conn.is_open()) << tls::last_error();
108 EXPECT_GT(conn.id(), 0);
109
110 tls::Conn active(std::move(conn)); // move-construct: transfer ownership
111 EXPECT_FALSE(conn.is_open());
112 ASSERT_EQ(active.send("GET / HTTP/1.0\r\n\r\n"), 0) << tls::last_error();
113 std::string all;
114 for (;;) {
115 const std::string chunk = active.recv(65536);
116 if (chunk.empty()) break;
117 all += chunk;
118 }
119 EXPECT_EQ(all.compare(0, 15, "HTTP/1.0 200 ok"), 0) << all.substr(0, 60);
120
121 tls::Conn sink;
122 sink = std::move(active); // move-assign onto a closed guard
123 EXPECT_FALSE(active.is_open());
124 EXPECT_EQ(sink.shutdown(), 0);
125 EXPECT_EQ(sink.close(), 0);
126 EXPECT_EQ(sink.close(), -1); // idempotent
127 // `tcp` (socket::Conn) closes the fd via its destructor here — no leak.
128}

131// Connect to a local TLS 1.3 server, GET /, and return true iff it answered 200 over the encrypted
132// channel — i.e. the FULL cheatah handshake (key exchange + leaf-cert verify + record cipher) and an
133// encrypted round trip all succeeded. The per-cert/per-cipher system tests below assert on this.
134bool handshake_gets_200(long long port, const std::string& ca_file) {
135 const long long fd = sock::tcp_connect("127.0.0.1", port);
136 if (fd < 0) return false;
137 sock::set_timeout(fd, 5000);
138 // Verify the server against its own self-signed cert (SAN localhost) supplied as the CA.
139 const long long s = tls::client_connect(fd, "localhost", false, ca_file);
140 if (s < 0) {
141 sock::close(fd);
142 return false;
143 }
144 tls::send(s, "GET / HTTP/1.0\r\n\r\n");
145 std::string all;
146 for (;;) {
147 const std::string c = tls::recv(s, 65536);
148 if (c.empty()) break;
149 all += c;
150 if (all.size() > 40) break;
151 }
152 tls::close(s);
153 sock::close(fd);
154 return all.compare(0, 12, "HTTP/1.0 200") == 0;
155}

226// A server that, right after the client's ClientHello, sends a plaintext TLS alert record instead of
227// a ServerHello must fail the handshake with a NAMED alert. This is adversarial crafted-byte input
228// (a 7-byte record: content_type 21 || version 0303 || length 0002 || level || description) — NOT a
229// mirrored handshake — and it drives alert_text()'s human-readable name table over every RFC 8446
230// alert description the client recognizes, so a refusal reports a specific cause, not a bare code.
231TEST(TlsSys, NamesEveryHandshakeAlert) {
232 // description code -> the exact substring alert_text() must place in last_error().
233 const std::vector<std::pair<int, std::string>> alerts = {
234 {0, "close_notify"}, {10, "unexpected_message"},
235 {20, "bad_record_mac"}, {22, "record_overflow"},
236 {42, "bad_certificate"}, {43, "unsupported_certificate"},
237 {47, "illegal_parameter"}, {48, "unknown_ca"},
238 {49, "access_denied"}, {50, "decode_error"},
239 {51, "decrypt_error"}, {70, "protocol_version"},
240 {71, "insufficient_security"},{80, "internal_error"},
241 {109, "missing_extension"}, {110, "unsupported_extension"},
242 {112, "unrecognized_name"}, {116, "certificate_required"},
243 {120, "no_application_protocol"},
244 {200, "unknown"}, // an unlisted code -> the default "unknown" name
245 };
246 for (const auto& [code, name] : alerts) {
247 const long long listen_fd = sock::tcp_listen("127.0.0.1", 0, 4);
248 ASSERT_GE(listen_fd, 0);
249 const long long port = sock::local_port(listen_fd);
250 std::thread peer([listen_fd, code]() {
251 const long long client = sock::accept(listen_fd);
252 if (client < 0) return;
253 sock::recv(client, 16384); // drain the ClientHello, then answer with a fatal alert
254 // Build the 7-byte alert record explicitly — a "\x00" in a string literal would
255 // terminate it early, so append each byte (type 21 || ver 0303 || len 0002 || lvl || desc).
256 std::string alert;
257 alert.push_back(static_cast<char>(21)); // content_type = alert
258 alert.push_back(static_cast<char>(0x03)); // legacy record version 0x0303
259 alert.push_back(static_cast<char>(0x03));
260 alert.push_back(static_cast<char>(0x00)); // length = 2
261 alert.push_back(static_cast<char>(0x02));
262 alert.push_back(static_cast<char>(2)); // level = fatal
263 alert.push_back(static_cast<char>(code & 0xFF)); // description
264 sock::sendall(client, alert);
265 sock::close(client);
266 });
267 const long long fd = sock::tcp_connect("127.0.0.1", port);
268 ASSERT_GE(fd, 0);
269 sock::set_timeout(fd, 2000);
270 const long long s = tls::client_connect(fd, "localhost");
271 EXPECT_LT(s, 0) << "an alert instead of ServerHello must fail the handshake (code " << code << ")";
272 EXPECT_NE(tls::last_error().find(name), std::string::npos)
273 << "alert " << code << " should be named '" << name << "': " << tls::last_error();
274 sock::close(fd);
275 peer.join();
276 sock::close(listen_fd);
277 }
278}
