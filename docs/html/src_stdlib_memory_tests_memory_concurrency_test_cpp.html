2// Original work; see ACKNOWLEDGMENTS.md for the open-source ideas we build upon.
3// GOLD adversarial CONCURRENCY suite for the `memory` module (suite MemoryConcurrency).
4//
5// ┌───────────────────────────────────────────────────────────────────────────────────────────┐
6// │ TDD-RED until the Owner scheduling engine is implemented. These use REAL std::thread over a │
7// │ shared Owner, so they cannot even be *linked* (Owner::rread/rwrite are declared-only) — and │
8// │ an exception thrown inside a spawned thread would std::terminate the runner. They are │
9// │ therefore built ONLY behind the CMake option CHEATAH_BUILD_MEMORY_TESTS (default OFF), which │
10// │ we flip ON as we build the engine (red → green). See stdlib/memory/tests/README.md. │
11// └───────────────────────────────────────────────────────────────────────────────────────────┘
12//
13// Design principle (per the user's ask): drive the object from many threads in a way whose *final*
14// state is DETERMINISTIC even though the interleaving is not. We never do "a bunch of rotations and
15// expect the same result"; we use commuting/exclusive updates and interleaving-invariant predicates,
16// so a correct engine ALWAYS yields the exact expected answer and a broken one (lost updates, torn
17// reads, missed drains, non-exclusive writes) is caught deterministically.

34// Spawn `n` threads running `body(i)`, join all.
35template <class F>
36void run_threads(int n, F body) {
37 std::vector<std::thread> ts;
38 ts.reserve(n);
39 for (int i = 0; i < n; ++i) ts.emplace_back([=] { body(i); });
40 for (auto& t : ts) t.join();
41}
42constexpr int kWriters = 8;
43constexpr int kIters = 20'000; // enough to expose a lost-update race, fast enough for the gate
44} // namespace
45
46// ── 1. Exclusive writes never lose an update: 8×N increments == 8N, exactly ───────────────
47TEST(MemoryConcurrency, ManyWritersDeterministicSum) {
48 auto o = mem::own<long long>(0);
49 run_threads(kWriters, [&](int) {
50 for (int k = 0; k < kIters; ++k) {
51 auto w = o.rwrite().acquire(); // exclusive: read-modify-write cannot interleave
52 w.write(w.read() + 1);
53 }
54 });
55 EXPECT_EQ(o.rread().acquire().read(),
56 static_cast<long long>(kWriters) * kIters); // == 400000, deterministic
57}
58
59// ── 2. Owner<ndarray>: readers NEVER see a torn (non-uniform) array; final is exact ───────
60// The array starts uniform (all equal). Every writer adds 1 to *every* element under one write
61// lease, so under a correct exclusive lease the array is uniform at every quiescent point. Readers
62// assert uniformity — the value they see varies (nondeterministic), but "all elements equal" must
63// ALWAYS hold. A missed drain / non-exclusive write would let a reader observe a half-updated array.
64TEST(MemoryConcurrency, OwnerOfNdArrayStaysUniformAndSumsExactly) {
65 constexpr std::size_t N = 256;
66 auto o = mem::own(nd::basic_ndarray<long long>({N}, 0));
67 std::atomic<bool> torn{false};
68 std::atomic<bool> stop{false};
69
70 // 4 reader threads: continuously assert the array is uniform.
71 std::vector<std::thread> readers;
72 for (int r = 0; r < 4; ++r) {
73 readers.emplace_back([&] {
74 while (!stop.load()) {
75 auto lease = o.rread().acquire();
76 const auto& a = lease.read();
77 const long long first = a.at({0}); // const read: at() (operator[] is non-const)
78 for (std::size_t i = 1; i < a.size(); ++i)
79 if (a.at({i}) != first) { torn = true; return; }
80 }
81 });
82 }
83 // Writers: each adds 1 to every element, kIters/50 times.
84 const int rounds = kIters / 50;
85 run_threads(kWriters, [&](int) {
86 for (int k = 0; k < rounds; ++k) {
87 auto w = o.rwrite().acquire();
88 const std::size_t n = w.read().size();
89 for (std::size_t i = 0; i < n; ++i) w.write(i, w.read().at({i}) + 1); // indexed setter
90 }
91 });
92 stop = true;
93 for (auto& t : readers) t.join();
94
95 EXPECT_FALSE(torn.load()) << "a reader observed a partially-updated (non-uniform) array";
96 auto final = o.rread().acquire();
97 const long long expected = static_cast<long long>(kWriters) * rounds;
98 for (std::size_t i = 0; i < N; ++i)
99 ASSERT_EQ(final.read().at({i}), expected); // const read via at()
100}

102// ── 3. Readers never see a torn write of a multi-field invariant (sum == a + b) ───────────
103TEST(MemoryConcurrency, ReadersNeverSeeATornWrite) {
104 struct Triple { long long a = 0, b = 0, sum = 0; };
105 auto o = mem::own(Triple{});
106 std::atomic<bool> torn{false}, stop{false};
107
108 std::thread reader([&] {
109 while (!stop.load()) {
110 auto r = o.rread().acquire();
111 const Triple& t = r.read();
112 if (t.sum != t.a + t.b) { torn = true; return; } // invariant must hold at every read
113 }
114 });
115 run_threads(kWriters, [&](int id) {
116 for (int k = 0; k < kIters; ++k) {
117 const long long a = id * 1000 + k, b = k;
118 auto w = o.rwrite().acquire();
119 w.write(Triple{a, b, a + b}); // set all three at once (one exclusive write)
120 }
121 });
122 stop = true;
123 reader.join();
124 EXPECT_FALSE(torn.load()) << "a reader observed a half-written Triple (sum != a + b)";
125}

127// ── 4. Immediate-write preempts under load; its effect lands; normal writers still finish ─
128// Adds commute, so the final total is deterministic regardless of WHEN the immediate-write fires.
129TEST(MemoryConcurrency, ImmediateWriteLandsUnderLoad) {
130 auto o = mem::own<long long>(0);
131 std::atomic<bool> go{false};
132 std::thread emergency([&] {
133 while (!go.load()) std::this_thread::yield();
134 auto w = o.rwrite<mem::immediate>().acquire(); // jumps the queue, preempts active writer
135 w.write(w.read() + 1'000'000);
136 });
137 go = true;
138 run_threads(kWriters, [&](int) {
139 for (int k = 0; k < kIters; ++k) { auto w = o.rwrite().acquire(); w.write(w.read() + 1); }
140 });
141 emergency.join();
142 EXPECT_EQ(o.rread().acquire().read(),
143 static_cast<long long>(kWriters) * kIters + 1'000'000);
144}

146// ── 5. A thread that reads then writes in a loop must never self-deadlock ──────────────────
147TEST(MemoryConcurrency, ReadThenWriteLoopNoDeadlock) {
148 auto o = mem::own<long long>(0);
149 run_threads(kWriters, [&](int) {
150 for (int k = 0; k < kIters / 5; ++k) {
151 { auto r = o.rread().acquire(); (void)r.read(); } // read, release
152 { auto w = o.rwrite().acquire(); w.write(w.read() + 1); } // then write — must not wait on self
153 }
154 });
155 EXPECT_EQ(o.rread().acquire().read(), static_cast<long long>(kWriters) * (kIters / 5));
156}

158// ── 6. Renewal across relocation: writers grow a string (reallocating), readers never dangle ─
159TEST(MemoryConcurrency, RenewalAcrossRelocationNeverDangles) {
160 auto o = mem::own(std::string("a"));
161 std::atomic<bool> stop{false}, corrupt{false};
162 std::thread reader([&] {
163 while (!stop.load()) {
164 auto r = o.rread().acquire();
165 const std::string& s = r.read(); // must point at the CURRENT buffer
166 for (char c : s) if (c != 'a') { corrupt = true; return; } // every byte is 'a', never garbage
167 }
168 });
169 for (int k = 0; k < 2000; ++k) { auto w = o.rwrite().acquire(); w.write(w.read() + "a"); } // grows/relocates
170 stop = true;
171 reader.join();
172 EXPECT_FALSE(corrupt.load()) << "a renewed reader saw freed/garbage bytes after a relocation";
173 EXPECT_EQ(o.rread().acquire().read().size(), 2001u);
174}

176// ── 7. Lease churn: hammer create/destroy so ASan/Valgrind (gate stages) can catch a leak ─
177TEST(MemoryConcurrency, LeaseChurnLeakHunt) {
178 auto o = mem::own<long long>(0);
179 run_threads(kWriters, [&](int) {
180 for (int k = 0; k < kIters; ++k) {
181 if (k & 1) { auto r = o.rread().acquire(); (void)r.read(); }
182 else { auto w = o.rwrite().acquire(); w.write(w.read() + 1); }
183 }
184 });
185 // Value check is secondary; the real assertion is "no bytes leaked" under ASan/Valgrind.
186 EXPECT_GE(o.rread().acquire().read(), 0);
187}

189// ── 8. Concurrent readers coexist (shared), and a write still drains them ─────────────────
190TEST(MemoryConcurrency, ManyReadersCoexistThenAWriteDrains) {
191 auto o = mem::own<long long>(7);
192 std::atomic<int> peak{0}, active{0};
193 std::atomic<bool> stop{false};
194 run_threads(6, [&](int id) {
195 if (id == 0) {
196 for (int k = 0; k < 200; ++k) { auto w = o.rwrite().acquire(); w.write(w.read() + 1); }
197 stop = true;
198 } else {
199 while (!stop.load()) {
200 auto r = o.rread().acquire();
201 int a = ++active;
202 int seen = peak.load();
203 while (a > seen && !peak.compare_exchange_weak(seen, a)) {}
204 std::this_thread::yield();
205 --active;
206 EXPECT_GE(r.read(), 7); // monotonic: writes only increment
207 }
208 }
209 });
210 EXPECT_GT(peak.load(), 1) << "read leases should have coexisted (shared), not serialized";
211 EXPECT_EQ(o.rread().acquire().read(), 207);
