1// Copyright (c) 2026 BigBrain LLC. MIT-licensed (see LICENSE).
2// Original work; see ACKNOWLEDGMENTS.md for the open-source ideas we build upon.
3// biome CLI end-to-end tests (suite BiomeCli) — the "Biome Standard" behavior of the
4// package manager. pkg-manager/biome.purr is compiled ONCE per suite with purrc, then
5// each test runs `cheatah biome.so <args>` inside its own scratch directory under
6// PURR_TEST_TMP, so no test depends on another's state.
7//
8// NOTE on exit codes: biome's run() computes a meaningful status, but the top-level
9// `run()` call discards its return value, so the PROCESS exit code is 0 even on
10// refusals (a known pre-existing wart). Error-path tests therefore assert on the
11// printed stdout text, never on the exit code.
12#include "e2e_harness.hpp"
13
14#include <unistd.h>
15
16#include <filesystem>
17
18#ifndef BIOME_PURR_PATH
19#define BIOME_PURR_PATH ""
20#endif
21#ifndef BIOME_STANDARDS_DIR
22#define BIOME_STANDARDS_DIR ""
23#endif
24
25namespace {
26
27namespace fs = std::filesystem;
28
29class BiomeCli : public ::testing::Test {
30 protected:
31 // Per-process path: under `ctest -j` each test runs as its own process of this binary,
32 // and every process recompiles biome.purr in SetUpTestSuite — a shared path makes those
33 // compilations race (one process loads another's half-written .so).
34 static std::string module_path() {
35 return std::string(PURR_TEST_TMP) + "/biome_cli_" + std::to_string(getpid()) + ".so";
36 }
37
38 // Compile pkg-manager/biome.purr once for the whole suite.
39 static void SetUpTestSuite() {
40 const std::string cmd = std::string(PURRC_PATH) + " \"" + BIOME_PURR_PATH + "\" -o \"" +
41 module_path() + "\"";
42 ASSERT_EQ(std::system(cmd.c_str()), 0) << "purrc failed to compile biome.purr";
43 ASSERT_TRUE(fs::exists(module_path()));
44 }
45
46 // A fresh, empty scratch directory for one test.
47 static std::string scratch(const std::string& name) {
48 const fs::path dir = fs::path(PURR_TEST_TMP) / "biome_cli" / name;
49 fs::remove_all(dir);
50 fs::create_directories(dir);
51 return dir.string();
52 }
53
54 // Run `biome <args>` with `dir` as the working directory; captures stdout+stderr.
55 static std::string biome(const std::string& dir, const std::string& args, int& rc) {
56 const std::string cmd = "cd \"" + dir + "\" && \"" + std::string(CHEATAH_RUNTIME_PATH) +
57 "\" \"" + module_path() + "\" " + args + " 2>&1";
58 return e2e::run_capture(cmd, rc);
59 }
60 static std::string biome(const std::string& dir, const std::string& args) {
61 int rc = -1;
62 const std::string out = biome(dir, args, rc);
63 EXPECT_EQ(rc, 0) << "biome " << args << " did not run cleanly:\n" << out;
64 return out;
65 }
66
67 static void expect_has(const std::string& hay, const std::string& needle) {
68 EXPECT_NE(hay.find(needle), std::string::npos) << "missing `" << needle << "` in:\n" << hay;
69 }
70 static void expect_lacks(const std::string& hay, const std::string& needle) {
71 EXPECT_EQ(hay.find(needle), std::string::npos) << "unexpected `" << needle << "` in:\n" << hay;
72 }
73
74 // Scaffold a project with `biome init proj` inside a fresh scratch dir and return
75 // the project directory (<scratch>/proj).
76 static std::string init_project(const std::string& test_name) {
77 const std::string dir = scratch(test_name);
78 const std::string out = biome(dir, "init proj");
79 EXPECT_NE(out.find("biome: created project proj"), std::string::npos) << out;
80 return dir + "/proj";
81 }
82};

84// `biome standards` lists the known sets: the newest carries the `* ` marker, with its
85// members' resolved tags, status and release date. Earlier sets stay listed as
86// `supported` — the table is append-only, so a released standard never disappears, and
87// asserting BOTH rows is what proves that rather than just the newest.
88TEST_F(BiomeCli, StandardsListsKnownSets) {
89 const std::string out = biome(scratch("standards_list"), "standards");
90 expect_has(out, "* 0.3.0-alpha (current, released 2026-07-27)");
91 expect_has(out, "cheatah v1.9.0-alpha"); // ljust(name, 16) columns
92 expect_has(out, "cheatah-gpu v0.5.0-alpha");
93 expect_has(out, "0.1.0-alpha (supported, released 2026-07-25)");
94 expect_has(out, "0.2.0-alpha (supported, released 2026-07-27)");
95 expect_has(out, "cheatah v1.7.0-alpha");
96}

98// `--emit-toml <ver>` reproduces the committed standards/*.toml byte-for-byte — the
99// drift check between the in-program table and the canonical files.
100TEST_F(BiomeCli, StandardsEmitTomlByteIdenticalToCommittedFile) {
101 const std::string committed =
102 e2e::read_file(std::string(BIOME_STANDARDS_DIR) + "/biome-standard-0.1.0-alpha.toml");
103 ASSERT_FALSE(committed.empty()) << "missing standards/biome-standard-0.1.0-alpha.toml";
104 const std::string out = biome(scratch("emit_toml"), "standards --emit-toml 0.1.0-alpha");
105 EXPECT_EQ(out, committed);
106}

108TEST_F(BiomeCli, StandardsEmitTomlUnknownVersion) {
109 const std::string out = biome(scratch("emit_toml_bad"), "standards --emit-toml 9.9.9");
110 expect_has(out, "biome: unknown Biome Standard: 9.9.9");

113// `biome init` pins the newest standard: cheatah.toml gets `standard = "0.3.0-alpha"`
114// under [cheatah] and NO manual `version =` override; the generated CMake pins the
115// toolchain tag the standard resolves.
116TEST_F(BiomeCli, InitScaffoldsStandardPinnedProject) {
117 const std::string proj = init_project("init");
118 const std::string toml = e2e::read_file(proj + "/cheatah.toml");
119 ASSERT_FALSE(toml.empty());
120 expect_has(toml, "[cheatah]");
121 expect_has(toml, "standard = \"0.3.0-alpha\"");
122 expect_lacks(toml, "\nversion ="); // no manual toolchain override in a fresh project
123 const std::string cml = e2e::read_file(proj + "/CMakeLists.txt");
124 expect_has(cml, "GIT_TAG v1.9.0-alpha");
125 EXPECT_TRUE(fs::exists(proj + "/src/main.purr"));
126 EXPECT_TRUE(fs::exists(proj + "/cmake/CPM.cmake"));
127}

129// Adding a member extension resolves its tag from the standard: manifest value,
130// regenerated CMake, and the confirmation message all carry v0.5.0-alpha.
131TEST_F(BiomeCli, AddStandardMemberExtension) {
132 const std::string proj = init_project("add_gpu");
133 const std::string out = biome(proj, "add cheatah-gpu");
134 expect_has(out, "biome: added cheatah-gpu v0.5.0-alpha — GPU arrays and compute kernels");
135 expect_has(out, "(from Biome Standard 0.3.0-alpha)");
136 const std::string toml = e2e::read_file(proj + "/cheatah.toml");
137 expect_has(toml, "[extensions]");
138 expect_has(toml, "cheatah-gpu = \"v0.5.0-alpha\"");
139 const std::string cml = e2e::read_file(proj + "/CMakeLists.txt");
140 expect_has(cml,
141 "CPMAddPackage(NAME cheatah-gpu GITHUB_REPOSITORY BrofessorDoucette/cheatah-gpu "
142 "GIT_TAG v0.5.0-alpha)");
143}

145// A known extension that is NOT a member of the pinned standard is refused, and
146// nothing is written. (Exit code stays 0 — the run() wart — so assert text only.)
147TEST_F(BiomeCli, AddNonMemberExtensionRefused) {
148 const std::string proj = init_project("add_plot");
149 const std::string before = e2e::read_file(proj + "/cheatah.toml");
150 const std::string out = biome(proj, "add cheatah-plot");
151 expect_has(out, "biome: cheatah-plot is not a member of Biome Standard 0.3.0-alpha");
152 EXPECT_EQ(e2e::read_file(proj + "/cheatah.toml"), before) << "refusal must not edit the manifest";
153 expect_lacks(e2e::read_file(proj + "/CMakeLists.txt"), "cheatah-plot");
154}

163// `biome remove` takes the extension back out of both the manifest and the CMake.
164TEST_F(BiomeCli, RemoveExtension) {
165 const std::string proj = init_project("remove_gpu");
166 biome(proj, "add cheatah-gpu");
167 ASSERT_NE(e2e::read_file(proj + "/cheatah.toml").find("cheatah-gpu"), std::string::npos);
168 const std::string out = biome(proj, "remove cheatah-gpu");
169 expect_has(out, "biome: removed cheatah-gpu");
170 expect_lacks(e2e::read_file(proj + "/cheatah.toml"), "cheatah-gpu");
171 expect_lacks(e2e::read_file(proj + "/CMakeLists.txt"), "cheatah-gpu");
172}

174// `biome list` marks added members `* ` with the standard's tag, shows non-members as
175// "(not in this standard)", and names the pinned standard in the header.
176TEST_F(BiomeCli, ListShowsMembershipAndStandard) {
177 const std::string proj = init_project("list");
178 biome(proj, "add cheatah-gpu");
179 const std::string out = biome(proj, "list");
180 expect_has(out, "Biome Standard 0.3.0-alpha");
181 expect_has(out, "* cheatah-gpu v0.5.0-alpha"); // added member: marked + tagged

183 expect_has(out, " cheatah-space (not in this standard)");

186// A hand-edited manual `version =` under [cheatah] overrides the TOOLCHAIN tag only:
187// generated CMake pins cheatah at v1.6.0-alpha while extensions still resolve their
188// tags from the standard.
189TEST_F(BiomeCli, ManualVersionOverridesToolchainTagOnly) {
190 const std::string dir = scratch("manual_override");
191 std::ofstream(dir + "/cheatah.toml")
192 << "[project]\nname = \"proj\"\n\n"
193 "[cheatah]\nversion = \"1.6.0-alpha\"\nstandard = \"0.1.0-alpha\"\n\n"
194 "[extensions]\n\n[dependencies]\n";
195 const std::string out = biome(dir, "add cheatah-gpu");
196 expect_has(out, "biome: added cheatah-gpu v0.5.0-alpha");
197 const std::string cml = e2e::read_file(dir + "/CMakeLists.txt");
198 expect_has(cml,
199 "CPMAddPackage(NAME cheatah GITHUB_REPOSITORY BrofessorDoucette/cheatah "
200 "GIT_TAG v1.6.0-alpha)"); // the manual override
201 expect_has(cml,
202 "CPMAddPackage(NAME cheatah-gpu GITHUB_REPOSITORY BrofessorDoucette/cheatah-gpu "
203 "GIT_TAG v0.5.0-alpha)"); // still from the standard
204 const std::string toml = e2e::read_file(dir + "/cheatah.toml");
205 expect_has(toml, "version = \"1.6.0-alpha\""); // the override survives re-serialization
206 expect_has(toml, "standard = \"0.1.0-alpha\"");
207}

209// A manifest pinning a standard this biome does not know: `add` and `configure` both
210// refuse with the version named, and write NOTHING (no manifest edit, no CMakeLists,
211// no cmake run). (Process exit code is still 0 — see the run() wart above.)
212TEST_F(BiomeCli, UnknownStandardRefusesToWrite) {
213 const std::string dir = scratch("unknown_standard");
214 const std::string manifest =
215 "[project]\nname = \"proj\"\n\n[cheatah]\nstandard = \"9.9.9\"\n\n"
216 "[extensions]\n\n[dependencies]\n";
217 std::ofstream(dir + "/cheatah.toml") << manifest;
218
219 const std::string add_out = biome(dir, "add cheatah-gpu");
220 expect_has(add_out, "biome: unknown Biome Standard: 9.9.9");
221 EXPECT_EQ(e2e::read_file(dir + "/cheatah.toml"), manifest) << "add must not edit the manifest";
222 EXPECT_FALSE(fs::exists(dir + "/CMakeLists.txt"));
223
224 const std::string cfg_out = biome(dir, "configure");
225 expect_has(cfg_out, "biome: unknown Biome Standard: 9.9.9");
226 EXPECT_FALSE(fs::exists(dir + "/CMakeLists.txt")) << "configure must not generate on refusal";
227 EXPECT_FALSE(fs::exists(dir + "/build")) << "configure must not run cmake on refusal";
228}

230// A legacy manifest with only `version = "1.6.0-alpha"` (no standard key) parses with
231// the default standard (the newest); the next save writes the standard line out.
232TEST_F(BiomeCli, LegacyManifestDefaultsStandard) {
233 const std::string dir = scratch("legacy_manifest");
234 std::ofstream(dir + "/cheatah.toml")
235 << "[project]\nname = \"proj\"\n\n[cheatah]\nversion = \"1.6.0-alpha\"\n\n"
236 "[extensions]\n\n[dependencies]\n";
237 const std::string out = biome(dir, "add cheatah-gpu"); // loads, validates, saves
238 expect_has(out, "biome: added cheatah-gpu v0.5.0-alpha");
239 const std::string toml = e2e::read_file(dir + "/cheatah.toml");
240 expect_has(toml, "standard = \"0.3.0-alpha\""); // written on the first save
241 expect_has(toml, "version = \"1.6.0-alpha\""); // the legacy pin is preserved
242}
