3// cheatah::parsers::json — a from-scratch JSON parser (pure C++, no deps).
4//
5// This header declares the public API; the implementation lives in json.cpp. The value model is
6// Node (json/node.hpp) — a class wrapping a std::variant; access its alternatives via .variant().
7//
8// Two parse paths trade copying for lifetime (chosen by the Builder, no runtime polymorphism):
9// * Parser::parse (pooled) is ZERO-COPY — an unescaped string is a String<std::string_view> into
10// the SOURCE text, so `text` (and the Parser's pools) must outlive the Document.
11// * the free parse() / Parser::parse_owning produce a SELF-CONTAINED Document — strings are
12// copied into owned String<std::string> — safe to return, cache, or outlive `text`.
13// An escaped string is always decoded into owned storage. No runtime polymorphism anywhere.
14

16#include <span>

27/**
28 * Read a Node's characters when it is a string (either backing), else an empty view.
29 *
30 * @complexity O(1)
31 * @alloc none
32 * @test Json.ParseObject
33 */
34[[nodiscard]] std::string_view to_view(const Node& value) noexcept;
35
36/**
37 * Parse `text` into a SELF-CONTAINED Document (owning containers AND owned strings — every string
38 * is copied, not a view), so the result is safe to return, cache, or outlive `text`. On success
39 * *ok is set true; on malformed input *ok is set false and the result is JSON null. For the
40 * zero-copy, source-viewing form (no string copies), reuse a Parser and call Parser::parse.
41 *
42 * Validate is a COMPILE-TIME switch: the default (true) does full bounds/structure checking and
43 * rejects malformed input; parse<false>(...) strips every such check from the binary via
44 * `if constexpr` (see the Parser docs) and does not write *ok — for trusted, known-well-formed
45 * input only, as feeding it malformed input is undefined behavior.
46 *
47 * @complexity O(n) in the input length
48 * @alloc allocates the owned document tree (arrays, objects, and copied strings)
49 * @test Json.ParseObject
50 */
51template <bool Validate = true>
52[[nodiscard]] Document parse(std::string_view text, bool* ok = nullptr);
53
54/**
55 * Serialize a Document to compact JSON text (string contents are re-escaped).
56 *
57 * @complexity O(nodes)
58 * @alloc the returned string
59 * @test Json.DumpRoundTrip
60 */
61[[nodiscard]] std::string dump(const Document& value);
62
63/**
64 * Serialize a Document by APPENDING to the caller's buffer — stream into a preallocated/reused
65 * std::string rather than allocating a fresh one. This is the efficient path (push_back/append
66 * into one growing buffer); it deliberately avoids std::stringstream, which adds formatting,
67 * locale, and virtual-streambuf overhead per write. Reserve `out` once and reuse it across calls.
68 *
69 * @complexity O(nodes)
70 * @alloc none of its own (grows `out` only if its capacity is exceeded)
71 * @test Json.DumpRoundTrip
72 */
73void dump(const Document& value, std::string& out);

75/**
76 * A reusable parser that owns reusable node/member POOLS. Parser::parse() builds a Document whose
77 * arrays/objects are VIEWS (ArrayView/ObjectView) into these pools — zero per-container heap
78 * allocation. Reusing ONE Parser across many parses amortizes the pool allocation/page-faults to
79 * ~0 after warm-up (the reusable-parser model), which is the whole point of option B.
80 *
81 * LIFETIME: the returned Document VIEWS this Parser's pools, so it is valid only until the next
82 * parse() on this Parser, and only while the Parser is alive. (For a self-contained, owning
83 * Document — e.g. for the cache — use the free parse() above instead.) No runtime polymorphism.
84 *
85 * VALIDATION: every parse method takes a compile-time `bool Validate` template parameter, defaulted
86 * to true. With Validate=true the grammar checks bounds/structure and rejects malformed input
87 * (result JSON null, *ok=false). With Validate=false those checks are guarded by `if constexpr` and
88 * therefore removed from the binary ENTIRELY — there is no runtime flag and no branch, and *ok is
89 * not written at all (an unchecked parse has no validity to report). The unchecked form is for
90 * trusted, known-well-formed input (e.g. our own cache); feeding it malformed input is undefined
91 * behavior. Call it as p.parse<false>(text) / p.parse_owning<false>(text).
92 *
93 * @complexity O(n) in the input length
94 * @alloc the pools, reused across parses (amortized ~0 after warm-up); owned only for escaped
95 * strings
96 * @test Json.ParseObject
97 */
98class Parser {
99public:
100 /**
101 * Parse @p text into a Document whose arrays/objects are VIEWS (ArrayView/ObjectView) into this
102 * Parser's reused pools — zero per-container allocation, amortized to ~0 across parses. The
103 * result is valid only until the next parse or dump() on this Parser, and while the Parser lives.
104 * @tparam Validate when true (default) reject malformed input; when false all bounds/structure
105 * checks are compiled out (trusted, known-well-formed input only — see the class doc).
106 * @param text the JSON source to parse.
107 * @param ok if non-null, set to true on success and false on a parse error (only written when
108 * Validate is true).
109 * @return the parsed Document (JSON null on error when validating).
110 * @complexity O(|text|)
111 * @alloc none after warm-up (reused pools); owned only for escaped strings
112 * @test Json.ParseObject
113 */
114 template <bool Validate = true>
115 [[nodiscard]] Document parse(std::string_view text, bool* ok = nullptr);
116
117 /**
118 * Parse @p text into a self-contained OWNING Document (OwnedArray/OwnedObject AND owned
119 * String<std::string> — strings are copied, not views), fully independent of this Parser and of
120 * @p text once returned. This is what the free parse() and the cache use.
121 * @tparam Validate as for parse().
122 * @param text the JSON source to parse.
123 * @param ok if non-null, set to true on success and false on a parse error (Validate=true only).
124 * @return a self-contained parsed Document (JSON null on error when validating).
125 * @complexity O(|text|)
126 * @alloc allocates the owned document tree (arrays, objects, and copied strings)
127 * @test Json.ParseObject
128 */
129 template <bool Validate = true>
130 [[nodiscard]] Document parse_owning(std::string_view text, bool* ok = nullptr);
131
132 /**
133 * Serialize @p value into the Parser's own REUSED buffer and return a view of it — no per-call
134 * allocation after warm-up. The view is valid until the next dump() on this Parser.
135 * @param value the document to serialize.
136 * @return a std::string_view of the serialized JSON (valid until the next dump()).
137 * @complexity O(output size)
138 * @alloc none after warm-up (the buffer is reused)
139 * @test Json.ParseObject
140 */
141 [[nodiscard]] std::string_view dump(const Document& value);
142
143private:
144 // ONE ITERATIVE grammar (this IS "all parsing in one class") — the recursion is unrolled into a
145 // loop over an explicit frame_stack_, so nesting depth costs heap, not C++ call frames (no stack
146 // overflow on adversarially deep input). It is parameterized on (1) a compile-time `bool
147 // Validate` that `if constexpr`-gates every bounds/structure check, and (2) a Builder policy
148 // (PoolBuilder -> ArrayView/ObjectView, or OwningBuilder -> OwnedArray/OwnedObject) driven as a
149 // stack machine (begin/add/finish). Compile-time dispatch; no runtime polymorphism. Defined and
150 // instantiated for both validation modes and both builders in json.cpp.
151 template <bool Validate, class Builder>
152 bool parse_value(Cursor& c, Node& out, Builder& b);
153
154 // One open container on the parse stack. The Builder owns the partial container itself; the
155 // grammar only needs to know whether it is an object (so it reads keys) and, for an object, the
156 // pending key awaiting its value.
157 struct Frame {
158 bool is_object;
159 Node key; // the in-progress object key (unused for arrays)
160 };
161
162 PoolBuilder pool_; // pooled construction policy, reused across parse() (view path)
163 std::vector<Frame> frame_stack_; // explicit recursion stack, reused across parses
164 std::string dump_buf_; // reused serialization buffer for dump()
165};
