1// Copyright (c) 2026 BigBrain LLC. MIT-licensed (see LICENSE).
2// Original work; see ACKNOWLEDGMENTS.md for the open-source ideas we build upon.
3//
4// Golden-master behavior lock for the cheatah frontend (lexer -> parser -> codegen).
5//
6// The v1.6 compiler-speed work is a PURE PERFORMANCE refactor: for a fixed .purr
7// input and fixed flags, the frontend must emit BYTE-IDENTICAL C++ before and
8// after. This test pins that invariant. It links cheatah::compiler and drives the
9// frontend in-process — NO purrc fork, NO C++ backend — so it runs in milliseconds
10// and isolates the transpiler from the (much slower) backend compile.
11//
12// For every program in a fixed on-disk corpus it serializes the FULL CodegenResult
13// (source / header_source / impl_source / modules / diagnostics) across every flag
14// combination purrc uses, and compares the bytes to a committed golden fixture under
15// tests/purrc/golden/. Any difference = a forbidden behavior change.
16//
17// Capture the goldens ONCE, on the pre-refactor baseline:
18// CHEATAH_GOLDEN_UPDATE=1 ctest -R 'golden-master'
19// then `git add tests/purrc/golden` and commit. Never regenerate during the
20// refactor — a diff is the point.
21//
22// The corpus (real .purr from the tree) spans structs / enums / interfaces / match /
23// kwargs / cpp-blocks / doc-comments / imports / multiline arrays — the branches most
24// likely to shift output under refactor. It is deliberately fork-free and fast so it
25// can gate every optimization commit with zero tolerance.

91std::string read_file(const std::string& path) {
92 std::ifstream f(path, std::ios::binary);
93 std::ostringstream ss;
94 ss << f.rdbuf();
95 return ss.str();
96}
97
98bool write_file(const std::string& path, const std::string& content) {
99 std::ofstream f(path, std::ios::binary | std::ios::trunc);
100 if (!f) return false;
101 f << content;
102 return static_cast<bool>(f);
103}

105// Slug for the golden filename: the relative path with separators/dots flattened.
106std::string slug(const std::string& rel) {
107 std::string s;
108 s.reserve(rel.size());
109 for (char c : rel) s += (c == '/' || c == '.') ? '_' : c;
110 return s;
111}
112
113// The module name codegen_library wraps output in, derived from the file stem.
114std::string module_name_of(const std::string& rel) {
115 const std::size_t slash = rel.find_last_of('/');
116 const std::size_t start = (slash == std::string::npos) ? 0 : slash + 1;
117 const std::size_t dot = rel.find('.', start);
118 return rel.substr(start, dot == std::string::npos ? std::string::npos : dot - start);
119}

121// Append one CodegenResult, fully, under a labeled section. Every observable field
122// is serialized so the lock covers diagnostics and module lists, not just source.
123void append_result(std::string& out, const std::string& label, const cheatah::CodegenResult& cg) {
124 out += "===== " + label + " =====\n";
125 out += "--- diagnostics (" + std::to_string(cg.diagnostics.size()) + ") ---\n";
126 for (const std::string& d : cg.diagnostics) out += d + "\n";
127 out += "--- modules (" + std::to_string(cg.modules.size()) + ") ---\n";
128 for (const std::string& m : cg.modules) out += m + "\n";
129 out += "--- source (" + std::to_string(cg.source.size()) + " bytes) ---\n";
130 out += cg.source;
131 out += "\n--- header_source (" + std::to_string(cg.header_source.size()) + " bytes) ---\n";
132 out += cg.header_source;
133 out += "\n--- impl_source (" + std::to_string(cg.impl_source.size()) + " bytes) ---\n";
134 out += cg.impl_source;
135 out += "\n";
136}

138// Serialize every flag combination purrc drives for this entry into one blob. The
139// combos are the distinct emit paths: remove_unused on/off and the #line path
140// (source_file empty vs set) for programs; transparent/opaque for libraries. Parsing
141// (incl. its diagnostics) is captured once up front and shared across combos.
142std::string serialize_entry(const CorpusEntry& e, const std::string& src) {
143 const cheatah::ParseResult pr = cheatah::parse_source(src);
144
145 std::string out;
146 out += "######## " + std::string(e.path) + " ########\n";
147 out += "parse diagnostics (" + std::to_string(pr.diagnostics.size()) + "):\n";
148 for (const cheatah::Diagnostic& d : pr.diagnostics)
149 out += " " + std::to_string(d.pos.line) + ":" + std::to_string(d.pos.column) + ": " + d.message + "\n";
150
151 if (e.mode == Mode::Program) {
152 for (bool remove_unused : {true, false}) {
153 for (const char* sf : {"", e.path}) {
154 const cheatah::CodegenResult cg = cheatah::codegen(pr.program, sf, remove_unused);
155 const std::string label = std::string("program remove_unused=") + (remove_unused ? "1" : "0") +
156 " source_file=" + (sf[0] ? "set" : "empty");
157 append_result(out, label, cg);
158 }
159 }
160 } else {
161 for (bool transparent : {true, false}) {
162 cheatah::LibOptions opts;
163 opts.module_name = module_name_of(e.path);
164 opts.transparent = transparent;
165 const cheatah::CodegenResult cg = cheatah::codegen_library(pr.program, opts);
166 append_result(out, std::string("library transparent=") + (transparent ? "1" : "0"), cg);
167 }
168 }
169 return out;
170}

172bool update_mode() {
173 const char* v = std::getenv("CHEATAH_GOLDEN_UPDATE");
174 return v != nullptr && v[0] == '1';
175}

177class GoldenMaster : public ::testing::TestWithParam<CorpusEntry> {};
178
179TEST_P(GoldenMaster, EmittedCppIsByteIdentical) {
180 const CorpusEntry e = GetParam();
181 const std::string src_path = std::string(CHEATAH_SRC_ROOT) + "/" + e.path;
182 const std::string src = read_file(src_path);
183 ASSERT_FALSE(src.empty()) << "corpus source missing or empty: " << src_path;
184
185 const std::string actual = serialize_entry(e, src);
186 const std::string golden_path = std::string(GOLDEN_DIR) + "/" + slug(e.path) + ".golden";
187
188 if (update_mode()) {
189 ASSERT_TRUE(write_file(golden_path, actual)) << "cannot write golden: " << golden_path;
190 GTEST_SKIP() << "wrote golden " << golden_path;
191 }
192
193 const std::string expected = read_file(golden_path);
194 ASSERT_FALSE(expected.empty())
195 << "golden missing: " << golden_path
196 << "\n capture it with: CHEATAH_GOLDEN_UPDATE=1 ctest -R golden-master";
197 EXPECT_EQ(actual, expected)
198 << "FRONTEND OUTPUT DRIFTED for " << e.path
199 << " — the refactor changed emitted C++. This is a behavior change and must be reverted."
200 << "\n (If the change is intentional AND reviewed, re-capture with CHEATAH_GOLDEN_UPDATE=1.)";
201}
202
203std::string entry_name(const ::testing::TestParamInfo<CorpusEntry>& info) {
204 return slug(info.param.path);
205}

207INSTANTIATE_TEST_SUITE_P(Corpus, GoldenMaster, ::testing::ValuesIn(kCorpus), entry_name);
