10TEST(CheatahParser, ParsesImportAndModuleCall) {
11 const ParseResult r = parse_source("import io\nio.print(\"meow\")\n");
12 ASSERT_TRUE(r.ok()) << (r.diagnostics.empty() ? "" : r.diagnostics.front().message);
13 ASSERT_EQ(r.program.body.size(), 2u);
14
15 auto* imp = dynamic_cast<Import*>(r.program.body[0].get());
16 ASSERT_NE(imp, nullptr);
17 ASSERT_EQ(imp->module.size(), 1u);
18 EXPECT_EQ(imp->module[0], "io");
19 EXPECT_TRUE(imp->alias.empty());
20
21 auto* es = dynamic_cast<ExprStmt*>(r.program.body[1].get());
22 ASSERT_NE(es, nullptr);
23 auto* call = dynamic_cast<Call*>(es->expr.get());
24 ASSERT_NE(call, nullptr);
25 ASSERT_EQ(call->args.size(), 1u);
26 auto* arg = dynamic_cast<StringLit*>(call->args[0].get());
27 ASSERT_NE(arg, nullptr);
28 EXPECT_EQ(arg->value, "meow");
29
30 auto* member = dynamic_cast<Member*>(call->callee.get());
31 ASSERT_NE(member, nullptr);
32 EXPECT_EQ(member->name, "print");
33 auto* obj = dynamic_cast<Ident*>(member->object.get());
34 ASSERT_NE(obj, nullptr);
35 EXPECT_EQ(obj->name, "io");
36}

133TEST(CheatahParser, OperatorPrecedence) {
134 // 1 + 2 * 3 -> (1 + (2 * 3))
135 const ParseResult r = parse_source("let x = 1 + 2 * 3\n");
136 ASSERT_TRUE(r.ok());
137 auto* let = dynamic_cast<Let*>(r.program.body[0].get());
138 ASSERT_NE(let, nullptr);
139 auto* top = dynamic_cast<Binary*>(let->value.get());
140 ASSERT_NE(top, nullptr);
141 EXPECT_EQ(top->op, "+");
142 auto* rhs = dynamic_cast<Binary*>(top->rhs.get());
143 ASSERT_NE(rhs, nullptr);
144 EXPECT_EQ(rhs->op, "*"); // multiplication binds tighter
145}

147// Regression: a stray top-level `}` (e.g. from an unrecognized continuation) must
148// produce a diagnostic and RETURN — parse_stmts must never spin forever on a token
149// that synchronize() can't advance past (which used to OOM the compiler).
150TEST(CheatahParser, StrayTopLevelBraceErrorsWithoutHanging) {
151 const ParseResult r = parse_source("import io\nio.print(1)\n}\n");
152 EXPECT_FALSE(r.ok()); // reports the stray brace
153 EXPECT_FALSE(r.diagnostics.empty());
154 EXPECT_GE(r.program.body.size(), 2u); // the valid statements before it still parsed
155}

157// Multi-line `elif`/`else` (on their own lines after the `if` block) parse like the
158// same-line form: the `elif` becomes a chained If in the else branch.
159TEST(CheatahParser, ParsesMultilineElifElse) {
160 const ParseResult r = parse_source(
161 "let r = 0\nif r < 0 {\nr = 1\n}\nelif r == 0 {\nr = 2\n}\nelse {\nr = 3\n}\n");
162 ASSERT_TRUE(r.ok()) << (r.diagnostics.empty() ? "" : r.diagnostics.front().message);
163 auto* iff = dynamic_cast<If*>(r.program.body.back().get());
164 ASSERT_NE(iff, nullptr);
165 ASSERT_EQ(iff->else_body.size(), 1u);
166 EXPECT_NE(dynamic_cast<If*>(iff->else_body[0].get()), nullptr);
167}
