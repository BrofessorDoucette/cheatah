8TEST(CheatahParser, ParsesImportAndModuleCall) {
9 const ParseResult r = parse_source("import io\nio.print(\"meow\")\n");
10 ASSERT_TRUE(r.ok()) << (r.diagnostics.empty() ? "" : r.diagnostics.front().message);
11 ASSERT_EQ(r.program.body.size(), 2u);
12
13 auto* imp = dynamic_cast<Import*>(r.program.body[0].get());
14 ASSERT_NE(imp, nullptr);
15 ASSERT_EQ(imp->module.size(), 1u);
16 EXPECT_EQ(imp->module[0], "io");
17 EXPECT_TRUE(imp->alias.empty());
18
19 auto* es = dynamic_cast<ExprStmt*>(r.program.body[1].get());
20 ASSERT_NE(es, nullptr);
21 auto* call = dynamic_cast<Call*>(es->expr.get());
22 ASSERT_NE(call, nullptr);
23 ASSERT_EQ(call->args.size(), 1u);
24 auto* arg = dynamic_cast<StringLit*>(call->args[0].get());
25 ASSERT_NE(arg, nullptr);
26 EXPECT_EQ(arg->value, "meow");
27
28 auto* member = dynamic_cast<Member*>(call->callee.get());
29 ASSERT_NE(member, nullptr);
30 EXPECT_EQ(member->name, "print");
31 auto* obj = dynamic_cast<Ident*>(member->object.get());
32 ASSERT_NE(obj, nullptr);
33 EXPECT_EQ(obj->name, "io");
34}

131TEST(CheatahParser, OperatorPrecedence) {
132 // 1 + 2 * 3 -> (1 + (2 * 3))
133 const ParseResult r = parse_source("let x = 1 + 2 * 3\n");
134 ASSERT_TRUE(r.ok());
135 auto* let = dynamic_cast<Let*>(r.program.body[0].get());
136 ASSERT_NE(let, nullptr);
137 auto* top = dynamic_cast<Binary*>(let->value.get());
138 ASSERT_NE(top, nullptr);
139 EXPECT_EQ(top->op, "+");
140 auto* rhs = dynamic_cast<Binary*>(top->rhs.get());
141 ASSERT_NE(rhs, nullptr);
142 EXPECT_EQ(rhs->op, "*"); // multiplication binds tighter
143}

145// Regression: a stray top-level `}` (e.g. from an unrecognized continuation) must
146// produce a diagnostic and RETURN — parse_stmts must never spin forever on a token
147// that synchronize() can't advance past (which used to OOM the compiler).
148TEST(CheatahParser, StrayTopLevelBraceErrorsWithoutHanging) {
149 const ParseResult r = parse_source("import io\nio.print(1)\n}\n");
150 EXPECT_FALSE(r.ok()); // reports the stray brace
151 EXPECT_FALSE(r.diagnostics.empty());
152 EXPECT_GE(r.program.body.size(), 2u); // the valid statements before it still parsed
153}
