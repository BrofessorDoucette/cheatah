5/**
6 * @file p384.hpp
7 * @brief cheatah `p384` — the NIST P-384 (secp384r1) elliptic curve: ECDSA
8 * signature VERIFICATION plus the DER/SPKI parsing needed to use it with
9 * TLS certificates. From scratch, no external libraries. `import p384`
10 * to use it.
11 *
12 * P-384 is the curve the big CAs' issuing chains are signed with (Sectigo,
13 * DigiCert, GlobalSign ECC roots), so it lets cheatah's TLS client validate the
14 * `ecdsa-with-SHA384` certificate chains real CDN-fronted hosts present, and
15 * P-384 leaf certificates (`ecdsa_secp384r1_sha384` CertificateVerify).
16 *
17 * Verify-only by design: certificate validation handles PUBLIC data, which is
18 * all TLS needs from this curve (cheatah's own TLS server signs with Ed25519,
19 * and JWT signing uses P-256/ES256). The field/point machinery is the shared
20 * width-generic core in `p256/ec_core.hpp` — the same battle-tested Montgomery
21 * arithmetic and Jacobian group law p256 runs, instantiated at 6 limbs.
22 *
23 * Byte conventions: scalars/coordinates are 48 big-endian bytes; an uncompressed
24 * public key point is the 96 bytes X||Y; a raw ECDSA signature is the 96 bytes
25 * r||s; a DER signature is SEQUENCE{ INTEGER r, INTEGER s } (the TLS/X.509 form).
26 */
27
28#include <string>
29#include <string_view>
30
31namespace cheatah::p384 {
32
33/**
34 * Verify an ECDSA/P-384 signature.
35 * @param pubkey_xy the public key as 96 bytes (X||Y), big-endian.
36 * @param msg_hash the message digest (e.g. 48 bytes of SHA-384); longer digests are
37 * truncated to the leftmost 384 bits (FIPS 186-4), shorter ones (e.g. SHA-256)
38 * are taken whole (X9.62 bits2int).
39 * @param sig_der the signature, DER-encoded SEQUENCE{INTEGER r, INTEGER s}.
40 * @return true iff the signature is valid for @p pubkey_xy over @p msg_hash.
41 * @complexity O(1) — two scalar multiplications, computed as one Strauss-Shamir double chain.
42 * @alloc a temporary raw r||s signature string.
43 * @test CheatahP384.VerifyKnownVector
44 */
45[[nodiscard]] bool verify_der(const std::string& pubkey_xy, const std::string& msg_hash,
46 const std::string& sig_der);
47
48/**
49 * Verify an ECDSA/P-384 signature given the raw 96-byte r||s form (the JWT
50 * ES384 layout).
51 * @param pubkey_xy the public key (96 bytes X||Y).
52 * @param msg_hash the digest (truncated to 384 bits if longer, taken whole if shorter).
53 * @param sig_raw the signature as 96 bytes r||s.
54 * @return true iff valid.
55 * @complexity O(1) — two scalar multiplications, computed as one Strauss-Shamir double chain.
56 * @alloc none.
57 * @test CheatahP384.VerifyKnownVector
58 */
59[[nodiscard]] bool verify_raw(const std::string& pubkey_xy, const std::string& msg_hash,
60 const std::string& sig_raw);
61
62/**
63 * Extract the P-384 public key point from a certificate / SubjectPublicKeyInfo
64 * DER: finds the uncompressed EC point of a `secp384r1` key and returns its
65 * 96 bytes (X||Y). Returns "" if the SPKI is not a P-384 uncompressed EC key.
66 * @param spki_or_cert_der the certificate or SPKI DER bytes.
67 * @return the 96-byte X||Y point, or "" if not a P-384 EC key.
68 * @complexity O(der length).
69 * @alloc the returned point.
70 * @test CheatahP384.SpkiExtractsPoint
71 */
72[[nodiscard]] std::string spki_ec_point(std::string_view spki_or_cert_der);
73
74} // namespace cheatah::p384
