24std::vector<double> as_matrix(const NDArray& a, std::size_t& rows, std::size_t& cols) {
25 if (a.ndim() != 2) throw std::runtime_error("linalg: expected a 2-D matrix");
26 rows = a.shape()[0];
27 cols = a.shape()[1];
28 std::vector<double> m(rows * cols);
29 for (std::size_t i = 0; i < rows; ++i)
30 for (std::size_t j = 0; j < cols; ++j) m[i * cols + j] = a.at({i, j});
31 return m;
32}
33std::vector<double> as_vector(const NDArray& a, std::size_t& n) {
34 if (a.ndim() == 1) {
35 n = a.shape()[0];
36 std::vector<double> v(n);
37 for (std::size_t i = 0; i < n; ++i) v[i] = a.at({i});
38 return v;
39 }
40 if (a.ndim() == 2 && (a.shape()[0] == 1 || a.shape()[1] == 1)) {
41 n = a.size();
42 std::vector<double> v;
43 v.reserve(n);
44 for (std::size_t i = 0; i < a.shape()[0]; ++i)
45 for (std::size_t j = 0; j < a.shape()[1]; ++j) v.push_back(a.at({i, j}));
46 return v;

48 throw std::runtime_error("linalg: expected a 1-D vector");

64// ---- LU decomposition with partial pivoting (in place on a copy) ----
65struct LU {
66 std::vector<double> a; // L (below diag, unit) + U (diag/above), row-major n×n
67 std::vector<std::size_t> piv;
68 double sign;
69 std::size_t n;
70 bool singular;
71};
72LU lu_decompose(std::vector<double> a, std::size_t n) {
73 std::vector<std::size_t> piv(n);
74 std::vector<double> vv(n);
75 double sign = 1.0;
76 bool singular = false;
77 for (std::size_t i = 0; i < n; ++i) {
78 double big = 0.0;
79 for (std::size_t j = 0; j < n; ++j) big = std::max(big, std::fabs(a[i * n + j]));
80 if (big == 0.0) { singular = true; big = 1.0; }
81 vv[i] = 1.0 / big;
82 }
83 for (std::size_t k = 0; k < n; ++k) {
84 double big = 0.0;
85 std::size_t imax = k;
86 for (std::size_t i = k; i < n; ++i) {
87 const double t = vv[i] * std::fabs(a[i * n + k]);
88 if (t > big) { big = t; imax = i; }
89 }
90 if (k != imax) {
91 for (std::size_t j = 0; j < n; ++j) std::swap(a[imax * n + j], a[k * n + j]);
92 sign = -sign;
93 vv[imax] = vv[k];
94 }
95 piv[k] = imax;
96 if (a[k * n + k] == 0.0) { a[k * n + k] = 1e-300; singular = true; }
97 for (std::size_t i = k + 1; i < n; ++i) {
98 const double f = a[i * n + k] / a[k * n + k];
99 a[i * n + k] = f;
100 for (std::size_t j = k + 1; j < n; ++j) a[i * n + j] -= f * a[k * n + j];
101 }
102 }
103 return {std::move(a), std::move(piv), sign, n, singular};
104}
105void lu_solve(const LU& lu, std::vector<double>& b) {
106 const std::size_t n = lu.n;
107 for (std::size_t k = 0; k < n; ++k) std::swap(b[k], b[lu.piv[k]]);
108 for (std::size_t i = 0; i < n; ++i) { // forward (unit L)
109 double s = b[i];
110 for (std::size_t j = 0; j < i; ++j) s -= lu.a[i * n + j] * b[j];
111 b[i] = s;
112 }
113 for (std::size_t i = n; i-- > 0;) { // back (U)
114 double s = b[i];
115 for (std::size_t j = i + 1; j < n; ++j) s -= lu.a[i * n + j] * b[j];
116 b[i] = s / lu.a[i * n + i];
117 }
118}

120// ---- one-sided Jacobi SVD: A(m×n) = U(m×n) diag(w) V(n×n)ᵀ ----
121struct SVDc {
122 std::vector<double> u, w, v;
123 std::size_t m, n;
124};
125SVDc svd_jacobi(std::vector<double> b, std::size_t m, std::size_t n) {
126 std::vector<double> v(n * n, 0.0);
127 for (std::size_t i = 0; i < n; ++i) v[i * n + i] = 1.0;
128 const double eps = 1e-15;
129 for (int sweep = 0; sweep < 80; ++sweep) {
130 bool rotated = false;
131 for (std::size_t p = 0; p < n; ++p) {
132 for (std::size_t q = p + 1; q < n; ++q) {
133 double alpha = 0, beta = 0, gamma = 0;
134 for (std::size_t i = 0; i < m; ++i) {
135 const double bp = b[i * n + p], bq = b[i * n + q];
136 alpha += bp * bp;
137 beta += bq * bq;
138 gamma += bp * bq;
139 }
140 if (gamma == 0.0 || std::fabs(gamma) <= eps * std::sqrt(alpha * beta)) continue;
141 rotated = true;
142 const double zeta = (beta - alpha) / (2.0 * gamma);
143 const double t = (zeta >= 0 ? 1.0 : -1.0) / (std::fabs(zeta) + std::sqrt(zeta * zeta + 1.0));
144 const double c = 1.0 / std::sqrt(t * t + 1.0), s = c * t;
145 for (std::size_t i = 0; i < m; ++i) {
146 const double bp = b[i * n + p], bq = b[i * n + q];
147 b[i * n + p] = c * bp - s * bq;
148 b[i * n + q] = s * bp + c * bq;
149 }
150 for (std::size_t i = 0; i < n; ++i) {
151 const double vp = v[i * n + p], vq = v[i * n + q];
152 v[i * n + p] = c * vp - s * vq;
153 v[i * n + q] = s * vp + c * vq;
154 }
155 }
156 }
157 if (!rotated) break;
158 }
159 std::vector<double> w(n), u(m * n);
160 for (std::size_t j = 0; j < n; ++j) {
161 double nrm = 0.0;
162 for (std::size_t i = 0; i < m; ++i) nrm += b[i * n + j] * b[i * n + j];
163 nrm = std::sqrt(nrm);
164 w[j] = nrm;
165 for (std::size_t i = 0; i < m; ++i) u[i * n + j] = nrm > 0 ? b[i * n + j] / nrm : 0.0;
166 }
167 std::vector<std::size_t> idx(n);
168 for (std::size_t i = 0; i < n; ++i) idx[i] = i;
169 std::sort(idx.begin(), idx.end(), [&](std::size_t x, std::size_t y) { return w[x] > w[y]; });
170 SVDc out{std::vector<double>(m * n), std::vector<double>(n), std::vector<double>(n * n), m, n};
171 for (std::size_t j = 0; j < n; ++j) {
172 const std::size_t src = idx[j];
173 out.w[j] = w[src];
174 for (std::size_t i = 0; i < m; ++i) out.u[i * n + j] = u[i * n + src];
175 for (std::size_t i = 0; i < n; ++i) out.v[i * n + j] = v[i * n + src];
176 }
177 return out;
178}

180// ---- cyclic Jacobi for a real symmetric matrix ----
181void jacobi_symmetric(std::vector<double> a, std::size_t n, std::vector<double>& values,
182 std::vector<double>& vectors) {
183 std::vector<double> v(n * n, 0.0);
184 for (std::size_t i = 0; i < n; ++i) v[i * n + i] = 1.0;
185 for (int sweep = 0; sweep < 100; ++sweep) {
186 double off = 0.0;
187 for (std::size_t p = 0; p < n; ++p)
188 for (std::size_t q = p + 1; q < n; ++q) off += a[p * n + q] * a[p * n + q];
189 if (off == 0.0) break;
190 for (std::size_t p = 0; p < n; ++p) {
191 for (std::size_t q = p + 1; q < n; ++q) {
192 const double apq = a[p * n + q];
193 if (apq == 0.0) continue;
194 const double theta = (a[q * n + q] - a[p * n + p]) / (2.0 * apq);
195 const double t = (theta >= 0 ? 1.0 : -1.0) / (std::fabs(theta) + std::sqrt(theta * theta + 1.0));
196 const double c = 1.0 / std::sqrt(t * t + 1.0), s = t * c, tau = s / (1.0 + c);
197 a[p * n + p] -= t * apq;
198 a[q * n + q] += t * apq;
199 a[p * n + q] = a[q * n + p] = 0.0;
200 for (std::size_t r = 0; r < n; ++r) {
201 if (r == p || r == q) continue;
202 const double arp = a[r * n + p], arq = a[r * n + q];
203 a[r * n + p] = a[p * n + r] = arp - s * (arq + tau * arp);
204 a[r * n + q] = a[q * n + r] = arq + s * (arp - tau * arq);
205 }
206 for (std::size_t r = 0; r < n; ++r) {
207 const double vrp = v[r * n + p], vrq = v[r * n + q];
208 v[r * n + p] = vrp - s * (vrq + tau * vrp);
209 v[r * n + q] = vrq + s * (vrp - tau * vrq);
210 }
211 }
212 }
213 }
214 std::vector<std::size_t> idx(n);
215 for (std::size_t i = 0; i < n; ++i) idx[i] = i;
216 std::sort(idx.begin(), idx.end(), [&](std::size_t x, std::size_t y) { return a[x * n + x] > a[y * n + y]; });
217 values.resize(n);
218 vectors.assign(n * n, 0.0);
219 for (std::size_t j = 0; j < n; ++j) {
220 values[j] = a[idx[j] * n + idx[j]];
221 for (std::size_t i = 0; i < n; ++i) vectors[i * n + j] = v[i * n + idx[j]];
222 }
223}

225bool is_symmetric(const std::vector<double>& a, std::size_t n) {
226 for (std::size_t i = 0; i < n; ++i)
227 for (std::size_t j = i + 1; j < n; ++j)
228 if (std::fabs(a[i * n + j] - a[j * n + i]) > 1e-12 * (1 + std::fabs(a[i * n + j])))
229 return false;
230 return true;

233// ---- general real eigenvalues: Hessenberg reduction + shifted QR ----
234std::vector<double> eigvals_general(std::vector<double> a, std::size_t n) {
235 // Householder reduction to upper Hessenberg.
236 for (std::size_t k = 1; k + 1 < n; ++k) {
237 double scale = 0.0;
238 for (std::size_t i = k; i < n; ++i) scale += std::fabs(a[i * n + (k - 1)]);
239 if (scale == 0.0) continue;
240 double h = 0.0;
241 std::vector<double> u(n, 0.0);
242 for (std::size_t i = k; i < n; ++i) {
243 u[i] = a[i * n + (k - 1)] / scale;
244 h += u[i] * u[i];
245 }
246 double g = (u[k] >= 0 ? -std::sqrt(h) : std::sqrt(h));
247 h -= u[k] * g;
248 u[k] -= g;
249 // A = (I - uuᵀ/h) A (I - uuᵀ/h)
250 for (std::size_t j = 0; j < n; ++j) { // right: columns
251 double f = 0.0;
252 for (std::size_t i = k; i < n; ++i) f += u[i] * a[j * n + i];
253 f /= h;
254 for (std::size_t i = k; i < n; ++i) a[j * n + i] -= f * u[i];
255 }
256 for (std::size_t i = 0; i < n; ++i) { // left: rows
257 double f = 0.0;
258 for (std::size_t j = k; j < n; ++j) f += u[j] * a[j * n + i];
259 f /= h;
260 for (std::size_t j = k; j < n; ++j) a[j * n + i] -= f * u[j];
261 }
262 a[k * n + (k - 1)] = scale * g;
263 for (std::size_t i = k + 1; i < n; ++i) a[i * n + (k - 1)] = 0.0;
264 }
265
266 // Shifted QR on the Hessenberg matrix (real eigenvalues; complex pairs throw).
267 std::vector<double> w(n, 0.0);
268 long long hi = static_cast<long long>(n) - 1;
269 const double eps = 1e-14;
270 int iter = 0;
271 while (hi >= 0) {
272 long long l = hi;
273 while (l > 0) {
274 const double s = std::fabs(a[(l - 1) * n + (l - 1)]) + std::fabs(a[l * n + l]);
275 if (std::fabs(a[l * n + (l - 1)]) <= eps * (s == 0 ? 1.0 : s)) break;
276 --l;
277 }
278 if (l == hi) { // 1×1 block -> real eigenvalue
279 w[hi] = a[hi * n + hi];
280 --hi;
281 iter = 0;
282 } else if (l == hi - 1) { // 2×2 block
283 const std::size_t p = static_cast<std::size_t>(hi - 1), q = static_cast<std::size_t>(hi);
284 const double app = a[p * n + p], aqq = a[q * n + q];
285 const double apq = a[p * n + q], aqp = a[q * n + p];
286 const double tr = app + aqq, det = app * aqq - apq * aqp;
287 const double disc = tr * tr - 4.0 * det;
288 if (disc < 0.0) throw std::runtime_error("linalg: complex eigenvalues (use eigh for symmetric matrices)");
289 const double sq = std::sqrt(disc);
290 w[p] = (tr + sq) / 2.0;
291 w[q] = (tr - sq) / 2.0;
292 hi -= 2;
293 iter = 0;
294 } else { // QR sweep with Wilkinson shift
295 if (++iter > 200) throw std::runtime_error("linalg: eigenvalue iteration did not converge");
296 const double shift = a[hi * n + hi];
297 for (long long i = l; i <= hi; ++i) a[i * n + i] -= shift;
298 // one explicit QR step via Givens rotations on the Hessenberg block
299 std::vector<double> cs, sn;
300 for (long long i = l; i < hi; ++i) {
301 const double x = a[i * n + i], y = a[(i + 1) * n + i];
302 const double r = std::hypot(x, y);
303 const double c = r == 0 ? 1.0 : x / r, s = r == 0 ? 0.0 : y / r;
304 cs.push_back(c);
305 sn.push_back(s);
306 for (long long j = i; j <= hi; ++j) {
307 const double t1 = a[i * n + j], t2 = a[(i + 1) * n + j];
308 a[i * n + j] = c * t1 + s * t2;
309 a[(i + 1) * n + j] = -s * t1 + c * t2;
310 }
311 }
312 for (long long i = l; i < hi; ++i) { // RQ: post-multiply
313 const double c = cs[static_cast<std::size_t>(i - l)], s = sn[static_cast<std::size_t>(i - l)];
314 for (long long j = l; j <= i + 1; ++j) {
315 const double t1 = a[j * n + i], t2 = a[j * n + (i + 1)];
316 a[j * n + i] = c * t1 + s * t2;
317 a[j * n + (i + 1)] = -s * t1 + c * t2;
318 }
319 }
320 for (long long i = l; i <= hi; ++i) a[i * n + i] += shift;
321 }
322 }
323 return w;
324}

331double dot(const NDArray& a, const NDArray& b) {
332 std::size_t n, m;
333 const std::vector<double> x = as_vector(a, n), y = as_vector(b, m);
334 if (n != m) throw std::runtime_error("linalg: dot dimension mismatch");
335 double s = 0.0;
336 for (std::size_t i = 0; i < n; ++i) s += x[i] * y[i];
337 return s;
338}

342NDArray outer(const NDArray& a, const NDArray& b) {

351NDArray matmul(const NDArray& a, const NDArray& b) {
352 std::size_t ar, ac, br, bc;
353 const std::vector<double> A = as_matrix(a, ar, ac);
354 const std::vector<double> B = as_matrix(b, br, bc);
355 if (ac != br) throw std::runtime_error("linalg: matmul inner dimension mismatch");
356 std::vector<double> C(ar * bc, 0.0);
357 for (std::size_t i = 0; i < ar; ++i) // ikj order -> contiguous inner loop (vectorizes)
358 for (std::size_t k = 0; k < ac; ++k) {
359 const double aik = A[i * ac + k];
360 for (std::size_t j = 0; j < bc; ++j) C[i * bc + j] += aik * B[k * bc + j];
361 }
362 return make_matrix(ar, bc, std::move(C));
363}

365NDArray matrix_power(const NDArray& a, long long p) {
366 std::size_t r, c;
367 as_matrix(a, r, c);
368 require_square(r, c);
369 std::vector<double> result(r * r, 0.0);
370 for (std::size_t i = 0; i < r; ++i) result[i * r + i] = 1.0; // identity
371 NDArray acc = make_matrix(r, r, result);
372 NDArray base = (p < 0) ? inv(a) : a;
373 long long e = p < 0 ? -p : p;
374 while (e > 0) {
375 if (e & 1) acc = matmul(acc, base);
376 base = matmul(base, base);
377 e >>= 1;
378 }
379 return acc;
380}

382NDArray kron(const NDArray& a, const NDArray& b) {
383 std::size_t ar, ac, br, bc;
384 const std::vector<double> A = as_matrix(a, ar, ac);
385 const std::vector<double> B = as_matrix(b, br, bc);
386 std::vector<double> K(ar * br * ac * bc, 0.0);
387 const std::size_t kc = ac * bc;
388 for (std::size_t i = 0; i < ar; ++i)
389 for (std::size_t j = 0; j < ac; ++j)
390 for (std::size_t p = 0; p < br; ++p)
391 for (std::size_t q = 0; q < bc; ++q)
392 K[(i * br + p) * kc + (j * bc + q)] = A[i * ac + j] * B[p * bc + q];
393 return make_matrix(ar * br, kc, std::move(K));
394}

404double norm(const NDArray& a) { // Frobenius (matrices) / L2 (vectors)
405 double s = 0.0;
406 if (a.ndim() <= 1) {
407 std::size_t n;
408 const std::vector<double> v = as_vector(a, n);
409 for (double x : v) s += x * x;
410 } else {
411 std::size_t r, c;
412 const std::vector<double> m = as_matrix(a, r, c);
413 for (double x : m) s += x * x;
414 }
415 return std::sqrt(s);
416}

418// ---- LU-based: solve / det / slogdet / inv / lstsq ----
419NDArray solve(const NDArray& a, const NDArray& b) {
420 std::size_t n, c;
421 std::vector<double> A = as_matrix(a, n, c);
422 require_square(n, c);
423 const LU lu = lu_decompose(std::move(A), n);
424 std::size_t bn;
425 std::vector<double> x = as_vector(b, bn);
426 if (bn != n) throw std::runtime_error("linalg: solve dimension mismatch");
427 lu_solve(lu, x);
428 return make_vector(std::move(x));

431double det(const NDArray& a) {
432 std::size_t n, c;
433 std::vector<double> A = as_matrix(a, n, c);
434 require_square(n, c);
435 const LU lu = lu_decompose(std::move(A), n);
436 double d = lu.sign;
437 for (std::size_t i = 0; i < n; ++i) d *= lu.a[i * n + i];
438 return d;

441SLogDet slogdet(const NDArray& a) {
442 std::size_t n, c;
443 std::vector<double> A = as_matrix(a, n, c);
444 require_square(n, c);
445 const LU lu = lu_decompose(std::move(A), n);
446 double sign = lu.sign, logabs = 0.0;
447 for (std::size_t i = 0; i < n; ++i) {
448 const double d = lu.a[i * n + i];
449 if (d < 0) sign = -sign;
450 logabs += std::log(std::fabs(d));
451 }
452 return {sign, logabs};
453}

455NDArray inv(const NDArray& a) {
456 std::size_t n, c;
457 std::vector<double> A = as_matrix(a, n, c);
458 require_square(n, c);
459 const LU lu = lu_decompose(std::move(A), n);
460 std::vector<double> result(n * n, 0.0);
461 for (std::size_t col = 0; col < n; ++col) {
462 std::vector<double> e(n, 0.0);
463 e[col] = 1.0;
464 lu_solve(lu, e);
465 for (std::size_t i = 0; i < n; ++i) result[i * n + col] = e[i];
466 }
467 return make_matrix(n, n, std::move(result));
468}

474// ---- Cholesky ----
475NDArray cholesky(const NDArray& a) {
476 std::size_t n, c;
477 const std::vector<double> A = as_matrix(a, n, c);
478 require_square(n, c);
479 std::vector<double> L(n * n, 0.0);
480 for (std::size_t i = 0; i < n; ++i) {
481 for (std::size_t j = 0; j <= i; ++j) {
482 double s = A[i * n + j];
483 for (std::size_t k = 0; k < j; ++k) s -= L[i * n + k] * L[j * n + k];
484 if (i == j) {
485 if (s <= 0.0) throw std::runtime_error("linalg: matrix is not positive-definite");
486 L[i * n + i] = std::sqrt(s);
487 } else {
488 L[i * n + j] = s / L[j * n + j];
489 }
490 }
491 }
492 return make_matrix(n, n, std::move(L));
493}

495// ---- Householder QR (reduced: Q is m×n, R is n×n) ----
496QR qr(const NDArray& a) {
497 std::size_t m, n;
498 std::vector<double> A = as_matrix(a, m, n);
499 if (m < n) throw std::runtime_error("linalg: qr requires rows >= cols");
500 std::vector<double> Q(m * m, 0.0);
501 for (std::size_t i = 0; i < m; ++i) Q[i * m + i] = 1.0;
502 for (std::size_t k = 0; k < n; ++k) {
503 double nrm = 0.0;
504 for (std::size_t i = k; i < m; ++i) nrm += A[i * n + k] * A[i * n + k];
505 nrm = std::sqrt(nrm);
506 if (nrm == 0.0) continue;
507 const double alpha = A[k * n + k] >= 0 ? -nrm : nrm;
508 std::vector<double> u(m, 0.0);
509 for (std::size_t i = k; i < m; ++i) u[i] = A[i * n + k];
510 u[k] -= alpha;
511 double unorm2 = 0.0;
512 for (std::size_t i = k; i < m; ++i) unorm2 += u[i] * u[i];
513 if (unorm2 == 0.0) continue;
514 for (std::size_t j = k; j < n; ++j) {
515 double s = 0.0;
516 for (std::size_t i = k; i < m; ++i) s += u[i] * A[i * n + j];
517 s = 2.0 * s / unorm2;
518 for (std::size_t i = k; i < m; ++i) A[i * n + j] -= s * u[i];
519 }
520 for (std::size_t j = 0; j < m; ++j) { // Q = Q · Hₖ
521 double s = 0.0;
522 for (std::size_t i = k; i < m; ++i) s += u[i] * Q[j * m + i];
523 s = 2.0 * s / unorm2;
524 for (std::size_t i = k; i < m; ++i) Q[j * m + i] -= s * u[i];
525 }
526 }
527 std::vector<double> Qr(m * n), Rr(n * n, 0.0); // reduced
528 for (std::size_t i = 0; i < m; ++i)
529 for (std::size_t j = 0; j < n; ++j) Qr[i * n + j] = Q[i * m + j];
530 for (std::size_t i = 0; i < n; ++i)
531 for (std::size_t j = i; j < n; ++j) Rr[i * n + j] = A[i * n + j];
532 return {make_matrix(m, n, std::move(Qr)), make_matrix(n, n, std::move(Rr))};
533}

535// ---- SVD and its derived quantities ----
536SVD svd(const NDArray& a) {
537 std::size_t m, n;
538 std::vector<double> A = as_matrix(a, m, n);
539 if (m < n) throw std::runtime_error("linalg: svd requires rows >= cols (transpose otherwise)");
540 const SVDc s = svd_jacobi(std::move(A), m, n);
541 // vh = Vᵀ
542 std::vector<double> vh(n * n);
543 for (std::size_t i = 0; i < n; ++i)
544 for (std::size_t j = 0; j < n; ++j) vh[i * n + j] = s.v[j * n + i];
545 return {make_matrix(m, n, s.u), make_vector(s.w), make_matrix(n, n, std::move(vh))};
546}

548NDArray pinv(const NDArray& a) {
549 std::size_t m, n;
550 std::vector<double> A = as_matrix(a, m, n);
551 if (m >= n) {
552 const SVDc s = svd_jacobi(std::move(A), m, n); // A = U(m×n) diag(w) V(n×n)ᵀ
553 const double tsh = 0.5 * std::sqrt(double(m + n + 1)) * (s.w.empty() ? 0 : s.w[0]) * 1e-15;
554 std::vector<double> p(n * m, 0.0); // pinv = V diag(1/w) Uᵀ -> n×m
555 for (std::size_t i = 0; i < n; ++i)
556 for (std::size_t j = 0; j < m; ++j) {
557 double acc = 0.0;
558 for (std::size_t k = 0; k < n; ++k)
559 if (s.w[k] > tsh) acc += s.v[i * n + k] * (s.u[j * n + k] / s.w[k]);
560 p[i * m + j] = acc;
561 }
562 return make_matrix(n, m, std::move(p));
563 }
564 // m < n: compute on Aᵀ (n×m, rows>=cols) then transpose the result.
565 std::vector<double> At(n * m);
566 for (std::size_t i = 0; i < m; ++i)
567 for (std::size_t j = 0; j < n; ++j) At[j * m + i] = A[i * n + j];
568 const SVDc s = svd_jacobi(std::move(At), n, m); // Aᵀ = U(n×m) diag(w) V(m×m)ᵀ
569 const double tsh = 0.5 * std::sqrt(double(n + m + 1)) * (s.w.empty() ? 0 : s.w[0]) * 1e-15;
570 std::vector<double> res(n * m, 0.0); // pinv(A) = (V diag(1/w) Uᵀ)ᵀ -> n×m
571 for (std::size_t i = 0; i < m; ++i)
572 for (std::size_t j = 0; j < n; ++j) {
573 double acc = 0.0;
574 for (std::size_t k = 0; k < m; ++k)
575 if (s.w[k] > tsh) acc += s.v[i * m + k] * (s.u[j * m + k] / s.w[k]);
576 res[j * m + i] = acc; // transpose into the n×m result
577 }
578 return make_matrix(n, m, std::move(res));
579}

581double cond(const NDArray& a) {
582 std::size_t m, n;
583 std::vector<double> A = as_matrix(a, m, n);
584 SVDc s;
585 if (m >= n) {
586 s = svd_jacobi(std::move(A), m, n);
587 } else {
588 std::vector<double> At(n * m);
589 for (std::size_t i = 0; i < m; ++i)
590 for (std::size_t j = 0; j < n; ++j) At[j * m + i] = A[i * n + j];
591 s = svd_jacobi(std::move(At), n, m);
592 }
593 const double wmin = s.w.empty() ? 0 : s.w.back();
594 return wmin == 0 ? std::numeric_limits<double>::infinity() : s.w.front() / wmin;
595}

597long long matrix_rank(const NDArray& a) {
598 std::size_t m, n;
599 std::vector<double> A = as_matrix(a, m, n);
600 const bool tr = m < n;
601 SVDc s;
602 if (tr) {
603 std::vector<double> At(n * m);
604 for (std::size_t i = 0; i < m; ++i)
605 for (std::size_t j = 0; j < n; ++j) At[j * m + i] = A[i * n + j];
606 s = svd_jacobi(std::move(At), n, m);
607 } else {
608 s = svd_jacobi(std::move(A), m, n);
609 }
610 const double tsh = 0.5 * std::sqrt(double(m + n + 1)) * (s.w.empty() ? 0 : s.w[0]) * 1e-15;
611 long long r = 0;
612 for (double w : s.w)
613 if (w > tsh) ++r;
614 return r;
615}

635 std::size_t n, c;

637 require_square(n, c);

639 std::vector<double> vals = eigvals_general(A, n);
640 std::sort(vals.begin(), vals.end(), std::greater<double>());
641 // General eigenvectors via inverse iteration would go here; not provided yet.

645 std::size_t n, c;

647 require_square(n, c);
