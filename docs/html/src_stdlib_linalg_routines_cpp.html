25// ---- extract / build contiguous row-major matrices & vectors ----
26//
27// IMPORTANT: read the shared buffer directly. The element accessor `a.at({i, j})`
28// constructs a `std::vector` index *per call* (one heap allocation per element), so
29// the old extractors did rows*cols allocations just to read a matrix. These pack
30// C-order with a flat `copy_n` when the array is already contiguous (the common
31// case — a freshly built matrix/vector), and a direct strided walk otherwise. No
32// per-element allocation either way.
33
34// Pack `a`'s elements into `out` (size a.size()) in C-order via direct buffer
35// indexing. Used only for the non-contiguous (view/broadcast/permuted) fallback.
36template <ndarray::Field T>
37void pack_corder(const ndarray::basic_ndarray<T>& a, T* out) {
38 const T* base = a.buffer()->data();
39 const auto& shp = a.shape();
40 const auto& st = a.strides();
41 const std::size_t nd = shp.size();
42 const std::ptrdiff_t off0 = static_cast<std::ptrdiff_t>(a.offset());
43 std::vector<std::size_t> idx(nd, 0);
44 const std::size_t total = a.size();
45 for (std::size_t lin = 0; lin < total; ++lin) {
46 std::ptrdiff_t off = off0;
47 for (std::size_t d = 0; d < nd; ++d)
48 off += static_cast<std::ptrdiff_t>(idx[d]) * st[d];
49 out[lin] = base[static_cast<std::size_t>(off)];
50 for (std::size_t d = nd; d-- > 0;) { // C-order increment
51 if (++idx[d] < shp[d]) break;
52 idx[d] = 0;
53 }
54 }
55}

143// ---- LU decomposition with partial pivoting (in place on a copy) ----
144struct LU {
145 std::vector<double> a; // L (below diag, unit) + U (diag/above), row-major n×n
146 std::vector<std::size_t> piv;
147 double sign;
148 std::size_t n;
149 bool singular;
150};
151LU lu_decompose(std::vector<double> a, std::size_t n) {
152 std::vector<std::size_t> piv(n);
153 std::vector<double> vv(n);
154 double sign = 1.0;
155 bool singular = false;
156 for (std::size_t i = 0; i < n; ++i) {
157 double big = 0.0;
158 for (std::size_t j = 0; j < n; ++j) big = std::max(big, std::fabs(a[i * n + j]));
159 if (big == 0.0) { singular = true; big = 1.0; }
160 vv[i] = 1.0 / big;
161 }
162 for (std::size_t k = 0; k < n; ++k) {
163 double big = 0.0;
164 std::size_t imax = k;
165 for (std::size_t i = k; i < n; ++i) {
166 const double t = vv[i] * std::fabs(a[i * n + k]);
167 if (t > big) { big = t; imax = i; }
168 }
169 if (k != imax) {
170 for (std::size_t j = 0; j < n; ++j) std::swap(a[imax * n + j], a[k * n + j]);
171 sign = -sign;
172 vv[imax] = vv[k];
173 }
174 piv[k] = imax;
175 if (a[k * n + k] == 0.0) { a[k * n + k] = 1e-300; singular = true; }
176 for (std::size_t i = k + 1; i < n; ++i) {
177 const double f = a[i * n + k] / a[k * n + k];
178 a[i * n + k] = f;
179 for (std::size_t j = k + 1; j < n; ++j) a[i * n + j] -= f * a[k * n + j];
180 }
181 }
182 return {std::move(a), std::move(piv), sign, n, singular};
183}
184void lu_solve(const LU& lu, std::vector<double>& b) {
185 const std::size_t n = lu.n;
186 for (std::size_t k = 0; k < n; ++k) std::swap(b[k], b[lu.piv[k]]);
187 for (std::size_t i = 0; i < n; ++i) { // forward (unit L)
188 double s = b[i];
189 for (std::size_t j = 0; j < i; ++j) s -= lu.a[i * n + j] * b[j];
190 b[i] = s;
191 }
192 for (std::size_t i = n; i-- > 0;) { // back (U)
193 double s = b[i];
194 for (std::size_t j = i + 1; j < n; ++j) s -= lu.a[i * n + j] * b[j];
195 b[i] = s / lu.a[i * n + i];
196 }
197}

199// ---- Golub–Reinsch SVD: A(m×n) = U(m×n) diag(w) V(n×n)ᵀ, requires m ≥ n ----
200struct SVDc {
201 std::vector<double> u, w, v;
202 std::size_t m, n;
203};
204// Overflow-safe √(a²+b²) for the QR sweeps. std::hypot is correctly-rounded and several
205// times slower; called once per Givens rotation (O(n²) of them) it dominated the
206// values-only SVD. This EISPACK form is plenty accurate and much faster.
207inline double pythag(double a, double b) {
208 const double aa = std::fabs(a), ab = std::fabs(b);
209 if (aa > ab) { const double r = ab / aa; return aa * std::sqrt(1.0 + r * r); }
210 if (ab == 0.0) return 0.0;
211 const double r = aa / ab;
212 return ab * std::sqrt(1.0 + r * r);
213}
214// The world-standard dense SVD (what LAPACK's dgesvd reduces to): Householder
215// bidiagonalization to an upper-bidiagonal B = Uᵦᵀ A Vᵦ, then diagonalization of B by
216// implicit-shift QR, accumulating the orthogonal factors. One reduction plus a
217// quadratically-converging QR sweep — vastly fewer flops than one-sided Jacobi's
218// repeated full passes. On input `a` is m×n row-major; on output it holds U (m×n).
219SVDc svd_golub_reinsch(std::vector<double> a_rm, std::size_t m, std::size_t n,
220 bool want_uv = true) {
221 // When @p want_uv is false only the singular values are produced: the U/V
222 // accumulation and the (dominant) U/V Givens rotations in the QR sweep are skipped
223 // — the same values-only fast path NumPy's `svd(compute_uv=False)` / `svdvals` take,
224 // and what `cond`/`matrix_rank` need.
225 // Work entirely COLUMN-MAJOR: U(r,c) = uc[c*m + r], V(r,c) = vc[c*n + r]. The bulk
226 // of Golub–Reinsch is the length-m LEFT Householder reflectors — the column
227 // reductions and their trailing-column updates. Column-major makes those unit-stride
228 // so -O3 -march=native vectorizes them (FMA over contiguous columns); in row-major
229 // they were stride-n and ran scalar, which is what left the bare SVD behind LAPACK.
230 // The QR sweep (rotating whole U/V columns) is contiguous for the same reason.
231 // Input arrives row-major; transpose it in once; uc holds U on output.
232 std::vector<double> uc(n * m), vc(n * n, 0.0), w(n, 0.0), rv1(n, 0.0), tbuf(m, 0.0);
233 for (std::size_t r = 0; r < m; ++r)
234 for (std::size_t c = 0; c < n; ++c) uc[c * m + r] = a_rm[r * n + c];
235 auto sign = [](double x, double s) { return s >= 0.0 ? std::fabs(x) : -std::fabs(x); };
236 // `g` and `scale` carry across iterations: the super-diagonal rv1[i] is the previous
237 // row-reflector's `scale * g`.
238 double g = 0.0, scale = 0.0, anorm = 0.0;
239
240 // --- Householder reduction to bidiagonal form (diagonal w, super-diagonal rv1) ---
241 for (std::size_t i = 0; i < n; ++i) {
242 const std::size_t l = i + 1;
243 rv1[i] = scale * g;
244 g = 0.0; scale = 0.0;
245 double s = 0.0;
246 double* Ui = &uc[i * m]; // column i — contiguous
247 for (std::size_t k = i; k < m; ++k) scale += std::fabs(Ui[k]);
248 if (scale != 0.0) { // left (column) reflector -> w[i]
249 for (std::size_t k = i; k < m; ++k) { Ui[k] /= scale; s += Ui[k] * Ui[k]; }
250 double f = Ui[i];
251 g = -sign(std::sqrt(s), f);
252 const double h = f * g - s;
253 Ui[i] = f - g;
254 for (std::size_t j = l; j < n; ++j) { // apply to trailing columns
255 double* Uj = &uc[j * m];
256 double sum = 0.0;
257 for (std::size_t k = i; k < m; ++k) sum += Ui[k] * Uj[k]; // contiguous → SIMD
258 const double fr = sum / h;
259 for (std::size_t k = i; k < m; ++k) Uj[k] += fr * Ui[k]; // contiguous → SIMD
260 }
261 for (std::size_t k = i; k < m; ++k) Ui[k] *= scale;
262 }
263 w[i] = scale * g;
264 g = 0.0; scale = 0.0; s = 0.0;
265 // right (row) reflector over columns l..n — length n, the minor half (strided)
266 if (l < n) {
267 for (std::size_t k = l; k < n; ++k) scale += std::fabs(uc[k * m + i]);
268 if (scale != 0.0) {
269 for (std::size_t k = l; k < n; ++k) { uc[k * m + i] /= scale; s += uc[k * m + i] * uc[k * m + i]; }
270 double f = uc[l * m + i];
271 g = -sign(std::sqrt(s), f);
272 const double h = f * g - s;
273 uc[l * m + i] = f - g;
274 for (std::size_t k = l; k < n; ++k) rv1[k] = uc[k * m + i] / h;
275 // Trailing update A(l:m, l:n) += (A·u)·rv1ᵀ, done COLUMN-by-column so the
276 // inner loops sweep contiguous rows of a column (vectorize) — the naive
277 // row-by-row form strode across columns (stride m) and ran scalar.
278 for (std::size_t j = l; j < m; ++j) tbuf[j] = 0.0;
279 for (std::size_t k = l; k < n; ++k) { // t[j] = Σ_k A(j,k)·u[k]
280 const double uk = uc[k * m + i];
281 const double* Ck = &uc[k * m];
282 for (std::size_t j = l; j < m; ++j) tbuf[j] += Ck[j] * uk;
283 }
284 for (std::size_t k = l; k < n; ++k) { // A(j,k) += t[j]·rv1[k]
285 const double r = rv1[k];
286 double* Ck = &uc[k * m];
287 for (std::size_t j = l; j < m; ++j) Ck[j] += tbuf[j] * r;
288 }
289 for (std::size_t k = l; k < n; ++k) uc[k * m + i] *= scale;
290 }
291 }
292 anorm = std::max(anorm, std::fabs(w[i]) + std::fabs(rv1[i]));
293 }
294
295 // --- accumulate the right-hand transformations into V (column-major vc) ---
296 if (want_uv)
297 for (std::size_t i = n; i-- > 0;) {
298 const std::size_t l = i + 1;
299 if (l < n) {
300 if (g != 0.0) {
301 for (std::size_t j = l; j < n; ++j) // V(j,i); double division guards overflow
302 vc[i * n + j] = (uc[j * m + i] / uc[l * m + i]) / g;
303 for (std::size_t j = l; j < n; ++j) {
304 double sum = 0.0;
305 for (std::size_t k = l; k < n; ++k) sum += uc[k * m + i] * vc[j * n + k];
306 for (std::size_t k = l; k < n; ++k) vc[j * n + k] += sum * vc[i * n + k];
307 }
308 }
309 for (std::size_t j = l; j < n; ++j) { vc[j * n + i] = 0.0; vc[i * n + j] = 0.0; }
310 }
311 vc[i * n + i] = 1.0;
312 g = rv1[i];
313 }
314
315 // --- accumulate the left-hand transformations into U (held in uc) ---
316 if (want_uv)
317 for (std::size_t i = n; i-- > 0;) { // min(m,n) == n since m >= n
318 const std::size_t l = i + 1;
319 g = w[i];
320 for (std::size_t j = l; j < n; ++j) uc[j * m + i] = 0.0;
321 double* Ui = &uc[i * m];
322 if (g != 0.0) {
323 g = 1.0 / g;
324 for (std::size_t j = l; j < n; ++j) {
325 double* Uj = &uc[j * m];
326 double sum = 0.0;
327 for (std::size_t k = l; k < m; ++k) sum += Ui[k] * Uj[k]; // contiguous → SIMD
328 const double f = (sum / Ui[i]) * g;
329 for (std::size_t k = i; k < m; ++k) Uj[k] += f * Ui[k]; // contiguous → SIMD
330 }
331 for (std::size_t k = i; k < m; ++k) Ui[k] *= g;
332 } else {
333 for (std::size_t k = i; k < m; ++k) Ui[k] = 0.0;
334 }
335 Ui[i] += 1.0;
336 }
337
338 // U (uc) and V (vc) are already column-major, so the QR sweep's whole-column
339 // rotations below are contiguous and vectorizable — no repacking needed.
340 // --- diagonalize the bidiagonal form: implicit-shift QR with deflation ---
341 const double eps = std::numeric_limits<double>::epsilon();
342 for (std::size_t k = n; k-- > 0;) {
343 for (int its = 0; its < 60; ++its) {
344 bool flag = true;
345 std::size_t l = k, nm = 0;
346 while (true) { // find a negligible super-diagonal to split at
347 if (l == 0) { flag = false; break; } // rv1[0] is structurally 0
348 if (std::fabs(rv1[l]) <= eps * anorm) { flag = false; break; }
349 nm = l - 1;
350 if (std::fabs(w[nm]) <= eps * anorm) break;
351 --l;
352 }
353 if (flag) { // cancel rv1[l] via Givens rotations in U
354 double c = 0.0, s = 1.0;
355 for (std::size_t i = l; i <= k; ++i) {
356 double f = s * rv1[i];
357 rv1[i] = c * rv1[i];
358 if (std::fabs(f) <= eps * anorm) break;
359 double gg = w[i];
360 double h = pythag(f, gg);
361 w[i] = h; h = 1.0 / h;
362 c = gg * h; s = -f * h;
363 if (want_uv) {
364 double* Unm = &uc[nm * m];
365 double* Ui = &uc[i * m];
366 for (std::size_t j = 0; j < m; ++j) {
367 const double y = Unm[j], z = Ui[j];
368 Unm[j] = y * c + z * s;
369 Ui[j] = z * c - y * s;
370 }
371 }
372 }
373 }
374 double z = w[k];
375 if (l == k) { // converged: make the singular value non-negative
376 if (z < 0.0) {
377 w[k] = -z;
378 if (want_uv) { double* Vk = &vc[k * n]; for (std::size_t j = 0; j < n; ++j) Vk[j] = -Vk[j]; }
379 }
380 break;
381 }
382 if (its == 59) throw std::runtime_error("linalg: SVD did not converge");
383 double x = w[l];
384 nm = k - 1;
385 double y = w[nm], gg = rv1[nm], h = rv1[k];
386 double f = ((y - z) * (y + z) + (gg - h) * (gg + h)) / (2.0 * h * y);
387 gg = pythag(f, 1.0);
388 f = ((x - z) * (x + z) + h * ((y / (f + sign(gg, f))) - h)) / x;
389 double c = 1.0, s = 1.0;
390 for (std::size_t j = l; j <= nm; ++j) { // QR sweep: chase the bulge
391 const std::size_t i = j + 1;
392 gg = rv1[i]; y = w[i]; h = s * gg; gg = c * gg;
393 z = pythag(f, h);
394 rv1[j] = z; c = f / z; s = h / z;
395 f = x * c + gg * s; gg = gg * c - x * s; h = y * s; y *= c;
396 if (want_uv) {
397 double* Vj = &vc[j * n];
398 double* Vi = &vc[i * n];
399 for (std::size_t jj = 0; jj < n; ++jj) { // rotate V columns j, i (contiguous)
400 const double vx = Vj[jj], vz = Vi[jj];
401 Vj[jj] = vx * c + vz * s;
402 Vi[jj] = vz * c - vx * s;
403 }
404 }
405 z = pythag(f, h);
406 w[j] = z;
407 if (z != 0.0) { z = 1.0 / z; c = f * z; s = h * z; }
408 f = c * gg + s * y; x = c * y - s * gg;
409 if (want_uv) {
410 double* Uj = &uc[j * m];
411 double* Ui = &uc[i * m];
412 for (std::size_t jj = 0; jj < m; ++jj) { // rotate U columns j, i (contiguous)
413 const double uy = Uj[jj], uz = Ui[jj];
414 Uj[jj] = uy * c + uz * s;
415 Ui[jj] = uz * c - uy * s;
416 }
417 }
418 }
419 rv1[l] = 0.0; rv1[k] = f; w[k] = x;
420 }
421 }

423 // singular values come out non-negative but unordered — sort descending, carrying
424 // the matching columns of U and V (read straight from the column-major buffers).
425 std::vector<std::size_t> idx(n);
426 for (std::size_t i = 0; i < n; ++i) idx[i] = i;
427 std::sort(idx.begin(), idx.end(), [&](std::size_t x, std::size_t y) { return w[x] > w[y]; });
428 SVDc out{std::vector<double>(want_uv ? m * n : 0), std::vector<double>(n),
429 std::vector<double>(want_uv ? n * n : 0), m, n};
430 for (std::size_t j = 0; j < n; ++j) out.w[j] = w[idx[j]];
431 if (want_uv)
432 for (std::size_t j = 0; j < n; ++j) {
433 const std::size_t src = idx[j];
434 const double* Uc = &uc[src * m];
435 for (std::size_t i = 0; i < m; ++i) out.u[i * n + j] = Uc[i];
436 const double* Vc = &vc[src * n];
437 for (std::size_t i = 0; i < n; ++i) out.v[i * n + j] = Vc[i];
438 }
439 return out;

442// ---- real symmetric eigensolver: Householder tridiagonalization (tred2) + ----
443// ---- implicit-shift QL (tql2). ------------------------------------------------
444// The O(n³) method LAPACK uses (one reduction + a QL sweep that converges in O(n)
445// rotations), far cheaper than cyclic Jacobi's repeated full-matrix sweeps. `a` is a
446// row-major n×n matrix ASSUMED symmetric (only the working triangle is used). Returns
447// eigenvalues DESCENDING in `values`, with the matching orthonormal eigenvector as
448// column j of the row-major `vectors` (vectors[i*n+j] = component i of eigenvector j).
449void symmetric_eig(std::vector<double> z, std::size_t n, std::vector<double>& values,
450 std::vector<double>& vectors, bool want_vectors = true) {
451 values.assign(n, 0.0);
452 vectors.assign(want_vectors ? n * n : 0, 0.0);
453 if (n == 0) return;
454 if (n == 1) { values[0] = z[0]; if (want_vectors) vectors[0] = 1.0; return; }
455
456 std::vector<double> d(n, 0.0), e(n, 0.0);
457
458 // --- tred2: reduce symmetric z -> tridiagonal (d diagonal, e subdiagonal),
459 // leaving the accumulated orthogonal transform in z. ---
460 for (std::size_t i = n - 1; i >= 1; --i) {
461 const std::size_t l = i - 1;
462 double h = 0.0;
463 if (l > 0) {
464 double scale = 0.0;
465 for (std::size_t k = 0; k <= l; ++k) scale += std::fabs(z[i * n + k]);
466 if (scale == 0.0) {
467 e[i] = z[i * n + l];
468 } else {
469 for (std::size_t k = 0; k <= l; ++k) {
470 z[i * n + k] /= scale;
471 h += z[i * n + k] * z[i * n + k];
472 }
473 double f = z[i * n + l];
474 double g = (f >= 0.0) ? -std::sqrt(h) : std::sqrt(h);
475 e[i] = scale * g;
476 h -= f * g;
477 z[i * n + l] = f - g;
478 f = 0.0;
479 for (std::size_t j = 0; j <= l; ++j) {
480 z[j * n + i] = z[i * n + j] / h; // store u/h in column i
481 g = 0.0;
482 for (std::size_t k = 0; k <= j; ++k) g += z[j * n + k] * z[i * n + k];
483 for (std::size_t k = j + 1; k <= l; ++k) g += z[k * n + j] * z[i * n + k];
484 e[j] = g / h;
485 f += e[j] * z[i * n + j];
486 }
487 const double hh = f / (h + h);
488 for (std::size_t j = 0; j <= l; ++j) {
489 f = z[i * n + j];
490 g = e[j] - hh * f;
491 e[j] = g;
492 for (std::size_t k = 0; k <= j; ++k)
493 z[j * n + k] -= (f * e[k] + g * z[i * n + k]);
494 }
495 }
496 } else {
497 e[i] = z[i * n + l];
498 }
499 d[i] = h;
500 }
501 d[0] = 0.0;
502 e[0] = 0.0;
503 if (want_vectors) {
504 for (std::size_t i = 0; i < n; ++i) { // accumulate the transform into z
505 if (d[i] != 0.0) {
506 for (std::size_t j = 0; j < i; ++j) {
507 double g = 0.0;
508 for (std::size_t k = 0; k < i; ++k) g += z[i * n + k] * z[k * n + j];
509 for (std::size_t k = 0; k < i; ++k) z[k * n + j] -= g * z[k * n + i];
510 }
511 }
512 d[i] = z[i * n + i];
513 z[i * n + i] = 1.0;
514 for (std::size_t j = 0; j < i; ++j) { z[j * n + i] = 0.0; z[i * n + j] = 0.0; }
515 }
516 } else {
517 for (std::size_t i = 0; i < n; ++i) d[i] = z[i * n + i]; // values only — skip Q
518 }
519
520 // --- tql2: implicit-shift QL on the tridiagonal (d, e), rotating z alongside. ---
521 for (std::size_t i = 1; i < n; ++i) e[i - 1] = e[i];
522 e[n - 1] = 0.0;
523 for (std::size_t l = 0; l < n; ++l) {
524 int iter = 0;
525 std::size_t m;
526 do {
527 for (m = l; m + 1 < n; ++m) {
528 const double dd = std::fabs(d[m]) + std::fabs(d[m + 1]);
529 if (std::fabs(e[m]) <= 2.2e-16 * dd) break;
530 }
531 if (m != l) {
532 if (iter++ == 50)
533 throw std::runtime_error("linalg: symmetric eigen QL did not converge");
534 double g = (d[l + 1] - d[l]) / (2.0 * e[l]);
535 double r = pythag(g, 1.0);
536 g = d[m] - d[l] + e[l] / (g + (g >= 0.0 ? std::fabs(r) : -std::fabs(r)));
537 double s = 1.0, c = 1.0, p = 0.0;
538 bool zeroed = false;
539 for (std::size_t i = m; i-- > l;) { // i = m-1 … l
540 double f = s * e[i];
541 const double b = c * e[i];
542 r = pythag(f, g);
543 e[i + 1] = r;
544 if (r == 0.0) { d[i + 1] -= p; e[m] = 0.0; zeroed = true; break; }
545 s = f / r;
546 c = g / r;
547 g = d[i + 1] - p;
548 r = (d[i] - g) * s + 2.0 * c * b;
549 p = s * r;
550 d[i + 1] = g + p;
551 g = c * r - b;
552 if (want_vectors)
553 for (std::size_t k = 0; k < n; ++k) { // rotate eigenvector columns
554 f = z[k * n + i + 1];
555 z[k * n + i + 1] = s * z[k * n + i] + c * f;
556 z[k * n + i] = c * z[k * n + i] - s * f;
557 }
558 }
559 if (!zeroed) { d[l] -= p; e[l] = g; e[m] = 0.0; }
560 }
561 } while (m != l);
562 }
563
564 // sort DESCENDING, carrying the matching eigenvector columns.
565 std::vector<std::size_t> idx(n);
566 for (std::size_t i = 0; i < n; ++i) idx[i] = i;
567 std::sort(idx.begin(), idx.end(), [&](std::size_t x, std::size_t y) { return d[x] > d[y]; });
568 for (std::size_t j = 0; j < n; ++j) values[j] = d[idx[j]];
569 if (want_vectors)
570 for (std::size_t j = 0; j < n; ++j)
571 for (std::size_t i = 0; i < n; ++i) vectors[i * n + j] = z[i * n + idx[j]];
572}

574bool is_symmetric(const std::vector<double>& a, std::size_t n) {
575 for (std::size_t i = 0; i < n; ++i)
576 for (std::size_t j = i + 1; j < n; ++j)
577 if (std::fabs(a[i * n + j] - a[j * n + i]) > 1e-12 * (1 + std::fabs(a[i * n + j])))
578 return false;
579 return true;
580}

582// Complex Hermitian eigensolver with REAL eigenvalues and COMPLEX eigenvectors,
583// reusing the real symmetric tridiagonal-QL solver via the standard 2n×2n real embedding:
584// for H = A + iB (A symmetric, B antisymmetric), the real symmetric matrix
585// M = [[A, -B], [B, A]]
586// has each eigenvalue of H twice, and a real eigenvector [x; y] of M corresponds to
587// the complex eigenvector x + iy of H (already unit-norm: |x|²+|y|² = 1). We take
588// one representative per duplicated pair. @p evecs (when requested) is row-major n×n
589// with column k the eigenvector for evals[k]; both come out sorted descending.
590void hermitian_eig(const std::vector<Cplx>& H, std::size_t n, std::vector<double>& evals,
591 std::vector<Cplx>& evecs, bool want_vectors) {
592 const std::size_t N = 2 * n;
593 std::vector<double> M(N * N, 0.0);
594 for (std::size_t i = 0; i < n; ++i)
595 for (std::size_t j = 0; j < n; ++j) {
596 const double re = H[i * n + j].real(), im = H[i * n + j].imag();
597 M[i * N + j] = re; // top-left A
598 M[(i + n) * N + (j + n)] = re; // bottom-right A
599 M[i * N + (j + n)] = -im; // top-right -B
600 M[(i + n) * N + j] = im; // bottom-left B
601 }
602 std::vector<double> w, V;
603 symmetric_eig(M, N, w, V, want_vectors); // 2n eigenvalues (desc, paired) + vectors
604 evals.resize(n);
605 for (std::size_t k = 0; k < n; ++k) evals[k] = w[2 * k]; // one of each equal pair
606 if (want_vectors) {
607 evecs.assign(n * n, Cplx{});
608 for (std::size_t k = 0; k < n; ++k) {
609 const std::size_t col = 2 * k;
610 for (std::size_t p = 0; p < n; ++p) {
611 const double x = V[p * N + col], y = V[(p + n) * N + col];
612 evecs[p * n + k] = Cplx(x, y); // column k = eigenvector for evals[k]
613 }
614 }
615 }
616}

618// Complex LU with partial pivoting, factored in place on M (row-major n×n): the unit
619// lower factor's multipliers are stored below the diagonal, U on/above it. Returns the
620// pivot vector. Factor ONCE, then `complex_lu_solve` for each right-hand side — inverse
621// iteration reuses the same (deliberately near-singular) M across several RHS.
622std::vector<std::size_t> complex_lu(std::vector<Cplx>& M, std::size_t n) {
623 std::vector<std::size_t> piv(n);
624 for (std::size_t k = 0; k < n; ++k) {
625 std::size_t p = k;
626 double best = std::abs(M[k * n + k]);
627 for (std::size_t i = k + 1; i < n; ++i) {
628 const double m = std::abs(M[i * n + k]);
629 if (m > best) { best = m; p = i; }
630 }
631 piv[k] = p;
632 if (p != k)
633 for (std::size_t j = 0; j < n; ++j) std::swap(M[k * n + j], M[p * n + j]);
634 const Cplx d = M[k * n + k];
635 for (std::size_t i = k + 1; i < n; ++i) {
636 const Cplx f = M[i * n + k] / d;
637 M[i * n + k] = f;
638 for (std::size_t j = k + 1; j < n; ++j) M[i * n + j] -= f * M[k * n + j];
639 }
640 }
641 return piv;
642}
643// Solve (already-factored) M·x = b in place on @p b (forward unit-L, then back-U).
644void complex_lu_solve(const std::vector<Cplx>& M, const std::vector<std::size_t>& piv,
645 std::vector<Cplx>& b, std::size_t n) {
646 for (std::size_t k = 0; k < n; ++k)
647 if (piv[k] != k) std::swap(b[k], b[piv[k]]);
648 for (std::size_t i = 0; i < n; ++i) {
649 Cplx s = b[i];
650 for (std::size_t j = 0; j < i; ++j) s -= M[i * n + j] * b[j];
651 b[i] = s;
652 }
653 for (std::size_t i = n; i-- > 0;) {
654 Cplx s = b[i];
655 for (std::size_t j = i + 1; j < n; ++j) s -= M[i * n + j] * b[j];
656 b[i] = s / M[i * n + i];
657 }
658}

660// Eigenvector of the real matrix @p A for (complex) eigenvalue @p lambda, by inverse
661// iteration. C = A − (λ + tiny complex shift)·I is made just non-singular by the
662// shift, then a few inverse-iteration steps converge to the eigenvector; the phase
663// is fixed so the largest-magnitude component is real-positive (a stable, if
664// arbitrary, choice — eigenvectors are only defined up to phase).
665std::vector<Cplx> eigvector_inverse_iteration(const std::vector<double>& A, std::size_t n,
666 Cplx lambda) {
667 double scale = 1.0;
668 for (double a : A) scale = std::max(scale, std::fabs(a));
669 const Cplx shifted = lambda + Cplx(scale * 1e-10, scale * 1e-10);
670 std::vector<Cplx> C(n * n);
671 for (std::size_t i = 0; i < n; ++i)
672 for (std::size_t j = 0; j < n; ++j)
673 C[i * n + j] = Cplx(A[i * n + j], 0.0) - (i == j ? shifted : Cplx{});
674 const auto normalize = [&](std::vector<Cplx>& x) {
675 double nrm = 0.0;
676 for (const Cplx& z : x) nrm += std::norm(z);
677 nrm = std::sqrt(nrm);
678 for (Cplx& z : x) z /= nrm;
679 };
680 const std::vector<std::size_t> piv = complex_lu(C, n); // factor ONCE, reuse per step
681 std::vector<Cplx> v(n, Cplx(1.0, 0.0));
682 normalize(v);
683 for (int it = 0; it < 5; ++it) {
684 complex_lu_solve(C, piv, v, n); // in place on v — no per-step copy or re-factor
685 normalize(v);
686 }
687 std::size_t mi = 0;
688 double mb = 0.0;
689 for (std::size_t i = 0; i < n; ++i) {
690 const double m = std::abs(v[i]);
691 if (m > mb) {
692 mb = m;
693 mi = i;
694 }
695 }
696 const Cplx phase = v[mi] / std::abs(v[mi]); // unit-norm v -> mb > 0
697 for (Cplx& z : v) z /= phase;
698 return v;
699}

701// ---- general eigenvalues: Hessenberg reduction + shifted QR ----
702// Real matrix in; COMPLEX spectrum out (a 2×2 block with negative discriminant is a
703// conjugate pair, not an error). The arithmetic stays real; only the extracted
704// eigenvalues are complex.
705std::vector<Cplx> eigvals_general(std::vector<double> a, std::size_t n) {
706 // Householder reduction to upper Hessenberg.
707 std::vector<double> u(n); // reflector, reused per column (entries < k unused)
708 for (std::size_t k = 1; k + 1 < n; ++k) {
709 double scale = 0.0;
710 for (std::size_t i = k; i < n; ++i) scale += std::fabs(a[i * n + (k - 1)]);
711 if (scale == 0.0) continue;
712 double h = 0.0;
713 for (std::size_t i = k; i < n; ++i) {
714 u[i] = a[i * n + (k - 1)] / scale;
715 h += u[i] * u[i];
716 }
717 double g = (u[k] >= 0 ? -std::sqrt(h) : std::sqrt(h));
718 h -= u[k] * g;
719 u[k] -= g;
720 // A = (I - uuᵀ/h) A (I - uuᵀ/h)
721 for (std::size_t j = 0; j < n; ++j) { // right: columns
722 double f = 0.0;
723 for (std::size_t i = k; i < n; ++i) f += u[i] * a[j * n + i];
724 f /= h;
725 for (std::size_t i = k; i < n; ++i) a[j * n + i] -= f * u[i];
726 }
727 for (std::size_t i = 0; i < n; ++i) { // left: rows
728 double f = 0.0;
729 for (std::size_t j = k; j < n; ++j) f += u[j] * a[j * n + i];
730 f /= h;
731 for (std::size_t j = k; j < n; ++j) a[j * n + i] -= f * u[j];
732 }
733 a[k * n + (k - 1)] = scale * g;
734 for (std::size_t i = k + 1; i < n; ++i) a[i * n + (k - 1)] = 0.0;
735 }
736
737 // Shifted QR on the Hessenberg matrix. A 1×1 block is a real eigenvalue; a 2×2
738 // block is two reals (disc ≥ 0) or a complex conjugate pair (disc < 0).
739 std::vector<Cplx> w(n);
740 std::vector<double> cs, sn; // Givens rotations, reused per QR sweep (clear keeps capacity)
741 long long hi = static_cast<long long>(n) - 1;
742 const double eps = 1e-14;
743 int iter = 0;
744 while (hi >= 0) {
745 long long l = hi;
746 while (l > 0) {
747 const double s = std::fabs(a[(l - 1) * n + (l - 1)]) + std::fabs(a[l * n + l]);
748 if (std::fabs(a[l * n + (l - 1)]) <= eps * (s == 0 ? 1.0 : s)) break;
749 --l;
750 }
751 if (l == hi) { // 1×1 block -> real eigenvalue
752 w[hi] = a[hi * n + hi];
753 --hi;
754 iter = 0;
755 } else if (l == hi - 1) { // 2×2 block
756 const std::size_t p = static_cast<std::size_t>(hi - 1), q = static_cast<std::size_t>(hi);
757 const double app = a[p * n + p], aqq = a[q * n + q];
758 const double apq = a[p * n + q], aqp = a[q * n + p];
759 const double tr = app + aqq, det = app * aqq - apq * aqp;
760 const double disc = tr * tr - 4.0 * det;
761 if (disc >= 0.0) { // two real eigenvalues
762 const double sq = std::sqrt(disc);
763 w[p] = (tr + sq) / 2.0;
764 w[q] = (tr - sq) / 2.0;
765 } else { // complex conjugate pair
766 const double im = std::sqrt(-disc) / 2.0;
767 w[p] = Cplx(tr / 2.0, im);
768 w[q] = Cplx(tr / 2.0, -im);
769 }
770 hi -= 2;
771 iter = 0;
772 } else { // QR sweep with Wilkinson shift
773 if (++iter > 200) throw std::runtime_error("linalg: eigenvalue iteration did not converge");
774 const double shift = a[hi * n + hi];
775 for (long long i = l; i <= hi; ++i) a[i * n + i] -= shift;
776 // one explicit QR step via Givens rotations on the Hessenberg block
777 cs.clear();
778 sn.clear();
779 for (long long i = l; i < hi; ++i) {
780 const double x = a[i * n + i], y = a[(i + 1) * n + i];
781 const double r = pythag(x, y);
782 const double c = r == 0 ? 1.0 : x / r, s = r == 0 ? 0.0 : y / r;
783 cs.push_back(c);
784 sn.push_back(s);
785 for (long long j = i; j <= hi; ++j) {
786 const double t1 = a[i * n + j], t2 = a[(i + 1) * n + j];
787 a[i * n + j] = c * t1 + s * t2;
788 a[(i + 1) * n + j] = -s * t1 + c * t2;
789 }
790 }
791 for (long long i = l; i < hi; ++i) { // RQ: post-multiply
792 const double c = cs[static_cast<std::size_t>(i - l)], s = sn[static_cast<std::size_t>(i - l)];
793 for (long long j = l; j <= i + 1; ++j) {
794 const double t1 = a[j * n + i], t2 = a[j * n + (i + 1)];
795 a[j * n + i] = c * t1 + s * t2;
796 a[j * n + (i + 1)] = -s * t1 + c * t2;
797 }
798 }
799 for (long long i = l; i <= hi; ++i) a[i * n + i] += shift;
800 }
801 }
802 return w;
803}

809// ---- products ----
810// The products only READ their operands, so they take a zero-copy `contig` pointer
811// (straight into the array's own buffer when it is contiguous — the common case)
812// and allocate nothing but the result.
813//
814// Reduction kernels use several independent accumulators. A single running sum
815// serializes the loop on floating-point-add latency (the compiler may not reassociate
816// FP adds without -ffast-math), so a plain `s += x[i]*y[i]` runs at ~one element per
817// FADD latency. Independent lanes break that dependency chain, letting -O3
818// -march=native issue SIMD + FMA and hit memory bandwidth instead of add latency.
819double ddot(const double* x, const double* y, std::size_t n) {
820 double s0 = 0, s1 = 0, s2 = 0, s3 = 0;
821 std::size_t i = 0;
822 for (; i + 4 <= n; i += 4) {
823 s0 += x[i] * y[i];
824 s1 += x[i + 1] * y[i + 1];
825 s2 += x[i + 2] * y[i + 2];
826 s3 += x[i + 3] * y[i + 3];
827 }
828 double s = (s0 + s1) + (s2 + s3);
829 for (; i < n; ++i) s += x[i] * y[i];
830 return s;
831}

844// Complex products. `dot`/`inner` are bilinear (Σ aᵢbᵢ, no conjugation, like numpy);
845// `vdot` is the conjugate-linear Hermitian inner product Σ conj(aᵢ)·bᵢ = ⟨a, b⟩.
846// Zero-copy reads (no `as_cvector`) with the same multi-accumulator reduction.
847Cplx cdot(const Cplx* x, const Cplx* y, std::size_t n, bool conjugate) {
848 Cplx s0{}, s1{}, s2{}, s3{};
849 std::size_t i = 0;
850 for (; i + 4 <= n; i += 4) {
851 if (conjugate) {
852 s0 += std::conj(x[i]) * y[i];
853 s1 += std::conj(x[i + 1]) * y[i + 1];
854 s2 += std::conj(x[i + 2]) * y[i + 2];
855 s3 += std::conj(x[i + 3]) * y[i + 3];
856 } else {
857 s0 += x[i] * y[i];
858 s1 += x[i + 1] * y[i + 1];
859 s2 += x[i + 2] * y[i + 2];
860 s3 += x[i + 3] * y[i + 3];
861 }
862 }
863 Cplx s = (s0 + s1) + (s2 + s3);
864 for (; i < n; ++i) s += (conjugate ? std::conj(x[i]) : x[i]) * y[i];
865 return s;
866}

895NDArray matmul(const NDArray& a, const NDArray& b) {
896 if (a.ndim() != 2 || b.ndim() != 2)
897 throw std::runtime_error("linalg: matmul expects 2-D matrices");
898 const std::size_t ar = a.shape()[0], ac = a.shape()[1];
899 const std::size_t br = b.shape()[0], bc = b.shape()[1];
900 if (ac != br) throw std::runtime_error("linalg: matmul inner dimension mismatch");
901 std::vector<double> sa, sb;
902 const double* A = contig(a, sa); // zero-copy when contiguous
903 const double* B = contig(b, sb);
904 std::vector<double> C(ar * bc, 0.0);
905 for (std::size_t i = 0; i < ar; ++i) // ikj order -> contiguous inner loop (vectorizes)
906 for (std::size_t k = 0; k < ac; ++k) {
907 const double aik = A[i * ac + k];
908 for (std::size_t j = 0; j < bc; ++j) C[i * bc + j] += aik * B[k * bc + j];
909 }
910 return make_matrix(ar, bc, std::move(C));
911}

914 if (a.ndim() != 2 || b.ndim() != 2)
915 throw std::runtime_error("linalg: matmul expects 2-D matrices");

917 const std::size_t br = b.shape()[0], bc = b.shape()[1];
918 if (ac != br) throw std::runtime_error("linalg: matmul inner dimension mismatch");

920 const Cplx* A = contig(a, sa); // zero-copy when contiguous
921 const Cplx* B = contig(b, sb);

923 for (std::size_t i = 0; i < ar; ++i)
924 for (std::size_t k = 0; k < ac; ++k) {
925 const Cplx aik = A[i * ac + k];
926 for (std::size_t j = 0; j < bc; ++j) C[i * bc + j] += aik * B[k * bc + j];
927 }
928 return make_cmatrix(ar, bc, std::move(C));

943NDArray matrix_power(const NDArray& a, long long p) {
944 if (a.ndim() != 2) throw std::runtime_error("linalg: expected a 2-D matrix");
945 const std::size_t r = a.shape()[0], c = a.shape()[1]; // dims only — no copy
946 require_square(r, c);
947 std::vector<double> result(r * r, 0.0);
948 for (std::size_t i = 0; i < r; ++i) result[i * r + i] = 1.0; // identity
949 NDArray acc = make_matrix(r, r, std::move(result));
950 NDArray base = (p < 0) ? inv(a) : a;
951 long long e = p < 0 ? -p : p;
952 while (e > 0) {
953 if (e & 1) acc = matmul(acc, base);
954 base = matmul(base, base);
955 e >>= 1;
956 }
957 return acc;
958}

978double trace(const NDArray& a) {
979 if (a.ndim() != 2) throw std::runtime_error("linalg: expected a 2-D matrix");
980 const std::size_t r = a.shape()[0], c = a.shape()[1];
981 // Read the diagonal straight from the buffer — no copy, even for a strided view.
982 const double* base = a.buffer()->data();
983 const std::ptrdiff_t off = static_cast<std::ptrdiff_t>(a.offset());
984 const std::ptrdiff_t step = a.strides()[0] + a.strides()[1]; // (i,i) advances by s0+s1
985 double s = 0.0;
986 for (std::size_t i = 0; i < std::min(r, c); ++i)
987 s += base[static_cast<std::size_t>(off + static_cast<std::ptrdiff_t>(i) * step)];
988 return s;
989}

1037NDArray inv(const NDArray& a) {
1038 std::size_t n, c;
1039 std::vector<double> A = as_matrix(a, n, c);
1040 require_square(n, c);
1041 const LU lu = lu_decompose(std::move(A), n);
1042 const std::vector<double>& M = lu.a; // L (unit, below diag) + U (on/above), row-major
1043 // Invert by solving L·U·X = P·I for the WHOLE identity at once. Doing the forward
1044 // and back substitution across all n columns turns each inner loop into a SAXPY
1045 // over a contiguous row (`X[i,:] -= M[i,j]·X[j,:]`), which auto-vectorizes — unlike
1046 // n separate single-RHS solves, whose substitution is a serial-reduction dot that
1047 // cannot vectorize (the reason a naive `inv` lost to LAPACK while `det` won).
1048 std::vector<double> X(n * n, 0.0);
1049 for (std::size_t i = 0; i < n; ++i) X[i * n + i] = 1.0; // identity
1050 for (std::size_t k = 0; k < n; ++k) // apply LU's row pivots: X = P·I
1051 if (lu.piv[k] != k)
1052 for (std::size_t col = 0; col < n; ++col) std::swap(X[k * n + col], X[lu.piv[k] * n + col]);
1053 for (std::size_t i = 0; i < n; ++i) // forward: unit-lower L·Y = P
1054 for (std::size_t j = 0; j < i; ++j) {
1055 const double f = M[i * n + j];
1056 for (std::size_t col = 0; col < n; ++col) X[i * n + col] -= f * X[j * n + col];
1057 }
1058 for (std::size_t i = n; i-- > 0;) { // back: upper U·X = Y
1059 for (std::size_t j = i + 1; j < n; ++j) {
1060 const double f = M[i * n + j];
1061 for (std::size_t col = 0; col < n; ++col) X[i * n + col] -= f * X[j * n + col];
1062 }
1063 const double d = M[i * n + i];
1064 for (std::size_t col = 0; col < n; ++col) X[i * n + col] /= d;
1065 }
1066 return make_matrix(n, n, std::move(X));
1067}

1073// ---- Cholesky ----
1074NDArray cholesky(const NDArray& a) {
1075 if (a.ndim() != 2) throw std::runtime_error("linalg: expected a 2-D matrix");
1076 const std::size_t n = a.shape()[0], c = a.shape()[1];
1077 require_square(n, c);
1078 std::vector<double> scratch;
1079 const double* A = contig(a, scratch); // read-only — zero-copy when contiguous
1080 std::vector<double> L(n * n, 0.0);
1081 for (std::size_t i = 0; i < n; ++i) {
1082 for (std::size_t j = 0; j <= i; ++j) {
1083 double s = A[i * n + j];
1084 for (std::size_t k = 0; k < j; ++k) s -= L[i * n + k] * L[j * n + k];
1085 if (i == j) {
1086 if (s <= 0.0) throw std::runtime_error("linalg: matrix is not positive-definite");
1087 L[i * n + i] = std::sqrt(s);
1088 } else {
1089 L[i * n + j] = s / L[j * n + j];
1090 }
1091 }
1092 }
1093 return make_matrix(n, n, std::move(L));
1094}

1096// ---- Householder QR (reduced: Q is m×n, R is n×n) ----
1097QR qr(const NDArray& a) {
1098 std::size_t m, n;
1099 std::vector<double> A = as_matrix(a, m, n);
1100 if (m < n) throw std::runtime_error("linalg: qr requires rows >= cols");
1101 std::vector<double> Q(m * m, 0.0);
1102 for (std::size_t i = 0; i < m; ++i) Q[i * m + i] = 1.0;
1103 std::vector<double> u(m); // Householder vector, reused per column (entries < k unused)
1104 for (std::size_t k = 0; k < n; ++k) {
1105 double nrm = 0.0;
1106 for (std::size_t i = k; i < m; ++i) nrm += A[i * n + k] * A[i * n + k];
1107 nrm = std::sqrt(nrm);
1108 if (nrm == 0.0) continue;
1109 const double alpha = A[k * n + k] >= 0 ? -nrm : nrm;
1110 for (std::size_t i = k; i < m; ++i) u[i] = A[i * n + k];
1111 u[k] -= alpha;
1112 double unorm2 = 0.0;
1113 for (std::size_t i = k; i < m; ++i) unorm2 += u[i] * u[i];
1114 if (unorm2 == 0.0) continue;
1115 for (std::size_t j = k; j < n; ++j) {
1116 double s = 0.0;
1117 for (std::size_t i = k; i < m; ++i) s += u[i] * A[i * n + j];
1118 s = 2.0 * s / unorm2;
1119 for (std::size_t i = k; i < m; ++i) A[i * n + j] -= s * u[i];
1120 }
1121 for (std::size_t j = 0; j < m; ++j) { // Q = Q · Hₖ
1122 double s = 0.0;
1123 for (std::size_t i = k; i < m; ++i) s += u[i] * Q[j * m + i];
1124 s = 2.0 * s / unorm2;
1125 for (std::size_t i = k; i < m; ++i) Q[j * m + i] -= s * u[i];
1126 }
1127 }
1128 std::vector<double> Qr(m * n), Rr(n * n, 0.0); // reduced
1129 for (std::size_t i = 0; i < m; ++i)
1130 for (std::size_t j = 0; j < n; ++j) Qr[i * n + j] = Q[i * m + j];
1131 for (std::size_t i = 0; i < n; ++i)
1132 for (std::size_t j = i; j < n; ++j) Rr[i * n + j] = A[i * n + j];
1133 return {make_matrix(m, n, std::move(Qr)), make_matrix(n, n, std::move(Rr))};
1134}

1164NDArray pinv(const NDArray& a) {
1165 std::size_t m, n;
1166 std::vector<double> A = as_matrix(a, m, n);
1167 if (m >= n) {
1168 const SVDc s = svd_golub_reinsch(std::move(A), m, n); // A = U(m×n) diag(w) V(n×n)ᵀ
1169 const double tsh = 0.5 * std::sqrt(double(m + n + 1)) * (s.w.empty() ? 0 : s.w[0]) * 1e-15;
1170 std::vector<double> p(n * m, 0.0); // pinv = V diag(1/w) Uᵀ -> n×m
1171 for (std::size_t i = 0; i < n; ++i)
1172 for (std::size_t j = 0; j < m; ++j) {
1173 double acc = 0.0;
1174 for (std::size_t k = 0; k < n; ++k)
1175 if (s.w[k] > tsh) acc += s.v[i * n + k] * (s.u[j * n + k] / s.w[k]);
1176 p[i * m + j] = acc;
1177 }
1178 return make_matrix(n, m, std::move(p));
1179 }
1180 // m < n: compute on Aᵀ (n×m, rows>=cols) then transpose the result.
1181 std::vector<double> At(n * m);
1182 for (std::size_t i = 0; i < m; ++i)
1183 for (std::size_t j = 0; j < n; ++j) At[j * m + i] = A[i * n + j];
1184 const SVDc s = svd_golub_reinsch(std::move(At), n, m); // Aᵀ = U(n×m) diag(w) V(m×m)ᵀ
1185 const double tsh = 0.5 * std::sqrt(double(n + m + 1)) * (s.w.empty() ? 0 : s.w[0]) * 1e-15;
1186 std::vector<double> res(n * m, 0.0); // pinv(A) = (V diag(1/w) Uᵀ)ᵀ -> n×m
1187 for (std::size_t i = 0; i < m; ++i)
1188 for (std::size_t j = 0; j < n; ++j) {
1189 double acc = 0.0;
1190 for (std::size_t k = 0; k < m; ++k)
1191 if (s.w[k] > tsh) acc += s.v[i * m + k] * (s.u[j * m + k] / s.w[k]);
1192 res[j * m + i] = acc; // transpose into the n×m result
1193 }
1194 return make_matrix(n, m, std::move(res));
1195}
