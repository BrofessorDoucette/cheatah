1// System-level test for the `socket` stdlib module: ONE cohesive program that
2// exercises EVERY purr-callable function in stdlib/socket/socket.hpp end to end
3// (purrc + the C++ backend + the runtime + the linked libcheatah_socket).
4//
5// Unlike the per-function compile-run probes (socket_cr_test.cpp), this stands up
6// real loopback TCP connections on 127.0.0.1 and does full round-trips inside a
7// SINGLE cheatah process, driving both the low-level BSD path and the high-level
8// convenience helpers:
9//
10// low-level : socket -> set_reuseaddr -> bind -> listen -> local_port
11// -> (client) socket -> connect -> accept
12// -> send / recv / sendall / recv -> close
13// high-level: tcp_listen -> local_port -> tcp_connect -> accept
14// -> sendall -> recv
15// error path: connect(-1, ...) deliberately fails -> last_error() is non-empty
16//
17// All 14 exported functions are called: tcp_listen, tcp_connect, accept, recv,
18// send, sendall, close, socket, set_reuseaddr, bind, listen, connect,
19// local_port, last_error.
20//
21// Why it doesn't deadlock in one process: for 127.0.0.1 the kernel completes the
22// TCP handshake on connect() and queues the connection on the listener's accept
23// backlog, so we connect() BEFORE accept() and accept() returns immediately. The
24// payloads are tiny (fit the send buffer), so send()/sendall() never block on a
25// reader. Ports are OS-assigned (bind to port 0) so nothing depends on a fixed
26// port or external network. Output is only booleans + fixed labels, so it is
27// fully deterministic and asserted byte-for-byte. The harness runs the program
28// under `timeout` so a regression that blocks fails fast instead of hanging.
29
30#include "e2e_harness.hpp"
31
32
33TEST(StdlibE2E, Socket) {
34 e2e::expect_e2e("socket_sys", R"PURR(import io
35import socket
36
37# ---- low-level BSD path: socket/set_reuseaddr/bind/listen/local_port ----
38let srv = socket.socket()
39let reuse = socket.set_reuseaddr(srv)
40let bound = socket.bind(srv, "127.0.0.1", 0)
41let lstn = socket.listen(srv, 8)
42let port = socket.local_port(srv)
43
44# client socket, connect BEFORE accept (kernel queues the loopback handshake)
45let cli = socket.socket()
46let conn = socket.connect(cli, "127.0.0.1", port)
47let peer = socket.accept(srv)
48
49# ---- per-connection I/O: send + sendall + recv ----
50let s1 = socket.send(cli, "AB")
51let got1 = socket.recv(peer, 16)
52let s2 = socket.sendall(peer, "PONG")
53let got2 = socket.recv(cli, 16)
54
55# ---- high-level convenience path: tcp_listen / tcp_connect ----
56let hl_srv = socket.tcp_listen("127.0.0.1", 0, 8)
57let hl_port = socket.local_port(hl_srv)
58let hl_cli = socket.tcp_connect("127.0.0.1", hl_port)
59let hl_peer = socket.accept(hl_srv)
60let hl_s = socket.sendall(hl_cli, "hi")
61let hl_got = socket.recv(hl_peer, 16)
62
63# ---- deliberate failure + last_error ----
64let fail = socket.connect(-1, "127.0.0.1", 1)
65let err = socket.last_error()
66
67# ---- verdicts (deterministic) ----
68io.print("socket_ok", srv >= 0)
69io.print("reuse_ok", reuse == 0)
70io.print("bind_ok", bound == 0)
71io.print("listen_ok", lstn == 0)
72io.print("port_ok", port > 0)
73io.print("connect_ok", conn == 0)
74io.print("accept_ok", peer >= 0)
75io.print("send_ok", s1 == 2)
76io.print("recv1_ok", got1 == "AB")
77io.print("sendall_ok", s2 == 0)
78io.print("recv2_ok", got2 == "PONG")
79io.print("tcp_listen_ok", hl_srv >= 0)
80io.print("tcp_connect_ok", hl_cli >= 0)
81io.print("hl_roundtrip_ok", hl_got == "hi")
82io.print("fail_ok", fail < 0)
83io.print("last_error_ok", err != "")
84
85let c1 = socket.close(peer)
86let c2 = socket.close(cli)
87let c3 = socket.close(srv)
88socket.close(hl_peer)
89socket.close(hl_cli)
90socket.close(hl_srv)
91io.print("close_ok", c1 == 0, c2 == 0, c3 == 0)
92)PURR",
