3/**
4 * @file ndarray.hpp
5 * @brief cheatah `ndarray` — our own numpy-flavored N-dimensional array (of
6 * doubles) with NumPy broadcasting, surfaced as a `NDArray` class plus free
7 * functions (a .purr program writes `ndarray.zeros([2, 3])`).
8 * See https://numpy.org/doc/stable/user/basics.broadcasting.html.
9 *
10 * `import ndarray` includes this header and links `libcheatah_ndarray`. Unit tests:
11 * `stdlib/tests/ndarray_test.cpp`; the suite runs under AddressSanitizer (the `asan`
12 * preset) and Valgrind (`security/run-valgrind.sh`) on every QA-gate run.
13 *
14 * @note Design (the "pointers + a bit of thinking"): the elements live in a shared
15 * buffer (`std::shared_ptr<std::vector<double>>`) and an array is a VIEW into
16 * it — {shape, strides, offset}. That makes reshape and **broadcast**
17 * zero-copy: to stretch a dimension of size 1 we give it a stride of 0, so
18 * every index along it reads the same element. Shared ownership = memory-safe,
19 * no manual frees. `size` below is the element count (product of dims).
20 */

28/**
29 * @brief An N-dimensional array of doubles: a view ({shape, strides, offset}) over
30 * a shared element buffer.
31 *
32 * Copies are cheap and share the buffer; reshape/broadcast produce new views without
33 * copying elements. Index math goes through @ref at, which bounds-checks.
34 */
35class NDArray {
36public:
37 /** Construct an empty 0-d array with a fresh empty buffer. @note O(1); allocates the empty shared buffer. @test CheatahNDArray.ToStringScalar */
38 NDArray();
39 /** Construct a contiguous array of @p shape filled with @p fill. @param shape the dimensions. @param fill value for every element. @note O(size); allocates a new NDArray buffer (shared_ptr<vector<double>>) of `product(shape)` elements; throws if the shape overflows size_t. @test CheatahNDArray.ShapeFactoriesAndReductions */
40 explicit NDArray(std::vector<std::size_t> shape, double fill = 0.0); // contiguous
41 /** Construct a view from explicit buffer/shape/strides/offset (used by views). @param data the shared element buffer. @param shape the dimensions. @param strides element strides per dimension. @param offset starting flat offset into @p data. @note O(1); shares @p data, no element copy. @test CheatahNDArray.BroadcastTo */
42 NDArray(std::shared_ptr<std::vector<double>> data, std::vector<std::size_t> shape,
43 std::vector<std::ptrdiff_t> strides, std::size_t offset);
44
45 /** The shape (dimensions). @return reference to the shape vector. @note O(1); no heap. @test CheatahNDArray.ShapeFactoriesAndReductions */
46 const std::vector<std::size_t>& shape() const { return shape_; }
47 /** The element strides. @return reference to the strides vector. @note O(1); no heap. @test CheatahNDArray.BroadcastTo */
48 const std::vector<std::ptrdiff_t>& strides() const { return strides_; }
49 /** The number of dimensions (rank). @return `shape().size()`. @note O(1); no heap. @test CheatahNDArray.BroadcastingAdd */
50 std::size_t ndim() const { return shape_.size(); }
51 /** The element count (product of dims; 1 for a 0-d scalar). @return the number of elements. @note O(ndim); no heap; throws on size overflow. @test CheatahNDArray.ShapeFactoriesAndReductions */
52 std::size_t size() const; // product of dims (1 for a 0-d scalar)
53
54 /** Read one element by multi-index (via strides). @param index one coordinate per dimension. @return the element value. @note O(ndim); no heap; bounds-checks rank and range, throwing on a bad index. @test CheatahNDArray.ShapeFactoriesAndReductions, CheatahNDArray.RejectsMaliciousShapesAndIndices */
55 double at(const std::vector<std::size_t>& index) const; // element via strides
56 /** The shared backing buffer. @return reference to the element buffer shared_ptr. @note O(1); no heap. @test CheatahNDArray.BroadcastingAdd */
57 const std::shared_ptr<std::vector<double>>& buffer() const { return data_; }
58 /** The flat offset into the buffer where this view starts. @return the offset. @note O(1); no heap. @test CheatahNDArray.BroadcastTo */
59 std::size_t offset() const { return offset_; }
60
61private:
62 std::shared_ptr<std::vector<double>> data_;
63 std::vector<std::size_t> shape_;
64 std::vector<std::ptrdiff_t> strides_; // element strides
65 std::size_t offset_ = 0;
66};
