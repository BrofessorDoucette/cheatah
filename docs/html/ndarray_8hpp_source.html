3/**
4 * @file ndarray.hpp
5 * @brief cheatah `ndarray` — our own numpy-flavored N-dimensional array (of
6 * doubles) with NumPy broadcasting, surfaced as a `NDArray` class plus free
7 * functions (a .purr program writes `ndarray.zeros([2, 3])`).
8 * See https://numpy.org/doc/stable/user/basics.broadcasting.html.
9 *
10 * `import ndarray` includes this header and links `libcheatah_ndarray`. Unit tests:
11 * `stdlib/tests/ndarray_test.cpp`; the suite runs under AddressSanitizer (the `asan`
12 * preset) and Valgrind (`security/run-valgrind.sh`) on every QA-gate run.
13 *
14 * @note Design (the "pointers + a bit of thinking"): the elements live in a shared
15 * buffer (`std::shared_ptr<std::vector<double>>`) and an array is a VIEW into
16 * it — {shape, strides, offset}. That makes reshape and **broadcast**
17 * zero-copy: to stretch a dimension of size 1 we give it a stride of 0, so
18 * every index along it reads the same element. Shared ownership = memory-safe,
19 * no manual frees. `size` below is the element count (product of dims).
20 */

28/**
29 * @brief An N-dimensional array of doubles: a view ({shape, strides, offset}) over
30 * a shared element buffer.
31 *
32 * Copies are cheap and share the buffer; reshape/broadcast produce new views without
33 * copying elements. Index math goes through @ref at, which bounds-checks.
34 */
35class NDArray {
36public:
37 /**
38 * Construct an empty 0-d array with a fresh empty buffer.
39 * @note O(1); allocates the empty shared buffer.
40 * @test CheatahNDArray.ToStringScalar
41 */
42 NDArray();
43 /**
44 * Construct a contiguous array of @p shape filled with @p fill.
45 * @param shape the dimensions.
46 * @param fill value for every element.
47 * @note O(size); allocates a new NDArray buffer (shared_ptr<vector<double>>) of
48 * `product(shape)` elements; throws if the shape overflows size_t.
49 * @test CheatahNDArray.ShapeFactoriesAndReductions
50 */
51 explicit NDArray(std::vector<std::size_t> shape, double fill = 0.0); // contiguous
52 /**
53 * Construct a view from explicit buffer/shape/strides/offset (used by views).
54 * @param data the shared element buffer.
55 * @param shape the dimensions.
56 * @param strides element strides per dimension.
57 * @param offset starting flat offset into @p data.
58 * @note O(1); shares @p data, no element copy.
59 * @test CheatahNDArray.BroadcastTo
60 */
61 NDArray(std::shared_ptr<std::vector<double>> data, std::vector<std::size_t> shape,
62 std::vector<std::ptrdiff_t> strides, std::size_t offset);
63
64 /**
65 * The shape (dimensions).
66 * @return reference to the shape vector.
67 * @note O(1); no heap.
68 * @test CheatahNDArray.ShapeFactoriesAndReductions
69 */
70 const std::vector<std::size_t>& shape() const { return shape_; }
71 /**
72 * The element strides.
73 * @return reference to the strides vector.
74 * @note O(1); no heap.
75 * @test CheatahNDArray.BroadcastTo
76 */
77 const std::vector<std::ptrdiff_t>& strides() const { return strides_; }
78 /**
79 * The number of dimensions (rank).
80 * @return `shape().size()`.
81 * @note O(1); no heap.
82 * @test CheatahNDArray.BroadcastingAdd
83 */
84 std::size_t ndim() const { return shape_.size(); }
85 /**
86 * The element count (product of dims; 1 for a 0-d scalar).
87 * @return the number of elements.
88 * @note O(ndim); no heap; throws on size overflow.
89 * @test CheatahNDArray.ShapeFactoriesAndReductions
90 */
91 std::size_t size() const; // product of dims (1 for a 0-d scalar)
92
93 /**
94 * Read one element by multi-index (via strides).
95 * @param index one coordinate per dimension.
96 * @return the element value.
97 * @note O(ndim); no heap; bounds-checks rank and range, throwing on a bad index.
98 * @test CheatahNDArray.ShapeFactoriesAndReductions,
99 * CheatahNDArray.RejectsMaliciousShapesAndIndices
100 */
101 double at(const std::vector<std::size_t>& index) const; // element via strides
102 /**
103 * The shared backing buffer.
104 * @return reference to the element buffer shared_ptr.
105 * @note O(1); no heap.
106 * @test CheatahNDArray.BroadcastingAdd
107 */
108 const std::shared_ptr<std::vector<double>>& buffer() const { return data_; }
109 /**
110 * The flat offset into the buffer where this view starts.
111 * @return the offset.
112 * @note O(1); no heap.
113 * @test CheatahNDArray.BroadcastTo
114 */
115 std::size_t offset() const { return offset_; }
116
117private:
118 std::shared_ptr<std::vector<double>> data_;
119 std::vector<std::size_t> shape_;
120 std::vector<std::ptrdiff_t> strides_; // element strides
121 std::size_t offset_ = 0;
122};
