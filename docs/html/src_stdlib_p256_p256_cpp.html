3// p256.cpp — NIST P-256 (secp256r1) ECDSA, from scratch. See p256.hpp.
4//
5// Representation: a 256-bit value is uint64_t[4], LEAST-significant limb first.
6// Arithmetic mod the field prime p and mod the group order n is done in
7// Montgomery form (R = 2^256); the Montgomery constants are derived from the
8// modulus at startup, so there are no hand-transcribed magic numbers to get
9// wrong. Points are Jacobian (X:Y:Z) with the curve a = -3.

39// ---- plain 256-bit helpers -------------------------------------------------
40bool is_zero(const fe& a) { return (a[0] | a[1] | a[2] | a[3]) == 0; }
41bool geq(const fe& a, const fe& b) { // a >= b
42 for (int i = 3; i >= 0; --i)
43 if (a[i] != b[i]) return a[i] > b[i];
44 return true;
45}
46// a - b (mod 2^256), returns borrow.
47u64 sub_borrow(fe& r, const fe& a, const fe& b) {
48 u128 br = 0;
49 for (int i = 0; i < 4; ++i) {
50 u128 d = (u128)a[i] - b[i] - br;
51 r[i] = (u64)d;
52 br = (d >> 64) & 1;
53 }
54 return (u64)br;
55}
56// a + b (mod 2^256), returns carry.
57u64 add_carry(fe& r, const fe& a, const fe& b) {
58 u128 c = 0;
59 for (int i = 0; i < 4; ++i) {
60 u128 s = (u128)a[i] + b[i] + c;
61 r[i] = (u64)s;
62 c = s >> 64;
63 }
64 return (u64)c;
65}

67// ---- Montgomery context for one modulus ------------------------------------
68struct Mont {
69 fe m; // the modulus
70 fe rr; // R^2 mod m (R = 2^256)
71 fe one; // R mod m (Montgomery form of 1)
72 u64 n0; // -m^{-1} mod 2^64
73};
74
75// CIOS Montgomery multiplication: r = a*b*R^-1 mod m.
76void mont_mul(fe& r, const fe& a, const fe& b, const Mont& M) {
77 u64 t[5] = {0, 0, 0, 0, 0};
78 for (int i = 0; i < 4; ++i) {
79 // t += a * b[i]
80 u128 carry = 0;
81 for (int j = 0; j < 4; ++j) {
82 u128 p = (u128)a[j] * b[i] + t[j] + carry;
83 t[j] = (u64)p;
84 carry = p >> 64;
85 }
86 u128 s = (u128)t[4] + carry;
87 t[4] = (u64)s;
88 u64 top = (u64)(s >> 64);
89 // m_mul = t[0] * n0 mod 2^64; t += m_mul * m; then shift right one limb
90 u64 mmul = (u64)((u128)t[0] * M.n0);
91 carry = 0;
92 {
93 u128 p = (u128)mmul * M.m[0] + t[0];
94 carry = p >> 64; // low limb becomes 0
95 }
96 for (int j = 1; j < 4; ++j) {
97 u128 p = (u128)mmul * M.m[j] + t[j] + carry;
98 t[j - 1] = (u64)p;
99 carry = p >> 64;
100 }
101 u128 s2 = (u128)t[4] + carry;
102 t[3] = (u64)s2;
103 t[4] = top + (u64)(s2 >> 64);
104 }
105 fe res = {t[0], t[1], t[2], t[3]};
106 // final conditional subtraction (t may be in [0, 2m))
107 if (t[4] != 0 || geq(res, M.m)) {
108 fe tmp;
109 sub_borrow(tmp, res, M.m);
110 res = tmp;
111 }
112 r = res;
113}
114void mont_add(fe& r, const fe& a, const fe& b, const Mont& M) {
115 fe s;
116 u64 c = add_carry(s, a, b);
117 if (c || geq(s, M.m)) {
118 fe t;
119 sub_borrow(t, s, M.m);
120 s = t;
121 }
122 r = s;
123}
124void mont_sub(fe& r, const fe& a, const fe& b, const Mont& M) {
125 fe d;
126 u64 br = sub_borrow(d, a, b);
127 if (br) {
128 fe t;
129 add_carry(t, d, M.m);
130 d = t;
131 }
132 r = d;
133}
134void to_mont(fe& r, const fe& a, const Mont& M) { mont_mul(r, a, M.rr, M); }
135void from_mont(fe& r, const fe& a, const Mont& M) {
136 fe one = {1, 0, 0, 0};
137 mont_mul(r, a, one, M);
138}
139// r = a^-1 mod m, via Fermat: a^(m-2). (m is prime for both p and n.)
140void mont_inv(fe& r, const fe& a, const Mont& M) {
141 fe exp;
142 sub_borrow(exp, M.m, fe{2, 0, 0, 0}); // m - 2
143 fe result = M.one; // Montgomery 1
144 fe base = a;
145 for (int i = 0; i < 256; ++i) {
146 if ((exp[i / 64] >> (i % 64)) & 1) mont_mul(result, result, base, M);
147 mont_mul(base, base, base, M);
148 }
149 r = result;
150}

152u64 inv64(u64 a) { // a^-1 mod 2^64 (a odd), Newton's iteration
153 u64 x = a; // correct to 3 bits
154 for (int i = 0; i < 5; ++i) x *= 2 - a * x;
155 return x;
156}
157Mont make_mont(const fe& m) {
158 Mont M;
159 M.m = m;
160 M.n0 = 0 - inv64(m[0]);
161 // rr = 2^512 mod m, by 512 doublings of 1 with conditional subtract.
162 fe x = {1, 0, 0, 0};
163 for (int i = 0; i < 512; ++i) {
164 fe d;
165 u64 c = add_carry(d, x, x);
166 if (c || geq(d, m)) {
167 fe t;
168 sub_borrow(t, d, m);
169 d = t;
170 }
171 x = d;
172 }
173 M.rr = x;
174 // one = R mod m = 2^256 mod m -> to_mont(1)
175 fe oneN = {1, 0, 0, 0};
176 mont_mul(M.one, oneN, M.rr, M);
177 return M;
178}

190// ---- bytes <-> limbs (32 big-endian bytes) ---------------------------------
191fe be_to_fe(const unsigned char* b) {
192 fe r{};
193 for (int limb = 0; limb < 4; ++limb) {
194 u64 v = 0;
195 const unsigned char* p = b + (3 - limb) * 8; // most-significant 8 bytes -> limb[3]
196 for (int k = 0; k < 8; ++k) v = (v << 8) | p[k];
197 r[limb] = v;
198 }
199 return r;
200}
201void fe_to_be(unsigned char* out, const fe& a) {
202 for (int limb = 0; limb < 4; ++limb) {
203 u64 v = a[limb];
204 unsigned char* p = out + (3 - limb) * 8;
205 for (int k = 7; k >= 0; --k) {
206 p[k] = (unsigned char)(v & 0xFF);
207 v >>= 8;
208 }
209 }
210}

212// ---- Jacobian points (coordinates in Montgomery form, mod p) ---------------
213struct Jac {
214 fe X, Y, Z;
215 bool inf;
216};
217Jac jac_infinity() { return Jac{Fp().one, Fp().one, {0, 0, 0, 0}, true}; }
218
219void jac_double(Jac& r, const Jac& q) {
220 const Mont& F = Fp();
221 if (q.inf || is_zero(q.Z)) {
222 r = jac_infinity();
223 return;
224 }
225 fe A, B, C, D, t1, t2;
226 mont_mul(A, q.X, q.X, F); // X^2
227 mont_mul(B, q.Y, q.Y, F); // Y^2
228 mont_mul(C, B, B, F); // Y^4
229 // D = 2*((X+B)^2 - A - C)
230 mont_add(t1, q.X, B, F);
231 mont_mul(t1, t1, t1, F);
232 mont_sub(t1, t1, A, F);
233 mont_sub(t1, t1, C, F);
234 mont_add(D, t1, t1, F);
235 // ZZ = Z^2 ; E = 3*(X - ZZ)*(X + ZZ) [uses a = -3]
236 fe ZZ;
237 mont_mul(ZZ, q.Z, q.Z, F);
238 mont_sub(t1, q.X, ZZ, F);
239 mont_add(t2, q.X, ZZ, F);
240 mont_mul(t1, t1, t2, F);
241 fe E;
242 mont_add(E, t1, t1, F);
243 mont_add(E, E, t1, F); // 3*(...)
244 // F2 = E^2 ; X3 = F2 - 2D
245 fe X3;
246 mont_mul(X3, E, E, F);
247 mont_sub(X3, X3, D, F);
248 mont_sub(X3, X3, D, F);
249 // Y3 = E*(D - X3) - 8C
250 fe Y3, eight;
251 mont_sub(t1, D, X3, F);
252 mont_mul(Y3, E, t1, F);
253 mont_add(eight, C, C, F);
254 mont_add(eight, eight, eight, F);
255 mont_add(eight, eight, eight, F); // 8C
256 mont_sub(Y3, Y3, eight, F);
257 // Z3 = 2*Y*Z
258 fe Z3;
259 mont_mul(Z3, q.Y, q.Z, F);
260 mont_add(Z3, Z3, Z3, F);
261 r = Jac{X3, Y3, Z3, false};
262}

264void jac_add(Jac& r, const Jac& a, const Jac& b) {
265 const Mont& F = Fp();
266 if (a.inf || is_zero(a.Z)) {
267 r = b;
268 return;
269 }
270 if (b.inf || is_zero(b.Z)) {
271 r = a;
272 return;
273 }
274 fe Z1Z1, Z2Z2, U1, U2, S1, S2;
275 mont_mul(Z1Z1, a.Z, a.Z, F);
276 mont_mul(Z2Z2, b.Z, b.Z, F);
277 mont_mul(U1, a.X, Z2Z2, F);
278 mont_mul(U2, b.X, Z1Z1, F);
279 fe t;
280 mont_mul(t, b.Z, Z2Z2, F);
281 mont_mul(S1, a.Y, t, F);
282 mont_mul(t, a.Z, Z1Z1, F);
283 mont_mul(S2, b.Y, t, F);
284 fe H, Rr;
285 mont_sub(H, U2, U1, F);
286 mont_sub(Rr, S2, S1, F);
287 if (is_zero(H)) {
288 if (is_zero(Rr)) {
289 jac_double(r, a);
290 return;
291 }
292 r = jac_infinity();
293 return;
294 }
295 fe HH, HHH, V;
296 mont_mul(HH, H, H, F);
297 mont_mul(HHH, HH, H, F);
298 mont_mul(V, U1, HH, F);
299 fe X3;
300 mont_mul(X3, Rr, Rr, F);
301 mont_sub(X3, X3, HHH, F);
302 mont_sub(X3, X3, V, F);
303 mont_sub(X3, X3, V, F);
304 fe Y3;
305 mont_sub(t, V, X3, F);
306 mont_mul(Y3, Rr, t, F);
307 fe s1hhh;
308 mont_mul(s1hhh, S1, HHH, F);
309 mont_sub(Y3, Y3, s1hhh, F);
310 fe Z3;
311 mont_mul(Z3, a.Z, b.Z, F);
312 mont_mul(Z3, Z3, H, F);
313 r = Jac{X3, Y3, Z3, false};
314}

316// Strauss-Shamir: u1*A + u2*B with ONE doubling chain (256 doublings total)
317// instead of two separate scalar multiplications (512). A 2-bit window over both
318// scalars uses a 16-entry combined table [i*A + j*B] so it also halves the adds.
319void jac_double_mul(Jac& r, const fe& u1, const Jac& A, const fe& u2, const Jac& B) {
320 Jac tbl[4][4]; // tbl[i][j] = i*A + j*B, i,j in {0..3}
321 tbl[0][0] = jac_infinity();
322 tbl[1][0] = A;
323 jac_double(tbl[2][0], A);
324 jac_add(tbl[3][0], tbl[2][0], A);
325 tbl[0][1] = B;
326 jac_double(tbl[0][2], B);
327 jac_add(tbl[0][3], tbl[0][2], B);
328 for (int i = 1; i < 4; ++i)
329 for (int j = 1; j < 4; ++j) jac_add(tbl[i][j], tbl[i][0], tbl[0][j]);
330
331 Jac acc = jac_infinity();
332 for (int i = 254; i >= 0; i -= 2) {
333 Jac t;
334 jac_double(t, acc);
335 acc = t;
336 jac_double(t, acc);
337 acc = t;
338 const unsigned a = (u1[i / 64] >> (i % 64)) & 0x3;
339 const unsigned b = (u2[i / 64] >> (i % 64)) & 0x3;
340 if (a || b) {
341 jac_add(t, acc, tbl[a][b]);
342 acc = t;
343 }
344 }
345 r = acc;
346}

348// affine x-coordinate (normal form) of a Jacobian point: x = X / Z^2.
349fe jac_affine_x(const Jac& q) {
350 const Mont& F = Fp();
351 fe zinv, zinv2, x;
352 mont_inv(zinv, q.Z, F);
353 mont_mul(zinv2, zinv, zinv, F);
354 mont_mul(x, q.X, zinv2, F);
355 fe out;
356 from_mont(out, x, F);
357 return out;
358}

360Jac affine_to_jac(const fe& x, const fe& y) {
361 const Mont& F = Fp();
362 Jac p;
363 to_mont(p.X, x, F);
364 to_mont(p.Y, y, F);
365 p.Z = F.one;
366 p.inf = false;
367 return p;
368}
369const Jac& base_point() {
370 static const Jac g = affine_to_jac(GX, GY);
371 return g;
372}

374// Fixed-base comb for k*G. G is constant, so we precompute (once) the 16-entry
375// table T[s] = sum over set bits i of s of (2^(64*i) * G). Then k*G is just 64
376// doublings + 64 adds (vs 256 doublings for a generic window) — the big win for
377// the per-message signing path. Selector at step j is bit j of each of the four
378// 64-bit limbs.
379const std::array<Jac, 16>& g_comb() {
380 static const std::array<Jac, 16> tbl = [] {
381 Jac gi[4];
382 gi[0] = affine_to_jac(GX, GY);
383 for (int i = 1; i < 4; ++i) {
384 Jac acc = gi[i - 1];
385 for (int b = 0; b < 64; ++b) { // gi[i] = 2^64 * gi[i-1]
386 Jac t;
387 jac_double(t, acc);
388 acc = t;
389 }
390 gi[i] = acc;
391 }
392 std::array<Jac, 16> t;
393 t[0] = jac_infinity();
394 for (unsigned s = 1; s < 16; ++s) {
395 Jac acc = jac_infinity();
396 for (int i = 0; i < 4; ++i)
397 if (s & (1u << i)) {
398 Jac r;
399 jac_add(r, acc, gi[i]);
400 acc = r;
401 }
402 t[s] = acc;
403 }
404 return t;
405 }();
406 return tbl;
407}

409void jac_mul_base(Jac& r, const fe& k) {
410 const std::array<Jac, 16>& T = g_comb();
411 Jac acc = jac_infinity();
412 for (int j = 63; j >= 0; --j) {
413 Jac t;
414 jac_double(t, acc);
415 acc = t;
416 const unsigned sel = ((k[0] >> j) & 1u) | (((k[1] >> j) & 1u) << 1) |
417 (((k[2] >> j) & 1u) << 2) | (((k[3] >> j) & 1u) << 3);
418 if (sel) {
419 jac_add(t, acc, T[sel]);
420 acc = t;
421 }
422 }
423 r = acc;
424}

426// Reduce a scalar already known to be < 2n into [0, n): a single conditional
427// subtraction of the group order n. Used for the FIPS 186-4 hash truncation and
428// for folding a curve x-coordinate (which lives in [0, p) < 2n) into a scalar.
429fe reduce_mod_n(const fe& v) {
430 if (geq(v, N)) {
431 fe t;
432 sub_borrow(t, v, N);
433 return t;
434 }
435 return v;
436}

438// reduce a 32-byte big-endian hash to a scalar in [0, n) (FIPS 186-4 leftmost bits).
439fe hash_to_scalar(const std::string& h) {
440 unsigned char buf[32] = {0};
441 const std::size_t take = h.size() < 32 ? h.size() : 32;
442 std::memcpy(buf, h.data(), take); // leftmost 32 bytes
443 return reduce_mod_n(be_to_fe(buf));
444}

446// ---- minimal DER helpers ---------------------------------------------------
447// Parse SEQUENCE{INTEGER r, INTEGER s} -> 32-byte big-endian r and s. false on error.
448bool der_to_rs(const std::string& der, unsigned char r[32], unsigned char s[32]) {
449 const unsigned char* p = (const unsigned char*)der.data();
450 std::size_t n = der.size(), i = 0;
451 auto read_int = [&](unsigned char out[32]) -> bool {
452 if (i >= n || p[i++] != 0x02) return false;
453 if (i >= n) return false;
454 std::size_t len = p[i++];
455 if (len & 0x80) return false; // P-256 ints are short-form
456 if (i + len > n || len == 0) return false;
457 const unsigned char* v = p + i;
458 // strip a leading zero (sign byte)
459 while (len > 1 && v[0] == 0) {
460 ++v;
461 --len;
462 }
463 if (len > 32) return false;
464 std::memset(out, 0, 32);
465 std::memcpy(out + (32 - len), v, len);
466 i += (std::size_t)(v - (p + i)) + len; // advance past the original field
467 return true;
468 };
469 if (i >= n || p[i++] != 0x30) return false;
470 if (i >= n) return false;
471 std::size_t seqlen = p[i++];
472 if (seqlen & 0x80) return false;
473 if (i + seqlen != n) return false;
474 return read_int(r) && read_int(s);
475}

482// used by verify/sign. The one-conditional-subtraction path (value in [n, p)) is a

495// Is (x, y) on the P-256 curve y^2 = x^3 - 3x + b (mod p)? Rejects an off-curve /
496// invalid-curve public key — SP 800-56A / FIPS 186 point validation, which the plain
497// coordinate-range check (x,y < p) does not catch.
498bool on_curve(const fe& x, const fe& y) {
499 const Mont& F = Fp();
500 fe xm, ym, x2, x3, tx, rhs, bm, y2;
501 to_mont(xm, x, F);
502 to_mont(ym, y, F);
503 mont_mul(x2, xm, xm, F); // x^2
504 mont_mul(x3, x2, xm, F); // x^3
505 mont_add(tx, xm, xm, F); // 2x
506 mont_add(tx, tx, xm, F); // 3x
507 mont_sub(rhs, x3, tx, F); // x^3 - 3x
508 to_mont(bm, CURVE_B, F);
509 mont_add(rhs, rhs, bm, F); // x^3 - 3x + b
510 mont_mul(y2, ym, ym, F); // y^2
511 return std::memcmp(y2.data(), rhs.data(), sizeof(fe)) == 0;
512}

514bool verify_raw(const std::string& pubkey_xy, const std::string& msg_hash,
515 const std::string& sig_raw) {
516 if (pubkey_xy.size() != 64 || sig_raw.size() != 64) return false;
517 fe r = be_to_fe((const unsigned char*)sig_raw.data());
518 fe s = be_to_fe((const unsigned char*)sig_raw.data() + 32);
519 if (is_zero(r) || is_zero(s) || geq(r, N) || geq(s, N)) return false;
520
521 const Mont& Fnn = Fn();
522 fe e = hash_to_scalar(msg_hash);
523 fe sm, em, rm, w, u1, u2;
524 to_mont(sm, s, Fnn);
525 mont_inv(w, sm, Fnn); // w = s^-1 (Montgomery)
526 to_mont(em, e, Fnn);
527 to_mont(rm, r, Fnn);
528 fe u1m, u2m;
529 mont_mul(u1m, em, w, Fnn);
530 mont_mul(u2m, rm, w, Fnn);
531 from_mont(u1, u1m, Fnn);
532 from_mont(u2, u2m, Fnn);
533
534 fe qx = be_to_fe((const unsigned char*)pubkey_xy.data());
535 fe qy = be_to_fe((const unsigned char*)pubkey_xy.data() + 32);
536 if (geq(qx, P) || geq(qy, P)) return false;
537 if (!on_curve(qx, qy)) return false; // reject off-curve / invalid-curve public keys
538 Jac Q = affine_to_jac(qx, qy);
539
540 Jac R;
541 jac_double_mul(R, u1, base_point(), u2, Q); // u1*G + u2*Q, one doubling chain
542 if (R.inf || is_zero(R.Z)) return false;
543 fe x = reduce_mod_n(jac_affine_x(R));
544 return std::memcmp(x.data(), r.data(), sizeof(fe)) == 0;
545}
546
547bool verify_der(const std::string& pubkey_xy, const std::string& msg_hash,
548 const std::string& sig_der) {
549 unsigned char r[32], s[32];
550 if (!der_to_rs(sig_der, r, s)) return false;
551 std::string raw;
552 raw.resize(64);
553 std::memcpy(raw.data(), r, 32);
554 std::memcpy(raw.data() + 32, s, 32);
555 return verify_raw(pubkey_xy, msg_hash, raw);
556}

558#ifdef CHEATAH_P256_TESTING
559// In test builds the RFC 6979 retry tail — normally a ~2^-128 event on real inputs —
560// is driven by rejecting the first `force_retries` otherwise-valid nonce candidates.
561// This whole parameter and its use compile out of release builds.
562std::string sign_raw_impl(const std::string& privkey, const std::string& msg_hash,
563 int force_retries) {
564#else
565std::string sign_raw(const std::string& privkey, const std::string& msg_hash) {
566#endif
567 if (privkey.size() != 32) return std::string();
568 const Mont& Fnn = Fn();
569 fe d = be_to_fe((const unsigned char*)privkey.data());
570 if (is_zero(d) || geq(d, N)) return std::string();
571 fe e = hash_to_scalar(msg_hash);
572
573 // RFC 6979 deterministic nonce generation (HMAC-SHA256).
574 unsigned char h1[32] = {0};
575 std::memcpy(h1, msg_hash.data(), msg_hash.size() < 32 ? msg_hash.size() : 32);
576 unsigned char x[32];
577 fe_to_be(x, d);
578 std::string V(32, '\x01'), K(32, '\x00');
579 auto bin = [](const fe&) {};
580 (void)bin;
581 auto hmac = [](const std::string& key, const std::string& msg) {
582 return hashlib::hmac_sha256(key, msg); // raw 32-byte digest
583 };
584 // K = HMAC(K, V || 0x00 || int2octets(x) || bits2octets(h1))
585 auto step = [&](unsigned char tag) {
586 std::string in = V;
587 if (tag != 0xFF) in.push_back((char)tag);
588 if (tag != 0xFF) {
589 in.append((const char*)x, 32);
590 in.append((const char*)h1, 32);
591 }
592 K = hmac(K, in);
593 V = hmac(K, V);
594 };
595 step(0x00);
596 step(0x01);
597 for (int attempt = 0; attempt < 64; ++attempt) {
598 V = hmac(K, V);
599 fe k = be_to_fe((const unsigned char*)V.data());
600 if (!is_zero(k) && !geq(k, N)) {
601 Jac R;
602 jac_mul_base(R, k); // fixed-base comb
603 if (!(R.inf || is_zero(R.Z))) {
604 fe rx = reduce_mod_n(jac_affine_x(R));
605#ifdef CHEATAH_P256_TESTING
606 if (force_retries > 0) {
607 --force_retries; // reject this valid candidate; take the retry tail
608 } else if (!is_zero(rx)) {
609#else
610 if (!is_zero(rx)) {
611#endif
612 // s = k^-1 (e + r*d) mod n
613 fe km, kinv, rm, dm, em, rd, sum, sm;
614 to_mont(km, k, Fnn);
615 mont_inv(kinv, km, Fnn);
616 to_mont(rm, rx, Fnn);
617 to_mont(dm, d, Fnn);
618 to_mont(em, e, Fnn);
619 mont_mul(rd, rm, dm, Fnn);
620 mont_add(sum, em, rd, Fnn);
621 mont_mul(sm, kinv, sum, Fnn);
622 fe sfinal;
623 from_mont(sfinal, sm, Fnn);
624 if (!is_zero(sfinal)) {
625 std::string out(64, '\0');
626 fe_to_be((unsigned char*)out.data(), rx);
627 fe_to_be((unsigned char*)out.data() + 32, sfinal);
628 return out;
629 }
630 }
631 }
632 }
633 // K = HMAC(K, V || 0x00); V = HMAC(K, V)
634 std::string in = V;
635 in.push_back('\x00');
636 K = hmac(K, in);
637 V = hmac(K, V);
638 }
639 return std::string();
640}

643std::string sign_raw(const std::string& privkey, const std::string& msg_hash) {
644 return sign_raw_impl(privkey, msg_hash, 0);
645}

657std::string public_from_private(const std::string& privkey) {
658 if (privkey.size() != 32) return std::string();
659 fe d = be_to_fe((const unsigned char*)privkey.data());
660 if (is_zero(d) || geq(d, N)) return std::string();
661 Jac Q;
662 jac_mul_base(Q, d); // d*G via the fixed-base comb
663 if (Q.inf || is_zero(Q.Z)) return std::string();
664 // affine x and y (normal form)
665 const Mont& F = Fp();
666 fe zinv, zinv2, zinv3, x, y;
667 mont_inv(zinv, Q.Z, F);
668 mont_mul(zinv2, zinv, zinv, F);
669 mont_mul(zinv3, zinv2, zinv, F);
670 mont_mul(x, Q.X, zinv2, F);
671 mont_mul(y, Q.Y, zinv3, F);
672 fe xo, yo;
673 from_mont(xo, x, F);
674 from_mont(yo, y, F);
675 std::string out(64, '\0');
676 fe_to_be((unsigned char*)out.data(), xo);
677 fe_to_be((unsigned char*)out.data() + 32, yo);
678 return out;
679}

681std::string spki_ec_point(std::string_view der) {
682 // Find the uncompressed-point marker: BIT STRING (03) <len> 00 04 <X(32)><Y(32)>.
683 // The OID id-ecPublicKey + prime256v1 precedes it; we anchor on the 0x04 point.
684 const unsigned char* p = (const unsigned char*)der.data();
685 const std::size_t n = der.size();
686 for (std::size_t i = 0; i + 2 + 65 <= n; ++i) {
687 // BIT STRING tag, then a length, then 00 (unused bits), then 04 (uncompressed)
688 if (p[i] == 0x03 && p[i + 2] == 0x00 && p[i + 3] == 0x04) {
689 const std::size_t len = p[i + 1];
690 if (len == 66 && i + 4 + 64 <= n) return std::string((const char*)p + i + 4, 64);
691 }
692 }
693 return std::string();
694}
