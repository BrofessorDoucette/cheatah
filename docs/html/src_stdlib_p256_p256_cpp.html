1// p256.cpp — NIST P-256 (secp256r1) ECDSA, from scratch. See p256.hpp.
2//
3// Representation: a 256-bit value is uint64_t[4], LEAST-significant limb first.
4// Arithmetic mod the field prime p and mod the group order n is done in
5// Montgomery form (R = 2^256); the Montgomery constants are derived from the
6// modulus at startup, so there are no hand-transcribed magic numbers to get
7// wrong. Points are Jacobian (X:Y:Z) with the curve a = -3.

37// ---- plain 256-bit helpers -------------------------------------------------
38bool is_zero(const fe& a) { return (a[0] | a[1] | a[2] | a[3]) == 0; }
39bool geq(const fe& a, const fe& b) { // a >= b
40 for (int i = 3; i >= 0; --i)
41 if (a[i] != b[i]) return a[i] > b[i];
42 return true;
43}
44// a - b (mod 2^256), returns borrow.
45u64 sub_borrow(fe& r, const fe& a, const fe& b) {
46 u128 br = 0;
47 for (int i = 0; i < 4; ++i) {
48 u128 d = (u128)a[i] - b[i] - br;
49 r[i] = (u64)d;
50 br = (d >> 64) & 1;
51 }
52 return (u64)br;
53}
54// a + b (mod 2^256), returns carry.
55u64 add_carry(fe& r, const fe& a, const fe& b) {
56 u128 c = 0;
57 for (int i = 0; i < 4; ++i) {
58 u128 s = (u128)a[i] + b[i] + c;
59 r[i] = (u64)s;
60 c = s >> 64;
61 }
62 return (u64)c;
63}

65// ---- Montgomery context for one modulus ------------------------------------
66struct Mont {
67 fe m; // the modulus
68 fe rr; // R^2 mod m (R = 2^256)
69 fe one; // R mod m (Montgomery form of 1)
70 u64 n0; // -m^{-1} mod 2^64
71};
72
73// CIOS Montgomery multiplication: r = a*b*R^-1 mod m.
74void mont_mul(fe& r, const fe& a, const fe& b, const Mont& M) {
75 u64 t[5] = {0, 0, 0, 0, 0};
76 for (int i = 0; i < 4; ++i) {
77 // t += a * b[i]
78 u128 carry = 0;
79 for (int j = 0; j < 4; ++j) {
80 u128 p = (u128)a[j] * b[i] + t[j] + carry;
81 t[j] = (u64)p;
82 carry = p >> 64;
83 }
84 u128 s = (u128)t[4] + carry;
85 t[4] = (u64)s;
86 u64 top = (u64)(s >> 64);
87 // m_mul = t[0] * n0 mod 2^64; t += m_mul * m; then shift right one limb
88 u64 mmul = (u64)((u128)t[0] * M.n0);
89 carry = 0;
90 {
91 u128 p = (u128)mmul * M.m[0] + t[0];
92 carry = p >> 64; // low limb becomes 0
93 }
94 for (int j = 1; j < 4; ++j) {
95 u128 p = (u128)mmul * M.m[j] + t[j] + carry;
96 t[j - 1] = (u64)p;
97 carry = p >> 64;
98 }
99 u128 s2 = (u128)t[4] + carry;
100 t[3] = (u64)s2;
101 t[4] = top + (u64)(s2 >> 64);
102 }
103 fe res = {t[0], t[1], t[2], t[3]};
104 // final conditional subtraction (t may be in [0, 2m))
105 if (t[4] != 0 || geq(res, M.m)) {
106 fe tmp;
107 sub_borrow(tmp, res, M.m);
108 res = tmp;
109 }
110 r = res;
111}
112void mont_add(fe& r, const fe& a, const fe& b, const Mont& M) {
113 fe s;
114 u64 c = add_carry(s, a, b);
115 if (c || geq(s, M.m)) {
116 fe t;
117 sub_borrow(t, s, M.m);
118 s = t;
119 }
120 r = s;
121}
122void mont_sub(fe& r, const fe& a, const fe& b, const Mont& M) {
123 fe d;
124 u64 br = sub_borrow(d, a, b);
125 if (br) {
126 fe t;
127 add_carry(t, d, M.m);
128 d = t;
129 }
130 r = d;
131}
132void to_mont(fe& r, const fe& a, const Mont& M) { mont_mul(r, a, M.rr, M); }
133void from_mont(fe& r, const fe& a, const Mont& M) {
134 fe one = {1, 0, 0, 0};
135 mont_mul(r, a, one, M);
136}
137// r = a^-1 mod m, via Fermat: a^(m-2). (m is prime for both p and n.)
138void mont_inv(fe& r, const fe& a, const Mont& M) {
139 fe exp;
140 sub_borrow(exp, M.m, fe{2, 0, 0, 0}); // m - 2
141 fe result = M.one; // Montgomery 1
142 fe base = a;
143 for (int i = 0; i < 256; ++i) {
144 if ((exp[i / 64] >> (i % 64)) & 1) mont_mul(result, result, base, M);
145 mont_mul(base, base, base, M);
146 }
147 r = result;
148}

150u64 inv64(u64 a) { // a^-1 mod 2^64 (a odd), Newton's iteration
151 u64 x = a; // correct to 3 bits
152 for (int i = 0; i < 5; ++i) x *= 2 - a * x;
153 return x;
154}
155Mont make_mont(const fe& m) {
156 Mont M;
157 M.m = m;
158 M.n0 = 0 - inv64(m[0]);
159 // rr = 2^512 mod m, by 512 doublings of 1 with conditional subtract.
160 fe x = {1, 0, 0, 0};
161 for (int i = 0; i < 512; ++i) {
162 fe d;
163 u64 c = add_carry(d, x, x);
164 if (c || geq(d, m)) {
165 fe t;
166 sub_borrow(t, d, m);
167 d = t;
168 }
169 x = d;
170 }
171 M.rr = x;
172 // one = R mod m = 2^256 mod m -> to_mont(1)
173 fe oneN = {1, 0, 0, 0};
174 mont_mul(M.one, oneN, M.rr, M);
175 return M;
176}

188// ---- bytes <-> limbs (32 big-endian bytes) ---------------------------------
189fe be_to_fe(const unsigned char* b) {
190 fe r{};
191 for (int limb = 0; limb < 4; ++limb) {
192 u64 v = 0;
193 const unsigned char* p = b + (3 - limb) * 8; // most-significant 8 bytes -> limb[3]
194 for (int k = 0; k < 8; ++k) v = (v << 8) | p[k];
195 r[limb] = v;
196 }
197 return r;
198}
199void fe_to_be(unsigned char* out, const fe& a) {
200 for (int limb = 0; limb < 4; ++limb) {
201 u64 v = a[limb];
202 unsigned char* p = out + (3 - limb) * 8;
203 for (int k = 7; k >= 0; --k) {
204 p[k] = (unsigned char)(v & 0xFF);
205 v >>= 8;
206 }
207 }
208}

210// ---- Jacobian points (coordinates in Montgomery form, mod p) ---------------
211struct Jac {
212 fe X, Y, Z;
213 bool inf;
214};
215Jac jac_infinity() { return Jac{Fp().one, Fp().one, {0, 0, 0, 0}, true}; }

217void jac_double(Jac& r, const Jac& q) {
218 const Mont& F = Fp();
219 if (q.inf || is_zero(q.Z)) {
220 r = jac_infinity();
221 return;
222 }
223 fe A, B, C, D, t1, t2;
224 mont_mul(A, q.X, q.X, F); // X^2
225 mont_mul(B, q.Y, q.Y, F); // Y^2
226 mont_mul(C, B, B, F); // Y^4
227 // D = 2*((X+B)^2 - A - C)
228 mont_add(t1, q.X, B, F);
229 mont_mul(t1, t1, t1, F);
230 mont_sub(t1, t1, A, F);
231 mont_sub(t1, t1, C, F);
232 mont_add(D, t1, t1, F);
233 // ZZ = Z^2 ; E = 3*(X - ZZ)*(X + ZZ) [uses a = -3]
234 fe ZZ;
235 mont_mul(ZZ, q.Z, q.Z, F);
236 mont_sub(t1, q.X, ZZ, F);
237 mont_add(t2, q.X, ZZ, F);
238 mont_mul(t1, t1, t2, F);
239 fe E;
240 mont_add(E, t1, t1, F);
241 mont_add(E, E, t1, F); // 3*(...)
242 // F2 = E^2 ; X3 = F2 - 2D
243 fe X3;
244 mont_mul(X3, E, E, F);
245 mont_sub(X3, X3, D, F);
246 mont_sub(X3, X3, D, F);
247 // Y3 = E*(D - X3) - 8C
248 fe Y3, eight;
249 mont_sub(t1, D, X3, F);
250 mont_mul(Y3, E, t1, F);
251 mont_add(eight, C, C, F);
252 mont_add(eight, eight, eight, F);
253 mont_add(eight, eight, eight, F); // 8C
254 mont_sub(Y3, Y3, eight, F);
255 // Z3 = 2*Y*Z
256 fe Z3;
257 mont_mul(Z3, q.Y, q.Z, F);
258 mont_add(Z3, Z3, Z3, F);
259 r = Jac{X3, Y3, Z3, false};
260}

262void jac_add(Jac& r, const Jac& a, const Jac& b) {
263 const Mont& F = Fp();
264 if (a.inf || is_zero(a.Z)) {
265 r = b;
266 return;
267 }
268 if (b.inf || is_zero(b.Z)) {
269 r = a;
270 return;
271 }
272 fe Z1Z1, Z2Z2, U1, U2, S1, S2;
273 mont_mul(Z1Z1, a.Z, a.Z, F);
274 mont_mul(Z2Z2, b.Z, b.Z, F);
275 mont_mul(U1, a.X, Z2Z2, F);
276 mont_mul(U2, b.X, Z1Z1, F);
277 fe t;
278 mont_mul(t, b.Z, Z2Z2, F);
279 mont_mul(S1, a.Y, t, F);
280 mont_mul(t, a.Z, Z1Z1, F);
281 mont_mul(S2, b.Y, t, F);
282 fe H, Rr;
283 mont_sub(H, U2, U1, F);
284 mont_sub(Rr, S2, S1, F);
285 if (is_zero(H)) {
286 if (is_zero(Rr)) {
287 jac_double(r, a);
288 return;
289 }
290 r = jac_infinity();
291 return;
292 }
293 fe HH, HHH, V;
294 mont_mul(HH, H, H, F);
295 mont_mul(HHH, HH, H, F);
296 mont_mul(V, U1, HH, F);
297 fe X3;
298 mont_mul(X3, Rr, Rr, F);
299 mont_sub(X3, X3, HHH, F);
300 mont_sub(X3, X3, V, F);
301 mont_sub(X3, X3, V, F);
302 fe Y3;
303 mont_sub(t, V, X3, F);
304 mont_mul(Y3, Rr, t, F);
305 fe s1hhh;
306 mont_mul(s1hhh, S1, HHH, F);
307 mont_sub(Y3, Y3, s1hhh, F);
308 fe Z3;
309 mont_mul(Z3, a.Z, b.Z, F);
310 mont_mul(Z3, Z3, H, F);
311 r = Jac{X3, Y3, Z3, false};
312}

314// Strauss-Shamir: u1*A + u2*B with ONE doubling chain (256 doublings total)
315// instead of two separate scalar multiplications (512). A 2-bit window over both
316// scalars uses a 16-entry combined table [i*A + j*B] so it also halves the adds.
317void jac_double_mul(Jac& r, const fe& u1, const Jac& A, const fe& u2, const Jac& B) {
318 Jac tbl[4][4]; // tbl[i][j] = i*A + j*B, i,j in {0..3}
319 tbl[0][0] = jac_infinity();
320 tbl[1][0] = A;
321 jac_double(tbl[2][0], A);
322 jac_add(tbl[3][0], tbl[2][0], A);
323 tbl[0][1] = B;
324 jac_double(tbl[0][2], B);
325 jac_add(tbl[0][3], tbl[0][2], B);
326 for (int i = 1; i < 4; ++i)
327 for (int j = 1; j < 4; ++j) jac_add(tbl[i][j], tbl[i][0], tbl[0][j]);
328
329 Jac acc = jac_infinity();
330 for (int i = 254; i >= 0; i -= 2) {
331 Jac t;
332 jac_double(t, acc);
333 acc = t;
334 jac_double(t, acc);
335 acc = t;
336 const unsigned a = (u1[i / 64] >> (i % 64)) & 0x3;
337 const unsigned b = (u2[i / 64] >> (i % 64)) & 0x3;
338 if (a || b) {
339 jac_add(t, acc, tbl[a][b]);
340 acc = t;
341 }
342 }
343 r = acc;
344}

346// affine x-coordinate (normal form) of a Jacobian point: x = X / Z^2.
347fe jac_affine_x(const Jac& q) {
348 const Mont& F = Fp();
349 fe zinv, zinv2, x;
350 mont_inv(zinv, q.Z, F);
351 mont_mul(zinv2, zinv, zinv, F);
352 mont_mul(x, q.X, zinv2, F);
353 fe out;
354 from_mont(out, x, F);
355 return out;
356}

358Jac affine_to_jac(const fe& x, const fe& y) {
359 const Mont& F = Fp();
360 Jac p;
361 to_mont(p.X, x, F);
362 to_mont(p.Y, y, F);
363 p.Z = F.one;
364 p.inf = false;
365 return p;
366}
367const Jac& base_point() {
368 static const Jac g = affine_to_jac(GX, GY);
369 return g;
370}

372// Fixed-base comb for k*G. G is constant, so we precompute (once) the 16-entry
373// table T[s] = sum over set bits i of s of (2^(64*i) * G). Then k*G is just 64
374// doublings + 64 adds (vs 256 doublings for a generic window) — the big win for
375// the per-message signing path. Selector at step j is bit j of each of the four
376// 64-bit limbs.
377const std::array<Jac, 16>& g_comb() {
378 static const std::array<Jac, 16> tbl = [] {
379 Jac gi[4];
380 gi[0] = affine_to_jac(GX, GY);
381 for (int i = 1; i < 4; ++i) {
382 Jac acc = gi[i - 1];
383 for (int b = 0; b < 64; ++b) { // gi[i] = 2^64 * gi[i-1]
384 Jac t;
385 jac_double(t, acc);
386 acc = t;
387 }
388 gi[i] = acc;
389 }
390 std::array<Jac, 16> t;
391 t[0] = jac_infinity();
392 for (unsigned s = 1; s < 16; ++s) {
393 Jac acc = jac_infinity();
394 for (int i = 0; i < 4; ++i)
395 if (s & (1u << i)) {
396 Jac r;
397 jac_add(r, acc, gi[i]);
398 acc = r;
399 }
400 t[s] = acc;
401 }
402 return t;
403 }();
404 return tbl;
405}

407void jac_mul_base(Jac& r, const fe& k) {
408 const std::array<Jac, 16>& T = g_comb();
409 Jac acc = jac_infinity();
410 for (int j = 63; j >= 0; --j) {
411 Jac t;
412 jac_double(t, acc);
413 acc = t;
414 const unsigned sel = ((k[0] >> j) & 1u) | (((k[1] >> j) & 1u) << 1) |
415 (((k[2] >> j) & 1u) << 2) | (((k[3] >> j) & 1u) << 3);
416 if (sel) {
417 jac_add(t, acc, T[sel]);
418 acc = t;
419 }
420 }
421 r = acc;
422}

424// Reduce a scalar already known to be < 2n into [0, n): a single conditional
425// subtraction of the group order n. Used for the FIPS 186-4 hash truncation and
426// for folding a curve x-coordinate (which lives in [0, p) < 2n) into a scalar.
427fe reduce_mod_n(const fe& v) {
428 if (geq(v, N)) {
429 fe t;
430 sub_borrow(t, v, N);
431 return t;
432 }
433 return v;
434}

436// reduce a 32-byte big-endian hash to a scalar in [0, n) (FIPS 186-4 leftmost bits).
437fe hash_to_scalar(const std::string& h) {
438 unsigned char buf[32] = {0};
439 const std::size_t take = h.size() < 32 ? h.size() : 32;
440 std::memcpy(buf, h.data(), take); // leftmost 32 bytes
441 return reduce_mod_n(be_to_fe(buf));
442}

444// ---- minimal DER helpers ---------------------------------------------------
445// Parse SEQUENCE{INTEGER r, INTEGER s} -> 32-byte big-endian r and s. false on error.
446bool der_to_rs(const std::string& der, unsigned char r[32], unsigned char s[32]) {
447 const unsigned char* p = (const unsigned char*)der.data();
448 std::size_t n = der.size(), i = 0;
449 auto read_int = [&](unsigned char out[32]) -> bool {
450 if (i >= n || p[i++] != 0x02) return false;
451 if (i >= n) return false;
452 std::size_t len = p[i++];
453 if (len & 0x80) return false; // P-256 ints are short-form
454 if (i + len > n || len == 0) return false;
455 const unsigned char* v = p + i;
456 // strip a leading zero (sign byte)
457 while (len > 1 && v[0] == 0) {
458 ++v;
459 --len;
460 }
461 if (len > 32) return false;
462 std::memset(out, 0, 32);
463 std::memcpy(out + (32 - len), v, len);
464 i += (std::size_t)(v - (p + i)) + len; // advance past the original field
465 return true;
466 };
467 if (i >= n || p[i++] != 0x30) return false;
468 if (i >= n) return false;
469 std::size_t seqlen = p[i++];
470 if (seqlen & 0x80) return false;
471 if (i + seqlen != n) return false;
472 return read_int(r) && read_int(s);
473}

493bool verify_raw(const std::string& pubkey_xy, const std::string& msg_hash,
494 const std::string& sig_raw) {
495 if (pubkey_xy.size() != 64 || sig_raw.size() != 64) return false;
496 fe r = be_to_fe((const unsigned char*)sig_raw.data());
497 fe s = be_to_fe((const unsigned char*)sig_raw.data() + 32);
498 if (is_zero(r) || is_zero(s) || geq(r, N) || geq(s, N)) return false;
499
500 const Mont& Fnn = Fn();
501 fe e = hash_to_scalar(msg_hash);
502 fe sm, em, rm, w, u1, u2;
503 to_mont(sm, s, Fnn);
504 mont_inv(w, sm, Fnn); // w = s^-1 (Montgomery)
505 to_mont(em, e, Fnn);
506 to_mont(rm, r, Fnn);
507 fe u1m, u2m;
508 mont_mul(u1m, em, w, Fnn);
509 mont_mul(u2m, rm, w, Fnn);
510 from_mont(u1, u1m, Fnn);
511 from_mont(u2, u2m, Fnn);
512
513 fe qx = be_to_fe((const unsigned char*)pubkey_xy.data());
514 fe qy = be_to_fe((const unsigned char*)pubkey_xy.data() + 32);
515 if (geq(qx, P) || geq(qy, P)) return false;
516 Jac Q = affine_to_jac(qx, qy);
517
518 Jac R;
519 jac_double_mul(R, u1, base_point(), u2, Q); // u1*G + u2*Q, one doubling chain
520 if (R.inf || is_zero(R.Z)) return false;
521 fe x = reduce_mod_n(jac_affine_x(R));
522 return std::memcmp(x.data(), r.data(), sizeof(fe)) == 0;
523}

525bool verify_der(const std::string& pubkey_xy, const std::string& msg_hash,
526 const std::string& sig_der) {
527 unsigned char r[32], s[32];
528 if (!der_to_rs(sig_der, r, s)) return false;
529 std::string raw;
530 raw.resize(64);
531 std::memcpy(raw.data(), r, 32);
532 std::memcpy(raw.data() + 32, s, 32);
533 return verify_raw(pubkey_xy, msg_hash, raw);
534}

537// In test builds the RFC 6979 retry tail — normally a ~2^-128 event on real inputs —
538// is driven by rejecting the first `force_retries` otherwise-valid nonce candidates.
539// This whole parameter and its use compile out of release builds.
540std::string sign_raw_impl(const std::string& privkey, const std::string& msg_hash,
541 int force_retries) {
542#else
543std::string sign_raw(const std::string& privkey, const std::string& msg_hash) {
544#endif
545 if (privkey.size() != 32) return std::string();
546 const Mont& Fnn = Fn();
547 fe d = be_to_fe((const unsigned char*)privkey.data());
548 if (is_zero(d) || geq(d, N)) return std::string();
549 fe e = hash_to_scalar(msg_hash);
550
551 // RFC 6979 deterministic nonce generation (HMAC-SHA256).
552 unsigned char h1[32] = {0};
553 std::memcpy(h1, msg_hash.data(), msg_hash.size() < 32 ? msg_hash.size() : 32);
554 unsigned char x[32];
555 fe_to_be(x, d);
556 std::string V(32, '\x01'), K(32, '\x00');
557 auto bin = [](const fe&) {};
558 (void)bin;
559 auto hmac = [](const std::string& key, const std::string& msg) {
560 return hashlib::hmac_sha256(key, msg); // raw 32-byte digest
561 };
562 // K = HMAC(K, V || 0x00 || int2octets(x) || bits2octets(h1))
563 auto step = [&](unsigned char tag) {
564 std::string in = V;
565 if (tag != 0xFF) in.push_back((char)tag);
566 if (tag != 0xFF) {
567 in.append((const char*)x, 32);
568 in.append((const char*)h1, 32);
569 }
570 K = hmac(K, in);
571 V = hmac(K, V);
572 };
573 step(0x00);
574 step(0x01);
575 for (int attempt = 0; attempt < 64; ++attempt) {
576 V = hmac(K, V);
577 fe k = be_to_fe((const unsigned char*)V.data());
578 if (!is_zero(k) && !geq(k, N)) {
579 Jac R;
580 jac_mul_base(R, k); // fixed-base comb
581 if (!(R.inf || is_zero(R.Z))) {
582 fe rx = reduce_mod_n(jac_affine_x(R));
583#ifdef CHEATAH_P256_TESTING
584 if (force_retries > 0) {
585 --force_retries; // reject this valid candidate; take the retry tail
586 } else if (!is_zero(rx)) {
587#else
588 if (!is_zero(rx)) {
589#endif
590 // s = k^-1 (e + r*d) mod n
591 fe km, kinv, rm, dm, em, rd, sum, sm;
592 to_mont(km, k, Fnn);
593 mont_inv(kinv, km, Fnn);
594 to_mont(rm, rx, Fnn);
595 to_mont(dm, d, Fnn);
596 to_mont(em, e, Fnn);
597 mont_mul(rd, rm, dm, Fnn);
598 mont_add(sum, em, rd, Fnn);
599 mont_mul(sm, kinv, sum, Fnn);
600 fe sfinal;
601 from_mont(sfinal, sm, Fnn);
602 if (!is_zero(sfinal)) {
603 std::string out(64, '\0');
604 fe_to_be((unsigned char*)out.data(), rx);
605 fe_to_be((unsigned char*)out.data() + 32, sfinal);
606 return out;
607 }
608 }
609 }
610 }
611 // K = HMAC(K, V || 0x00); V = HMAC(K, V)
612 std::string in = V;
613 in.push_back('\x00');
614 K = hmac(K, in);
615 V = hmac(K, V);
616 }
617 return std::string();
618}

635std::string public_from_private(const std::string& privkey) {
636 if (privkey.size() != 32) return std::string();
637 fe d = be_to_fe((const unsigned char*)privkey.data());
638 if (is_zero(d) || geq(d, N)) return std::string();
639 Jac Q;
640 jac_mul_base(Q, d); // d*G via the fixed-base comb
641 if (Q.inf || is_zero(Q.Z)) return std::string();
642 // affine x and y (normal form)
643 const Mont& F = Fp();
644 fe zinv, zinv2, zinv3, x, y;
645 mont_inv(zinv, Q.Z, F);
646 mont_mul(zinv2, zinv, zinv, F);
647 mont_mul(zinv3, zinv2, zinv, F);
648 mont_mul(x, Q.X, zinv2, F);
649 mont_mul(y, Q.Y, zinv3, F);
650 fe xo, yo;
651 from_mont(xo, x, F);
652 from_mont(yo, y, F);
653 std::string out(64, '\0');
654 fe_to_be((unsigned char*)out.data(), xo);
655 fe_to_be((unsigned char*)out.data() + 32, yo);
656 return out;
657}

659std::string spki_ec_point(std::string_view der) {
660 // Find the uncompressed-point marker: BIT STRING (03) <len> 00 04 <X(32)><Y(32)>.
661 // The OID id-ecPublicKey + prime256v1 precedes it; we anchor on the 0x04 point.
662 const unsigned char* p = (const unsigned char*)der.data();
663 const std::size_t n = der.size();
664 for (std::size_t i = 0; i + 2 + 65 <= n; ++i) {
665 // BIT STRING tag, then a length, then 00 (unused bits), then 04 (uncompressed)
666 if (p[i] == 0x03 && p[i + 2] == 0x00 && p[i + 3] == 0x04) {
667 const std::size_t len = p[i + 1];
668 if (len == 66 && i + 4 + 64 <= n) return std::string((const char*)p + i + 4, 64);
669 }
670 }
671 return std::string();
672}
