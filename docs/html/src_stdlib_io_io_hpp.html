31/// Streamable<T>: T can be written to a `std::ostream` — the ONLY thing str(),
32/// print(), format() and File::write() actually need. Naming the requirement turns
33/// a deep `operator<<` instantiation error into a clear "constraint Streamable not
34/// satisfied" message, without narrowing what those templates already accept.
35template <typename T>
36concept Streamable = requires(std::ostream& os, const T& value) {
37 { os << value } -> std::convertible_to<std::ostream&>;
38};
39
40/// HasStr<T>: T exposes a `str()` method that returns a Streamable value. This is
41/// the hook a struct (or a built-in object like NDArray) implements to become
42/// printable — `io.print` calls it to get something it can stream.
43template <typename T>
44concept HasStr = requires(const T& value) {
45 { value.str() } -> Streamable;
46};
47
48// Printable<T>: what `print`/`str` actually require — NOT "is it Streamable", but
49// "can it be turned into something Streamable, then streamed". True when T streams
50// directly, exposes a HasStr `str()`, or is a list/dict whose elements are
51// themselves Printable (checked recursively, so nested containers work).
52template <typename T>
53struct printable_trait : std::bool_constant<Streamable<T> || HasStr<T>> {};
54template <typename T, typename A>
55struct printable_trait<std::vector<T, A>> : printable_trait<T> {};
56template <typename K, typename V, typename H, typename E, typename A>
57struct printable_trait<std::unordered_map<K, V, H, E, A>>
58 : std::bool_constant<printable_trait<K>::value && printable_trait<V>::value> {};
59template <typename T>
60concept Printable = printable_trait<std::remove_cvref_t<T>>::value;

139/**
140 * `str()` for a complex number — Python-style `a+bj` / `a-bj` (not `std::complex`'s
141 * default `(a,b)`), so a complex scalar and a complex `ndarray` element read alike.
142 * Negative zero in either part is flushed to `+0` (a conjugate prints `1+0j`).
143 * @param z the complex value.
144 * @return the `a±bj` rendering.
145 * @test CheatahIo.StrRendersComplex
146 * @systest StdlibE2E.Io
147 */
148template <std::floating_point T>
149std::string str(const std::complex<T>& z) {
150 const auto nz = [](T x) -> T { return x == T{0} ? T{0} : x; };
151 std::ostringstream os;
152 os << nz(z.real());
153 if (z.imag() < T{0}) {
154 os << '-' << nz(-z.imag()) << 'j';
155 } else {
156 os << '+' << nz(z.imag()) << 'j';
157 }
158 return os.str();
159}
160/**
161 * `str()` for a list — Python `[a, b, c]`. Elements are rendered with repr(), so a
162 * `list[str]` prints with quotes (`['a', 'b']`), matching Python.
163 * @param v the list (its element type must be Printable).
164 * @return the bracketed rendering.
165 * @test CheatahIo.StrRendersContainersAndObjects
166 * @systest StdlibE2E.Io
167 */
168template <typename T>
169 requires Printable<T>
170std::string str(const std::vector<T>& v) {
171 std::ostringstream os;
172 os << '[';
173 for (std::size_t i = 0; i < v.size(); ++i) {
174 if (i != 0) os << ", ";
175 os << repr(v[i]);
176 }
177 os << ']';
178 return os.str();
179}
180/**
181 * `str()` for a dict — Python `{k: v, …}`. Iteration order is unspecified (it is a
182 * hash map). Keys and values are rendered with repr().
183 * @param m the dict (key and value types must be Printable).
184 * @return the brace-wrapped rendering.
185 * @test CheatahIo.StrRendersContainersAndObjects
186 * @systest StdlibE2E.Io
187 */
188template <typename K, typename V, typename H, typename E, typename A>
189 requires(Printable<K> && Printable<V>)
190std::string str(const std::unordered_map<K, V, H, E, A>& m) {
191 std::ostringstream os;
192 os << '{';
193 bool first = true;
194 for (const auto& [k, val] : m) {
195 if (!first) os << ", ";
196 first = false;
197 os << repr(k) << ": " << repr(val);
198 }
199 os << '}';
200 return os.str();
201}

203/**
204 * Python `print(*args)`: space-separated, newline-terminated, to stdout (sep=' ', end='\n').
205 *
206 * Inserts a single space between consecutive arguments (none before the first) and always
207 * ends with a trailing `\n`; with no arguments it writes just that newline (blank line).
208 * @param args zero or more streamable values.
209 * @complexity O(total output length).
210 * @alloc each arg is routed through str() so values format the Python way (e.g. bool →
211 * True/False), allocating temporary strings.
212 * @test CheatahIo.PrintWritesSpaceSeparatedLine, CheatahIo.PrintNoArgsIsJustNewline
213 * @crtest IoCompileRun.Print
214 * @systest StdlibE2E.Io
215 */
216template <Printable... Args>
217void print(const Args&... args) {
218 std::size_t i = 0;
219 ((std::cout << (i++ ? " " : "") << str(args)), ...);
220 std::cout << '\n';
221}

317namespace detail {
318/**
319 * Base case of the recursive formatter: emit the remainder of @p fmt verbatim.
320 * @param os destination stream.
321 * @param fmt remaining format text (no placeholders left to fill).
322 */
323void format_into(std::ostringstream& os, std::string_view fmt);
324/**
325 * Recursive step: write text up to the next `{}`, substitute @p arg, recurse on the rest.
326 * @param os destination stream.
327 * @param fmt remaining format text.
328 * @param arg value substituted at the next `{}` placeholder.
329 * @param rest values for the remaining placeholders.
330 */
331template <Streamable T, typename... Rest>
332void format_into(std::ostringstream& os, std::string_view fmt, const T& arg, const Rest&... rest) {
333 const std::size_t brace = fmt.find("{}");
334 if (brace == std::string_view::npos) {
335 os << fmt; // more args than placeholders — drop the extras
336 return;
337 }
338 os << fmt.substr(0, brace) << arg;
339 format_into(os, fmt.substr(brace + 2), rest...);
340}
341} // namespace detail

343/**
344 * Sequential `{}` substitution — the common case of Python's str.format() / f-strings.
345 *
346 * Replaces each `{}` left-to-right with the corresponding argument (streamed via its
347 * `operator<<`); surplus arguments are silently dropped, and any `{}` left without an
348 * argument is emitted literally rather than raising. Does not support indexed or named
349 * fields (`{0}`, `{name}`) or escaped braces (`{{`).
350 * @param fmt format string with `{}` placeholders.
351 * @param args values substituted left-to-right (extras dropped, missing placeholders left
352 * as-is).
353 * @return the formatted string.
354 * @complexity O(len(fmt) + total arg output).
355 * @alloc allocates the result string (via an ostringstream).
356 * @test CheatahIo.FormatSubstitutesBraces, CheatahIo.FormatMultiArgAndExtraArgs
357 * @crtest IoCompileRun.Format
358 * @systest StdlibE2E.Io
359 */
360template <Streamable... Args>
361std::string format(std::string_view fmt, const Args&... args) {
362 std::ostringstream os;
363 detail::format_into(os, fmt, args...);
364 return os.str();
365}

383/**
384 * @brief A Python-like file object over `std::fstream`.
385 *
386 * RAII closes on scope exit — the C++ analog of `with open(...) as f:`. Move-only
387 * (copying a file handle is meaningless), like Python file objects.
388 */
389class File {
390public:
391 /**
392 * Construct a closed file (no stream attached).
393 *
394 * Leaves the underlying stream default-constructed and unopened, so is_open() is false
395 * until a later open(); read/write calls on it are no-ops that fail silently.
396 * @complexity O(1).
397 * @alloc none.
398 * @test CheatahIo.FileIsOpenAndClose
399 * @crtest IoCompileRun.IsOpenAndClose
400 * @systest StdlibE2E.Io
401 */
402 File() = default;
403 /**
404 * Open @p path in @p mode (the open() free function's workhorse).
405 *
406 * Translates the Python mode and opens the stream; failure (e.g. missing file in `r`)
407 * is not thrown — it leaves is_open() false, so callers should check before using it.
408 * @param path filesystem path.
409 * @param mode Python-style mode (`r`/`w`/`a`, optional `+`/`b`).
410 * @complexity O(1) plus the OS open.
411 * @alloc none.
412 * @test CheatahIo.FileWriteThenReadWhole
413 * @crtest IoCompileRun.OpenWriteRead
414 * @systest StdlibE2E.Io
415 */
416 File(const std::string& path, std::string_view mode);
417 File(const File&) = delete;
418 File& operator=(const File&) = delete;
419 /** Move-construct, taking over the other handle (the moved-from File becomes closed). */
420 File(File&&) = default;
421 /**
422 * Move-assign, taking over the other handle (the moved-from File becomes closed).
423 * @return reference to this File.
424 */
425 File& operator=(File&&) = default;
426 /**
427 * Close the stream if still open.
428 * @complexity O(1).
429 * @alloc none.
430 * @test CheatahIo.FileIsOpenAndClose
431 * @crtest IoCompileRun.IsOpenAndClose
432 * @systest StdlibE2E.Io

434 ~File();

437 * (Re)open @p path in @p mode.
438 *
439 * Opens the stream on @p path; it does not first close an already-open handle, so reuse
440 * this on a closed File. Mode follows Python: `r` read, `w` truncate-write, `a` append,
441 * `+` adds the opposite direction, `b` binary; an unrecognized/empty mode defaults to `r`.
442 * @param path filesystem path.
443 * @param mode Python-style mode string.
444 * @complexity O(1) plus the OS open.
445 * @alloc none.
446 * @test CheatahIo.FileWriteThenReadWhole
447 * @crtest IoCompileRun.OpenWriteRead
448 * @systest StdlibE2E.Io

452 * Is the underlying stream open?

454 * @complexity O(1).
455 * @alloc none.
456 * @test CheatahIo.FileIsOpenAndClose
457 * @crtest IoCompileRun.IsOpenAndClose
458 * @systest StdlibE2E.Io

460 bool is_open() const;
461 /**
462 * Close the underlying stream (no-op if already closed).
463 * @complexity O(1).
464 * @alloc none.
465 * @test CheatahIo.FileIsOpenAndClose
466 * @crtest IoCompileRun.IsOpenAndClose
467 * @systest StdlibE2E.Io

469 void close();

471 /**
472 * Read the whole remaining file.
473 *
474 * Drains the stream buffer from the current position to EOF in one shot, so a prior
475 * readline()/read() returns only what is left; returns "" at EOF or on a closed file.
476 * @return the remaining bytes as one string.
477 * @complexity O(n) in bytes read.
478 * @alloc allocates the returned string (buffered through a stringstream).
479 * @test CheatahIo.FileWriteThenReadWhole
480 * @crtest IoCompileRun.OpenWriteRead
481 * @systest StdlibE2E.Io
482 */
483 std::string read(); // whole remaining file
484 /**
485 * Read the next line.
486 *
487 * Consumes through the next `\n` (which is discarded). Because both a genuine empty line
488 * and EOF yield "", the return value alone cannot distinguish them — check is_open()/EOF
489 * separately if that matters.
490 * @return the line with its newline stripped; `""` at EOF.
491 * @complexity O(line length).
492 * @alloc allocates the returned string.
493 * @test CheatahIo.FileReadlineThenReadlines
494 * @crtest IoCompileRun.Readline
495 * @systest StdlibE2E.Io
496 */
497 std::string readline(); // next line, no newline; "" at EOF
498 /**
499 * Read all remaining lines.
500 *
501 * Repeatedly getlines from the current position until EOF, pushing each newline-stripped
502 * line; returns an empty vector at EOF, and a trailing newline does not produce a final
503 * empty element.
504 * @return a vector of lines (newlines stripped).
505 * @complexity O(n) in bytes.
506 * @alloc allocates the vector and each line string.
507 * @test CheatahIo.FileReadlineThenReadlines
508 * @crtest IoCompileRun.Readlines
509 * @systest StdlibE2E.Io
510 */
511 std::vector<std::string> readlines(); // all remaining lines
512
513 /**
514 * Write a streamable value to the file.
515 *
516 * Streams @p value through `operator<<` exactly as written — no separator and no trailing
517 * newline are added (unlike print()), so the caller supplies any `\n`; data may stay
518 * buffered until the stream is flushed or the File is closed.
519 * @param value any streamable value.
520 * @complexity O(output length).
521 * @alloc none (writes straight to the stream buffer).
522 * @test CheatahIo.FileWriteThenReadWhole, CheatahIo.FileAppendMode
523 * @crtest IoCompileRun.OpenWriteRead
524 * @systest StdlibE2E.Io
525 */
526 template <Streamable T>
527 void write(const T& value) { stream_ << value; }

532 std::fstream stream_;
533};
