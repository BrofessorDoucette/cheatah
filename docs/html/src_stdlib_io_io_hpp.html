32/// Streamable<T>: T can be written to a `std::ostream` — the ONLY thing str(),
33/// print(), format() and File::write() actually need. Naming the requirement turns
34/// a deep `operator<<` instantiation error into a clear "constraint Streamable not
35/// satisfied" message, without narrowing what those templates already accept.
36template <typename T>
37concept Streamable = requires(std::ostream& os, const T& value) {
38 { os << value } -> std::convertible_to<std::ostream&>;
39};
40
41/// HasStr<T>: T exposes a `str()` method that returns a Streamable value. This is
42/// the hook a struct (or a built-in object like NDArray) implements to become
43/// printable — `io.print` calls it to get something it can stream.
44template <typename T>
45concept HasStr = requires(const T& value) {
46 { value.str() } -> Streamable;
47};
48
49// Printable<T>: what `print`/`str` actually require — NOT "is it Streamable", but
50// "can it be turned into something Streamable, then streamed". True when T streams
51// directly, exposes a HasStr `str()`, or is a list/dict whose elements are
52// themselves Printable (checked recursively, so nested containers work).
53template <typename T>
54struct printable_trait : std::bool_constant<Streamable<T> || HasStr<T>> {};
55template <typename T, typename A>
56struct printable_trait<std::vector<T, A>> : printable_trait<T> {};
57template <typename K, typename V, typename H, typename E, typename A>
58struct printable_trait<std::unordered_map<K, V, H, E, A>>
59 : std::bool_constant<printable_trait<K>::value && printable_trait<V>::value> {};
60template <typename T>
61concept Printable = printable_trait<std::remove_cvref_t<T>>::value;

63/**

66 * Renders @p value via its `operator<<` into a fresh `ostringstream`, so the result
67 * matches whatever that stream insertion produces (e.g. default float precision).
68 * @param value the value to render.

70 * @complexity O(n) in the output length.

126 *
127 * Renders @p value with exactly @p places digits after the decimal point, correctly

174/**
175 * `str()` for a complex number — Python-style `a+bj` / `a-bj` (not `std::complex`'s
176 * default `(a,b)`), so a complex scalar and a complex `ndarray` element read alike.
177 * Negative zero in either part is flushed to `+0` (a conjugate prints `1+0j`).
178 * @param z the complex value.
179 * @return the `a±bj` rendering.
180 * @test CheatahIo.StrRendersComplex
181 * @systest StdlibE2E.Io
182 */
183template <std::floating_point T>
184std::string str(const std::complex<T>& z) {
185 const auto nz = [](T x) -> T { return x == T{0} ? T{0} : x; };
186 std::ostringstream os;
187 os << nz(z.real());
188 if (z.imag() < T{0}) {
189 os << '-' << nz(-z.imag()) << 'j';
190 } else {
191 os << '+' << nz(z.imag()) << 'j';
192 }
193 return os.str();
194}
195/**
196 * `str()` for a list — Python `[a, b, c]`. Elements are rendered with repr(), so a
197 * `list[str]` prints with quotes (`['a', 'b']`), matching Python.
198 * @param v the list (its element type must be Printable).
199 * @return the bracketed rendering.
200 * @test CheatahIo.StrRendersContainersAndObjects
201 * @systest StdlibE2E.Io
202 */
203template <typename T>
204 requires Printable<T>
205std::string str(const std::vector<T>& v) {
206 std::ostringstream os;
207 os << '[';
208 for (std::size_t i = 0; i < v.size(); ++i) {
209 if (i != 0) os << ", ";
210 os << repr(v[i]);
211 }
212 os << ']';
213 return os.str();
214}
215/**
216 * `str()` for a dict — Python `{k: v, …}`. Iteration order is unspecified (it is a
217 * hash map). Keys and values are rendered with repr().
218 * @param m the dict (key and value types must be Printable).
219 * @return the brace-wrapped rendering.
220 * @test CheatahIo.StrRendersContainersAndObjects
221 * @systest StdlibE2E.Io
222 */
223template <typename K, typename V, typename H, typename E, typename A>
224 requires(Printable<K> && Printable<V>)
225std::string str(const std::unordered_map<K, V, H, E, A>& m) {
226 std::ostringstream os;
227 os << '{';
228 bool first = true;
229 for (const auto& [k, val] : m) {
230 if (!first) os << ", ";
231 first = false;
232 os << repr(k) << ": " << repr(val);
233 }
234 os << '}';
235 return os.str();
236}

238/**
239 * Python `print(*args)`: space-separated, newline-terminated, to stdout (sep=' ', end='\n').
240 *
241 * Inserts a single space between consecutive arguments (none before the first) and always
242 * ends with a trailing `\n`; with no arguments it writes just that newline (blank line).
243 * Output is meant to be NICE AND READABLE by default: a struct (which the compiler gives a
244 * `cheatah_pretty_print` member) is rendered over multiple indented lines, e.g.
245 * `Point(\n x = 1,\n y = 2\n)`. Use @ref rprint to print a struct in its compact form.
246 * @param args zero or more printable values.
247 * @complexity O(total output length).
248 * @alloc each arg is routed through str()/its pretty-printer, allocating temporary strings.
249 * @test CheatahIo.PrintWritesSpaceSeparatedLine, CheatahIo.PrintNoArgsIsJustNewline
250 * @crtest IoCompileRun.Print
251 * @systest StdlibE2E.Io
252 */
253template <Printable... Args>
254void print(const Args&... args) {
255 std::size_t i = 0;
256 // A struct exposes a `cheatah_pretty_print` member (the compiler generates it): use it for
257 // the readable multi-line layout. Everything else uses str() (the compact Python form).
258 auto one = [&](const auto& v) {
259 std::cout << (i++ ? " " : "");
260 if constexpr (requires(std::ostream& o) { v.cheatah_pretty_print(o, 0LL); })
261 v.cheatah_pretty_print(std::cout, 0LL);
262 else
263 std::cout << str(v);
264 };
265 (one(args), ...);
266 std::cout << '\n';
267}

269/**
270 * Python `print` but RAW: a struct prints in its COMPACT `Name(field=value, …)` form (exactly
271 * as stored) instead of the pretty multi-line layout @ref print uses; otherwise identical
272 * (space-separated, newline-terminated). Reach for it when you want a struct exactly as it is
273 * rather than the default human-readable formatting.
274 * @param args zero or more printable values.
275 * @complexity O(total output length).
276 * @alloc each arg is routed through str(), allocating temporary strings.
277 * @test CheatahIo.RprintIsCompact
278 * @crtest IoCompileRun.Rprint
279 * @systest StdlibE2E.Io
280 */
281template <Printable... Args>
282void rprint(const Args&... args) {
283 std::size_t i = 0;
284 ((std::cout << (i++ ? " " : "") << str(args)), ...);
285 std::cout << '\n';
286}

382namespace detail {
383/**
384 * Base case of the recursive formatter: emit the remainder of @p fmt verbatim.
385 * @param os destination stream.
386 * @param fmt remaining format text (no placeholders left to fill).
387 */
388void format_into(std::ostringstream& os, std::string_view fmt);
389/**
390 * Recursive step: write text up to the next `{}`, substitute @p arg, recurse on the rest.
391 * @param os destination stream.
392 * @param fmt remaining format text.
393 * @param arg value substituted at the next `{}` placeholder.
394 * @param rest values for the remaining placeholders.
395 */
396template <Streamable T, typename... Rest>
397void format_into(std::ostringstream& os, std::string_view fmt, const T& arg, const Rest&... rest) {
398 const std::size_t brace = fmt.find("{}");
399 if (brace == std::string_view::npos) {
400 os << fmt; // more args than placeholders — drop the extras
401 return;
402 }
403 os << fmt.substr(0, brace) << arg;
404 format_into(os, fmt.substr(brace + 2), rest...);
405}
406} // namespace detail

408/**
409 * Sequential `{}` substitution — the common case of Python's str.format() / f-strings.
410 *
411 * Replaces each `{}` left-to-right with the corresponding argument (streamed via its
412 * `operator<<`); surplus arguments are silently dropped, and any `{}` left without an
413 * argument is emitted literally rather than raising. Does not support indexed or named
414 * fields (`{0}`, `{name}`) or escaped braces (`{{`).
415 * @param fmt format string with `{}` placeholders.
416 * @param args values substituted left-to-right (extras dropped, missing placeholders left
417 * as-is).
418 * @return the formatted string.
419 * @complexity O(len(fmt) + total arg output).
420 * @alloc allocates the result string (via an ostringstream).
421 * @test CheatahIo.FormatSubstitutesBraces, CheatahIo.FormatMultiArgAndExtraArgs
422 * @crtest IoCompileRun.Format
423 * @systest StdlibE2E.Io
424 */
425template <Streamable... Args>
426std::string format(std::string_view fmt, const Args&... args) {
427 std::ostringstream os;
428 detail::format_into(os, fmt, args...);
429 return os.str();
430}

448/**
449 * @brief A Python-like file object over `std::fstream`.

457 * Construct a closed file (no stream attached).

461 * @complexity O(1).
462 * @alloc none.

464 * @crtest IoCompileRun.IsOpenAndClose

473 * @param path filesystem path.
474 * @param mode Python-style mode (`r`/`w`/`a`, optional `+`/`b`).
475 * @complexity O(1) plus the OS open.
476 * @alloc none.
477 * @test CheatahIo.FileWriteThenReadWhole
478 * @crtest IoCompileRun.OpenWriteRead
479 * @systest StdlibE2E.Io

481 File(const std::string& path, std::string_view mode);

485 File(File&&) = default;
486 /**
487 * Move-assign, taking over the other handle (the moved-from File becomes closed).
488 * @return reference to this File.

490 File& operator=(File&&) = default;

494 * @alloc none.

496 * @crtest IoCompileRun.IsOpenAndClose

536 /**
537 * Read the whole remaining file.
538 *
539 * Drains the stream buffer from the current position to EOF in one shot, so a prior
540 * readline()/read() returns only what is left; returns "" at EOF or on a closed file.
541 * @return the remaining bytes as one string.
542 * @complexity O(n) in bytes read.
543 * @alloc allocates the returned string (buffered through a stringstream).

553 * and EOF yield "", the return value alone cannot distinguish them — check is_open()/EOF
554 * separately if that matters.
555 * @return the line with its newline stripped; `""` at EOF.
556 * @complexity O(line length).

558 * @test CheatahIo.FileReadlineThenReadlines
559 * @crtest IoCompileRun.Readline
560 * @systest StdlibE2E.Io
561 */

563 /**
564 * Read all remaining lines.
565 *
566 * Repeatedly getlines from the current position until EOF, pushing each newline-stripped
567 * line; returns an empty vector at EOF, and a trailing newline does not produce a final
568 * empty element.
569 * @return a vector of lines (newlines stripped).
570 * @complexity O(n) in bytes.
571 * @alloc allocates the vector and each line string.
572 * @test CheatahIo.FileReadlineThenReadlines
573 * @crtest IoCompileRun.Readlines
574 * @systest StdlibE2E.Io
575 */
576 std::vector<std::string> readlines(); // all remaining lines
577
578 /**
579 * Write a streamable value to the file.
580 *
581 * Streams @p value through `operator<<` exactly as written — no separator and no trailing
582 * newline are added (unlike print()), so the caller supplies any `\n`; data may stay
583 * buffered until the stream is flushed or the File is closed.
584 * @param value any streamable value.
585 * @complexity O(output length).
586 * @alloc none (writes straight to the stream buffer).
587 * @test CheatahIo.FileWriteThenReadWhole, CheatahIo.FileAppendMode
588 * @crtest IoCompileRun.OpenWriteRead
589 * @systest StdlibE2E.Io
590 */
591 template <Streamable T>
592 void write(const T& value) { stream_ << value; }
593
594private:
595 /// Map a Python mode string (`r`/`w`/`a`, optional `+`/`b`) to `std::ios::openmode`.
596 static std::ios::openmode translate_mode(std::string_view mode);
597 std::fstream stream_;
598};
