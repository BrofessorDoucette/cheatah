301/**
302 * True division — the cheatah `/` operator (like Python 3): **always floating-point**,
303 * so `6 / 2` is `3.0`, not `3`, and integer operands never silently truncate. Use the
304 * `//` operator (@ref floordiv) when you want integer/floor division.
305 * @param a numerator.
306 * @param b denominator.
307 * @return `double(a) / double(b)`.
308 * @complexity O(1).
309 * @alloc none.
310 * @test CheatahBuiltins.Division
311 * @crtest BuiltinsCompileRun.TrueDivision
312 * @systest StdlibE2E.Builtins
313 */
314template <typename A, typename B>
315 requires std::is_arithmetic_v<A> && std::is_arithmetic_v<B>
316double truediv(A a, B b) {
317 return static_cast<double>(a) / static_cast<double>(b);
318}
319/**
320 * Floor division — the cheatah `//` operator (like Python): the quotient floored toward
321 * −∞. Integer operands give an integer (`7 // 2 == 3`, `-7 // 2 == -4`, flooring the way
322 * Python does, not truncating toward zero like raw C++); a floating operand gives a
323 * floored double (`7.0 // 2 == 3.0`).
324 * @param a numerator.
325 * @param b denominator.
326 * @return `floor(a / b)`, integral for integral operands.
327 * @complexity O(1).
328 * @alloc none.
329 * @test CheatahBuiltins.Division
330 * @crtest BuiltinsCompileRun.FloorDivision
331 * @systest StdlibE2E.Builtins
332 */
333template <std::integral A, std::integral B>
334std::common_type_t<A, B> floordiv(A a, B b) {
335 // A controlled error, not UB: integer divide-by-zero is undefined in C++ (SIGFPE/trap), so guard
336 // it so pure-cheatah `x // 0` raises rather than corrupting the process. (Float `//` is IEEE-safe.)
337 if (b == 0) throw std::domain_error("integer floor division by zero");
338 std::common_type_t<A, B> q = a / b; // C++ truncates toward zero…
339 if ((a % b != 0) && ((a < 0) != (b < 0))) --q; // …adjust to floor toward −∞
340 return q;
341}
342/**
343 * Floor division (`//`) for floating-point operands: floors the quotient toward −∞.
344 *
345 * Selected when at least one operand is floating-point (the all-integer case uses the
346 * @ref floordiv overload above). Mirrors Python, where `7.0 // 2.0 == 3.0`.
347 * @param a numerator.
348 * @param b denominator.
349 * @return `std::floor(double(a) / double(b))`.
350 * @complexity O(1).
351 * @alloc none.
352 * @test CheatahBuiltins.Division
353 * @crtest BuiltinsCompileRun.FloorDivision
354 * @systest StdlibE2E.Builtins
355 */
356template <typename A, typename B>
357 requires(std::is_arithmetic_v<A> && std::is_arithmetic_v<B> &&
358 !(std::integral<A> && std::integral<B>))
359double floordiv(A a, B b) {
360 return std::floor(static_cast<double>(a) / static_cast<double>(b));
361}

404/**
405 * Append @p x to list @p v in place (Python `list.append`).
406 *
407 * Grows @p v by one, converting @p x to the list's element type. Usable as a

503/**
504 * Python FLOOR-mod for integers: the result takes the DIVISOR's sign (-7 % 3 == 2), unlike
505 * raw C++ `%`. Backs the `%` operator.
506 * @param a the dividend.
507 * @param b the divisor; @p b == 0 throws std::domain_error (integer modulo by zero).
508 * @return a mod b with the sign of @p b (Python floor-mod semantics).
509 * @complexity O(1).
510 * @alloc none.
511 * @test CheatahBuiltins.Mod
512 * @crtest LangFeatures.Modulo
513 */
514template <class A, class B>
515 requires(std::is_integral_v<A> && std::is_integral_v<B>)
516std::common_type_t<A, B> mod(A a, B b) {
517 using R = std::common_type_t<A, B>;
518 // A controlled error, not UB: integer `% 0` is undefined in C++ (SIGFPE/trap) — guard it so
519 // pure-cheatah `x % 0` raises rather than corrupting the process. (Float `%` is IEEE-safe.)
520 if (b == 0) throw std::domain_error("integer modulo by zero");
521 const R r = static_cast<R>(a) % static_cast<R>(b);
522 return (r != 0 && ((r < 0) != (b < 0))) ? r + static_cast<R>(b) : r;
523}

525/**
526 * Python floor-mod for floats (either operand): fmod adjusted to the divisor's sign,
527 * mirroring `7.5 % 2 == 1.5` and `-7.5 % 2 == 0.5`.
528 * @param a the dividend.
529 * @param b the divisor.
530 * @return a mod b as a double, with the sign of @p b (Python floor-mod semantics).
531 * @complexity O(1).
532 * @alloc none.
533 * @test CheatahBuiltins.Mod
534 * @crtest LangFeatures.Modulo
535 */
536template <class A, class B>
537 requires(!std::is_integral_v<A> || !std::is_integral_v<B>)
538double mod(A a, B b) {
539 const double r = std::fmod(static_cast<double>(a), static_cast<double>(b));
540 return (r != 0.0 && ((r < 0.0) != (static_cast<double>(b) < 0.0))) ? r + static_cast<double>(b)
541 : r;
542}

544// ---- indexing & slicing (Python `seq[i]` / `seq[i:j]`) ----
545//
546// The compiler lowers value-position `seq[i]` to `index(seq, i)` and `seq[a:b]`
547// to `slice(seq, a, b)` (a missing bound becomes 0 / `slice_end`). Indices may be
548// negative (counted from the end). Indexing a string yields a length-1 string
549// (Python semantics), so `s[i] == "<"` type-checks; slicing yields the same kind.

551/// Sentinel for an omitted slice upper bound (`s[a:]`): "to the end".
552inline constexpr long long slice_end = std::numeric_limits<long long>::max();

558/**
559 * Element at @p i of a string — a length-1 string (Python `s[i]`).
560 * Negative @p i counts from the end; out-of-range throws `std::out_of_range`.
561 * @param s the string.
562 * @param i the index (may be negative).
563 * @return the one-character string at @p i.
564 * @complexity O(1).
565 * @alloc none (1-char small-string optimization).
566 * @test CheatahBuiltins.IndexString
567 * @crtest BuiltinsCompileRun.IndexString
568 * @systest StdlibE2E.Builtins
569 */
570inline std::string index(const std::string& s, long long i) {
571 const long long n = static_cast<long long>(s.size());
572 i = detail::norm_index(i, n);
573 if (i < 0 || i >= n) throw std::out_of_range("string index out of range");
574 return std::string(1, s[static_cast<std::size_t>(i)]);
575}

620 * Value for @p key in a dict (Python `d[key]`), by value.
621 * @param m the dict.
622 * @param key the key to look up.
623 * @return a copy of the mapped value (throws `std::out_of_range` if absent).
624 * @complexity O(1) average.
625 * @alloc none beyond copying the value.
626 * @test CheatahBuiltins.IndexDict
627 * @crtest BuiltinsCompileRun.IndexDict
628 * @systest StdlibE2E.Builtins

632V index(const std::unordered_map<K, V, H, E, A>& m, const Key& key) {
633 const auto it = m.find(key);

638/**
639 * Substring `s[lo:hi]` (Python slice semantics: clamped, negatives from the end).
640 * @param s the string.
641 * @param lo start index (default 0 at the call site).
642 * @param hi end index, or @ref slice_end for "to the end".
643 * @return the slice (empty if `lo >= hi` after clamping).
644 * @complexity O(hi-lo).
645 * @alloc the result string.
646 * @test CheatahBuiltins.SliceString
647 * @crtest BuiltinsCompileRun.SliceString
648 * @systest StdlibE2E.Builtins
649 */
650inline std::string slice(const std::string& s, long long lo, long long hi) {
651 const long long n = static_cast<long long>(s.size());
652 lo = detail::norm_index(lo, n);
653 hi = (hi == slice_end) ? n : detail::norm_index(hi, n);
654 if (lo < 0) lo = 0;
655 if (hi > n) hi = n;
656 if (lo >= hi) return std::string();
657 return s.substr(static_cast<std::size_t>(lo), static_cast<std::size_t>(hi - lo));
658}

660/**
661 * Sub-list `xs[lo:hi]` (Python slice semantics), returned as a new list.
662 * @param c the sequence.
663 * @param lo start index.
664 * @param hi end index, or @ref slice_end for "to the end".
665 * @return the slice as a `std::vector` of the element type.
666 * @complexity O(hi-lo).
667 * @alloc the result vector.
668 * @test CheatahBuiltins.SliceList
669 * @crtest BuiltinsCompileRun.SliceList
670 * @systest StdlibE2E.Builtins
671 */
672template <typename C>
673 requires requires(const C& c) { c.data(); c.size(); } // contiguous seq, not a map
674auto slice(const C& c, long long lo, long long hi) -> std::vector<std::decay_t<decltype(c[0])>> {
675 const long long n = static_cast<long long>(c.size());
676 lo = detail::norm_index(lo, n);
677 hi = (hi == slice_end) ? n : detail::norm_index(hi, n);
678 if (lo < 0) lo = 0;
679 if (hi > n) hi = n;
680 std::vector<std::decay_t<decltype(c[0])>> out;
681 for (long long k = lo; k < hi; ++k) out.push_back(c[static_cast<std::size_t>(k)]);
682 return out;
683}
