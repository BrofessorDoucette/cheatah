3/**
4 * @file builtins.hpp
5 * @brief cheatah `builtins` — Python's always-available built-ins (no `import`):
6 * length, character/representation conversions, and hashing.
7 *
8 * The compiler auto-includes this header and resolves bare calls like `len("x")`
9 * to `builtins::len`. The mathematical built-ins (`abs`/`min`/`max`/`round`/`pow`)
10 * live in the `math` module. Unit tests: `stdlib/tests/builtins_test.cpp`; the
11 * suite runs under AddressSanitizer (the `asan` preset) and Valgrind
12 * (`security/run-valgrind.sh`) on every QA-gate run.

23/// Sized<C>: C reports a `.size()` — strings and STL containers (list/dict/array).
24template <typename C>
25concept Sized = requires(const C& c) {
26 { c.size() } -> std::convertible_to<std::size_t>;
27};
28
29/**
30 * Length / element count.
31 *
32 * Returns the element count of any `Sized` container by forwarding to its
33 * `.size()`; for strings this is the byte length, not a Unicode code-point count.
34 * @param c a string or sized container.
35 * @return `c.size()`.
36 * @complexity O(1).
37 * @alloc none.
38 * @test CheatahBuiltins.LenOrdChr
39 * @crtest BuiltinsCompileRun.Len
40 * @systest StdlibE2E.Builtins
41 */
42template <Sized C>
43std::size_t len(const C& c) { return c.size(); }
44/**
45 * Length of a C-string / string literal.
46 *
47 * Returns the byte length of the view; any embedded NUL bytes are counted (the
48 * length comes from the view, not from a terminating NUL).
49 * @param s the string.
50 * @return its byte length.
51 * @complexity O(1).
52 * @alloc none.
53 * @test CheatahBuiltins.LenOrdChr
54 * @crtest BuiltinsCompileRun.Len
55 * @systest StdlibE2E.Builtins
56 */
57std::size_t len(std::string_view s);

59/**
60 * Code point of the first byte.
61 *
62 * Returns the unsigned byte value of `s[0]` (range 0–255), ignoring any
63 * trailing bytes; an empty string yields 0 rather than throwing.
64 * @param s a one-character string.
65 * @return its byte value (0 if empty).
66 * @complexity O(1).
67 * @alloc none.
68 * @test CheatahBuiltins.LenOrdChr
69 * @crtest BuiltinsCompileRun.Ord
70 * @systest StdlibE2E.Builtins
71 */
72int ord(std::string_view s);
73/**
74 * Character for a code point.
75 *
76 * Builds a one-byte string from the low 8 bits of @p codepoint (it is narrowed
77 * to `char`), so values outside 0–255 wrap modulo 256 rather than producing
78 * multi-byte output.
79 * @param codepoint a byte value.
80 * @return the one-character string.
81 * @complexity O(1).
82 * @alloc none (1-char small-string optimization).
83 * @test CheatahBuiltins.LenOrdChr
84 * @crtest BuiltinsCompileRun.Chr
85 * @systest StdlibE2E.Builtins
86 */
87std::string chr(int codepoint);

89/**
90 * Hex representation.
91 *
92 * Formats @p value in base 16 with lowercase digits and a `0x` prefix; negatives
93 * are rendered as a leading `-` before the prefix (e.g. `-0x1f`), and 0 is `0x0`.
94 * @param value the integer.
95 * @return `"0x…"` (with sign).
96 * @complexity O(log @p value).
97 * @alloc allocates the result string.
98 * @test CheatahBuiltins.BaseReprs
99 * @crtest BuiltinsCompileRun.Hex
100 * @systest StdlibE2E.Builtins
101 */
102std::string hex(long long value);
103/**
104 * Octal representation.
105 *
106 * Formats @p value in base 8 with a `0o` prefix; negatives get a leading `-`
107 * before the prefix (e.g. `-0o17`), and 0 is `0o0`.
108 * @param value the integer.
109 * @return `"0o…"` (with sign).
110 * @complexity O(log @p value).
111 * @alloc allocates the result string.
112 * @test CheatahBuiltins.BaseReprs
113 * @crtest BuiltinsCompileRun.Oct
114 * @systest StdlibE2E.Builtins
115 */
116std::string oct(long long value);
117/**
118 * Binary representation.
119 *
120 * Formats @p value in base 2 with a `0b` prefix; negatives get a leading `-`
121 * before the prefix (e.g. `-0b101`), and 0 is `0b0`.
122 * @param value the integer.
123 * @return `"0b…"` (with sign).
124 * @complexity O(log @p value).
125 * @alloc allocates the result string.
126 * @test CheatahBuiltins.BaseReprs
127 * @crtest BuiltinsCompileRun.Bin
128 * @systest StdlibE2E.Builtins
129 */
130std::string bin(long long value);

132/**
133 * Printable-ASCII repr (non-printables/`\`/`'` escaped, single-quoted).
134 *
135 * Wraps @p s in single quotes, passing through printable ASCII (bytes 32–126)
136 * verbatim while escaping `\` and `'` as `\\`/`\'` and emitting any other byte as
137 * a two-digit `\xHH` hex escape.
138 * @param s input.
139 * @return the quoted repr.
140 * @complexity O(n).
141 * @alloc allocates the result string.
142 * @test CheatahBuiltins.Ascii
143 * @crtest BuiltinsCompileRun.Ascii
144 * @systest StdlibE2E.Builtins
145 */
146std::string ascii(std::string_view s);

148/**
149 * Truthiness of a string.
150 *
151 * A string is truthy when it has at least one byte; a whitespace-only or
152 * `"0"`/`"false"` string is still truthy (only emptiness is false).
153 * @param s input.
154 * @return false iff @p s is empty.
155 * @complexity O(1).
156 * @alloc none.
157 * @test CheatahBuiltins.Conversions
158 * @crtest BuiltinsCompileRun.BoolFromString
159 * @systest StdlibE2E.Builtins
160 */
161bool to_bool(std::string_view s);
162/**
163 * Truthiness of a number.
164 * @param x any arithmetic value.
165 * @return @p x != 0.
166 * @complexity O(1).
167 * @alloc none.
168 * @test CheatahBuiltins.Conversions
169 * @crtest BuiltinsCompileRun.BoolFromNonzero
170 * @systest StdlibE2E.Builtins
171 */
172template <typename T>
173 requires std::is_arithmetic_v<T>
174bool to_bool(T x) { return x != T{}; }
175/**
176 * Parse a base-10 integer.
177 *
178 * Parses leading whitespace and an optional sign followed by decimal digits via
179 * `std::stoll`; it stops at the first non-digit (so trailing junk is ignored),
180 * throws on no parseable digits, and throws on out-of-range values.
181 * @param s the integer string.
182 * @return its value (throws on bad input).
183 * @complexity O(n).
184 * @alloc allocates a temporary `std::string` for the parse.
185 * @test CheatahBuiltins.Conversions
186 * @crtest BuiltinsCompileRun.IntFromString
187 * @systest StdlibE2E.Builtins
188 */
189long long to_int(std::string_view s);
190/**
191 * Truncate a double to an integer.
192 *
193 * Truncates toward zero (drops the fractional part rather than rounding), so
194 * `2.9` becomes 2 and `-2.9` becomes -2; values outside `long long` range are
195 * undefined behavior.
196 * @param x the value.
197 * @return @p x toward zero.
198 * @complexity O(1).
199 * @alloc none.
200 * @test CheatahBuiltins.Conversions
201 * @crtest BuiltinsCompileRun.IntFromFloat
202 * @systest StdlibE2E.Builtins
203 */
204long long to_int(double x);
205/**
206 * Parse a float.
207 *
208 * Parses leading whitespace and a floating-point literal via `std::stod`,
209 * accepting decimal, scientific (`1e9`), `inf`, and `nan` forms; it stops at the
210 * first unparsed character, throws when nothing parses, and throws on overflow.
211 * @param s a floating-point string.
212 * @return its value (throws on bad input).
213 * @complexity O(n).
214 * @alloc allocates a temporary `std::string` for the parse.
215 * @test CheatahBuiltins.Conversions
216 * @crtest BuiltinsCompileRun.FloatFromString
217 * @systest StdlibE2E.Builtins
218 */
219double to_float(std::string_view s);
220/**
221 * Widen an integer to a double.
222 * @param x the value.
223 * @return @p x as a `double`.
224 * @complexity O(1).
225 * @alloc none.
226 * @test CheatahBuiltins.ToFloatFromInt
227 * @crtest BuiltinsCompileRun.FloatFromInt
228 * @systest StdlibE2E.Builtins
229 */
230double to_float(long long x);

232/**
233 * Content hash of a string.
234 *
235 * Hashes the bytes via `std::hash<std::string_view>`, so equal contents hash
236 * equally; the value is implementation-defined and not stable across runs or
237 * compilers (do not persist it).
238 * @param s input.
239 * @return a `std::size_t` hash.
240 * @complexity O(n).
241 * @alloc none.
242 * @test CheatahBuiltins.Hash
243 * @systest StdlibE2E.Builtins
244 * @note No @crtest: compile-run coverage is intentionally skipped because the
245 * hash value is implementation-defined and has no portable expected stdout.
246 */
247std::size_t hash(std::string_view s);
248/**
249 * Hash of any hashable value.
250 *
251 * Defers to `std::hash<T>` for the static type of @p x, so it requires a
252 * specialization to exist for `T`; like the string overload, the result is
253 * implementation-defined and unstable across runs.
254 * @param x the value.
255 * @return `std::hash<T>{}(x)`.
256 * @complexity O(1) for scalars.
257 * @alloc none.
258 * @test CheatahBuiltins.Hash
259 * @systest StdlibE2E.Builtins
260 * @note No @crtest: compile-run coverage is intentionally skipped because the
261 * hash value is implementation-defined and has no portable expected stdout.
262 */
263template <typename T>
264std::size_t hash(const T& x) { return std::hash<T>{}(x); }
