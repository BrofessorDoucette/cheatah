47// ---- errors -------------------------------------------------------------------------------------
48//
49// `raise` throws an Error and `except` catches one. An Error carries a KIND alongside its message, so a
50// handler can select what it knows how to deal with (`except e of "index"`) and let everything else keep
51// travelling — which is the whole difference between recovering from a failure and swallowing one.
52//
53// The kind is a plain string, not a class hierarchy, because cheatah has no inheritance: "is-a" is a
54// concept, and a runtime taxonomy of errors is a discriminated value, not a base class. Kinds are open —
55// any string works — so a library can name its own failures without every caller having to know them.
56//
57// An Error is still a `str` wherever one is expected: it converts and compares as its MESSAGE, so
58// `io.print(e)` and `e == "boom"` read exactly as they did when a handler bound a bare string.
59
60/// Conventional kinds raised from the language core. Libraries are free to define their own.
61inline constexpr const char* kErrorKindError = "error"; ///< `raise "msg"` — unclassified
62inline constexpr const char* kErrorKindIndex = "index"; ///< subscript out of range
63inline constexpr const char* kErrorKindKey = "key"; ///< dict key absent
64inline constexpr const char* kErrorKindArithmetic = "arithmetic"; ///< divide / modulo by zero
65inline constexpr const char* kErrorKindUnknown = "unknown"; ///< a throw of a type we cannot inspect

67/**
68 * A raised error: a `kind` naming what went wrong and a human `message`.
69 *
70 * Derives from `std::runtime_error` so it interoperates with C++ code that catches `std::exception` —
71 * that inheritance is a C++ implementation detail and is not visible from cheatah, where an Error is an
72 * ordinary value with two string fields.
73 */
74class Error : public std::runtime_error {
75public:
76 /**
77 * An unclassified error — what `raise "msg"` builds. Kind is @ref kErrorKindError.
78 * @param message the human-readable description.
79 * @complexity O(message).
80 * @alloc copies the message (twice: the base class keeps its own).
81 */
82 explicit Error(std::string message)
83 : std::runtime_error(message), kind_(kErrorKindError), message_(std::move(message)) {}
84
85 /**
86 * A classified error — what `raise Error("kind", "msg")` builds.
87 * @param kind the open-ended kind string a handler selects on (`except e of "kind"`).
88 * @param message the human-readable description.
89 * @complexity O(kind + message).
90 * @alloc copies both strings.
91 */
92 Error(std::string kind, std::string message)
93 : std::runtime_error(message), kind_(std::move(kind)), message_(std::move(message)) {}
94
95 /**
96 * @brief What CLASS of failure this is — the string `except … of` matches against.
97 * @return the kind, by const reference; never empty for an Error built through these constructors.
98 * @complexity O(1).
99 * @alloc none.
100 */
101 const std::string& kind() const noexcept { return kind_; }
102
103 /**
104 * @brief The human-readable description. This is also what @ref str and `operator<<` yield, so a
105 * caught error prints as its message rather than as a struct.
106 * @return the message, by const reference.
107 * @complexity O(1).
108 * @alloc none.
109 */
110 const std::string& message() const noexcept { return message_; }
111
112 // Deliberately NOT implicitly convertible to std::string. It would read nicely, but `str()` is a
113 // heavily overloaded set and an implicit conversion makes half of it ambiguous the moment an Error
114 // is printed. The `str` overload and the comparisons below give the same ergonomics explicitly.
115
116private:
117 std::string kind_;
118 std::string message_;
119};

121/**
122 * @brief Compare an error against a string — by MESSAGE, so `e == "boom"` reads the way it did when a
123 * handler bound a bare string. Compare `e.kind()` when you mean the kind.

125 * @param s the message to compare against.
126 * @return true when the error's message is exactly @p s.
127 * @complexity O(min(len)).

160/**
161 * The error currently being handled, normalized to an @ref Error.
162 *
163 * Called from inside a `catch (...)`, where `throw;` re-raises the in-flight exception so it can be
164 * inspected by type. This is what lets ONE handler shape cover a raised Error, a `std::exception` from
165 * any C++ library, and a throw of some type we have never heard of — the last of which used to travel
166 * straight past every handler and abort the process.
167 *
168 * @return the in-flight exception as an Error: a raised Error verbatim, a `std::out_of_range` as kind
169 * "index", a `std::domain_error` as "arithmetic", any other `std::exception` as "error", and
170 * anything else as "unknown".
171 * @complexity O(1) plus the message copy.
172 * @alloc copies the kind and message.
173 */
174inline Error current_error() {
175 try {
176 throw;
177 } catch (const Error& e) {
178 return e;
179 } catch (const std::out_of_range& e) {
180 return Error(kErrorKindIndex, e.what());
181 } catch (const std::domain_error& e) {
182 return Error(kErrorKindArithmetic, e.what());
183 } catch (const std::exception& e) {
184 return Error(kErrorKindError, e.what());
185 } catch (...) {
186 return Error(kErrorKindUnknown, "unknown error");
187 }
188}

190/// Runs its action when the scope ends, however it ends — the body of a `finally`.
191template <std::invocable F>
192class Finally {
193public:
194 /**
195 * @brief Take ownership of the action to run at scope exit.
196 * @param f the callable to invoke from the destructor.
197 * @complexity O(1).
198 * @alloc moves @p f into the guard.
199 */
200 explicit Finally(F f) : f_(std::move(f)) {}
201 Finally(const Finally&) = delete;
202 Finally& operator=(const Finally&) = delete;
203
204 /**
205 * @brief Run the action. Reached on every exit path — normal fall-through, `return`, `break`, or an
206 * exception unwinding through the scope, which is the whole point of a guard over a
207 * duplicated block.
208 * @complexity that of the action.
209 * @alloc that of the action.
210 */
211 ~Finally() {
212 // A `finally` that throws while an exception is already unwinding would terminate the process,
213 // which is a worse outcome than losing the second error — so it is swallowed here.
214 try {
215 f_();
216 } catch (...) {
217 }
218 }
219
220private:
221 F f_;
222};
223
224/**
225 * @brief Build a scope guard around @p f — how `finally { … }` lowers.
226 * @param f the callable to run when the enclosing scope ends.
227 * @return the guard; keep it alive for the scope you want covered.
228 * @complexity O(1).
229 * @alloc moves @p f into the returned guard.
230 */
231template <std::invocable F>
232Finally<F> make_finally(F f) {
233 return Finally<F>(std::move(f));
234}

532/**
533 * True division — the cheatah `/` operator (like Python 3): **always floating-point**,
534 * so `6 / 2` is `3.0`, not `3`, and integer operands never silently truncate. Use the
535 * `//` operator (@ref floordiv) when you want integer/floor division.
536 * @param a numerator.
537 * @param b denominator.
538 * @return `double(a) / double(b)`.
539 * @complexity O(1).
540 * @alloc none.
541 * @test CheatahBuiltins.Division
542 * @crtest BuiltinsCompileRun.TrueDivision
543 * @systest StdlibE2E.Builtins
544 */
545template <typename A, typename B>
546 requires std::is_arithmetic_v<A> && std::is_arithmetic_v<B>
547double truediv(A a, B b) {
548 return static_cast<double>(a) / static_cast<double>(b);
549}
550/**
551 * Floor division — the cheatah `//` operator (like Python): the quotient floored toward
552 * −∞. Integer operands give an integer (`7 // 2 == 3`, `-7 // 2 == -4`, flooring the way
553 * Python does, not truncating toward zero like raw C++); a floating operand gives a
554 * floored double (`7.0 // 2 == 3.0`).
555 * @param a numerator.
556 * @param b denominator.
557 * @return `floor(a / b)`, integral for integral operands.
558 * @complexity O(1).
559 * @alloc none.
560 * @test CheatahBuiltins.Division
561 * @crtest BuiltinsCompileRun.FloorDivision
562 * @systest StdlibE2E.Builtins
563 */
564template <std::integral A, std::integral B>
565std::common_type_t<A, B> floordiv(A a, B b) {
566 // A controlled error, not UB: integer divide-by-zero is undefined in C++ (SIGFPE/trap), so guard
567 // it so pure-cheatah `x // 0` raises rather than corrupting the process. (Float `//` is IEEE-safe.)
568 if (b == 0) throw std::domain_error("integer floor division by zero");
569 std::common_type_t<A, B> q = a / b; // C++ truncates toward zero…
570 if ((a % b != 0) && ((a < 0) != (b < 0))) --q; // …adjust to floor toward −∞
571 return q;
572}
573/**

635/**
636 * Append @p x to list @p v in place (Python `list.append`).
637 *
638 * Grows @p v by one, converting @p x to the list's element type. Usable as a
639 * method (`xs.append(x)`) or a bare call (`append(xs, x)`); the list is taken by

642 * @param x the value to append.
643 * @complexity amortized O(1).

651void append(std::vector<T>& v, U&& x) {

734/**
735 * Python FLOOR-mod for integers: the result takes the DIVISOR's sign (-7 % 3 == 2), unlike
736 * raw C++ `%`. Backs the `%` operator.
737 * @param a the dividend.
738 * @param b the divisor; @p b == 0 throws std::domain_error (integer modulo by zero).
739 * @return a mod b with the sign of @p b (Python floor-mod semantics).
740 * @complexity O(1).
741 * @alloc none.
742 * @test CheatahBuiltins.Mod
743 * @crtest LangFeatures.Modulo
744 */
745template <class A, class B>
746 requires(std::is_integral_v<A> && std::is_integral_v<B>)
747std::common_type_t<A, B> mod(A a, B b) {
748 using R = std::common_type_t<A, B>;
749 // A controlled error, not UB: integer `% 0` is undefined in C++ (SIGFPE/trap) — guard it so
750 // pure-cheatah `x % 0` raises rather than corrupting the process. (Float `%` is IEEE-safe.)
751 if (b == 0) throw std::domain_error("integer modulo by zero");
752 const R r = static_cast<R>(a) % static_cast<R>(b);
753 return (r != 0 && ((r < 0) != (b < 0))) ? r + static_cast<R>(b) : r;
754}
755
756/**
757 * Python floor-mod for floats (either operand): fmod adjusted to the divisor's sign,
758 * mirroring `7.5 % 2 == 1.5` and `-7.5 % 2 == 0.5`.
759 * @param a the dividend.
760 * @param b the divisor.
761 * @return a mod b as a double, with the sign of @p b (Python floor-mod semantics).
762 * @complexity O(1).
763 * @alloc none.
764 * @test CheatahBuiltins.Mod
765 * @crtest LangFeatures.Modulo
766 */
767template <class A, class B>
768 requires(!std::is_integral_v<A> || !std::is_integral_v<B>)
769double mod(A a, B b) {
770 const double r = std::fmod(static_cast<double>(a), static_cast<double>(b));
771 return (r != 0.0 && ((r < 0.0) != (static_cast<double>(b) < 0.0))) ? r + static_cast<double>(b)
772 : r;
773}

775// ---- indexing & slicing (Python `seq[i]` / `seq[i:j]`) ----
776//
777// The compiler lowers value-position `seq[i]` to `index(seq, i)` and `seq[a:b]`
778// to `slice(seq, a, b)` (a missing bound becomes 0 / `slice_end`). Indices may be
779// negative (counted from the end). Indexing a string yields a length-1 string
780// (Python semantics), so `s[i] == "<"` type-checks; slicing yields the same kind.
781
782/// Sentinel for an omitted slice upper bound (`s[a:]`): "to the end".
783inline constexpr long long slice_end = std::numeric_limits<long long>::max();
784
785namespace detail {
786inline long long norm_index(long long i, long long n) { return i < 0 ? i + n : i; }
787} // namespace detail
788
789/**
790 * Element at @p i of a string — a length-1 string (Python `s[i]`).
791 * Negative @p i counts from the end; out-of-range throws `std::out_of_range`.
792 * @param s the string.
793 * @param i the index (may be negative).
794 * @return the one-character string at @p i.
795 * @complexity O(1).
796 * @alloc none (1-char small-string optimization).
797 * @test CheatahBuiltins.IndexString
798 * @crtest BuiltinsCompileRun.IndexString
799 * @systest StdlibE2E.Builtins
800 */
801inline std::string index(const std::string& s, long long i) {
802 const long long n = static_cast<long long>(s.size());
803 i = detail::norm_index(i, n);
804 if (i < 0 || i >= n) throw std::out_of_range("string index out of range");
805 return std::string(1, s[static_cast<std::size_t>(i)]);
806}
807
808/**
809 * Element at @p i of a list/array (Python `xs[i]`), by CONST REFERENCE.
810 * Negative @p i counts from the end; out-of-range throws `std::out_of_range`.
811 *
812 * Returning a reference — not a copy — is what makes `xs[i].field` free: reading one field of a
813 * heap-owning element (a struct with strings/lists) no longer deep-copies the whole element. Value
814 * semantics are UNCHANGED at the `.purr` level, because codegen binds a subscript with plain `auto`
815 * (`let e = xs[i]` still copies), and the const-ness preserves cheatah's "list elements are
816 * read-only" rule — whole-element `xs[i] = v` assignment goes through a different path.
817 *
818 * Lifetime: the reference is into @p c, so it is valid as long as @p c is and is not mutated.
819 * Subscripting a temporary container is safe in the expression that does it (the temporary outlives
820 * the full-expression); binding that reference to a name that outlives the statement is not, and
821 * codegen never emits such a binding.
822 * @param c the sequence.
823 * @param i the index (may be negative).
824 * @return a const reference to the element at @p i.
825 * @complexity O(1).
826 * @alloc none.
827 * @test CheatahBuiltins.IndexList
828 * @crtest BuiltinsCompileRun.IndexList
829 * @systest StdlibE2E.Builtins
830 */
831template <typename C>
832 requires requires(const C& c) { c.data(); c.size(); } // contiguous seq (vector/array), not a map
833auto index(const C& c, long long i) -> const std::decay_t<decltype(c[0])>& {
834 const long long n = static_cast<long long>(c.size());
835 i = detail::norm_index(i, n);
836 if (i < 0 || i >= n) throw std::out_of_range("index out of range");
837 return c[static_cast<std::size_t>(i)];
838}

840/**
841 * Element at @p i of a `list[bool]` (Python `xs[i]`), by value.
842 * `std::vector<bool>` is the one sequence type the contiguous overload above
843 * cannot accept: it is bit-packed, so it has proxy references and no `.data()`.
844 * Same semantics — negative @p i counts from the end; out-of-range throws.
845 * @param c the bool list.
846 * @param i the index (may be negative).
847 * @return the element at @p i.
848 * @complexity O(1).
849 * @alloc none.
850 * @test CheatahBuiltins.IndexBoolList
851 * @crtest BuiltinsCompileRun.IndexBoolList
852 * @systest StdlibE2E.Builtins
853 */
854inline bool index(const std::vector<bool>& c, long long i) {
855 const long long n = static_cast<long long>(c.size());
856 i = detail::norm_index(i, n);
857 if (i < 0 || i >= n) throw std::out_of_range("index out of range");
858 return c[static_cast<std::size_t>(i)];
859}

861/**
862 * Value for @p key in a dict (Python `d[key]`), by CONST REFERENCE.
863 * Same rationale and lifetime rules as the sequence overload above: `d[key].field` stops
864 * deep-copying the mapped value, while `let v = d[key]` still copies.
865 * @param m the dict.
866 * @param key the key to look up.
867 * @return a const reference to the mapped value; an absent key raises kind `"key"`, which is distinct
868 * from the `"index"` a sequence subscript raises — a missing dict entry and a walked-off-the-end
869 * list are different mistakes and a handler should be able to take one without the other.
870 * @complexity O(1) average.
871 * @alloc none.
872 * @test CheatahBuiltins.IndexDict
873 * @crtest BuiltinsCompileRun.IndexDict
874 * @systest StdlibE2E.Builtins
875 */
876template <typename K, typename V, typename H, typename E, typename A, typename Key>
877 requires requires(const std::unordered_map<K, V, H, E, A>& m, const Key& key) { m.find(key); }
878const V& index(const std::unordered_map<K, V, H, E, A>& m, const Key& key) {
879 const auto it = m.find(key);
880 if (it == m.end()) throw Error(kErrorKindKey, "key not found");
881 return it->second;
882}

884/**
885 * Substring `s[lo:hi]` (Python slice semantics: clamped, negatives from the end).
886 * @param s the string.
887 * @param lo start index (default 0 at the call site).
888 * @param hi end index, or @ref slice_end for "to the end".
889 * @return the slice (empty if `lo >= hi` after clamping).
890 * @complexity O(hi-lo).
891 * @alloc the result string.
892 * @test CheatahBuiltins.SliceString
893 * @crtest BuiltinsCompileRun.SliceString
894 * @systest StdlibE2E.Builtins
895 */
896inline std::string slice(const std::string& s, long long lo, long long hi) {
897 const long long n = static_cast<long long>(s.size());
898 lo = detail::norm_index(lo, n);
899 hi = (hi == slice_end) ? n : detail::norm_index(hi, n);
900 if (lo < 0) lo = 0;
901 if (hi > n) hi = n;
902 if (lo >= hi) return std::string();
903 return s.substr(static_cast<std::size_t>(lo), static_cast<std::size_t>(hi - lo));
904}

906/**
907 * Sub-list `xs[lo:hi]` (Python slice semantics), returned as a new list.
908 * @param c the sequence.
909 * @param lo start index.
910 * @param hi end index, or @ref slice_end for "to the end".
911 * @return the slice as a `std::vector` of the element type.
912 * @complexity O(hi-lo).
913 * @alloc the result vector.
914 * @test CheatahBuiltins.SliceList
915 * @crtest BuiltinsCompileRun.SliceList
916 * @systest StdlibE2E.Builtins
917 */
918template <typename C>
919 requires requires(const C& c) { c.data(); c.size(); } // contiguous seq, not a map
920auto slice(const C& c, long long lo, long long hi) -> std::vector<std::decay_t<decltype(c[0])>> {
921 const long long n = static_cast<long long>(c.size());
922 lo = detail::norm_index(lo, n);
923 hi = (hi == slice_end) ? n : detail::norm_index(hi, n);
924 if (lo < 0) lo = 0;
925 if (hi > n) hi = n;
926 // assign() from the source range sizes the buffer ONCE — a push_back loop re-tests capacity every
927 // iteration and reallocates O(log n) times, which also keeps the loop from ever vectorizing. For
928 // trivially-copyable elements libstdc++ lowers this to a single memmove.
929 // Clamping with a ternary rather than guarding the assign with an `if` keeps every line here
930 // unconditionally executed (the coverage gate demands 100% lines), and assign() is happy with an
931 // empty range, so a reversed lo/hi simply yields an empty list exactly as the old loop did.
932 const long long stop = hi < lo ? lo : hi;
933 std::vector<std::decay_t<decltype(c[0])>> out;
934 out.assign(c.begin() + static_cast<std::size_t>(lo), c.begin() + static_cast<std::size_t>(stop));
935 return out;
936}
