28/// Sized<C>: C reports a `.size()` — strings and STL containers (list/dict/array).

34/// Value<T>: a cheatah value — movable, so it can be stored, passed, and returned.
35/// This is the baseline concept purrc stamps on every emitted function/method
36/// parameter, so no cheatah code ever instantiates a fully unconstrained template
37/// (keeps compile errors comprehensible). See `constrain-all-templates` policy.

244/**
245 * Content hash of a string.
246 *
247 * Hashes the bytes via `std::hash<std::string_view>`, so equal contents hash
248 * equally; the value is implementation-defined and not stable across runs or
249 * compilers (do not persist it).
250 * @param s input.
251 * @return a `std::size_t` hash.
252 * @complexity O(n).
253 * @alloc none.
254 * @test CheatahBuiltins.Hash
255 * @systest StdlibE2E.Builtins
256 * @note No @crtest: compile-run coverage is intentionally skipped because the
257 * hash value is implementation-defined and has no portable expected stdout.
258 */
259std::size_t hash(std::string_view s);
260/**
261 * Hash of any hashable value.
262 *
263 * Defers to `std::hash<T>` for the static type of @p x, so it requires a
264 * specialization to exist for `T`; like the string overload, the result is
265 * implementation-defined and unstable across runs.
266 * @param x the value.
267 * @return `std::hash<T>{}(x)`.
268 * @complexity O(1) for scalars.
269 * @alloc none.
270 * @test CheatahBuiltins.Hash
271 * @systest StdlibE2E.Builtins
272 * @note No @crtest: compile-run coverage is intentionally skipped because the
273 * hash value is implementation-defined and has no portable expected stdout.
274 */
275template <typename T>
276 requires requires(const T& x) { std::hash<T>{}(x); }
277std::size_t hash(const T& x) { return std::hash<T>{}(x); }

279// ---- collection + method-style helpers ----
280//
281// These power cheatah's growable lists and the method-call syntax `obj.f(a)`,
282// which lowers to `cheatah::builtins::f(obj, a)`. Free-function form works too:
283// `append(xs, x)` and `xs.append(x)` are the same call.

285/**
286 * Append @p x to list @p v in place (Python `list.append`).
287 *
288 * Grows @p v by one, converting @p x to the list's element type. Usable as a
289 * method (`xs.append(x)`) or a bare call (`append(xs, x)`); the list is taken by
290 * reference, so the caller's list is mutated.
291 * @param v the list to grow.
292 * @param x the value to append.
293 * @complexity amortized O(1).
294 * @alloc reallocates @p v when it outgrows its capacity.
295 * @test CheatahBuiltins.Append
296 * @crtest BuiltinsCompileRun.Append
297 * @systest StdlibE2E.Builtins
298 */
299template <typename T, typename U>
300 requires std::convertible_to<std::remove_cvref_t<U>, T>
301void append(std::vector<T>& v, U&& x) {
302 v.push_back(static_cast<T>(std::forward<U>(x)));
303}

351// ---- indexing & slicing (Python `seq[i]` / `seq[i:j]`) ----
352//
353// The compiler lowers value-position `seq[i]` to `index(seq, i)` and `seq[a:b]`
354// to `slice(seq, a, b)` (a missing bound becomes 0 / `slice_end`). Indices may be
355// negative (counted from the end). Indexing a string yields a length-1 string
356// (Python semantics), so `s[i] == "<"` type-checks; slicing yields the same kind.
357
358/// Sentinel for an omitted slice upper bound (`s[a:]`): "to the end".
359inline constexpr long long slice_end = std::numeric_limits<long long>::max();
360
361namespace detail {
362inline long long norm_index(long long i, long long n) { return i < 0 ? i + n : i; }
363} // namespace detail
364
365/**
366 * Element at @p i of a string — a length-1 string (Python `s[i]`).
367 * Negative @p i counts from the end; out-of-range throws `std::out_of_range`.
368 * @param s the string.
369 * @param i the index (may be negative).
370 * @return the one-character string at @p i.
371 * @complexity O(1).
372 * @alloc the 1-char result (small-string optimized).
373 * @test CheatahBuiltins.IndexString
374 * @crtest BuiltinsCompileRun.IndexString
375 * @systest StdlibE2E.Builtins
376 */
377inline std::string index(const std::string& s, long long i) {
378 const long long n = static_cast<long long>(s.size());
379 i = detail::norm_index(i, n);
380 if (i < 0 || i >= n) throw std::out_of_range("string index out of range");
381 return std::string(1, s[static_cast<std::size_t>(i)]);
382}

384/**
385 * Element at @p i of a list/array (Python `xs[i]`), by value.
386 * Negative @p i counts from the end; out-of-range throws `std::out_of_range`.
387 * @param c the sequence.
388 * @param i the index (may be negative).
389 * @return a copy of the element at @p i.
390 * @complexity O(1).
391 * @alloc none beyond copying the element.
392 * @test CheatahBuiltins.IndexList
393 * @crtest BuiltinsCompileRun.IndexList
394 * @systest StdlibE2E.Builtins
395 */
396template <typename C>
397 requires requires(const C& c) { c.data(); c.size(); } // contiguous seq (vector/array), not a map
398auto index(const C& c, long long i) -> std::decay_t<decltype(c[0])> {
399 const long long n = static_cast<long long>(c.size());
400 i = detail::norm_index(i, n);
401 if (i < 0 || i >= n) throw std::out_of_range("index out of range");
402 return c[static_cast<std::size_t>(i)];
403}

405/**
406 * Value for @p key in a dict (Python `d[key]`), by value.
407 * @param m the dict.
408 * @param key the key to look up.
409 * @return a copy of the mapped value (throws `std::out_of_range` if absent).
410 * @complexity O(1) average.
411 * @alloc none beyond copying the value.
412 * @test CheatahBuiltins.IndexDict
413 * @crtest BuiltinsCompileRun.IndexDict
414 * @systest StdlibE2E.Builtins
415 */
416template <typename K, typename V, typename H, typename E, typename A, typename Key>
417 requires requires(const std::unordered_map<K, V, H, E, A>& m, const Key& key) { m.find(key); }
418V index(const std::unordered_map<K, V, H, E, A>& m, const Key& key) {
419 const auto it = m.find(key);
420 if (it == m.end()) throw std::out_of_range("key not found");
421 return it->second;
422}

424/**
425 * Substring `s[lo:hi]` (Python slice semantics: clamped, negatives from the end).
426 * @param s the string.
427 * @param lo start index (default 0 at the call site).
428 * @param hi end index, or @ref slice_end for "to the end".
429 * @return the slice (empty if `lo >= hi` after clamping).
430 * @complexity O(hi-lo).
431 * @alloc the result string.
432 * @test CheatahBuiltins.SliceString
433 * @crtest BuiltinsCompileRun.SliceString
434 * @systest StdlibE2E.Builtins
435 */
436inline std::string slice(const std::string& s, long long lo, long long hi) {
437 const long long n = static_cast<long long>(s.size());
438 lo = detail::norm_index(lo, n);
439 hi = (hi == slice_end) ? n : detail::norm_index(hi, n);
440 if (lo < 0) lo = 0;
441 if (hi > n) hi = n;
442 if (lo >= hi) return std::string();
443 return s.substr(static_cast<std::size_t>(lo), static_cast<std::size_t>(hi - lo));
444}

446/**
447 * Sub-list `xs[lo:hi]` (Python slice semantics), returned as a new list.
448 * @param c the sequence.
449 * @param lo start index.
450 * @param hi end index, or @ref slice_end for "to the end".
451 * @return the slice as a `std::vector` of the element type.
452 * @complexity O(hi-lo).
453 * @alloc the result vector.
454 * @test CheatahBuiltins.SliceList
455 * @crtest BuiltinsCompileRun.SliceList
456 * @systest StdlibE2E.Builtins
457 */
458template <typename C>
459 requires requires(const C& c) { c.data(); c.size(); } // contiguous seq, not a map
460auto slice(const C& c, long long lo, long long hi) -> std::vector<std::decay_t<decltype(c[0])>> {
461 const long long n = static_cast<long long>(c.size());
462 lo = detail::norm_index(lo, n);
463 hi = (hi == slice_end) ? n : detail::norm_index(hi, n);
464 if (lo < 0) lo = 0;
465 if (hi > n) hi = n;
466 std::vector<std::decay_t<decltype(c[0])>> out;
467 for (long long k = lo; k < hi; ++k) out.push_back(c[static_cast<std::size_t>(k)]);
468 return out;
469}
