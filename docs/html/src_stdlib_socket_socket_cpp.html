21// Suppress SIGPIPE on send() to a closed peer. Linux passes MSG_NOSIGNAL per-call; macOS/
22// BSD have no such flag and instead use the SO_NOSIGPIPE socket option (set in socket()).
23// Define MSG_NOSIGNAL to 0 where it's absent (macOS, Windows) so the send() calls stay portable.
24#ifndef MSG_NOSIGNAL
25#define MSG_NOSIGNAL 0
26#endif

32// Winsock must be initialized once per process before any socket call. A function-local
33// static does it lazily and tears it down at exit.
34void ensure_winsock() {
35 struct WinsockInit {
36 WinsockInit() { WSADATA d; WSAStartup(MAKEWORD(2, 2), &d); }
37 ~WinsockInit() { WSACleanup(); }
38 };
39 static WinsockInit init;
40}

47// Resolve host:port to an IPv4 TCP address. Returns true and fills `out`/`len` on
48// success. Used by bind/connect so "localhost", dotted IPs, and DNS names all work.
49bool resolve(const std::string& host, long long port, sockaddr_storage& out, socklen_t& len) {
50 ensure_winsock();
51 addrinfo hints{};
52 hints.ai_family = AF_INET;
53 hints.ai_socktype = SOCK_STREAM;
54 hints.ai_protocol = IPPROTO_TCP;
55 const std::string service = std::to_string(port);
56 addrinfo* res = nullptr;
57 if (::getaddrinfo(host.c_str(), service.c_str(), &hints, &res) != 0 || res == nullptr) {
58 errno = EADDRNOTAVAIL;
59 return false;
60 }
61 std::memcpy(&out, res->ai_addr, res->ai_addrlen);
62 len = static_cast<socklen_t>(res->ai_addrlen);
63 ::freeaddrinfo(res);
64 return true;
65}

69long long socket() {
70 ensure_winsock();
71 const auto fd = ::socket(AF_INET, SOCK_STREAM, IPPROTO_TCP);
72#if defined(_WIN32)
73 if (fd == INVALID_SOCKET) return -1; // Winsock signals failure with INVALID_SOCKET
74#else
75#ifdef SO_NOSIGPIPE
76 // macOS/BSD: ask the kernel not to raise SIGPIPE on this socket (Linux uses
77 // MSG_NOSIGNAL per send() instead — SO_NOSIGPIPE isn't defined there).
78 if (fd >= 0) {
79 int on = 1;
80 ::setsockopt(fd, SOL_SOCKET, SO_NOSIGPIPE, &on, sizeof(on));
81 }
82#endif
83#endif
84 return static_cast<long long>(fd);
85}

87long long set_reuseaddr(long long fd) {
88 int yes = 1;
89 // optval is const void* on POSIX, const char* on Winsock — char* converts to both.
90 return ::setsockopt(as_fd(fd), SOL_SOCKET, SO_REUSEADDR,
91 reinterpret_cast<const char*>(&yes), sizeof(yes));
92}

94long long bind(long long fd, const std::string& host, long long port) {
95 sockaddr_storage addr{};
96 socklen_t len = 0;
97 if (!resolve(host, port, addr, len)) return -1;
98 return ::bind(as_fd(fd), reinterpret_cast<sockaddr*>(&addr), len);
99}

120long long local_port(long long fd) {
121 sockaddr_storage addr{};
122 socklen_t len = sizeof(addr);
123 if (::getsockname(as_fd(fd), reinterpret_cast<sockaddr*>(&addr), &len) != 0) return -1;
124 // sin_port sits at the same offset for IPv4/IPv6, and we only ever make
125 // AF_INET sockets, so reading it back is safe.
126 return ntohs(reinterpret_cast<sockaddr_in*>(&addr)->sin_port);
127}

129long long send(long long fd, const std::string& data) {
130 // send() returns ssize_t on POSIX, int on Winsock — long long holds both.
131 const long long n = ::send(as_fd(fd), data.data(),
132 static_cast<int>(data.size()), MSG_NOSIGNAL);
133 return n;
134}
135
136long long sendall(long long fd, const std::string& data) {
137 std::size_t sent = 0;
138 while (sent < data.size()) {
139 const long long n = ::send(as_fd(fd), data.data() + sent,
140 static_cast<int>(data.size() - sent), MSG_NOSIGNAL);
141 if (n <= 0) return -1;
142 sent += static_cast<std::size_t>(n);
143 }
144 return 0;
145}

147std::string recv(long long fd, long long bufsize) {
148 if (bufsize <= 0) return std::string();
149 std::string buf(static_cast<std::size_t>(bufsize), '\0');
150 const long long n = ::recv(as_fd(fd), buf.data(), static_cast<int>(buf.size()), 0);
151 if (n <= 0) return std::string();
152 buf.resize(static_cast<std::size_t>(n));
153 return buf;
154}

164long long shutdown(long long fd) {
165 // Half-close both directions WITHOUT releasing the fd: a blocking recv() on
166 // ANOTHER thread returns immediately (EOF). Used to wake a reader for a clean
167 // shutdown; the fd is still owned by the caller and must be close()d afterwards.
168#if defined(_WIN32)
169 return ::shutdown(as_fd(fd), SD_BOTH);
170#else
171 return ::shutdown(as_fd(fd), SHUT_RDWR);
172#endif
173}

175long long tcp_listen(const std::string& host, long long port, long long backlog) {
176 // If socket() fails (-1), the bind below fails too and we fall through to the
177 // error path — no separate early return needed.
178 long long fd = socket();
179 set_reuseaddr(fd);
180 if (bind(fd, host, port) != 0 || listen(fd, backlog) != 0) {
181 close(fd);
182 return -1;
183 }
184 return fd;
185}
186
187long long tcp_connect(const std::string& host, long long port) {
188 long long fd = socket();
189 if (connect(fd, host, port) != 0) {
190 close(fd);
191 return -1;
192 }
193 return fd;
194}

196long long set_timeout(long long fd, long long timeout_ms) {
197 ensure_winsock();
198#if defined(_WIN32)
199 const DWORD ms = timeout_ms > 0 ? static_cast<DWORD>(timeout_ms) : 0;
200 if (::setsockopt(as_fd(fd), SOL_SOCKET, SO_RCVTIMEO,
201 reinterpret_cast<const char*>(&ms), sizeof ms) != 0) return -1;
202 if (::setsockopt(as_fd(fd), SOL_SOCKET, SO_SNDTIMEO,
203 reinterpret_cast<const char*>(&ms), sizeof ms) != 0) return -1;
204#else
205 timeval tv{};
206 if (timeout_ms > 0) {
207 tv.tv_sec = timeout_ms / 1000;
208 tv.tv_usec = static_cast<suseconds_t>((timeout_ms % 1000) * 1000);
209 }
210 if (::setsockopt(as_fd(fd), SOL_SOCKET, SO_RCVTIMEO, &tv, sizeof tv) != 0) return -1;
211 if (::setsockopt(as_fd(fd), SOL_SOCKET, SO_SNDTIMEO, &tv, sizeof tv) != 0) return -1;
212#endif
213 return 0;
214}

228Conn& Conn::operator=(Conn&& other) noexcept {
229 if (this != &other) {
230 if (fd_ >= 0) cheatah::socket::close(fd_);
231 fd_ = other.fd_;
232 other.fd_ = -1;
233 }
234 return *this;
235}

237 if (fd_ >= 0) cheatah::socket::close(fd_);
