19// Suppress SIGPIPE on send() to a closed peer. Linux passes MSG_NOSIGNAL per-call; macOS/
20// BSD have no such flag and instead use the SO_NOSIGPIPE socket option (set in socket()).
21// Define MSG_NOSIGNAL to 0 where it's absent (macOS, Windows) so the send() calls stay portable.
22#ifndef MSG_NOSIGNAL
23#define MSG_NOSIGNAL 0
24#endif

30// Winsock must be initialized once per process before any socket call. A function-local
31// static does it lazily and tears it down at exit.
32void ensure_winsock() {
33 struct WinsockInit {
34 WinsockInit() { WSADATA d; WSAStartup(MAKEWORD(2, 2), &d); }
35 ~WinsockInit() { WSACleanup(); }
36 };
37 static WinsockInit init;
38}

45// Resolve host:port to an IPv4 TCP address. Returns true and fills `out`/`len` on
46// success. Used by bind/connect so "localhost", dotted IPs, and DNS names all work.
47bool resolve(const std::string& host, long long port, sockaddr_storage& out, socklen_t& len) {
48 ensure_winsock();
49 addrinfo hints{};
50 hints.ai_family = AF_INET;
51 hints.ai_socktype = SOCK_STREAM;
52 hints.ai_protocol = IPPROTO_TCP;
53 const std::string service = std::to_string(port);
54 addrinfo* res = nullptr;
55 if (::getaddrinfo(host.c_str(), service.c_str(), &hints, &res) != 0 || res == nullptr) {
56 errno = EADDRNOTAVAIL;
57 return false;
58 }
59 std::memcpy(&out, res->ai_addr, res->ai_addrlen);
60 len = static_cast<socklen_t>(res->ai_addrlen);
61 ::freeaddrinfo(res);
62 return true;
63}

67long long socket() {
68 ensure_winsock();
69 const auto fd = ::socket(AF_INET, SOCK_STREAM, IPPROTO_TCP);
70#if defined(_WIN32)
71 if (fd == INVALID_SOCKET) return -1; // Winsock signals failure with INVALID_SOCKET
72#else
73#ifdef SO_NOSIGPIPE
74 // macOS/BSD: ask the kernel not to raise SIGPIPE on this socket (Linux uses
75 // MSG_NOSIGNAL per send() instead — SO_NOSIGPIPE isn't defined there).
76 if (fd >= 0) {
77 int on = 1;
78 ::setsockopt(fd, SOL_SOCKET, SO_NOSIGPIPE, &on, sizeof(on));
79 }
80#endif
81#endif
82 return static_cast<long long>(fd);
83}

85long long set_reuseaddr(long long fd) {
86 int yes = 1;
87 // optval is const void* on POSIX, const char* on Winsock — char* converts to both.
88 return ::setsockopt(as_fd(fd), SOL_SOCKET, SO_REUSEADDR,
89 reinterpret_cast<const char*>(&yes), sizeof(yes));
90}

118long long local_port(long long fd) {
119 sockaddr_storage addr{};
120 socklen_t len = sizeof(addr);
121 if (::getsockname(as_fd(fd), reinterpret_cast<sockaddr*>(&addr), &len) != 0) return -1;
122 // sin_port sits at the same offset for IPv4/IPv6, and we only ever make
123 // AF_INET sockets, so reading it back is safe.
124 return ntohs(reinterpret_cast<sockaddr_in*>(&addr)->sin_port);
125}

127long long send(long long fd, const std::string& data) {
128 // send() returns ssize_t on POSIX, int on Winsock — long long holds both.
129 const long long n = ::send(as_fd(fd), data.data(),
130 static_cast<int>(data.size()), MSG_NOSIGNAL);
131 return n;
132}

134long long sendall(long long fd, const std::string& data) {
135 std::size_t sent = 0;
136 while (sent < data.size()) {
137 const long long n = ::send(as_fd(fd), data.data() + sent,
138 static_cast<int>(data.size() - sent), MSG_NOSIGNAL);
139 if (n <= 0) return -1;
140 sent += static_cast<std::size_t>(n);
141 }
142 return 0;
143}

145std::string recv(long long fd, long long bufsize) {
146 if (bufsize <= 0) return std::string();
147 std::string buf(static_cast<std::size_t>(bufsize), '\0');
148 const long long n = ::recv(as_fd(fd), buf.data(), static_cast<int>(buf.size()), 0);
149 if (n <= 0) return std::string();
150 buf.resize(static_cast<std::size_t>(n));

162long long tcp_listen(const std::string& host, long long port, long long backlog) {
163 // If socket() fails (-1), the bind below fails too and we fall through to the
164 // error path — no separate early return needed.
165 long long fd = socket();
166 set_reuseaddr(fd);
167 if (bind(fd, host, port) != 0 || listen(fd, backlog) != 0) {
168 close(fd);
169 return -1;
170 }
171 return fd;
172}

174long long tcp_connect(const std::string& host, long long port) {
175 long long fd = socket();
176 if (connect(fd, host, port) != 0) {
177 close(fd);
178 return -1;
179 }
180 return fd;
181}
