5/**
6 * @file ed25519.hpp
7 * @brief cheatah `ed25519` — public-key signatures (RFC 8032 Ed25519), implemented
8 * from scratch with NO external crypto dependency. `import ed25519` to use it.
9 *
10 * `import ed25519` includes this header AND links `libcheatah_ed25519` (which links
11 * `libcheatah_hashlib` for the SHA-512 Ed25519 needs internally). Python has no
12 * public-key crypto in its standard library, so this is a focused cheatah module,
13 * sibling to `hashlib`: a secret seed signs, the derived public key verifies, and a
14 * verifier holding only the public key cannot forge.
15 *
16 * Keys and signatures are passed as lowercase hex (printable, file-storable); the
17 * MESSAGE is raw bytes. The cheatah runtime links this module to verify a `.so`'s
18 * Ed25519 signature before loading it (`CHEATAH_VERIFY=strict`).
19 *
20 * SECURITY NOTES
21 * - `verify()` touches only public data (public key, message, signature) — no secret
22 * is handled, so there is no secret-dependent timing in the verify path.
23 * - `sign()` / `generate()` handle the secret seed; they are intended for OFFLINE use
24 * (the `purrc --sign` / `--keygen` tooling), not on an untrusted host.
25 * - The secret seed must stay secret. Anyone with it can sign as you.
26 *
27 * Unit tests: `stdlib/tests/ed25519_test.cpp` (RFC 8032 known-answer vectors,
28 * sign/verify round-trip, tamper + wrong-key rejection); the suite runs under
29 * AddressSanitizer and Valgrind on every QA-gate run.
30 */
31#include <string>
32#include <string_view>
33
34namespace cheatah::ed25519 {
35
36/**
37 * Generate a fresh 32-byte secret seed from the OS CSPRNG, as 64 hex chars.
38 *
39 * The returned seed is the private key; derive its public key with public_key(). NOT
40 * reproducible (uses os-level entropy) and must be kept secret. Throws
41 * `std::runtime_error` if the OS randomness source cannot be read.
42 * @return a 64-char lowercase hex string (32 random bytes).
43 * @complexity O(1) plus a syscall.
44 * @alloc allocates the result string.
45 * @test CheatahEd25519.GenerateRoundTrip
46 * @crtest Ed25519CompileRun.Generate
47 * @systest StdlibE2E.Ed25519
48 */
49std::string generate();
50
51/**
52 * Derive the 32-byte public key (hex) from a 32-byte secret seed (hex).
53 *
54 * Computes the Ed25519 public key per RFC 8032 (SHA-512 of the seed, clamp, scalar
55 * multiply the base point). Throws `std::invalid_argument` if @p secret_hex is not
56 * exactly 64 hex chars.
57 * @param secret_hex the secret seed as 64 hex chars.
58 * @return the public key as 64 hex chars.
59 * @complexity O(1) (one fixed-base scalar multiplication).
60 * @alloc allocates the result string and a temporary decoded-seed string.
61 * @test CheatahEd25519.KnownVectors, CheatahEd25519.GenerateRoundTrip
62 * @crtest Ed25519CompileRun.PublicKey
63 * @systest StdlibE2E.Ed25519
64 */
65std::string public_key(std::string_view secret_hex);
66
67/**
68 * Sign @p message with the secret seed @p secret_hex; returns a 64-byte signature (hex).
69 *
70 * RFC 8032 Ed25519 signing (deterministic — no RNG, so the same seed and message
71 * always produce the same signature). The message is raw bytes of any length. Throws
72 * `std::invalid_argument` if @p secret_hex is not 64 hex chars.
73 * @param secret_hex the secret seed as 64 hex chars.
74 * @param message the bytes to sign.
75 * @return the signature as 128 hex chars (64 bytes).
76 * @complexity O(n) in the message length (two message-length SHA-512s) plus two fixed-base
77 * scalar multiplications (public-key derivation and R).
78 * @alloc allocates the signature string and two message-sized internal hash-input buffers
79 * (plus small seed/digest temporaries).
80 * @test CheatahEd25519.KnownVectors, CheatahEd25519.SignVerifyRoundTrip
81 * @crtest Ed25519CompileRun.SignVerify
82 * @systest StdlibE2E.Ed25519
83 */
84std::string sign(std::string_view secret_hex, std::string_view message);
85
86/**
87 * Verify that @p signature_hex is a valid Ed25519 signature of @p message under the
88 * public key @p public_hex. Returns true iff valid.
89 *
90 * Touches only public data, so it is safe to run on an untrusted host. Returns false
91 * (never throws) for a malformed public key/signature or a hex-decoding failure, so a
92 * caller can treat any non-true result as "reject".
93 * @param public_hex the public key as 64 hex chars.
94 * @param message the signed bytes.
95 * @param signature_hex the signature as 128 hex chars.
96 * @return true iff the signature verifies.
97 * @complexity O(n) in the message length plus two scalar multiplications.
98 * @alloc allocates internal buffers (the decoded key/signature strings and a message-sized
99 * hash-input buffer).
100 * @test CheatahEd25519.KnownVectors, CheatahEd25519.RejectsTamperAndWrongKey
101 * @crtest Ed25519CompileRun.SignVerify
102 * @systest StdlibE2E.Ed25519
103 */
104bool verify(std::string_view public_hex, std::string_view message, std::string_view signature_hex);
105
106} // namespace cheatah::ed25519
