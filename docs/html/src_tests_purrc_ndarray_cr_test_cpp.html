1// Compile-run unit tests for the `ndarray` module: one test per function. Each
2// writes a tiny .purr that calls a single ndarray function, compiles it with purrc,
3// runs it under the cheatah runtime, and asserts the exact stdout. Complements the
4// in-process unit tests (stdlib/tests/ndarray_test.cpp) and the per-module
5// system-level test (StdlibE2E.Ndarray).
6//
7// Skipped (not callable from .purr):
8// - broadcast_to / broadcast_shapes: take std::vector<std::size_t>, but cheatah
9// `list[int]` lowers to std::vector<long long>, which doesn't convert.
10// - shape_of returns std::vector<long long>, which io.print can't stream directly;
11// it is exercised here by indexing the returned list (ShapeOf below).
12#include "e2e_harness.hpp"
13
14TEST(NdarrayCompileRun, Array) {
15 e2e::expect_e2e("ndarray_array", R"PURR(import io
16import ndarray
17io.print(ndarray.to_string(ndarray.array([1.0, 2.0, 3.0])))
18)PURR", "[1, 2, 3]\n");
19}
