3// x509_test — offline unit tests for the from-scratch X.509 certificate parser + path validator
4// (stdlib/tls/x509.hpp), which makes the TLS client MITM-proof. Fixtures are throwaway OpenSSL
5// certs embedded as DER hex in x509_test_certs.hpp.
6//
7// Regenerate fixtures (from a scratch dir):
8// openssl genrsa -out rsa_ca.key 2048
9// openssl req -x509 -new -key rsa_ca.key -sha256 -days 3650 -subj "/CN=cheatah Test RSA CA" \
10// -addext "basicConstraints=critical,CA:TRUE" -out rsa_ca.pem
11// ... (RSA/EC/Ed25519 leaves for example.test, a *.wild.test leaf, a root->intermediate->leaf
12// chain, a non-CA "intermediate", and an RSA+SHA-512 unsupported-algorithm leaf;
13// SHA-384-era sets: a secp384r1 CA + leaf signed -sha384, a prime256v1 intermediate
14// signed -sha384 by that P-384 CA with a -sha256 P-256 leaf under it (the Sectigo shape),
15// an RSA CA + leaf signed -sha384, a prime256v1 CA whose leaf is signed -sha384, and a
16// multi-SAN leaf with subjectAltName=DNS:first.test,DNS:second.test,DNS:third.test) ...
17// then `openssl x509 -in X.pem -outform DER | xxd -p` for each.

32std::string unhex(const std::string& h) {
33 auto v = [](char c) { return c <= '9' ? c - '0' : (c | 0x20) - 'a' + 10; };
34 std::string out;
35 out.reserve(h.size() / 2);
36 for (std::size_t i = 0; i + 1 < h.size(); i += 2)
37 out.push_back(static_cast<char>((v(h[i]) << 4) | v(h[i + 1])));
38 return out;
39}

41// A trust store containing the given CA certificates (DER hex).
42x::TrustStore store_of(std::initializer_list<std::string> ca_hex) {
43 x::TrustStore s;
44 for (const std::string& h : ca_hex) {
45 x::Cert c;
46 EXPECT_TRUE(x::parse_cert(unhex(h), c));
47 s.by_subject[c.subject].push_back(c);
48 }
49 return s;
50}

52std::string b64encode(const std::string& in) {
53 static const char* T = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";
54 std::string out;
55 std::size_t i = 0;
56 for (; i + 3 <= in.size(); i += 3) {
57 const unsigned n = (static_cast<unsigned char>(in[i]) << 16) |
58 (static_cast<unsigned char>(in[i + 1]) << 8) |
59 static_cast<unsigned char>(in[i + 2]);
60 out.push_back(T[(n >> 18) & 63]);
61 out.push_back(T[(n >> 12) & 63]);
62 out.push_back(T[(n >> 6) & 63]);
63 out.push_back(T[n & 63]);
64 }
65 if (i < in.size()) {
66 unsigned n = static_cast<unsigned char>(in[i]) << 16;
67 if (i + 1 < in.size()) n |= static_cast<unsigned char>(in[i + 1]) << 8;
68 out.push_back(T[(n >> 18) & 63]);
69 out.push_back(T[(n >> 12) & 63]);
70 out.push_back(i + 1 < in.size() ? T[(n >> 6) & 63] : '=');
71 out.push_back('=');
72 }
73 return out;
74}

111// A certificate with a malformed validity time (a non-digit in notBefore or notAfter) is rejected.
112TEST(CheatahX509, ParseRejectsBadValidityTime) {
113 const std::string base = unhex(x509test::kRsaLeaf);
114 const std::string utctime("\x17\x0d", 2); // UTCTime tag + length 13 (YYMMDDHHMMSSZ)
115 const std::size_t t1 = base.find(utctime);
116 ASSERT_NE(t1, std::string::npos);
117 x::Cert c;
118 std::string bad_nb = base;
119 bad_nb[t1 + 2] = 'Q'; // first year digit -> non-digit -> parse_time fails on notBefore
120 EXPECT_FALSE(x::parse_cert(bad_nb, c));
121
122 const std::size_t t2 = base.find(utctime, t1 + 2); // the second UTCTime = notAfter
123 ASSERT_NE(t2, std::string::npos);
124 std::string bad_na = base;
125 bad_na[t2 + 2] = 'Q'; // notBefore stays valid; notAfter fails to parse
126 EXPECT_FALSE(x::parse_cert(bad_na, c));
127}

328TEST(CheatahX509, LoadTrustFilesTriesPathsInOrder) {
329 const std::string dir = ::testing::TempDir();
330 const std::string good = dir + "/cheatah_x509_ca.pem";
331 const std::string empty = dir + "/cheatah_x509_empty.pem";
332 { std::ofstream(good) << "-----BEGIN CERTIFICATE-----\n"
