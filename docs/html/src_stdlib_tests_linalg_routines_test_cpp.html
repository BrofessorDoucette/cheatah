99TEST(LinalgRoutines, ComplexProducts) {
100 const la::CNDArray a = cvec({C(1, 2), C(3, -1)});
101 const la::CNDArray b = cvec({C(0, 1), C(2, 0)});
102 // Bilinear dot (no conjugation): (1+2j)(0+1j) + (3-1j)(2) = (-2+1j) + (6-2j) = 4-1j.
103 EXPECT_TRUE(cclose(la::dot(a, b), C(4, -1)));
104 // Hermitian inner product (conjugate the first): conj(a)·b = (1-2j)(0+1j)+(3+1j)(2) = (2+1j)+(6+2j) = 8+3j.
105 EXPECT_TRUE(cclose(la::vdot(a, b), C(8, 3)));
106 // vdot(a,a) is the real squared norm ‖a‖² = 1+4+9+1 = 15.
107 EXPECT_TRUE(cclose(la::vdot(a, a), C(15, 0)));
108
109 // Conjugate transpose (Hermitian adjoint): transpose + conjugate every entry.
110 const la::CNDArray M = cmat(2, 2, {C(1, 1), C(2, 0), C(0, 0), C(3, -1)});
111 const la::CNDArray H = la::conj_transpose(M); // [[1-1j, 0],[2, 3+1j]]
112 EXPECT_TRUE(cclose(cget(H, {0, 0}), C(1, -1)));
113 EXPECT_TRUE(cclose(cget(H, {0, 1}), C(0, 0)));
114 EXPECT_TRUE(cclose(cget(H, {1, 0}), C(2, 0)));
115 EXPECT_TRUE(cclose(cget(H, {1, 1}), C(3, 1)));
116
117 // Complex matmul: M · Mᴴ is Hermitian; check entry (0,0) = |1+1j|² + |2|² = 2 + 4 = 6.
118 const la::CNDArray P = la::matmul(M, H);
119 EXPECT_TRUE(cclose(cget(P, {0, 0}), C(6, 0)));
120 EXPECT_TRUE(cclose(cget(P, {1, 1}), C(10, 0))); // |0|² + |3-1j|² = 0 + 10
121
122 // as_cvector accepts a 2-D N×1 / 1×N as a flat vector (like the real path)…
123 const la::CNDArray col = cmat(2, 1, {C(1, 0), C(0, 1)});
124 EXPECT_TRUE(cclose(la::dot(col, col), C(0, 0))); // 1·1 + i·i = 1 − 1 = 0
125 // …and rejects a genuine 2-D matrix where a vector is required.
126 EXPECT_THROW(la::vdot(M, M), std::runtime_error);
127}

129TEST(LinalgRoutines, GeneralEigVectors) {
130 // For a real matrix, eig() returns complex eigenvalues AND eigenvectors (via
131 // inverse iteration). Verify A·v_k = λ_k·v_k for each column (phase-independent),
132 // building a complex copy Ac of A so we can multiply the complex eigenvectors.
133 const auto check = [](const std::vector<double>& data, std::size_t n) {
134 const nd::NDArray A = mat(n, n, data);
135 std::vector<C> cdata;
136 for (double x : data) cdata.push_back(C(x, 0.0));
137 const la::CNDArray Ac = cmat(n, n, cdata);
138 const la::EigC e = la::eig(A);
139 const la::CNDArray AV = la::matmul(Ac, e.vectors);
140 for (std::size_t k = 0; k < n; ++k) {
141 const C lam = cget(e.values, {(long long)k});
142 for (std::size_t r = 0; r < n; ++r) {
143 EXPECT_TRUE(cclose(cget(AV, {(long long)r, (long long)k}),
144 lam * cget(e.vectors, {(long long)r, (long long)k}), 1e-5));
145 }
146 }
147 };
148 check({2, 1, 0, 3}, 2); // real eigenvalues 3, 2 (upper-triangular)
149 check({0, -1, 1, 0}, 2); // complex conjugate pair ±i (rotation)
150 check({0, 1, 2, 0}, 2); // eigenvalues ±√2; forces a pivot row-swap in the solve
151 check({2, 1, 1, 1, 2, 1, 1, 1, 2}, 3); // symmetric -> real eigenvalues 4,1,1
152}

155 // H = [[2, 1+i],[1-i, 3]] is Hermitian (conj_transpose(H) == H); eigenvalues 4, 1.
156 const la::CNDArray H = cmat(2, 2, {C(2, 0), C(1, 1), C(1, -1), C(3, 0)});

163 EXPECT_TRUE(close(nd::get(e.values, {1}), 1.0, 1e-6));
164 // Verify the eigenpairs: H·V should equal V·diag(λ), independent of eigenvector

172 }

175 const la::CNDArray vk = cvec({cget(e.vectors, {0, k}), cget(e.vectors, {1, k})});

359TEST(LinalgRoutines, ComplexDotFourPlusElements) {
360 // 4+ elements drives the multi-accumulator loop in cdot (both dot and the conjugating vdot).
361 const la::CNDArray a = cvec({C(1, 1), C(2, 0), C(0, 1), C(1, -1), C(2, 2)});
362 const la::CNDArray b = cvec({C(1, 0), C(0, 1), C(1, 1), C(2, 0), C(0, -1)});
363 C dotref{}, vdotref{};
364 for (long long i = 0; i < 5; ++i) {
365 const C ai = cget(a, {i}), bi = cget(b, {i});
366 dotref += ai * bi;
367 vdotref += std::conj(ai) * bi;

409// Exercise the WIDE-UNROLL main loops of the multi-accumulator/blocked kernels: the
410// existing tests use tiny matrices that only ever run the scalar remainder, leaving the
411// 4-/8-wide vectorized bodies (ddot, real+complex matmul row-blocking, cholesky/qr/
412// tred2/trace reductions) uncovered. These use n≥8 so the main loops run.
413TEST(LinalgRoutines, WideKernelPaths) {
414 // 8×8 SPD: diag 10, off-diag 1 (= 9·I + J). Eigenvalues {17, 9×7}, trace 80.
415 std::vector<double> a8(64);
416 for (std::size_t i = 0; i < 8; ++i)
417 for (std::size_t j = 0; j < 8; ++j) a8[i * 8 + j] = (i == j) ? 10.0 : 1.0;
418 const nd::NDArray A = mat(8, 8, a8);
419
420 // dot over ≥8 elements → ddot 8-wide body.
421 EXPECT_DOUBLE_EQ(la::dot(nd::array(std::vector<double>(10, 1.0)),
422 nd::array(std::vector<double>(10, 2.0))), 20.0);
423 // trace 8×8 → trace 4-wide body.
424 EXPECT_DOUBLE_EQ(la::trace(A), 80.0);
425 // matmul 8×8 → real 4-row block. A·I == A.
426 std::vector<double> id8(64, 0.0);
427 for (std::size_t i = 0; i < 8; ++i) id8[i * 8 + i] = 1.0;
428 const nd::NDArray AI = la::matmul(A, mat(8, 8, id8));
429 EXPECT_DOUBLE_EQ(nd::get(AI, {0, 0}), 10.0);
430 EXPECT_DOUBLE_EQ(nd::get(AI, {1, 0}), 1.0);
431 // cholesky 8×8 (j reaches ≥4 → 4-wide inner dot). Reconstruct A = L·Lᵀ.
432 const nd::NDArray L = la::cholesky(A);
433 double a00 = 0;
434 for (long long k = 0; k < 8; ++k) a00 += nd::get(L, {0, k}) * nd::get(L, {0, k});
435 EXPECT_NEAR(a00, 10.0, 1e-9);
436 // qr 8×4 → reflect 4-wide body. R upper-triangular, Q·R == A_panel.
437 std::vector<double> p(32);
438 for (std::size_t i = 0; i < 8; ++i)
439 for (std::size_t j = 0; j < 4; ++j) p[i * 4 + j] = a8[i * 8 + j];
440 const la::QR qr = la::qr(mat(8, 4, p));
441 EXPECT_NEAR(nd::get(qr.r, {1, 0}), 0.0, 1e-9); // upper-triangular
442 // eigvalsh 8×8 → tred2 mat-vec 4-wide body. Largest eigenvalue 17, sum 80.
443 const nd::NDArray w = la::eigvalsh(A);
444 EXPECT_NEAR(nd::get(w, {0}), 17.0, 1e-7);
445 double sw = 0;
446 for (long long i = 0; i < 8; ++i) sw += nd::get(w, {i});
447 EXPECT_NEAR(sw, 80.0, 1e-7);
448 // eig() on a symmetric matrix → the reuse-the-extracted-A symmetric branch.
449 const la::EigC e = la::eig(A);
450 EXPECT_NEAR(cget(e.values, {0}).real(), 17.0, 1e-6);
451
452 // complex matmul 8×8 → complex 4-row block.
453 std::vector<C> z(64), zi(64, C{0, 0});
454 for (std::size_t i = 0; i < 8; ++i) { z[i * 8 + i] = C{2, 0}; zi[i * 8 + i] = C{1, 0}; }
455 const la::CNDArray Z = cmat(8, 8, z), I = cmat(8, 8, zi);
456 EXPECT_TRUE(cclose(cget(la::matmul(Z, I), {3, 3}), C{2, 0}));
457}
