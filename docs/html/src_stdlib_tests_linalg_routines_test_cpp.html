45TEST(LinalgRoutines, CholeskyAndQR) {
46 const nd::NDArray A = mat(2, 2, {4, 2, 2, 3}); // SPD
47 const nd::NDArray L = la::cholesky(A);
48 const nd::NDArray LLt = la::matmul(L, la::matmul(la::inv(L), A)); // == A trivially; check L Lᵀ:
49 // verify L·Lᵀ == A
50 const nd::NDArray Lt = mat(2, 2, {nd::get(L, {0, 0}), nd::get(L, {1, 0}),
51 nd::get(L, {0, 1}), nd::get(L, {1, 1})});
52 const nd::NDArray rec = la::matmul(L, Lt);
53 EXPECT_TRUE(close(nd::get(rec, {0, 0}), 4.0));
54 EXPECT_TRUE(close(nd::get(rec, {1, 1}), 3.0));
55
56 const la::QR qr = la::qr(mat(3, 2, {1, 0, 1, 1, 0, 1}));
57 const nd::NDArray QtQ = la::matmul(la::pinv(qr.q), qr.q); // Q has orthonormal cols
58 EXPECT_TRUE(close(nd::get(QtQ, {0, 0}), 1.0, 1e-6));
59 const nd::NDArray reQR = la::matmul(qr.q, qr.r); // == original
60 EXPECT_TRUE(close(nd::get(reQR, {0, 0}), 1.0, 1e-6));
61 EXPECT_TRUE(close(nd::get(reQR, {1, 1}), 1.0, 1e-6));
