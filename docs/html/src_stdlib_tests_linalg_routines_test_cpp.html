48TEST(LinalgRoutines, MatmulIntoReusesBuffer) {
49 const nd::NDArray a = mat(2, 3, {1, 2, 3, 4, 5, 6});
50 const nd::NDArray b = mat(3, 2, {7, 8, 9, 10, 11, 12});
51 nd::NDArray out = nd::zeros({2, 2});
52 const double* const before = out.buffer()->data() + out.offset(); // capture the buffer identity
53 la::matmul(out, a, b); // [[58,64],[139,154]] into out
54 EXPECT_EQ(out.buffer()->data() + out.offset(), before); // SAME buffer — no reallocation
55 EXPECT_DOUBLE_EQ(nd::get(out, {0, 0}), 58);
56 EXPECT_DOUBLE_EQ(nd::get(out, {0, 1}), 64);
57 EXPECT_DOUBLE_EQ(nd::get(out, {1, 0}), 139);
58 EXPECT_DOUBLE_EQ(nd::get(out, {1, 1}), 154);

60 const nd::NDArray c = la::matmul(a, b);
61 EXPECT_DOUBLE_EQ(nd::get(out, {1, 1}), nd::get(c, {1, 1}));

65 nd::NDArray sq = mat(2, 2, {1, 2, 3, 4});

357TEST(LinalgRoutines, GeneralEigVectors) {
358 // For a real matrix, eig() returns complex eigenvalues AND eigenvectors (via
359 // inverse iteration). Verify A·v_k = λ_k·v_k for each column (phase-independent),
360 // building a complex copy Ac of A so we can multiply the complex eigenvectors.
361 const auto check = [](const std::vector<double>& data, std::size_t n) {
362 const nd::NDArray A = mat(n, n, data);
363 std::vector<C> cdata;
364 for (double x : data) cdata.push_back(C(x, 0.0));
365 const la::CNDArray Ac = cmat(n, n, cdata);
366 const la::EigC e = la::eig(A);
367 const la::CNDArray AV = la::matmul(Ac, e.vectors);
368 for (std::size_t k = 0; k < n; ++k) {
369 const C lam = cget(e.values, {(long long)k});
370 for (std::size_t r = 0; r < n; ++r) {
371 EXPECT_TRUE(cclose(cget(AV, {(long long)r, (long long)k}),
372 lam * cget(e.vectors, {(long long)r, (long long)k}), 1e-5));
373 }
374 }
375 };
376 check({2, 1, 0, 3}, 2); // real eigenvalues 3, 2 (upper-triangular)
377 check({0, -1, 1, 0}, 2); // complex conjugate pair ±i (rotation)
378 check({0, 1, 2, 0}, 2); // eigenvalues ±√2; forces a pivot row-swap in the solve
379 check({2, 1, 1, 1, 2, 1, 1, 1, 2}, 3); // symmetric -> real eigenvalues 4,1,1
380}

382TEST(LinalgRoutines, ComplexHermitianEigh) {
383 // H = [[2, 1+i],[1-i, 3]] is Hermitian (conj_transpose(H) == H); eigenvalues 4, 1.
384 const la::CNDArray H = cmat(2, 2, {C(2, 0), C(1, 1), C(1, -1), C(3, 0)});
385 const nd::NDArray w = la::eigvalsh(H); // real, descending
386 EXPECT_TRUE(close(nd::get(w, {0}), 4.0, 1e-6));
387 EXPECT_TRUE(close(nd::get(w, {1}), 1.0, 1e-6));
388
389 const la::EighC e = la::eigh(H);
390 EXPECT_TRUE(close(nd::get(e.values, {0}), 4.0, 1e-6));
391 EXPECT_TRUE(close(nd::get(e.values, {1}), 1.0, 1e-6));
392 // Verify the eigenpairs: H·V should equal V·diag(λ), independent of eigenvector
393 // phase. So column k of H·V equals λ_k · column k of V.
394 const la::CNDArray HV = la::matmul(H, e.vectors);

637// Exercise the WIDE-UNROLL main loops of the multi-accumulator/blocked kernels: the
638// existing tests use tiny matrices that only ever run the scalar remainder, leaving the
639// 4-/8-wide vectorized bodies (ddot, real+complex matmul row-blocking, cholesky/qr/
640// tred2/trace reductions) uncovered. These use n≥8 so the main loops run.
641TEST(LinalgRoutines, WideKernelPaths) {
642 // 8×8 SPD: diag 10, off-diag 1 (= 9·I + J). Eigenvalues {17, 9×7}, trace 80.
643 std::vector<double> a8(64);
644 for (std::size_t i = 0; i < 8; ++i)
645 for (std::size_t j = 0; j < 8; ++j) a8[i * 8 + j] = (i == j) ? 10.0 : 1.0;
646 const nd::NDArray A = mat(8, 8, a8);
647
648 // dot over ≥8 elements → ddot 8-wide body.
649 EXPECT_DOUBLE_EQ(la::dot(nd::array(std::vector<double>(10, 1.0)),
650 nd::array(std::vector<double>(10, 2.0))), 20.0);
651 // trace 8×8 → trace 4-wide body.
652 EXPECT_DOUBLE_EQ(la::trace(A), 80.0);
653 // matmul 8×8 → real 4-row block. A·I == A.
654 std::vector<double> id8(64, 0.0);
655 for (std::size_t i = 0; i < 8; ++i) id8[i * 8 + i] = 1.0;
656 const nd::NDArray AI = la::matmul(A, mat(8, 8, id8));
657 EXPECT_DOUBLE_EQ(nd::get(AI, {0, 0}), 10.0);
658 EXPECT_DOUBLE_EQ(nd::get(AI, {1, 0}), 1.0);
659 // cholesky 8×8 (j reaches ≥4 → 4-wide inner dot). Reconstruct A = L·Lᵀ.
660 const nd::NDArray L = la::cholesky(A);
661 double a00 = 0;
662 for (long long k = 0; k < 8; ++k) a00 += nd::get(L, {0, k}) * nd::get(L, {0, k});
663 EXPECT_NEAR(a00, 10.0, 1e-9);
664 // qr 8×4 → reflect 4-wide body. R upper-triangular, Q·R == A_panel.
665 std::vector<double> p(32);
666 for (std::size_t i = 0; i < 8; ++i)
667 for (std::size_t j = 0; j < 4; ++j) p[i * 4 + j] = a8[i * 8 + j];
668 const la::QR qr = la::qr(mat(8, 4, p));
669 EXPECT_NEAR(nd::get(qr.r, {1, 0}), 0.0, 1e-9); // upper-triangular
670 // eigvalsh 8×8 → tred2 mat-vec 4-wide body. Largest eigenvalue 17, sum 80.
671 const nd::NDArray w = la::eigvalsh(A);
672 EXPECT_NEAR(nd::get(w, {0}), 17.0, 1e-7);
673 double sw = 0;
674 for (long long i = 0; i < 8; ++i) sw += nd::get(w, {i});
675 EXPECT_NEAR(sw, 80.0, 1e-7);
676 // eig() on a symmetric matrix → the reuse-the-extracted-A symmetric branch.
677 const la::EigC e = la::eig(A);
678 EXPECT_NEAR(cget(e.values, {0}).real(), 17.0, 1e-6);
679
680 // complex matmul 8×8 → complex 4-row block.
681 std::vector<C> z(64), zi(64, C{0, 0});
682 for (std::size_t i = 0; i < 8; ++i) { z[i * 8 + i] = C{2, 0}; zi[i * 8 + i] = C{1, 0}; }
683 const la::CNDArray Z = cmat(8, 8, z), I = cmat(8, 8, zi);
684 EXPECT_TRUE(cclose(cget(la::matmul(Z, I), {3, 3}), C{2, 0}));
685}
