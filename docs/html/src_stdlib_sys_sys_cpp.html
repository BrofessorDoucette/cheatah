1// cheatah `sys` module implementation — see sys.hpp for the interface.
2
3#include "sys.hpp"
4
5namespace cheatah::sys {
6
7// Definition of the storage declared `extern` in the header. Empty until the
8// cheatah runtime forwards the program's arguments via cheatah_set_argv().
9std::vector<std::string> argv;
10
11void set_argv(int argc, char** argv_) {
12 argv.clear();
13 if (argc < 0 || argv_ == nullptr) return;
14 argv.reserve(static_cast<std::size_t>(argc));
15 for (int i = 0; i < argc; ++i) {
16 argv.emplace_back(argv_[i] ? argv_[i] : "");
17 }
18}
19
20} // namespace cheatah::sys

22// ---------------------------------------------------------------------------
23// The stable, exported entry point the cheatah runtime resolves with dlsym and
24// calls (before purr_main) to hand the program its command-line arguments. It is
25// `extern "C"` so the runtime can find it by a fixed name, and given default
26// visibility so it survives in the loaded module's dynamic symbol table. Only
27// programs that `import sys` carry this symbol; for the rest the runtime simply
28// finds nothing and forwards no arguments.
29#if defined(_WIN32)
30#define CHEATAH_SYS_EXPORT extern "C" __declspec(dllexport)
31#else
32#define CHEATAH_SYS_EXPORT extern "C" __attribute__((visibility("default")))
33#endif
34
35CHEATAH_SYS_EXPORT void cheatah_set_argv(int argc, char** argv) {
36 cheatah::sys::set_argv(argc, argv);
37}
