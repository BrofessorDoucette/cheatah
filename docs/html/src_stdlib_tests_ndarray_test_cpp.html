11// Security hardening: malicious/buggy shapes and indices must throw, not corrupt
12// memory (negative dims -> huge size; product overflow -> under-allocation; OOB
13// index -> out-of-bounds read). Matters once untrusted .purr can reach these.
14TEST(CheatahNDArray, RejectsMaliciousShapesAndIndices) {
15 EXPECT_THROW(nd::zeros({-1}), std::runtime_error); // negative dimension
16 EXPECT_THROW(nd::full({-3, 2}, 1.0), std::runtime_error); // negative dimension
17 const long long big = 1LL << 40; // product 2^120 wraps size_t
18 EXPECT_THROW(nd::zeros({big, big, big}), std::runtime_error);
19 EXPECT_THROW(nd::get(nd::array({1.0, 2.0}), {5}), std::runtime_error); // OOB index
20 EXPECT_THROW(nd::get(nd::array({1.0, 2.0}), {-1}), std::runtime_error); // negative index
21 EXPECT_THROW(nd::get(nd::reshape(nd::array({1.0, 2.0, 3.0, 4.0}), {2, 2}), {0}),
22 std::runtime_error); // wrong-rank index
23}

100 // A 0-d complex scalar formats without brackets.

104TEST(CheatahNDArray, ComplexConstructAndParts) {
105 using C = std::complex<double>;
106 const nd::NDArray re = nd::array({0.0, 1.0, 2.0});
107 const nd::NDArray im = nd::array({1.0, 0.0, -3.0});
108 const nd::basic_ndarray<C> z = nd::complex(re, im); // [0+1j, 1+0j, 2-3j]
109 EXPECT_EQ(nd::to_string(z), "[0+1j, 1+0j, 2-3j]");
110 EXPECT_EQ(nd::get(z, {2}), C(2, -3));
111 // real / imag pull the parts back out as real arrays.
112 EXPECT_EQ(nd::to_string(nd::real(z)), "[0, 1, 2]");
113 EXPECT_EQ(nd::to_string(nd::imag(z)), "[1, 0, -3]");
114 // conj negates the imaginary part; the "1+0j" element keeps a clean +0 (not -0).
115 EXPECT_EQ(nd::to_string(nd::conj(z)), "[0-1j, 1+0j, 2+3j]");
116 // On a real array: conj is identity, real is a copy, imag is all zeros.
117 EXPECT_EQ(nd::to_string(nd::conj(re)), "[0, 1, 2]");
118 EXPECT_EQ(nd::to_string(nd::real(re)), "[0, 1, 2]");
119 EXPECT_EQ(nd::to_string(nd::imag(re)), "[0, 0, 0]");
120 // complex() broadcasts a scalar imaginary part against the real vector.
121 EXPECT_EQ(nd::to_string(nd::complex(re, nd::scalar(5.0))), "[0+5j, 1+5j, 2+5j]");
122 // A strided (non-contiguous) view exercises map_array's odometer fallback.
123 const nd::basic_ndarray<C> zb = nd::broadcast_to(nd::scalar(C(1, 2)), {3});
124 EXPECT_EQ(nd::to_string(nd::conj(zb)), "[1-2j, 1-2j, 1-2j]");
125}

137// Cover both element-wise paths: the vectorized contiguous fast path (matching
138// shapes, no broadcast) and the C-order odometer fallback (a strided/broadcast
139// view), plus the strided-reduction (sum) fallback.
140TEST(CheatahNDArray, ContiguousFastPathAndStridedReduce) {
141 // Same-shape, contiguous operands -> the std::transform(unseq) fast path.
142 const nd::NDArray a = nd::array({1.0, 2.0, 3.0, 4.0});
143 const nd::NDArray b = nd::array({10.0, 20.0, 30.0, 40.0});
144 const nd::NDArray c = nd::add(a, b);
145 EXPECT_DOUBLE_EQ(nd::get(c, {0}), 11.0);
146 EXPECT_DOUBLE_EQ(nd::get(c, {3}), 44.0);
147 EXPECT_DOUBLE_EQ(nd::get(nd::mul(a, b), {1}), 40.0);
148 // Sum of a NON-contiguous (broadcast, stride-0) view -> the odometer fallback.
149 const nd::NDArray v = nd::broadcast_to(nd::scalar(2.0), {3}); // [2, 2, 2], stride 0
150 EXPECT_DOUBLE_EQ(nd::sum(v), 6.0);
151}
