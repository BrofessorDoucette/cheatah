12// Security hardening: malicious/buggy shapes and indices must throw, not corrupt
13// memory (negative dims -> huge size; product overflow -> under-allocation; OOB
14// index -> out-of-bounds read). Matters once untrusted .purr can reach these.

20 EXPECT_THROW(nd::get(nd::array({1.0, 2.0}), {5}), std::runtime_error); // OOB index
21 EXPECT_THROW(nd::get(nd::array({1.0, 2.0}), {-1}), std::runtime_error); // negative index
22 EXPECT_THROW(nd::get(nd::reshape(nd::array({1.0, 2.0, 3.0, 4.0}), {2, 2}), {0}),
23 std::runtime_error); // wrong-rank index
24}

121TEST(CheatahNDArray, ComplexConstructAndParts) {
122 using C = std::complex<double>;
123 const nd::NDArray re = nd::array({0.0, 1.0, 2.0});
124 const nd::NDArray im = nd::array({1.0, 0.0, -3.0});
125 const nd::basic_ndarray<C> z = nd::complex(re, im); // [0+1j, 1+0j, 2-3j]
126 EXPECT_EQ(nd::to_string(z), "[0+1j, 1+0j, 2-3j]");
127 EXPECT_EQ(nd::get(z, {2}), C(2, -3));
128 // real / imag pull the parts back out as real arrays.
129 EXPECT_EQ(nd::to_string(nd::real(z)), "[0, 1, 2]");
130 EXPECT_EQ(nd::to_string(nd::imag(z)), "[1, 0, -3]");
131 // conj negates the imaginary part; the "1+0j" element keeps a clean +0 (not -0).
132 EXPECT_EQ(nd::to_string(nd::conj(z)), "[0-1j, 1+0j, 2+3j]");
133 // On a real array: conj is identity, real is a copy, imag is all zeros.
134 EXPECT_EQ(nd::to_string(nd::conj(re)), "[0, 1, 2]");
135 EXPECT_EQ(nd::to_string(nd::real(re)), "[0, 1, 2]");
136 EXPECT_EQ(nd::to_string(nd::imag(re)), "[0, 0, 0]");
137 // complex() broadcasts a scalar imaginary part against the real vector.
138 EXPECT_EQ(nd::to_string(nd::complex(re, nd::scalar(5.0))), "[0+5j, 1+5j, 2+5j]");
139 // A strided (non-contiguous) view exercises map_array's odometer fallback.
140 const nd::basic_ndarray<C> zb = nd::broadcast_to(nd::scalar(C(1, 2)), {3});
141 EXPECT_EQ(nd::to_string(nd::conj(zb)), "[1-2j, 1-2j, 1-2j]");
142}

154// Cover both element-wise paths: the vectorized contiguous fast path (matching
155// shapes, no broadcast) and the C-order odometer fallback (a strided/broadcast
156// view), plus the strided-reduction (sum) fallback.
157TEST(CheatahNDArray, ContiguousFastPathAndStridedReduce) {
158 // Same-shape, contiguous operands -> the std::transform(unseq) fast path.
159 const nd::NDArray a = nd::array({1.0, 2.0, 3.0, 4.0});
160 const nd::NDArray b = nd::array({10.0, 20.0, 30.0, 40.0});
161 const nd::NDArray c = nd::add(a, b);
162 EXPECT_DOUBLE_EQ(nd::get(c, {0}), 11.0);
163 EXPECT_DOUBLE_EQ(nd::get(c, {3}), 44.0);
164 EXPECT_DOUBLE_EQ(nd::get(nd::mul(a, b), {1}), 40.0);
165 // Sum of a NON-contiguous (broadcast, stride-0) view -> the odometer fallback.
166 const nd::NDArray v = nd::broadcast_to(nd::scalar(2.0), {3}); // [2, 2, 2], stride 0
167 EXPECT_DOUBLE_EQ(nd::sum(v), 6.0);
168}
