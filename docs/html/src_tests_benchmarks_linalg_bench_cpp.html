13namespace nd = cheatah::ndarray;
14namespace la = cheatah::linalg;

16namespace {

18// A deterministic, well-conditioned m×n design-matrix entry (matches the LSQ e2e
19// test): small values with a strong diagonal on the top n×n block.
20double design(long long i, long long j, long long n) {
21 double v = static_cast<double>(((i * 7 + j * 3 + 1) % 7) - 3);
22 if (i < n && i == j) v += static_cast<double>(10 * n);
23 return v;
24}

26// 1-D dot product across a sweep of vector lengths. `items/bytes_processed` let
27// Google Benchmark report throughput.
28void BM_Dot(benchmark::State& state) {
29 const std::size_t n = static_cast<std::size_t>(state.range(0));
30 const nd::NDArray a = nd::ones({static_cast<long long>(n)});
31 const nd::NDArray b = nd::full({static_cast<long long>(n)}, 2.0);
32 for (auto _ : state) {
33 benchmark::DoNotOptimize(la::dot(a, b));
34 }
35 state.SetItemsProcessed(state.iterations() * static_cast<std::int64_t>(n));
36 state.SetBytesProcessed(state.iterations() * static_cast<std::int64_t>(n) * 2 * sizeof(double));
37}
38BENCHMARK(BM_Dot)->Arg(64)->Arg(1024)->Arg(65536);
39
40// Square matrix multiply across a sweep of dimensions — the O(n^3) hot path.
41void BM_Matmul(benchmark::State& state) {
42 const long long n = state.range(0);
43 const nd::NDArray a = nd::reshape(nd::ones({n * n}), {n, n});
44 const nd::NDArray b = nd::reshape(nd::full({n * n}, 2.0), {n, n});
45 for (auto _ : state) {
46 benchmark::DoNotOptimize(la::matmul(a, b));
47 }
48 state.SetItemsProcessed(state.iterations() * n * n * n);
49}
50BENCHMARK(BM_Matmul)->Arg(16)->Arg(64)->Arg(128);

52// Least-squares regression (lstsq = pinv·y, SVD-based) across the dimensionalities
53// the system-level tests cover: 2, 3, 5, 10, 15 parameters, each over-determined
54// with m = n + 5 samples. This is the hot path for the regression e2e tests.
55void BM_LeastSquares(benchmark::State& state) {
56 const long long n = state.range(0);
57 const long long m = n + 5;
58 std::vector<double> xdata(static_cast<std::size_t>(m * n));
59 for (long long i = 0; i < m; ++i)
60 for (long long j = 0; j < n; ++j)
61 xdata[static_cast<std::size_t>(i * n + j)] = design(i, j, n);
62 const nd::NDArray X = nd::reshape(nd::array(xdata), {m, n});
63 const nd::NDArray y = nd::reshape(nd::ones({m}), {m, 1}); // m×1 targets
64 for (auto _ : state) {
65 benchmark::DoNotOptimize(la::lstsq(X, y));
66 }
67 state.SetLabel(std::to_string(n) + "D");
68}
69BENCHMARK(BM_LeastSquares)->Arg(2)->Arg(3)->Arg(5)->Arg(10)->Arg(15);
