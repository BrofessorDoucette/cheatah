1// Compile-run unit tests for the `statistics` module: one test per function.
2// Each writes a tiny .purr that calls a single statistics function over a fixed
3// list, compiles it with purrc, runs it under the cheatah runtime, and asserts
4// the exact stdout. Complements the in-process unit tests
5// (stdlib/tests/statistics_test.cpp) and the per-module system-level test
6// (StdlibE2E.Statistics).
7#include "e2e_harness.hpp"
8
9// Fixed dataset {2,4,4,4,5,5,7,9}: sum 40, count 8, mean 5, pvariance 4,
10// pstdev 2. Sample stats use {1,2,3,4,5}: variance 2.5, stdev sqrt(2.5),
11// median 3.
12
13TEST(StatisticsCompileRun, Sum) {
14 e2e::expect_e2e("statistics_sum", R"PURR(import io
15import statistics
16let xs = [2.0, 4.0, 4.0, 4.0, 5.0, 5.0, 7.0, 9.0]
17io.print(statistics.sum(xs))
18)PURR", "40\n");
19}
