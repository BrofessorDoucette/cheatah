1// System-level "real app" test: a multi-module grade-report program that only
2// passes if io + statistics + string + builtins + math all cooperate end to end.
3//
4// Unlike the per-module compile-run suites (which exercise one function each),
5// this is a small but genuine mini-app: it owns a fixed dataset of student
6// scores, classifies each score pass/fail against a threshold and into a letter
7// grade, builds an aligned table, and prints a summary computed from the data.
8// The program is fully DETERMINISTIC (no clocks/PIDs/RNG), so its stdout is
9// asserted byte-for-byte.
10//
11// Modules exercised together:
12// - io : print, str, format
13// - statistics : mean, median, stdev
14// - string : center, ljust, rjust
15// - builtins : len, float (to_float)
16// - math : round, min, max
17//
18// Note: purrc treats a newline as a statement terminator, so every call stays
19// on a single source line; wide rows are assembled via `+` concatenation.
20#include "e2e_harness.hpp"
21
22TEST(SystemApps, GradeReport) {
23 e2e::expect_e2e("app_grades", R"PURR(import io
24import string
25import math
26import statistics
27import builtins
28
29# --- classify a numeric score into a letter grade ---
30fn letter(score) {
31 if score >= 90.0 { return "A" }
32 if score >= 80.0 { return "B" }
33 if score >= 70.0 { return "C" }
34 if score >= 60.0 { return "D" }
35 return "F"
36}
37
38# --- fixed, deterministic dataset ---
39let names = ["Ada", "Bjarne", "Cy", "Dennis", "Edsger"]
40let scores = [91.5, 67.0, 100.0, 82.25, 58.5]
41let pass_mark = 70.0
42
43# ---------------- header ----------------
44io.print(string.center(" GRADE REPORT ", 40, "="))
45let head = string.ljust("NAME", 10) + " " + string.rjust("SCORE", 7) + " " + string.rjust("GRADE", 6) + " STATUS"
46io.print(head)
47io.print(string.ljust("", 40, "-"))
48
49# ---------------- per-student table rows ----------------
50let passed = 0
51let hi = scores[0]
52let lo = scores[0]
53let i = 0
54while i < len(names) {
55 let nm = names[i]
56 let sc = scores[i]
57 hi = math.max(hi, sc)
58 lo = math.min(lo, sc)
59 let status = "FAIL"
60 if sc >= pass_mark {
61 status = "PASS"
62 passed = passed + 1
63 }
64 let col_name = string.ljust(nm, 10)
65 let col_score = string.rjust(io.str(math.round(sc)), 7)
66 let col_grade = string.rjust(letter(sc), 6)
67 io.print(col_name + " " + col_score + " " + col_grade + " " + status)
68 i = i + 1
69}
70
71# ---------------- summary ----------------
72io.print(string.ljust("", 40, "-"))
73let n = len(scores)
74let rate = math.round(100.0 * float(passed) / float(n))
75let stdev2 = math.round(statistics.stdev(scores) * 100.0) / 100.0
76io.print(io.format("students : {}", n))
77io.print(io.format("passed : {}/{} ({} pct)", passed, n, rate))
78io.print(io.format("mean : {}", statistics.mean(scores)))
79io.print(io.format("median : {}", statistics.median(scores)))
80io.print(io.format("stdev : {}", stdev2))
81io.print(io.format("range : {} .. {}", math.round(lo), math.round(hi)))
82io.print(string.center(" END ", 40, "="))
83)PURR",
84 "============= GRADE REPORT =============\n"
85 "NAME SCORE GRADE STATUS\n"
86 "----------------------------------------\n"
87 "Ada 92 A PASS\n"
88 "Bjarne 67 D FAIL\n"
89 "Cy 100 A PASS\n"
90 "Dennis 82 B PASS\n"
91 "Edsger 59 F FAIL\n"
92 "----------------------------------------\n"
93 "students : 5\n"
94 "passed : 3/5 (60 pct)\n"
95 "mean : 79.85\n"
96 "median : 82.25\n"
97 "stdev : 17.09\n"
98 "range : 59 .. 100\n"
99 "================= END ==================\n");
100}
