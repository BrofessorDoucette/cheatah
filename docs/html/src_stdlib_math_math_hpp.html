3/**
4 * @file math.hpp
5 * @brief cheatah `math` — scalar math: Python's `math` module plus the
6 * `abs`/`min`/`max`/`pow` built-ins. `import math` to use it.
7 *
8 * Every function here is **pure and allocation-free** (operates on `double` /
9 * `long long` by value). Unit tests: `stdlib/tests/math_test.cpp`. The whole
10 * suite runs under AddressSanitizer (the `asan` preset) and Valgrind
11 * (`security/run-valgrind.sh`) on every QA-gate run.
12 *
13 * Doc convention (see also the other stdlib headers): each function documents
14 * its runtime complexity with @complexity, its heap allocation with @alloc, and
15 * the @test that covers it.
16 */

24// Concepts naming what each scalar op needs, so a misuse fails with the concept's
25// name rather than a deep template error (see constrain-all-templates policy).
26/// Numeric<T>: an arithmetic type — the int/float family `abs`/`pow` operate on.
27template <typename T>
28concept Numeric = std::is_arithmetic_v<T>;
29/// Ordered<T>: `<`-comparable, so `min`/`max` can pick the smaller/larger.
30template <typename T>
31concept Ordered = requires(const T& a, const T& b) {
32 { a < b } -> std::convertible_to<bool>;
33};

44/**
45 * Absolute value.
46 * @param x any signed value.
47 * @return |@p x|.
48 * @complexity O(1) time.
49 * @alloc none.
50 * @test CheatahMath.BuiltinLikeOps
51 * @crtest MathCompileRun.Abs
52 * @systest StdlibE2E.Math
53 */
54template <Numeric T>
55T abs(T x) { return x < T{} ? -x : x; }

57/**
58 * Smallest of two-or-more values (variadic; the overloads chain to fold extra args).
59 *
60 * Returns a reference bound to whichever argument compares smaller; on a tie
61 * (neither `b < a`) it returns @p a, the first argument. Because the result is
62 * a reference into the caller's arguments, it dangles if the operands are
63 * temporaries that outlive the call expression.
64 * @param a,b the values to compare (`operator<` required).
65 * @return a reference to the minimum.
66 * @complexity O(n) in the argument count.
67 * @alloc none.
68 * @test CheatahMath.BuiltinLikeOps
69 * @crtest MathCompileRun.Min
70 * @systest StdlibE2E.Math
71 */
72template <Ordered T>
73const T& min(const T& a, const T& b) { return (b < a) ? b : a; }
74/**
75 * Smallest of three-or-more values (folds the extra args onto the two-argument overload).
76 * @param a,b the first two values.
77 * @param rest the remaining values (`operator<` required).
78 * @return a reference to the minimum.
79 * @test CheatahMath.BuiltinLikeOps
80 * @crtest MathCompileRun.Min
81 * @systest StdlibE2E.Math
82 */
83template <Ordered T, Ordered... Rest>
84const T& min(const T& a, const T& b, const Rest&... rest) { return min(min(a, b), rest...); }
85
86/**
87 * Largest of two-or-more values (variadic; the overloads chain to fold extra args).
88 *
89 * Returns a reference bound to whichever argument compares larger; on a tie
90 * (neither `a < b`) it returns @p a, the first argument. As with min, the
91 * returned reference dangles if the operands are temporaries.
92 * @param a,b the values to compare (`operator<` required).
93 * @return a reference to the maximum.
94 * @complexity O(n) in the argument count.
95 * @alloc none.
96 * @test CheatahMath.BuiltinLikeOps
97 * @crtest MathCompileRun.Max
98 * @systest StdlibE2E.Math
99 */
100template <Ordered T>
101const T& max(const T& a, const T& b) { return (a < b) ? b : a; }
102/**
103 * Largest of three-or-more values (folds the extra args onto the two-argument overload).
104 * @param a,b the first two values.
105 * @param rest the remaining values (`operator<` required).
106 * @return a reference to the maximum.
107 * @test CheatahMath.BuiltinLikeOps
108 * @crtest MathCompileRun.Max
109 * @systest StdlibE2E.Math
110 */
111template <Ordered T, Ordered... Rest>
112const T& max(const T& a, const T& b, const Rest&... rest) { return max(max(a, b), rest...); }

114/**
115 * Power.
116 *
117 * Both operands are cast to `double` and forwarded to `std::pow`, so this
118 * follows IEEE-754 semantics (e.g. `pow(0, 0)` is 1, and a negative base with a
119 * non-integer exponent yields NaN); integer arguments lose exactness beyond
120 * 2^53.
121 * @param base the base.
122 * @param exp the exponent.
123 * @return @p base raised to @p exp (computed as `double`).
124 * @complexity O(1) time.
125 * @alloc none.
126 * @test CheatahMath.BuiltinLikeOps
127 * @crtest MathCompileRun.Pow
128 * @systest StdlibE2E.Math
129 */
130template <Numeric Base, Numeric Exp>
131double pow(Base base, Exp exp) {
132 return std::pow(static_cast<double>(base), static_cast<double>(exp));
133}
