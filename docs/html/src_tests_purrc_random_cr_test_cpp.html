1// Compile-run unit tests for the `random` module: one test per function. Each
2// writes a tiny .purr that calls a single random function, compiles it with
3// purrc, runs it under the cheatah runtime, and asserts the exact stdout.
4// Complements the in-process unit tests (stdlib/tests/random_test.cpp) and the
5// per-module system-level test (StdlibE2E.Random).
6//
7// RNG is non-deterministic until seeded, so every program seeds first and then
8// asserts a DETERMINISTIC property (degenerate range, in-bounds membership, or
9// same-seed reproducibility) — never a raw random value.
10#include "e2e_harness.hpp"
11
12// seed: same seed => same stream, proven by drawing twice and comparing.
13TEST(RandomCompileRun, Seed) {
14 e2e::expect_e2e("random_seed", R"PURR(import io
15import random
16random.seed(42)
17let a = random.random()
18random.seed(42)
19let b = random.random()
20io.print(a == b)
21)PURR", "True\n");
22}
23
24// random: seeded value lies in the half-open unit interval [0, 1).
25TEST(RandomCompileRun, Random) {
26 e2e::expect_e2e("random_random", R"PURR(import io
27import random
28random.seed(42)
29let r = random.random()
30io.print(0.0 <= r and r < 1.0)
31)PURR", "True\n");
32}
33
34// uniform: reproducible under a fixed seed and within the requested bounds.
35TEST(RandomCompileRun, Uniform) {
36 e2e::expect_e2e("random_uniform", R"PURR(import io
37import random
38random.seed(42)
39let a = random.uniform(-2.0, 3.0)
40random.seed(42)
41let b = random.uniform(-2.0, 3.0)
42io.print(a == b and -2.0 <= a and a <= 3.0)
43)PURR", "True\n");
44}

46// randint: a degenerate range is fully deterministic — randint(5, 5) == 5.
