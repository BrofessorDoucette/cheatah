14// Kinds of every token except the trailing EndOfInput, for compact assertions.
15std::vector<TokenKind> kinds(const LexResult& r) {
16 std::vector<TokenKind> out;
17 for (const auto& t : r.tokens) {
18 if (t.kind == TokenKind::EndOfInput) {
19 break;
20 }
21 out.push_back(t.kind);
22 }
23 return out;
24}

41TEST(CheatahLexer, ClassifiesKeywordsVsIdentifiers) {
42 // `let` and `fn` are keywords; `meow` and `print` are identifiers (print is a
43 // runtime builtin function, not a language keyword).
44 const LexResult r = tokenize("let meow = fn");
45 EXPECT_TRUE(r.ok());
46 EXPECT_EQ(kinds(r), (std::vector<TokenKind>{
47 TokenKind::Keyword, // let
48 TokenKind::Identifier, // meow
49 TokenKind::Assign, // =
50 TokenKind::Keyword, // fn
51 }));
52 EXPECT_EQ(r.tokens[0].text, "let");
53 EXPECT_EQ(r.tokens[1].text, "meow");
54}
