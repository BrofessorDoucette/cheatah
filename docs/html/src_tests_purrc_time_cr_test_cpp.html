1// Compile-run unit tests for the `time` module: one test per function. Each writes
2// a tiny .purr that calls a single time function, compiles it with purrc, runs it
3// under the cheatah runtime, and asserts the exact stdout. Complements the
4// in-process unit tests (stdlib/tests/time_test.cpp) and the per-module
5// system-level test (StdlibE2E.Time).
6//
7// Clock values are non-deterministic, so every program prints a DETERMINISTIC
8// boolean property (ordering / sign) instead of any raw clock value.
9#include "e2e_harness.hpp"
10
11TEST(TimeCompileRun, Time) {
12 e2e::expect_e2e("time_time", R"PURR(import io
13import time
14io.print(time.time() > 0.0)
15)PURR", "True\n");
16}
