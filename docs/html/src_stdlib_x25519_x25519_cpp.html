5// X25519 (RFC 7748) from scratch. Field elements of GF(2^255 - 19) are 16 limbs of 16 bits
6// in int64 lanes (the compact TweetNaCl-style representation): wide enough that schoolbook
7// multiplication cannot overflow before the carry pass, small enough to stay readable.
8// EVERYTHING here is constant-time in the secret scalar: fixed 255 ladder steps, arithmetic
9// conditional swaps (no branches on secret bits), and a fixed carry/reduce schedule.
10
11namespace cheatah::x25519 {
12namespace {
13
14using i64 = std::int64_t;
15using Fe = i64[16]; // one field element: 16 little-endian 16-bit limbs
16
17// Propagate carries so every limb fits 16 bits again; the top carry re-enters at limb 0
18// multiplied by 38 (= 2 * 19, because 2^256 = 38 mod p). @complexity O(1) @alloc none
19// @test CheatahX25519.Rfc7748Vector1
20void carry(Fe o) {
21 for (int i = 0; i < 16; ++i) {
22 o[i] += (1LL << 16);
23 const i64 c = o[i] >> 16;
24 o[(i + 1) * (i < 15)] += c - 1 + 37 * (c - 1) * (i == 15);
25 o[i] -= c << 16;
26 }
27}
28
29// Constant-time conditional swap: when b == 1 swap (p, q), when b == 0 leave them — by
30// XOR masking, never by branching on b (b derives from a SECRET scalar bit).
31// @complexity O(1) @alloc none @test CheatahX25519.Rfc7748Vector2
32void cswap(Fe p, Fe q, i64 b) {
33 const i64 mask = ~(b - 1);
34 for (int i = 0; i < 16; ++i) {
35 const i64 t = mask & (p[i] ^ q[i]);
36 p[i] ^= t;
37 q[i] ^= t;
38 }
39}
40
41// o = a + b (no carry; callers carry after multiplication). @complexity O(1) @alloc none
42// @test CheatahX25519.Rfc7748Vector1
43void add(Fe o, const Fe a, const Fe b) {
44 for (int i = 0; i < 16; ++i) o[i] = a[i] + b[i];
45}
46
47// o = a - b. @complexity O(1) @alloc none @test CheatahX25519.Rfc7748Vector1
48void sub(Fe o, const Fe a, const Fe b) {
49 for (int i = 0; i < 16; ++i) o[i] = a[i] - b[i];
50}
51
52// o = a * b mod p: schoolbook product, fold the high half back with *38, then two carry
53// passes restore the limb bounds. @complexity O(1) @alloc none @test CheatahX25519.Rfc7748Vector1
54void mul(Fe o, const Fe a, const Fe b) {
55 i64 t[31] = {0};
56 for (int i = 0; i < 16; ++i)
57 for (int j = 0; j < 16; ++j) t[i + j] += a[i] * b[j];
58 for (int i = 0; i < 15; ++i) t[i] += 38 * t[i + 16];
59 for (int i = 0; i < 16; ++i) o[i] = t[i];
60 carry(o);
61 carry(o);
62}
63
64// o = a^2 mod p. @complexity O(1) @alloc none @test CheatahX25519.Rfc7748Vector1
65void sqr(Fe o, const Fe a) { mul(o, a, a); }
66
67// o = z^-1 mod p by Fermat: z^(p-2), the standard fixed square-and-multiply chain (the two
68// skipped indices 2 and 4 encode p-2's binary form). Constant-time: fixed 254 iterations.
69// @complexity O(1) @alloc none @test CheatahX25519.DiffieHellman
70void invert(Fe o, const Fe z) {
71 Fe c;
72 for (int i = 0; i < 16; ++i) c[i] = z[i];
73 for (int i = 253; i >= 0; --i) {
74 sqr(c, c);
75 if (i != 2 && i != 4) mul(c, c, z);
76 }
77 for (int i = 0; i < 16; ++i) o[i] = c[i];
78}

80// Freeze to the canonical representative in [0, p) and serialize to 32 little-endian bytes.
81// The two trial subtractions of p use arithmetic selection (no data-dependent branch).
82// @complexity O(1) @alloc none @test CheatahX25519.Rfc7748Vector1
83void pack(unsigned char out[32], const Fe n) {
84 Fe t, m;
85 for (int i = 0; i < 16; ++i) t[i] = n[i];
86 carry(t);
87 carry(t);
88 carry(t);
89 for (int rep = 0; rep < 2; ++rep) {
90 m[0] = t[0] - 0xffed;
91 for (int i = 1; i < 15; ++i) {
92 m[i] = t[i] - 0xffff - ((m[i - 1] >> 16) & 1);
93 m[i - 1] &= 0xffff;
94 }
95 m[15] = t[15] - 0x7fff - ((m[14] >> 16) & 1);
96 const i64 borrow = (m[15] >> 16) & 1;
97 m[14] &= 0xffff;
98 cswap(t, m, 1 - borrow);
99 }
100 for (int i = 0; i < 16; ++i) {
101 out[2 * i] = static_cast<unsigned char>(t[i] & 0xff);
102 out[2 * i + 1] = static_cast<unsigned char>(t[i] >> 8);
103 }
104}

106// Parse 32 little-endian bytes into limbs; the top bit is MASKED OFF per RFC 7748.
107// @complexity O(1) @alloc none @test CheatahX25519.Rfc7748Vector1
108void unpack(Fe o, const unsigned char in[32]) {
109 for (int i = 0; i < 16; ++i) o[i] = in[2 * i] + (static_cast<i64>(in[2 * i + 1]) << 8);
110 o[15] &= 0x7fff;
111}

113// The X25519 scalar multiplication: clamp the scalar, then 255 Montgomery-ladder steps over
114// the u-coordinate only (x/z pairs), each step one cswap + the fixed add/sub/mul schedule
115// with a24 = 121665. @complexity O(1) @alloc none @test CheatahX25519.Rfc7748Vector1
116void scalarmult(unsigned char out[32], const unsigned char scalar[32], const unsigned char point[32]) {
117 unsigned char z[32];
118 for (int i = 0; i < 32; ++i) z[i] = scalar[i];
119 z[0] &= 248; // clamp: clear the low 3 bits (cofactor),
120 z[31] &= 127; // clear the top bit,
121 z[31] |= 64; // set bit 254.
122
123 Fe x, a, b, c, d, e, f;
124 static const Fe k121665 = {0xDB41, 1};
125 unpack(x, point);
126 for (int i = 0; i < 16; ++i) {
127 b[i] = x[i];
128 a[i] = c[i] = d[i] = 0;
129 }
130 a[0] = d[0] = 1;
131
132 for (int i = 254; i >= 0; --i) {
133 const i64 bit = (z[i >> 3] >> (i & 7)) & 1;
134 cswap(a, b, bit);
135 cswap(c, d, bit);
136 add(e, a, c);
137 sub(a, a, c);
138 add(c, b, d);
139 sub(b, b, d);
140 sqr(d, e);
141 sqr(f, a);
142 mul(a, c, a);
143 mul(c, b, e);
144 add(e, a, c);
145 sub(a, a, c);
146 sqr(b, a);
147 sub(c, d, f);
148 mul(a, c, k121665);
149 add(a, a, d);
150 mul(c, c, a);
151 mul(a, d, f);
152 mul(d, b, x);
153 sqr(b, e);
154 cswap(a, b, bit);
155 cswap(c, d, bit);
156 }
157 invert(c, c);
158 mul(a, a, c);
159 pack(out, a);
160}

164// 64-char lowercase/uppercase hex -> 32 bytes; false on any malformed input.
165// @complexity O(1) @alloc none @test CheatahX25519.RejectsMalformed
166bool hex32(std::string_view hex, unsigned char out[32]) {
167 if (hex.size() != 64) return false;
168 for (int i = 0; i < 32; ++i) {
169 unsigned v = 0;
170 for (int k = 0; k < 2; ++k) {
171 const char ch = hex[2 * i + k];
172 v <<= 4;
173 if (ch >= '0' && ch <= '9') v |= static_cast<unsigned>(ch - '0');
174 else if (ch >= 'a' && ch <= 'f') v |= static_cast<unsigned>(ch - 'a' + 10);
175 else if (ch >= 'A' && ch <= 'F') v |= static_cast<unsigned>(ch - 'A' + 10);
176 else return false;
177 }
178 out[i] = static_cast<unsigned char>(v);
179 }
180 return true;
181}

183// 32 bytes -> 64-char lowercase hex. @complexity O(1) @alloc the returned string
184// @test CheatahX25519.Rfc7748Vector1
185std::string hex_of(const unsigned char in[32]) {
186 static constexpr char kHex[] = "0123456789abcdef";
187 std::string out;
188 out.resize(64);
189 for (int i = 0; i < 32; ++i) {
190 out[2 * i] = kHex[in[i] >> 4];
191 out[2 * i + 1] = kHex[in[i] & 0xF];
192 }
193 return out;
194}

198std::string x25519(std::string_view scalar_hex, std::string_view point_hex) {
199 unsigned char scalar[32], point[32], out[32];
200 if (!hex32(scalar_hex, scalar) || !hex32(point_hex, point)) return "";
201 scalarmult(out, scalar, point);
202 unsigned char acc = 0; // contributory check: an all-zero shared secret is rejected
203 for (int i = 0; i < 32; ++i) acc |= out[i];
204 if (acc == 0) return "";
205 return hex_of(out);
206}

208std::string x25519_base(std::string_view scalar_hex) {
209 unsigned char scalar[32], out[32];
210 if (!hex32(scalar_hex, scalar)) return "";
211 static const unsigned char kBase[32] = {9};
212 scalarmult(out, scalar, kBase);
213 return hex_of(out);
214}
