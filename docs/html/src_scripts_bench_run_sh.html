scripts/bench_run.sh

1#!/usr/bin/env bash
2# bench_run.sh — the ONE way to invoke cheatah_benchmarks.
3#
4# Google Benchmark's defaults are a trap for a project that publishes numbers: run the binary
5# bare and you get ONE repetition, no statistics, no interleaving, and output that looks
6# exactly as authoritative as a careful measurement. Every caller in this repo used to pass
7# its own ad-hoc flag set, and whether a given number was rigorous depended on which script
8# happened to produce it.
9#
10# So there are three named profiles, and the name says what the numbers are worth:
11#
12# smoke 1 rep, 0.05s, no interleaving. DOES THIS RUN AT ALL — timings are discarded.
13# The QA gate's pass (scripts/bench_smoke.sh) lives here, and its speed is a
14# deliberate property: 447 cases must not cost the gate minutes for statistics
15# nobody reads. bench_main.cpp is told to skip its "not publishable" banner.
16#
17# gate 5 reps (15 on confirmation), interleaved, aggregates only. IS THIS A REGRESSION —
18# enough repetitions for a stable median, and the pair-sides-interleaved ordering
19# that keeps drift from masquerading as a ratio change.
20#
21# publish N reps, interleaved, all samples kept, stamped Markdown emitted. NUMBERS THAT
22# GO IN A DOC. Interleaving scatters every case's repetitions across the run, so a
23# pair's two sides are never measured as consecutive blocks with drift between
24# them. Each row is a median with its IQR. This is the profile bench_main.cpp
25# stamps `publishable: true`, and the only one whose output may reach a document.
26#
27# Usage:
28# scripts/bench_run.sh smoke [filter]
29# scripts/bench_run.sh gate [filter] [reps] [min_time]
30# scripts/bench_run.sh publish <suite-name> [filter] [reps]
31#
32# publish writes the stamped Markdown table to docs/bench/<suite-name>.md and the raw
33# Google Benchmark JSON beside it as docs/bench/<suite-name>.json.
34set -uo pipefail
35cd "$(git rev-parse --show-toplevel)"
36
37BIN=build/release/bin/cheatah_benchmarks
38PROFILE="${1:-}"
39
40bold() { printf '\n\033[1m[bench-run] %s\033[0m\n' "$*"; }
41die() { printf '\033[31m[bench-run] %s\033[0m\n' "$*" >&2; exit 1; }
42
43[ -n "$PROFILE" ] || die "usage: bench_run.sh {smoke|gate|publish} …"
44if [ ! -x "$BIN" ]; then
45 bold "building the benchmarks (release)…"
46 cmake --preset release -DCHEATAH_BUILD_BENCHMARKS=ON >/tmp/bench_run_cfg.log 2>&1 \
47 || die "configure failed — see /tmp/bench_run_cfg.log"
48 cmake --build --preset release --target cheatah_benchmarks -j"$(nproc)" \
49 >/tmp/bench_run_build.log 2>&1 || die "build failed — see /tmp/bench_run_build.log"
50fi
51
52# The commit reaches bench_main.cpp by environment rather than a compile definition: a -D
53# baked in at configure time silently goes stale, and a stamp naming the wrong commit is
54# worse than one admitting it does not know. `dirty` is recorded, and the doc lint refuses
55# to publish a table stamped from a dirty tree.
56COMMIT="$(git rev-parse --short HEAD)"
57# Dirtiness is judged against THIS SUITE'S OWN WATCHED SOURCES, not the whole tree.
58#
59# A whole-tree check is what a first attempt reaches for and it is wrong in a way that costs
60# real time: editing any file anywhere — a README, an unrelated script — marks every later
61# measurement untraceable, so you re-measure to clear it, and the next edit dirties it again.
62# What the stamp actually claims is "this commit's source produced this number", and the
63# source of a number is exactly what `watch:` already names. Markdown is excluded outright:
64# a document cannot change what code measures, so it must never invalidate a measurement.
65# (`stdlib/regex/` contains a README; without this, editing it marks the regex ENGINE stale.)
66#
67# `update-index --refresh` first — stale stat info alone makes --quiet report a phantom diff.
68git update-index --refresh >/dev/null 2>&1 || true
69dirty_check() { # dirty_check <comma-separated watch paths> -> 0 clean, 1 dirty
70 local spec=() p
71 if [ -z "${1:-}" ]; then return 0; fi # no watch paths: nothing to judge against
72 local IFS=,
73 for p in $1; do
74 p="$(printf '%s' "$p" | sed 's/^ *//;s/ *$//')"
75 [ -n "$p" ] && spec+=("$p")
76 done
77 [ "${#spec[@]}" -gt 0 ] || return 0
78 git diff --quiet -- "${spec[@]}" ':!*.md' ':!docs/bench' && return 0 || return 1
79}
80export CHEATAH_BENCH_COMMIT="$COMMIT"
81
82case "$PROFILE" in
83smoke)
84 FILTER="${2:-.*}"
85 export CHEATAH_BENCH_SMOKE=1
86 exec "$BIN" --benchmark_filter="$FILTER" --benchmark_min_time=0.05s
87 ;;
88
89gate)
90 FILTER="${2:-.*}"
91 REPS="${3:-5}"
92 MIN_TIME="${4:-0.2s}"
93 exec "$BIN" --benchmark_filter="$FILTER" \
94 --benchmark_repetitions="$REPS" --benchmark_min_time="$MIN_TIME" \
95 --benchmark_enable_random_interleaving=true \
96 --benchmark_report_aggregates_only=true --benchmark_format=csv
97 ;;
98
99publish)
100 SUITE="${2:?publish needs a suite name: bench_run.sh publish <suite> [filter] [reps]}"
101
102 # ---- the suite registry -------------------------------------------------------------
103 # THIS BLOCK IS THE DEFINITION OF EVERY PUBLISHED TABLE. Filter, layout, watched sources
104 # and (for curated tables) the exact row selection all live here, so `bench_run.sh publish
105 # <suite>` is a complete, reviewable command and re-running it reproduces the document.
106 # Curation that lives in a Markdown edit cannot be reproduced or reviewed; curation that
107 # lives here shows up in a diff.
108 #
109 # WATCH is what scripts/bench_table.purr uses to decide the table has gone stale: list the
110 # sources whose change invalidates the measurement. Too narrow and staleness never fires;
111 # too broad and it cries wolf on every commit. Aim at the code the benchmark exercises.
112 CHEATAH_BENCH_LAYOUT=""; CHEATAH_BENCH_WATCH=""; CHEATAH_BENCH_ROWS=""
113 case "$SUITE" in
114 crypto-vs-openssl)
115 SUITE_FILTER='Crypto'
116 CHEATAH_BENCH_LAYOUT=throughput
117 CHEATAH_BENCH_WATCH='stdlib/aead/, stdlib/hashlib/, tests/benchmarks/crypto_openssl_bench.cpp'
118 ;;
119 fixarray-vs-glm)
120 SUITE_FILTER='_(fixed|glm)$'
121 CHEATAH_BENCH_LAYOUT=opstype
122 CHEATAH_BENCH_WATCH='stdlib/fixarray/, tests/benchmarks/fixed_glm_bench.cpp'
123 ;;
124 fixarray-vs-glm-highlights)
125 # Narrowed to exactly the curated cases: the full fixed/glm sweep is ~280 cases and
126 # this table shows six of them, so filtering here turns a six-minute run into seconds
127 # without changing a single published number.
128 SUITE_FILTER='^BM_(identity_mat4f|matmul_mat4f|add_mat4f|inverse_mat4d|abs_vec4f|dot_vec4f)_(fixed|glm)$'
129 CHEATAH_BENCH_LAYOUT=highlights
130 CHEATAH_BENCH_WATCH='stdlib/fixarray/, tests/benchmarks/fixed_glm_bench.cpp'
131 # Six operations that show where structure pays, plus two deliberately kept because
132 # they land INSIDE the parity band — a highlights table that only showed wins would
133 # be a different kind of dishonest.
134 CHEATAH_BENCH_ROWS='BM_identity_mat4f=mat4f::identity();BM_matmul_mat4f=mat4f * mat4f;BM_add_mat4f=mat4f + mat4f;BM_inverse_mat4d=inverse(mat4d);BM_abs_vec4f=abs(vec4f);BM_dot_vec4f=dot(vec4f, vec4f)'
135 ;;
136 linalg-vs-eigen)
137 SUITE_FILTER='^BM_(dot|matmul|solve|inv)_(cheatah|eigen)'
138 CHEATAH_BENCH_LAYOUT=pairs
139 CHEATAH_BENCH_WATCH='stdlib/linalg/, tests/benchmarks/eigen_compare_bench.cpp'
140 ;;
141 integrity)
142 # The load-time integrity cost the security page publishes: four tiers (off, checksum,
143 # signature, + runtime manifest) at two module sizes, driven through verify_module itself
144 # rather than a process spawn.
145 SUITE_FILTER='^BM_Integrity'
146 CHEATAH_BENCH_LAYOUT=solo
147 CHEATAH_BENCH_WATCH='runtime/, stdlib/hashlib/, stdlib/ed25519/, tests/benchmarks/integrity_bench.cpp'
148 ;;
149 p256)
150 SUITE_FILTER='^BM_P256_'
151 CHEATAH_BENCH_LAYOUT=solo
152 CHEATAH_BENCH_WATCH='stdlib/p256/, tests/benchmarks/p256_bench.cpp'
153 ;;
154 *)
155 # An unregistered suite is allowed for ad-hoc exploration, but it gets no layout and
156 # no watch — so bench_table.purr will refuse to publish it, which is the right default.
157 SUITE_FILTER='.*'
158 printf '\033[33m[bench-run] %s is not in the suite registry — ad-hoc run, not publishable\033[0m\n' "$SUITE"
159 ;;
160 esac
161 export CHEATAH_BENCH_LAYOUT CHEATAH_BENCH_WATCH CHEATAH_BENCH_ROWS
162 dirty_check "$CHEATAH_BENCH_WATCH" || CHEATAH_BENCH_COMMIT="$COMMIT (dirty)"
163 export CHEATAH_BENCH_COMMIT
164
165 FILTER="${3:-$SUITE_FILTER}"
166 ROUNDS="${4:-9}"
167 OUT_DIR=docs/bench
168 mkdir -p "$OUT_DIR"
169
170 bold "publish: suite=$SUITE reps=$ROUNDS filter=$FILTER"
171 # ONE pass, with repetitions and random interleaving.
172 #
173 # This used to run N separate single-repetition "rounds" first and write a JSON sidecar
174 # per round. That was cost without a consumer: nothing read the sidecars, and the paired
175 # per-round ratios they were meant to enable are not what bench_main.cpp computes — it
176 # reports a median per case. The striation that matters is already here:
177 # --benchmark_enable_random_interleaving shuffles the flat list of repetition slots across
178 # every filtered case, so a pair's two sides are scattered through the run rather than
179 # measured as two consecutive blocks. Keeping a round loop that nothing consumed would
180 # have made a full fixarray publish take an hour to produce the same table.
181 CHEATAH_BENCH_SUITE="$SUITE" \
182 CHEATAH_BENCH_TABLE="$OUT_DIR/$SUITE.md" \
183 "$BIN" --benchmark_filter="$FILTER" \
184 --benchmark_repetitions="$ROUNDS" --benchmark_min_time=0.3s \
185 --benchmark_enable_random_interleaving=true \
186 --benchmark_out_format=json --benchmark_out="$OUT_DIR/$SUITE.json" \
187 --benchmark_format=console >/dev/null 2>&1 \
188 || die "measurement failed"
189 bold "wrote $OUT_DIR/$SUITE.md (+ $SUITE.json, the raw Google Benchmark output)"
190 ;;
191
192*)
193 die "unknown profile '$PROFILE' — expected smoke, gate, or publish"
194 ;;
195esac
