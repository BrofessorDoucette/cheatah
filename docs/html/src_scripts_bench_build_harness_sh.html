scripts/bench/build-harness.sh

1#!/usr/bin/env bash
2# build-harness.sh — compile a .purr benchmark harness that imports the shared `stamp` module.
3#
4# purrc resolves `import stamp` to an emitted LIBRARY HEADER, not to a .purr on disk, so the
5# module has to be emitted once before any harness that imports it will compile. That is the
6# same two-step docs/build-docs.sh performs for the biome module; this script exists so the
7# three benchmark harnesses do not each reinvent it.
8#
9# bash scripts/bench/build-harness.sh <harness.purr> <out.so>
10set -euo pipefail
11cd "$(git rev-parse --show-toplevel)"
12
13SRC="${1:?usage: build-harness.sh <harness.purr> <out.so>}"
14OUT="${2:?usage: build-harness.sh <harness.purr> <out.so>}"
15
16PURRC=build/release/bin/purrc
17[ -x "$PURRC" ] || { echo "build-harness: missing $PURRC — build the release preset first"; exit 1; }
18
19LIBDIR=build/bench-lib
20STAMP_HPP="$LIBDIR/stamp.hpp"
21if [ ! -f "$STAMP_HPP" ] || [ scripts/bench/stamp.purr -nt "$STAMP_HPP" ] || [ "$PURRC" -nt "$STAMP_HPP" ]; then
22 mkdir -p "$LIBDIR"
23 # --transparent: header-only, so a harness links nothing extra. Library mode also drops a
24 # top-level entry call, which stamp.purr does not have anyway.
25 "$PURRC" --emit-library --transparent scripts/bench/stamp.purr -o "$STAMP_HPP"
26fi
27
28"$PURRC" "$SRC" -o "$OUT" --import-root "$(pwd)/$LIBDIR"
