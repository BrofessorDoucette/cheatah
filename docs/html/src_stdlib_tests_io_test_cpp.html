15TEST(CheatahIo, StrRendersComplex) {
16 using C = std::complex<double>;
17 EXPECT_EQ(io::str(C(8, 3)), "8+3j"); // positive imaginary
18 EXPECT_EQ(io::str(C(1, -2)), "1-2j"); // negative imaginary
19 EXPECT_EQ(io::str(C(1, 0)), "1+0j"); // +0 (not "-0") after a conjugate
20 EXPECT_EQ(io::str(std::conj(C(1, 0))), "1+0j");
21 // repr matches str (numbers are not quoted), and a list of complex prints readably.
22 EXPECT_EQ(io::repr(C(2, -5)), "2-5j");
23 EXPECT_EQ(io::str(std::vector<C>{C(0, 1), C(2, 0)}), "[0+1j, 2+0j]");
24}

26// The io concept must accept exactly what the templates already take (streamable
27// types) and reject the rest, so a bad call gives a clear diagnostic.
28static_assert(io::Streamable<int>);
29static_assert(io::Streamable<std::string>);
30static_assert(io::Streamable<const char*>);
31static_assert(!io::Streamable<std::vector<int>>);
32
33// print() requires Printable, NOT Streamable: a value is printable if it streams
34// directly, has a `str()`, or is a list/dict of printable elements.
35namespace {
36struct WithStr { std::string str() const { return "W"; } };
37struct NotPrintable { int z; }; // not streamable, no str()
38} // namespace
39static_assert(io::Printable<int>);
40static_assert(io::Printable<std::vector<int>>);
41static_assert(io::Printable<std::vector<std::vector<std::string>>>);
42static_assert(io::Printable<std::unordered_map<std::string, int>>);
43static_assert(io::Printable<WithStr>);
44static_assert(!io::Printable<NotPrintable>); // no stream op, no str()
45static_assert(!io::Printable<std::vector<NotPrintable>>); // element not printable

64// A stand-in for a cheatah struct: it both streams (operator<<, the compact form rprint/str
65// use) and exposes a cheatah_pretty_print member (the pretty form print uses).
66struct PrettyStub {

69 os << std::string(static_cast<std::size_t>(indent), ' ') << "P(\n x = 1\n)";

90TEST(CheatahIo, StrRendersContainersAndObjects) {

96 EXPECT_EQ(io::str(WithStr{}), "W"); // a type with a str() method
