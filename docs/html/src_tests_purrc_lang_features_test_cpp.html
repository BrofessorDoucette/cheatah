2// growable lists (append) + dict mutation, method-call syntax, and string/list
3// slicing with negative indices. Each writes a .purr, compiles it with purrc,
4// runs it under the cheatah runtime, and asserts the exact stdout.

9let sum = 0

12 if i == 10 { break }
13 sum = sum + i
14}
15io.print(sum)
16)PURR",
17 "40\n"); // 1+2+3+4+6+7+8+9
18}

79TEST(LangFeatures, ParamsPassByReference) {
80 // Parameters bind by reference (Python object semantics): a function that
81 // mutates a struct field, an ndarray element, or appends to a list through
82 // its parameter changes the CALLER'S object. A literal argument still works
83 // (binds as a temporary).
84 e2e::expect_e2e("lang_byref_params", R"PURR(import io
85import ndarray
86
87struct Counter { hits: int }
88
89fn bump(c: Counter) {
90 c.hits += 1
91}
92
93fn scale_in_place(a, factor: float) {
94 a *= factor
95}
96
97fn push_two(xs: list) {
98 xs.append(2)
99}
100
101let c = Counter(0)
102bump(c)
103bump(c)
104let a = ndarray.array([1.0, 2.0])
105scale_in_place(a, 10.0)
106let xs = [1]
107push_two(xs)
108io.print(c.hits, ndarray.to_string(a), xs)
109)PURR", "2 [10, 20] [1, 2]\n");
110}

198TEST(LangFeatures, ReturnTypeHints) {
199 // Optional Python-style `-> Type` return hints. When present the function lowers with
200 // that concrete C++ return type (the backend enforces it); when absent the return stays
201 // `auto`. Mixed here: int, float, an ndarray, and an untyped function all interoperate.
202 e2e::expect_e2e("lang_return_hints", R"PURR(import io
203import ndarray
204fn add(a : int, b : int) -> int {
205 return a + b
206}
207fn half(x : float) -> float {
208 return x / 2.0
209}
210fn untyped(x) {
211 return x + 1
212}
213fn ones(n : int) -> ndarray<float> {
214 let a = ndarray.zeros([n])
215 for i in range(0, n) {
216 a[i] = 1.0
217 }
218 return a
219}
220io.print(add(2, 3))
221io.print(half(9.0))
222io.print(untyped(41))
223io.print(ndarray.to_string(ones(3)))
224)PURR",
225 "5\n4.5\n42\n[1, 1, 1]\n");
226}

228// Explicit template arguments on a call/construction, including NON-TYPE (integer-literal)
229// args: `f<3>(x)` and the mixed `f<int, 4>(x)`. A top-level cpp{} block supplies the C++
230// template fixtures so the program compiles AND runs; we also assert the emitted C++ carries
231// the literal (and the mapped type) verbatim. Guards the ambiguity with `<` as a comparison:
232// the args only commit when the angle list closes and is immediately followed by `(`.
233TEST(LangFeatures, NonTypeTemplateArgs) {
234 const std::string gen = e2e::expect_e2e_source("lang_nontype_targs", R"PURR(import io
235cpp {
236template <long long N>
237long long times_n(long long x) { return x * N; }
238
239template <typename T, long long N>
240T scale_n(T x) { return x * static_cast<T>(N); }
241}
242io.print(times_n<3>(7))
243io.print(scale_n<int, 4>(5))
244)PURR",
245 "21\n20\n");
246 // The single non-type arg is spliced verbatim.
247 EXPECT_NE(gen.find("times_n<3>"), std::string::npos)
248 << "expected the non-type template arg `<3>` in the emitted C++";
249 // Mixed: the type arg maps (int -> long long) and the non-type literal passes through.
250 EXPECT_NE(gen.find("scale_n<long long, 4>"), std::string::npos)
251 << "expected mixed `<long long, 4>` type + non-type args in the emitted C++";
252}

254// Compile-time `if constexpr (cond) {…}` — kept C++-style (parenthesised condition) and
255// lowered verbatim to C++ `if constexpr`, so the live branch is chosen at COMPILE time from
256// a constant condition. The constexpr-ness threads down the `else if constexpr` chain. A
257// cpp{} block supplies the constexpr fixtures so the program compiles AND runs; we also
258// assert the emitted C++ carries `if constexpr (` on both arms. Boolean logic (`and`,
259// comparisons) in the condition goes through the ordinary expression path.
260TEST(LangFeatures, IfConstexpr) {
261 const std::string gen = e2e::expect_e2e_source("lang_if_constexpr", R"PURR(import io
262cpp {
263constexpr int kMode = 2;
264constexpr bool kOn = true;
265}
266fn pick() {
267 if constexpr (kMode == 1 and kOn) {
268 return "one"
269 } else if constexpr (kMode == 2) {
270 return "two"
271 } else {
272 return "other"
273 }
274}
275io.print(pick())
276)PURR",
277 "two\n");
278 // The leading `if` lowers to a compile-time branch...
279 const std::size_t first = gen.find("if constexpr (");
280 ASSERT_NE(first, std::string::npos)
281 << "expected `if constexpr (` in the emitted C++";
282 // ...and the `else if constexpr` arm inherits it (a second occurrence).
283 EXPECT_NE(gen.find("if constexpr (", first + 1), std::string::npos)
284 << "the else-if arm should also lower to `if constexpr`";
285}
