1// End-to-end tests for the newer language features: break/continue, elif, match,
2// growable lists (append) + dict mutation, method-call syntax, and string/list
3// slicing with negative indices. Each writes a .purr, compiles it with purrc,
4// runs it under the cheatah runtime, and asserts the exact stdout.
5#include "e2e_harness.hpp"
6
7TEST(LangFeatures, BreakContinue) {
8 e2e::expect_e2e("lang_break_continue", R"PURR(import io
9let sum = 0
10for i in range(1, 20) {
11 if i == 5 { continue }
12 if i == 10 { break }
13 sum = sum + i
14}
15io.print(sum)
16)PURR",
17 "40\n"); // 1+2+3+4+6+7+8+9
18}
19
20TEST(LangFeatures, ElifChain) {
21 e2e::expect_e2e("lang_elif", R"PURR(import io
22fn classify(n) {
23 if n < 0 { return "neg" }
24 elif n == 0 { return "zero" }
25 elif n < 10 { return "small" }
26 else { return "big" }
27}
28for n in [-3, 0, 7, 99] {
29 io.print(classify(n))
30}
31)PURR",
32 "neg\nzero\nsmall\nbig\n");
33}
34
35TEST(LangFeatures, Match) {
36 e2e::expect_e2e("lang_match", R"PURR(import io
37for x in [1, 2, 3] {
38 match x {
39 case 1 { io.print("one") }
40 case 2 { io.print("two") }
41 case _ { io.print("many") }
42 }
43}
44)PURR",
45 "one\ntwo\nmany\n");
46}

48TEST(LangFeatures, AppendAndDictMutation) {
49 e2e::expect_e2e("lang_append", R"PURR(import io
50let xs: list[int] = []
51append(xs, 1)
52xs.append(2)
53xs.append(3)
54io.print(len(xs), xs[2])
55let counts: dict[str, int] = {}
56counts["x"] = 1
57counts["x"] = counts["x"] + 5
58io.print(counts["x"])
59)PURR",
60 "3 3\n6\n");
61}

63TEST(LangFeatures, MethodPredicates) {

66io.print("hello".endswith("lo"))

70}

72TEST(LangFeatures, StringSlicingAndIndex) {

78io.print(s[:5])
79io.print(s[-5:])
80io.print(s[0] == "h")
