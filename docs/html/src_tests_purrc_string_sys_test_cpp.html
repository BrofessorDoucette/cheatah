1// System-level test for the stdlib `string` module: one cohesive text-processing
2// pipeline that exercises EVERY public function in stdlib/string/string.hpp in a
3// single program (not isolated per-function prints — that is what the compile-run
4// suite string_cr_test.cpp does).
5//
6// The program takes a messy, multi-line "inventory note", then:
7// - normalizes it (splitlines, strip/lstrip/rstrip)
8// - re-cases it (upper, lower, capitalize, title, swapcase, capwords)
9// - probes it (startswith, endswith, contains, find, rfind, count)
10// - tokenizes it (split-on-sep, split-on-whitespace, join, replace)
11// - classifies tokens (isdigit, isalpha, isalnum, isupper, islower, isspace)
12// - formats a table (ljust, rjust, center, zfill)
13//
14// Every function in the header appears: case ops (upper/lower/capitalize/title/
15// swapcase), trimming (strip/lstrip/rstrip), search-test (startswith/endswith/
16// contains/find/rfind/count), transform (replace/split[2 overloads]/splitlines/
17// capwords/join), padding (ljust/rjust/center/zfill), classifiers (isdigit/
18// isalpha/isalnum/isspace/isupper/islower).
19//
20// List-returning split/splitlines are join()ed before printing; booleans go
21// through io.str (print as True/False). Fully DETERMINISTIC — stdout is asserted
22// byte-for-byte. Note: purrc treats a newline as a statement terminator, so each
23// call stays on one source line and wide rows are built with `+` concatenation;
24// only \n \t \" \\ escapes are supported by the lexer (no \r).
25#include "e2e_harness.hpp"
26
27TEST(StdlibE2E, String) {
28 e2e::expect_e2e("string_sys", R"PURR(import io
29import string
30
31let raw = " RaW inVENTORY \nsku001, sku002 , Cat-Food\nQTY: 42 units\n"
32
33# ---------------- normalize ----------------
34io.print(string.center(" REPORT ", 36, "="))
35
36# splitlines breaks the raw note into logical lines
37let lines = string.splitlines(raw)
38io.print(string.join(" / ", lines))
39
40# strip / lstrip / rstrip clean the first line's edges
41let header = lines[0]
42io.print("strip :" + string.strip(header))
43io.print("lstrip :" + string.lstrip(header))
44io.print("rstrip :" + string.rstrip(header) + "|")
45
46# ---------------- case ops ----------------
47let clean = string.strip(header)
48io.print("upper :" + string.upper(clean))
49io.print("lower :" + string.lower(clean))
50io.print("cap :" + string.capitalize(clean))
51io.print("title :" + string.title(clean))
52io.print("swap :" + string.swapcase(clean))
53io.print("capwords:" + string.capwords(raw))
54
55# ---------------- search / test ----------------
56let line2 = string.strip(lines[1])
57io.print("starts :" + io.str(string.startswith(line2, "sku")))
58io.print("ends :" + io.str(string.endswith(line2, "Food")))
59io.print("has :" + io.str(string.contains(line2, "Cat")))
60io.print("find :" + io.str(string.find(line2, "sku")))
61io.print("rfind :" + io.str(string.rfind(line2, "sku")))
62io.print("count :" + io.str(string.count(line2, "sku")))
63
64# ---------------- transform: split on comma ----------------
65let parts = string.split(line2, ",")
66io.print("nparts :" + io.str(len(parts)))
67io.print(string.ljust("", 36, "-"))
68
69# clean each comma-part, then split each on whitespace (ws split)
70let i = 0
71while i < len(parts) {
72 let tok = string.strip(parts[i])
73 let words = string.split(tok)
74 let joined = string.join("_", words)
75 let norm = string.replace(joined, "-", "")
76 # classify the normalized token
77 let kind = "mixed"
78 if string.isdigit(norm) {
79 kind = "digit"
80 }
81 if string.isalpha(norm) {
82 kind = "alpha"
83 }
84 if string.isalnum(norm) {
85 if kind == "mixed" {
86 kind = "alnum"
87 }
88 }
89 let col = string.ljust(string.lower(norm), 12)
90 io.print(col + string.rjust(kind, 8))
91 i = i + 1
92}
93io.print(string.ljust("", 36, "-"))
94
95# ---------------- numeric field: line 3 "QTY: 42 units" ----------------
96let line3 = lines[2]
97let qparts = string.split(line3)
98let qty = qparts[1]
99io.print("qty.isdigit :" + io.str(string.isdigit(qty)))
100io.print("qty.zfill :" + string.zfill(qty, 6))
101io.print("qty.center :" + string.center(qty, 8, "."))
102
103# ---------------- classifier sweep ----------------
104io.print("upper? ABC :" + io.str(string.isupper("ABC")))
105io.print("lower? abc :" + io.str(string.islower("abc")))
106io.print("space? :" + io.str(string.isspace(" \t")))
107
108io.print(string.center(" END ", 36, "="))
109)PURR",
110 "============== REPORT ==============\n"
111 " RaW inVENTORY / sku001, sku002 , Cat-Food / QTY: 42 units\n"
112 "strip :RaW inVENTORY\n"
113 "lstrip :RaW inVENTORY \n"
114 "rstrip : RaW inVENTORY|\n"
115 "upper :RAW INVENTORY\n"
116 "lower :raw inventory\n"
117 "cap :Raw inventory\n"
118 "title :Raw Inventory\n"
119 "swap :rAw INventory\n"

123 "has :True\n"
124 "find :0\n"

126 "count :2\n"

128 "------------------------------------\n"
129 "sku001 alnum\n"
130 "sku002 alnum\n"
131 "catfood alpha\n"
132 "------------------------------------\n"
133 "qty.isdigit :True\n"
134 "qty.zfill :000042\n"
135 "qty.center :...42...\n"
136 "upper? ABC :True\n"
137 "lower? abc :True\n"
138 "space? :True\n"
139 "=============== END ================\n");
