3// rsa_verify.hpp — minimal RSA-PSS (rsa_pss_rsae_sha256) signature verification for the TLS 1.3
4// CertificateVerify, so servers with an RSA leaf certificate complete the handshake. Verification
5// ONLY (no key generation, no signing): a small big-unsigned-integer (modexp with the public
6// exponent), an RSA public-key extractor from the certificate's SubjectPublicKeyInfo DER, and
7// EMSA-PSS-VERIFY (MGF1 + salt, both SHA-256). No OpenSSL — SHA-256 comes from hashlib. Header-only
8// and `inline`, included by exactly one translation unit (tls.cpp).
9//
10// Scope: rsa_pss_rsae_sha256 only (SignatureScheme 0x0804) — the scheme a TLS 1.3 server with an RSA
11// certificate uses for CertificateVerify. Salt length = hash length = 32 (per the scheme). The bignum
12// is correctness-first (schoolbook multiply + bit-serial reduction); the public exponent is tiny
13// (typically 65537), so a handful of modular squarings per handshake is negligible.

25// A big unsigned integer as little-endian 32-bit limbs (limb[0] = least significant); no trailing
26// zero limbs (the empty vector is the value 0).
27using Big = std::vector<std::uint32_t>;
28
29inline void trim(Big& a) {
30 while (!a.empty() && a.back() == 0) a.pop_back();
31}

33// Big from big-endian bytes (leading zero bytes are ignored).
34inline Big from_be(const unsigned char* p, std::size_t n) {
35 std::size_t s = 0;
36 while (s < n && p[s] == 0) ++s;
37 const std::size_t len = n - s;
38 Big a((len + 3) / 4, 0);
39 for (std::size_t i = 0; i < len; ++i) {
40 const unsigned char b = p[n - 1 - i]; // i-th byte from the least significant end
41 a[i / 4] |= static_cast<std::uint32_t>(b) << (8 * (i % 4));
42 }
43 trim(a);
44 return a;
45}
46inline Big from_be(const std::string& s) {
47 return from_be(reinterpret_cast<const unsigned char*>(s.data()), s.size());
48}

50inline std::size_t bitlen(const Big& a) {
51 if (a.empty()) return 0;
52 std::uint32_t top = a.back();
53 std::size_t b = 0;
54 while (top) {
55 ++b;
56 top >>= 1;
57 }
58 return (a.size() - 1) * 32 + b;
59}
60inline std::size_t bytelen(const Big& a) { return (bitlen(a) + 7) / 8; }

62// Big to big-endian bytes, left-padded/truncated to exactly `outlen` bytes (caller ensures it fits).
63inline std::string to_be(const Big& a, std::size_t outlen) {
64 std::string out(outlen, '\0');
65 for (std::size_t i = 0; i < a.size(); ++i)
66 for (int b = 0; b < 4; ++b) {
67 const std::size_t byteidx = i * 4 + b; // from the least significant end
68 if (byteidx >= outlen) continue;
69 out[outlen - 1 - byteidx] = static_cast<char>((a[i] >> (8 * b)) & 0xff);
70 }
71 return out;
72}

74inline int cmp(const Big& a, const Big& b) {
75 if (a.size() != b.size()) return a.size() < b.size() ? -1 : 1;
76 for (std::size_t i = a.size(); i-- > 0;)
77 if (a[i] != b[i]) return a[i] < b[i] ? -1 : 1;
78 return 0;
79}

81inline Big shl1(const Big& a) {
82 Big r(a.size(), 0);
83 std::uint32_t carry = 0;
84 for (std::size_t i = 0; i < a.size(); ++i) {
85 const std::uint64_t v = (static_cast<std::uint64_t>(a[i]) << 1) | carry;
86 r[i] = static_cast<std::uint32_t>(v);
87 carry = static_cast<std::uint32_t>(v >> 32);
88 }
89 if (carry) r.push_back(carry);
90 trim(r);
91 return r;
92}

94// a - b, assuming a >= b.
95inline Big sub(const Big& a, const Big& b) {
96 Big r(a.size(), 0);
97 std::int64_t borrow = 0;
98 for (std::size_t i = 0; i < a.size(); ++i) {
99 const std::int64_t bi = i < b.size() ? static_cast<std::int64_t>(b[i]) : 0;
100 std::int64_t v = static_cast<std::int64_t>(a[i]) - bi - borrow;
101 if (v < 0) {
102 v += (static_cast<std::int64_t>(1) << 32);
103 borrow = 1;
104 } else {
105 borrow = 0;
106 }
107 r[i] = static_cast<std::uint32_t>(v);
108 }
109 trim(r);
110 return r;
111}

113inline Big mul(const Big& a, const Big& b) {
114 if (a.empty() || b.empty()) return {};
115 Big r(a.size() + b.size(), 0);
116 for (std::size_t i = 0; i < a.size(); ++i) {
117 std::uint64_t carry = 0;
118 for (std::size_t j = 0; j < b.size(); ++j) {
119 const std::uint64_t cur = static_cast<std::uint64_t>(r[i + j]) +
120 static_cast<std::uint64_t>(a[i]) * b[j] + carry;
121 r[i + j] = static_cast<std::uint32_t>(cur);
122 carry = cur >> 32;
123 }
124 r[i + b.size()] = static_cast<std::uint32_t>(r[i + b.size()] + carry);
125 }
126 trim(r);
127 return r;
128}

130// x mod m, via bit-serial long division (correctness-first; called a few dozen times per handshake).
131inline Big mod(const Big& x, const Big& m) {
132 Big r; // 0
133 for (std::size_t bi = x.size() * 32; bi-- > 0;) {
134 const std::uint32_t bit = (x[bi / 32] >> (bi % 32)) & 1u;
135 r = shl1(r);
136 if (bit) {
137 if (r.empty())
138 r.push_back(1);
139 else
140 r[0] |= 1u;
141 }
142 if (cmp(r, m) >= 0) r = sub(r, m);
143 }
144 return r;
145}

147inline Big mulmod(const Big& a, const Big& b, const Big& m) { return mod(mul(a, b), m); }

149// base^e mod m (left-to-right square-and-multiply; e is small for RSA verification).
150inline Big modexp(Big base, const Big& e, const Big& m) {
151 base = mod(base, m);
152 Big result;
153 result.push_back(1);
154 const std::size_t bits = e.size() * 32;
155 std::size_t top = 0;
156 bool found = false;
157 for (std::size_t bi = bits; bi-- > 0;)
158 if ((e[bi / 32] >> (bi % 32)) & 1u) {
159 top = bi;
160 found = true;
161 break;
162 }
163 if (!found) return result; // e == 0 -> 1
164 for (std::size_t bi = top + 1; bi-- > 0;) {
165 result = mulmod(result, result, m);
166 if ((e[bi / 32] >> (bi % 32)) & 1u) result = mulmod(result, base, m);
167 }
168 return result;
169}

171// ---- DER: read a length octet sequence at `p` (advances p); SIZE_MAX on error ----
172inline std::size_t der_len(const std::string& d, std::size_t& p) {
173 if (p >= d.size()) return static_cast<std::size_t>(-1);
174 const unsigned char b = static_cast<unsigned char>(d[p++]);
175 if (!(b & 0x80)) return b; // short form
176 const int n = b & 0x7f;
177 if (n == 0 || n > 4 || p + static_cast<std::size_t>(n) > d.size()) return static_cast<std::size_t>(-1);
178 std::size_t L = 0;
179 for (int i = 0; i < n; ++i) L = (L << 8) | static_cast<unsigned char>(d[p++]);
180 return L;
181}

183// Extract the RSA public key (modulus n, exponent e as big-endian bytes) from a certificate's
184// SubjectPublicKeyInfo: find the rsaEncryption OID (1.2.840.113549.1.1.1), then the BIT STRING wrapping
185// RSAPublicKey ::= SEQUENCE { INTEGER n, INTEGER e }. Returns false when the cert key is not RSA.
186inline bool parse_rsa_pubkey(const std::string& der, std::string& n, std::string& e) {
187 static const unsigned char kOid[] = {0x06, 0x09, 0x2A, 0x86, 0x48, 0x86,
188 0xF7, 0x0D, 0x01, 0x01, 0x01};
189 std::size_t pos = static_cast<std::size_t>(-1);
190 for (std::size_t i = 0; i + sizeof kOid <= der.size(); ++i)
191 if (std::memcmp(der.data() + i, kOid, sizeof kOid) == 0) {
192 pos = i;
193 break;
194 }
195 if (pos == static_cast<std::size_t>(-1)) return false;
196 std::size_t i = pos + sizeof kOid;
197 if (i + 1 < der.size() && static_cast<unsigned char>(der[i]) == 0x05 &&
198 static_cast<unsigned char>(der[i + 1]) == 0x00)
199 i += 2; // skip the NULL algorithm parameters
200 if (i >= der.size() || static_cast<unsigned char>(der[i]) != 0x03) return false; // BIT STRING
201 ++i;
202 if (der_len(der, i) == static_cast<std::size_t>(-1)) return false;
203 if (i >= der.size() || static_cast<unsigned char>(der[i]) != 0x00) return false; // 0 unused bits
204 ++i;
205 if (i >= der.size() || static_cast<unsigned char>(der[i]) != 0x30) return false; // SEQUENCE
206 ++i;
207 if (der_len(der, i) == static_cast<std::size_t>(-1)) return false;
208 if (i >= der.size() || static_cast<unsigned char>(der[i]) != 0x02) return false; // INTEGER n
209 ++i;
210 const std::size_t nl = der_len(der, i);
211 if (nl == static_cast<std::size_t>(-1) || i + nl > der.size()) return false;
212 n = der.substr(i, nl);
213 i += nl;
214 if (i >= der.size() || static_cast<unsigned char>(der[i]) != 0x02) return false; // INTEGER e
215 ++i;
216 const std::size_t el = der_len(der, i);
217 if (el == static_cast<std::size_t>(-1) || i + el > der.size()) return false;
218 e = der.substr(i, el);
219 auto strip = [](std::string& s) {
220 std::size_t k = 0;
221 while (k + 1 < s.size() && static_cast<unsigned char>(s[k]) == 0) ++k;
222 s = s.substr(k);
223 };
224 strip(n);
225 strip(e);
226 return !n.empty() && !e.empty();
227}

229// MGF1 with SHA-256 (RFC 8017): the mask-generation function for PSS.
230inline std::string mgf1_sha256(const std::string& seed, std::size_t mask_len) {
231 std::string T;
232 std::uint32_t counter = 0;
233 while (T.size() < mask_len) {
234 const unsigned char c[4] = {static_cast<unsigned char>(counter >> 24),
235 static_cast<unsigned char>(counter >> 16),
236 static_cast<unsigned char>(counter >> 8),
237 static_cast<unsigned char>(counter)};
238 T += hashlib::sha256_digest(seed + std::string(reinterpret_cast<const char*>(c), 4));
239 ++counter;
240 }
241 T.resize(mask_len);
242 return T;
243}

245// EMSA-PSS-VERIFY (RFC 8017 §9.1.2) with SHA-256 and salt length 32. `em` is the encoded message of
246// `em_bits` bits (length ceil(em_bits/8)); `m_hash` is SHA-256 of the signed content.
247inline bool pss_verify(const std::string& m_hash, const std::string& em, std::size_t em_bits) {
248 const std::size_t hlen = 32, slen = 32;
249 const std::size_t em_len = (em_bits + 7) / 8;
250 if (em.size() != em_len) return false;
251 if (em_len < hlen + slen + 2) return false;
252 if (static_cast<unsigned char>(em[em_len - 1]) != 0xbc) return false;
253 const std::size_t db_len = em_len - hlen - 1;
254 std::string masked_db = em.substr(0, db_len);
255 const std::string H = em.substr(db_len, hlen);
256 const std::size_t top_bits = 8 * em_len - em_bits;
257 if (top_bits > 0) {
258 const unsigned char mask = static_cast<unsigned char>(0xFF << (8 - top_bits));
259 if (static_cast<unsigned char>(masked_db[0]) & mask) return false;
260 }
261 const std::string db_mask = mgf1_sha256(H, db_len);
262 std::string db(db_len, '\0');
263 for (std::size_t i = 0; i < db_len; ++i)
264 db[i] = static_cast<char>(static_cast<unsigned char>(masked_db[i]) ^
265 static_cast<unsigned char>(db_mask[i]));
266 if (top_bits > 0)
267 db[0] = static_cast<char>(static_cast<unsigned char>(db[0]) & (0xFF >> top_bits));
268 const std::size_t ps_len = db_len - slen - 1;
269 for (std::size_t i = 0; i < ps_len; ++i)
270 if (static_cast<unsigned char>(db[i]) != 0) return false;
271 if (static_cast<unsigned char>(db[ps_len]) != 0x01) return false;
272 const std::string salt = db.substr(ps_len + 1); // slen bytes
273 const std::string mprime = std::string(8, '\0') + m_hash + salt;
274 return hashlib::sha256_digest(mprime) == H;
275}

277// Verify a TLS 1.3 CertificateVerify RSA-PSS (rsa_pss_rsae_sha256) signature: recover EM = sig^e mod n
278// with the certificate's RSA public key, then EMSA-PSS-VERIFY against SHA-256(`message`).
279inline bool verify_pss_sha256(const std::string& cert_der, const std::string& message,
280 const std::string& sig) {
281 std::string nb, eb;
282 if (!parse_rsa_pubkey(cert_der, nb, eb)) return false;
283 const Big n = from_be(nb), e = from_be(eb);
284 if (n.empty() || e.empty()) return false;
285 if (sig.size() != nb.size()) return false; // signature length must equal the modulus length
286 const Big s = from_be(sig);
287 if (cmp(s, n) >= 0) return false; // signature representative out of range
288 const Big m = modexp(s, e, n);
289 const std::size_t em_bits = bitlen(n) - 1; // RFC 8017: emBits = modBits - 1
290 const std::size_t em_len = (em_bits + 7) / 8;
291 if (bytelen(m) > em_len) return false;
292 const std::string em = to_be(m, em_len);
293 return pss_verify(hashlib::sha256_digest(message), em, em_bits);
294}
