1// Copyright (c) 2026 BigBrain LLC. MIT-licensed (see LICENSE).
2// Original work; see ACKNOWLEDGMENTS.md for the open-source ideas we build upon.
3// Compile-run tests for OPT-IN sized integer storage types (the memory-footprint feature).
4//
5// A declaration may opt into an explicit width — `i8`/`u32`/… (abbreviated) or `int8`/`uint32`/…
6// (full spelling), both the SAME <cstdint> exact-width type by construction — anywhere a type
7// annotation appears: scalars, `list`/`dict`/`array` elements, `ndarray` elements, and struct
8// fields. The stored form shrinks (proven with the in-language `sizeof` builtin); arithmetic still
9// promotes, and `int` itself is UNCHANGED (still 64-bit) so standalone integers never regress.
10//
11// Each test asserts BOTH the emitted C++ (the declared std::intN_t really reaches storage) and the
12// program's stdout (numeric — never a char, the i8/u8 gotcha these types must not fall into).
13#include "e2e_harness.hpp"
14
15using e2e::expect_e2e;
16using e2e::expect_e2e_source;
17using e2e::expect_compile_fail;
18
19// A scalar width, BOTH spellings — the abbreviated and full names deduce to one canonical type.
20TEST(SizedIntCompileRun, ScalarBothSpellings) {
21 std::string src = expect_e2e_source("sized_scalar", R"PURR(import io
22let a: i8 = 5
23let b: int8 = -7
24let c: u32 = 4000000000
25let d: uint32 = 42
26io.print(a, b, c, d)
27)PURR", "5 -7 4000000000 42\n");
28 EXPECT_NE(src.find("std::int8_t a = 5LL"), std::string::npos) << src;
29 EXPECT_NE(src.find("std::int8_t b = (-7LL)"), std::string::npos) << src; // int8 == i8
30 EXPECT_NE(src.find("std::uint32_t c = 4000000000LL"), std::string::npos) << src;
31 EXPECT_NE(src.find("std::uint32_t d = 42LL"), std::string::npos) << src; // uint32 == u32
32}
33
34// The ORIGINAL <cstdint> spelling (`int16_t`/`uint8_t`) names the SAME type as `i16`/`u8` — a C
35// programmer writes the name they already know, with no difference in the emitted storage.
36TEST(SizedIntCompileRun, CstdintSpelling) {
37 std::string src = expect_e2e_source("sized_cstdint", R"PURR(import io
38let a: int16_t = 300
39let b: uint8_t = 200
40let xs: list<int32_t> = [1, 2, 3]
41io.print(a, b)
42io.print(xs)
43io.print(sizeof(a), sizeof(b))
44)PURR", "300 200\n[1, 2, 3]\n2 1\n");
45 EXPECT_NE(src.find("std::int16_t a = 300LL"), std::string::npos) << src; // int16_t == i16
46 EXPECT_NE(src.find("std::uint8_t b = 200LL"), std::string::npos) << src; // uint8_t == u8
47 EXPECT_NE(src.find("std::vector<std::int32_t> xs"), std::string::npos) << src;
48}

50// A narrow list literal must be BUILT AS the declared vector, not CTAD'd to vector<long long>
51// (which would not assign). Elements print numerically even for the u8 char-sized case.
52TEST(SizedIntCompileRun, ListNarrow) {
53 std::string src = expect_e2e_source("sized_list", R"PURR(import io
54let xs: list<i32> = [1, 2, 3]
55let bs: list<u8> = [250, 4, 9]
56io.print(xs)
57io.print(bs)
58)PURR", "[1, 2, 3]\n[250, 4, 9]\n");
59 EXPECT_NE(src.find("std::vector<std::int32_t> xs = std::vector<std::int32_t>{1LL, 2LL, 3LL}"),
60 std::string::npos) << src;
61 EXPECT_NE(src.find("std::vector<std::uint8_t> bs = std::vector<std::uint8_t>{250LL, 4LL, 9LL}"),
62 std::string::npos) << src;
63}

83// A narrow-element ndarray is now CONSTRUCTIBLE: a declared `ndarray<i16>` drives the element type
84// (the initializer is converted for you), and the param reference / return type carry the width.
85TEST(SizedIntCompileRun, NdarrayNarrowElement) {
86 std::string src = expect_e2e_source("sized_ndarray", R"PURR(import io
87import ndarray
88fn first(a: ndarray<i16>) -> i16 {
89 return 0
90}
91fn main() {
92 let v: ndarray<i16> = ndarray.array([1, 2, 3])
93 io.print(v)
94 io.print(first(v))
95}
96main()
97)PURR", "[1, 2, 3]\n0\n");
98 EXPECT_NE(src.find("basic_ndarray<std::int16_t>& a"), std::string::npos) << src; // param ref
99 EXPECT_NE(src.find("std::int16_t first("), std::string::npos) << src; // return width
100 EXPECT_NE(src.find("astype<std::int16_t>"), std::string::npos) << src; // driven construction
101}

114// A struct of narrow fields PACKS (the whole memory point) and prints its i8/u8 fields as numbers.
115TEST(SizedIntCompileRun, StructPacksAndPrintsNumerically) {
116 std::string src = expect_e2e_source("sized_struct", R"PURR(import io
117struct Cell {
118 x: u8
119 y: u8
120}
121fn main() {
122 let c: Cell = Cell(3, 7)
123 io.print(c)
124 io.print(sizeof(Cell))
125}
126main()
127)PURR", "Cell(\n x = 3,\n y = 7\n)\n2\n"); // two u8 pack to 2 bytes (vs 16 as long long)
128 EXPECT_NE(src.find("std::uint8_t x;"), std::string::npos) << src;
129 EXPECT_NE(src.find("+this->x"), std::string::npos) << src; // promoted so it prints as a number
130}

132// The footprint win, proven in-language: element and struct sizes are the narrow widths, and
133// `int` itself is untouched (still 8 bytes) so standalone integers never grow or shrink.
134TEST(SizedIntCompileRun, SizeofProvesFootprint) {
135 expect_e2e("sized_footprint", R"PURR(import io
136let a: i8 = 1
137io.print(sizeof(a))
138io.print(sizeof(i8), sizeof(i16), sizeof(i32), sizeof(i64))
139io.print(sizeof(u8), sizeof(u32))
140io.print(sizeof(int))
141)PURR", "1\n1 2 4 8\n1 4\n8\n");
142}

144// Arithmetic on narrow operands PROMOTES to 64-bit for free — 100 + 100 as i8 is 200, not the
145// wrapped -56 an i8 add would give. Speed and range of the compute are unchanged; only storage
146// is narrow. (This is the guarantee that keeps sized ints from silently corrupting results.)
147TEST(SizedIntCompileRun, ArithmeticPromotesToWide) {
148 expect_e2e("sized_promote", R"PURR(import io
149let a: i8 = 100
150let b: i8 = 100
151io.print(a + b)
152)PURR", "200\n");
153}

155// Storing a wide value into a narrow scalar TRUNCATES at the width (like C / a NumPy fixed dtype):
156// 300 into a u8 keeps the low byte, 44. Opt-in, documented, and never a hidden runtime check.
157TEST(SizedIntCompileRun, ScalarStoreTruncates) {
158 expect_e2e("sized_truncate", R"PURR(import io
159let w: u8 = 300
160io.print(w)
161let big: u64 = 5000000000
162io.print(big)
163)PURR", "44\n5000000000\n");
164}

166// A narrow RETURN type binds (`-> i16` => std::int16_t); the body's `n + n` promotes to compute
167// then narrows to the return. dbl(50) -> 100. (A narrow SCALAR param stays a generic forwarding
168// reference — as every cheatah scalar param does — so the width there documents intent and drives
169// the return, while storage widths bind on lets / fields / containers.)
170TEST(SizedIntCompileRun, NarrowReturnBinds) {
171 std::string src = expect_e2e_source("sized_param", R"PURR(import io
172fn dbl(n: i16) -> i16 {
173 return n + n
174}
175io.print(dbl(50))
176)PURR", "100\n");
177 EXPECT_NE(src.find("std::int16_t dbl("), std::string::npos) << src; // narrow return type
178}
