1// Copyright (c) 2026 BigBrain LLC. MIT-licensed (see LICENSE).
2// Original work; see ACKNOWLEDGMENTS.md for the open-source ideas we build upon.
3// System-level test: least-squares regression written in cheatah, using the
4// native ndarray + linalg standard library. For each dimensionality n we build a
5// well-conditioned m×n design matrix X and a known coefficient vector β, form the
6// exact targets y = X·β in cheatah (linalg.matmul), recover β̂ = lstsq(X, y), and
7// confirm β̂ == β. The whole regression runs as a compiled .purr program; this test
8// just compiles it with purrc, runs it under the cheatah runtime, and checks output.
9

34std::string run_capture(const std::string& cmd, int& exit_code) {
35 std::string out;
36 FILE* pipe = popen(cmd.c_str(), "r");
37 if (pipe == nullptr) {
38 exit_code = -1;
39 return out;
40 }
41 std::array<char, 256> buf{};
42 while (std::fgets(buf.data(), static_cast<int>(buf.size()), pipe) != nullptr) {
43 out += buf.data();
44 }
45 const int status = pclose(pipe);
46 exit_code = WIFEXITED(status) ? WEXITSTATUS(status) : -1;
47 return out;
48}
49
50// A deterministic, well-conditioned design-matrix entry: small values in [-3, 3]
51// with a strong diagonal on the top n×n block (so X has full column rank and the
52// least-squares problem is well-conditioned). m = n + 5 over-determines the fit.
53double design(int i, int j, int n) {
54 double v = static_cast<double>(((i * 7 + j * 3 + 1) % 7) - 3);
55 if (i < n && i == j) v += static_cast<double>(10 * n);
56 return v;
57}
58
59void expect_lsq_recovers(int n) {
60 const int m = n + 5;
61
62 std::ostringstream xs, bs;
63 xs << std::fixed << std::setprecision(1);
64 bs << std::fixed << std::setprecision(1);
65 for (int i = 0; i < m; ++i)
66 for (int j = 0; j < n; ++j) xs << ((i || j) ? ", " : "") << design(i, j, n);
67 for (int i = 0; i < n; ++i) bs << (i ? ", " : "") << (static_cast<double>(i + 1) * 0.5);
68
69 const std::string tmp = PURR_TEST_TMP;
70 const std::string purr = tmp + "/lsq" + std::to_string(n) + ".purr";
71 const std::string mod = tmp + "/lsq" + std::to_string(n) + ".so";
72 {
73 std::ofstream f(purr);
74 f << "import io\nimport ndarray\nimport linalg\n";
75 f << "let Xflat = [" << xs.str() << "]\n";
76 f << "let beta = [" << bs.str() << "]\n";
77 f << "let X = ndarray.reshape(ndarray.array(Xflat), [" << m << ", " << n << "])\n";
78 f << "let b = ndarray.reshape(ndarray.array(beta), [" << n << ", 1])\n";
79 f << "let y = linalg.matmul(X, b)\n"; // exact targets, m×1
80 f << "let est = linalg.lstsq(X, y)\n"; // recovered coefficients, n×1
81 f << "let maxerr = 0.0\n";
82 f << "for i in range(" << n << ") {\n";
83 f << " let d = ndarray.get(est, [i, 0]) - beta[i]\n";
84 f << " if d < 0.0 { d = 0.0 - d }\n";
85 f << " if d > maxerr { maxerr = d }\n";
86 f << "}\n";
87 f << "io.print(\"dim\", " << n << ", \"recovered\", maxerr < 0.000001)\n";
88 }
89
90 const std::string compile =
91 std::string(PURRC_PATH) + " \"" + purr + "\" -o \"" + mod + "\"";
92 ASSERT_EQ(std::system(compile.c_str()), 0) << "purrc failed to compile (n=" << n << ")";
93
94 int rc = -1;
95 const std::string out = run_capture(std::string(CHEATAH_RUNTIME_PATH) + " \"" + mod + "\"", rc);
96 EXPECT_EQ(rc, 0);
97 EXPECT_EQ(out, "dim " + std::to_string(n) + " recovered True\n");
98}
99
100} // namespace

102TEST(LinalgLsqE2E, Recovers2D) { expect_lsq_recovers(2); }
103TEST(LinalgLsqE2E, Recovers3D) { expect_lsq_recovers(3); }
