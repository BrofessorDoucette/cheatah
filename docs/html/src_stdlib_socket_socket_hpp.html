5/**
6 * @file socket.hpp
7 * @brief cheatah `socket` — a small wrapper around BSD/POSIX TCP sockets,
8 * in the spirit of Python's `socket`. `import socket` to use it.
9 *
10 * The recommended API is the owning `socket::Conn` / `socket::Listener` guards (from
11 * `socket.open(host, port)` / `socket.serve(host, port, backlog)`), which close their fd at
12 * scope exit. A **flat, file-descriptor-based** API is also available for hand-built servers:
13 * pass the integer fd from `socket()` / `tcp_listen()` / `accept()` to the other calls (an
14 * unclosed fd there is a resource leak — prefer the guards). IPv4 + TCP only; host names are
15 * resolved with `getaddrinfo` (so `"localhost"`, `"127.0.0.1"`, and DNS names all work).
16 * Errors are a negative return (or empty string for `recv`); `last_error()` gives the `errno` text.
17 *
18 * `import socket` includes this header AND links `libcheatah_socket`. Unit tests:
19 * `stdlib/tests/socket_test.cpp`; the suite runs under AddressSanitizer (the
20 * `asan` preset) and Valgrind (`security/run-valgrind.sh`) on every QA-gate run.
21 *
22 * @note Every call is a thin wrapper over one or two syscalls. Only `recv` and
23 * `last_error` allocate (their returned `std::string`); the fd/status calls
24 * return a `long long` and do not allocate.
25 */
26#include <string>
27
28namespace cheatah::socket {

30// ---- high-level convenience (what a server/client usually wants) ----

32/**
33 * Create a TCP socket bound to @p host:@p port and put it in the listening state (sets
34 * `SO_REUSEADDR`).
35 *
36 * Performs socket()/set_reuseaddr/bind/listen in one shot; on any failure it closes the
37 * partially-created socket and returns -1, so the caller never leaks an fd. On success the
38 * returned fd is owned by the caller and must be passed to close() when done.
39 * @param host interface to bind ("127.0.0.1", "0.0.0.0", …).
40 * @param port TCP port (0 = let the OS pick — read it back with local_port()).
41 * @param backlog pending-connection queue length.
42 * @return the listening fd, or -1 on error.
43 * @complexity O(1) (a few syscalls).
44 * @alloc none.
45 * @test CheatahSocket.Loopback
46 * @crtest SocketCompileRun.TcpListen
47 * @systest StdlibE2E.Socket
48 */
49long long tcp_listen(const std::string& host, long long port, long long backlog);
50
51/**
52 * Create a TCP socket and connect it to @p host:@p port.
53 *
54 * The host is resolved via `getaddrinfo`, so names, "localhost", and dotted IPs all work; the
55 * connect blocks until the handshake completes or fails. On failure the socket is closed and
56 * -1 is returned; on success the caller owns the connected fd and must close() it.
57 * @param host destination host (name or IP).
58 * @param port destination port.
59 * @return the connected fd, or -1 on error.
60 * @complexity O(1) + DNS resolution.
61 * @alloc none.
62 * @test CheatahSocket.Loopback
63 * @crtest SocketCompileRun.TcpConnect
64 * @systest StdlibE2E.Socket
65 */
66long long tcp_connect(const std::string& host, long long port);

68// ---- per-connection I/O ----

70/**
71 * Accept one pending connection.
72 *
73 * Blocks until a client connects, then returns a new fd for that one connection (the listening
74 * fd stays open for further accepts). The returned client fd is owned by the caller and must be
75 * closed separately; the peer address is discarded.
76 * @param fd a listening fd.
77 * @return the connected client fd, or -1 on error.
78 * @complexity O(1) syscall (blocks until a client arrives).
79 * @alloc none.
80 * @test CheatahSocket.Loopback
81 * @crtest SocketCompileRun.Accept
82 * @systest StdlibE2E.Socket
83 */
84long long accept(long long fd);
85
86/**
87 * Receive up to @p bufsize bytes.
88 *
89 * Blocks for one `recv` and returns whatever bytes arrive (possibly fewer than @p bufsize); the
90 * result is binary-safe, so a returned string may contain embedded NULs and its length is the
91 * true byte count. A clean EOF (peer closed) and an error both yield "", so check last_error()
92 * to tell them apart; @p bufsize <= 0 also returns "" without touching the socket.
93 * @param fd a connected fd.
94 * @param bufsize maximum bytes to read.
95 * @return the bytes read (binary-safe), or "" on EOF/error.
96 * @complexity O(@p bufsize).
97 * @alloc allocates the returned string.
98 * @test CheatahSocket.Loopback
99 * @crtest SocketCompileRun.Recv
100 * @systest StdlibE2E.Socket
101 */
102std::string recv(long long fd, long long bufsize);
103
104/**
105 * Send some of @p data (one `send`).
106 *
107 * Issues a single `send`, which may transmit fewer bytes than supplied (a partial send); the
108 * caller is responsible for re-sending the remainder, or use sendall() to loop automatically.
109 * @param fd a connected fd.
110 * @param data bytes to send.
111 * @return bytes actually sent, or -1 on error.
112 * @complexity O(n).
113 * @alloc none (`MSG_NOSIGNAL`, so a broken pipe never raises `SIGPIPE`).
114 * @test CheatahSocket.Sendall
115 * @crtest SocketCompileRun.Send
116 * @systest StdlibE2E.Socket
117 */
118long long send(long long fd, const std::string& data);
119
120/**
121 * Send @p data in full, looping until all bytes are written.
122 *
123 * Repeatedly calls `send` on the unsent remainder until every byte is written, so unlike send()
124 * there are no partial sends to handle; it aborts with -1 the moment a `send` returns <= 0
125 * (error or peer hang-up), in which case some bytes may already have been transmitted.
126 * @param fd a connected fd.
127 * @param data bytes to send.
128 * @return 0 on success, -1 on error.
129 * @complexity O(n).
130 * @alloc none.
131 * @test CheatahSocket.Sendall
132 * @crtest SocketCompileRun.Sendall
133 * @systest StdlibE2E.Socket
134 */
135long long sendall(long long fd, const std::string& data);
136
137/**
138 * Close a socket.
139 *
140 * Releases the fd back to the OS; after this the fd is invalid and must not be reused. Closing an
141 * already-closed or never-opened fd fails with -1 (EBADF), which is how the BadFd test exercises
142 * the error path.
143 * @param fd the fd to close.
144 * @return 0 on success, -1 on error.
145 * @complexity O(1) syscall.
146 * @alloc none.
147 * @test CheatahSocket.BadFd
148 * @crtest SocketCompileRun.Close
149 * @systest StdlibE2E.Socket
150 */
151long long close(long long fd);
152
153/**
154 * Half-close @p fd in both directions (::shutdown SHUT_RDWR) WITHOUT releasing it.
155 * A blocking recv() on another thread returns immediately (EOF) — the safe way to
156 * wake a reader for a clean shutdown. The fd is still owned by the caller and must
157 * be close()d afterwards.
158 * @param fd the fd to half-close.
159 * @return 0 on success, -1 on error.
160 * @complexity O(1) syscall.
161 * @alloc none.
162 */
163long long shutdown(long long fd);

181/**
182 * Enable `SO_REUSEADDR` on @p fd.
183 *
184 * Lets a subsequent bind() reuse a local address still lingering in TIME_WAIT, so a restarted
185 * server can re-listen on the same port immediately; call it before bind().
186 * @param fd the socket.
187 * @return 0 on success, -1 on error.
188 * @complexity O(1).
189 * @alloc none.

191 * @crtest SocketCompileRun.SetReuseaddr

194long long set_reuseaddr(long long fd);

196/**
197 * Bind @p fd to @p host:@p port.
198 *
199 * Resolves @p host via `getaddrinfo` and assigns the resulting local address to the socket; a
200 * resolution failure returns -1 with errno set to EADDRNOTAVAIL (see the ResolveFailure test).
201 * @param fd the socket.
202 * @param host interface to bind.
203 * @param port TCP port (0 = OS-assigned).
204 * @return 0 on success, -1 on error.
205 * @complexity O(1) + resolution.
206 * @alloc none.
207 * @test CheatahSocket.ListenLowLevel, CheatahSocket.ResolveFailure
208 * @crtest SocketCompileRun.Bind
209 * @systest StdlibE2E.Socket
210 */
211long long bind(long long fd, const std::string& host, long long port);

213/**
214 * Mark @p fd as a passive (listening) socket.
215 *
216 * Switches an already-bound socket into the listening state so accept() can pull connections from
217 * it; @p backlog caps how many fully-established connections may queue before new ones are refused.
218 * @param fd the socket.
219 * @param backlog queue length.
220 * @return 0 on success, -1 on error.
221 * @complexity O(1).
222 * @alloc none.
223 * @test CheatahSocket.ListenLowLevel
224 * @crtest SocketCompileRun.Listen
225 * @systest StdlibE2E.Socket
226 */
227long long listen(long long fd, long long backlog);
228
229/**
230 * Connect @p fd to @p host:@p port.
231 *
232 * Resolves @p host and blocks until the TCP handshake succeeds or fails; a refused connection
233 * returns -1 with errno ECONNREFUSED (see the ConnectRefused test). Unlike tcp_connect() it does
234 * not close the fd on failure — the caller still owns @p fd.
235 * @param fd the socket.
236 * @param host destination.
237 * @param port destination port.
238 * @return 0 on success, -1 on error.
239 * @complexity O(1) + DNS.
240 * @alloc none.
241 * @test CheatahSocket.ConnectRefused
242 * @crtest SocketCompileRun.Connect
243 * @systest StdlibE2E.Socket
244 */
245long long connect(long long fd, const std::string& host, long long port);
246
247/**
248 * The local TCP port @p fd is bound to (useful after binding to port 0).
249 *
250 * Reads the address actually assigned via `getsockname` and returns its port in host byte order;
251 * this is the way to discover the ephemeral port the OS chose when you bound to port 0.
252 * @param fd a bound socket.
253 * @return the port, or -1 on error.
254 * @complexity O(1) syscall.
255 * @alloc none.
256 * @test CheatahSocket.Loopback
257 * @crtest SocketCompileRun.LocalPort
258 * @systest StdlibE2E.Socket
259 */
260long long local_port(long long fd);
261
262/**
263 * Bound both blocking directions of @p fd by @p timeout_ms (SO_RCVTIMEO + SO_SNDTIMEO), so a
264 * silent peer cannot hang a recv/send forever. A recv that times out returns "" (check
265 * last_error() to distinguish from EOF). @p timeout_ms <= 0 clears the timeouts (block forever).
266 *
267 * @param fd a socket.
268 * @param timeout_ms the per-operation bound in milliseconds.
269 * @return 0 on success, -1 on error.
270 * @complexity O(1) (two setsockopt calls).
271 * @alloc none.
272 * @test CheatahSocket.SetTimeout
273 * @crtest SocketCompileRun.SetTimeout
274 * @systest StdlibE2E.Socket
275 */
276long long set_timeout(long long fd, long long timeout_ms);

278/**
279 * The message for the current `errno`.
280 *
281 * Returns the human-readable text for the thread's current `errno`; call it right after a function
282 * reports failure (a -1 return, or "" from recv), since any later syscall may overwrite `errno`.
283 * @return `strerror(errno)`.
284 * @complexity O(1).
285 * @alloc allocates the returned string.
286 * @test CheatahSocket.ConnectRefused
287 * @crtest SocketCompileRun.LastError
288 * @systest StdlibE2E.Socket
289 */
290std::string last_error();

292// ---- owning RAII connections (the `with`-friendly, leak-proof API) ----
293
294/**
295 * @brief An owning TCP connection — a socket fd whose destructor closes it.
296 *
297 * The RAII counterpart to the fd-based calls above, and the C++/cheatah analog of a
298 * Python socket used in a `with` block. A `Conn` owns exactly one fd; when it is
299 * destroyed (scope exit, including a `return`/`break`/exception out of a `with` body)
300 * or explicitly close()d, the fd is released — so a connection opened with
301 * `with socket.open(host, port) as c { … }` cannot leak. Move-only: copying would give
302 * two owners of one fd and double-close it, so the copy operations are deleted and a
303 * moved-from `Conn` is left closed.
304 */
305class Conn {
306public:
307 /**
308 * Construct a closed connection (owns no fd).
309 * @complexity O(1).
310 * @alloc none.
311 * @test CheatahSocket.ConnDefaultIsClosed
312 */
313 Conn() = default;
314 /**
315 * Adopt an already-connected fd (e.g. from tcp_connect()/accept()); the `Conn` now owns it.
316 * @param fd a connected fd to take ownership of (-1 for a closed connection).
317 * @complexity O(1).
318 * @alloc none.
319 * @test CheatahSocket.ConnGuardClosesOnScopeExit
320 */
321 explicit Conn(long long fd) : fd_(fd) {}
322 Conn(const Conn&) = delete;
323 Conn& operator=(const Conn&) = delete;
324 /**
325 * Move-construct, taking over @p other's fd (the moved-from `Conn` becomes closed).
326 * @param other the connection to move from.
327 * @complexity O(1).
328 * @alloc none.
329 * @test CheatahSocket.ConnMoveTransfersOwnership
330 */
331 Conn(Conn&& other) noexcept : fd_(other.fd_) { other.fd_ = -1; }
332 /**
333 * Move-assign, closing this fd first, then taking over @p other's (which becomes closed).
334 * @param other the connection to move from.
335 * @return reference to this connection.
336 * @complexity O(1).
337 * @alloc none.
338 * @test CheatahSocket.ConnMoveTransfersOwnership
339 */
340 Conn& operator=(Conn&& other) noexcept;
341 /**
342 * Close the fd if still open.
343 * @complexity O(1) syscall.
344 * @alloc none.
345 * @test CheatahSocket.ConnGuardClosesOnScopeExit
346 */
347 ~Conn();
348
349 /**
350 * Is a connection open?
351 * @return true iff this owns an open fd.
352 * @complexity O(1).
353 * @alloc none.
354 * @test CheatahSocket.ConnDefaultIsClosed
355 */
356 bool is_open() const { return fd_ >= 0; }
357 /**
358 * The raw fd, for the low-level calls or to hand to tls.open(conn.fd(), …).
359 * @return the owned fd, or -1 when closed.
360 * @complexity O(1).
361 * @alloc none.
362 * @test CheatahSocket.ConnGuardClosesOnScopeExit
363 */
364 long long fd() const { return fd_; }
365 /**
366 * Send some of @p data (one send(); see the free send()).
367 * @param data bytes to send.
368 * @return bytes actually sent, or -1 on error.
369 * @complexity O(n).
370 * @alloc none.
371 * @test CheatahSocket.ConnLoopback
372 */
373 long long send(const std::string& data);
374 /**
375 * Send @p data in full, looping until every byte is written (see the free sendall()).
376 * @param data bytes to send.
377 * @return 0 on success, -1 on error.
378 * @complexity O(n).
379 * @alloc none.
380 * @test CheatahSocket.ConnLoopback
381 */
382 long long sendall(const std::string& data);
383 /**
384 * Receive up to @p bufsize bytes (see the free recv()).
385 * @param bufsize maximum bytes to read.
386 * @return the bytes read (binary-safe), or "" on EOF/error.
387 * @complexity O(@p bufsize).
388 * @alloc allocates the returned string.
389 * @test CheatahSocket.ConnLoopback
390 */
391 std::string recv(long long bufsize);
392 /**
393 * Bound both blocking directions by @p timeout_ms (see the free set_timeout()).
394 * @param timeout_ms per-operation bound in milliseconds (<= 0 clears it).
395 * @return 0 on success, -1 on error.
396 * @complexity O(1).
397 * @alloc none.
398 * @test CheatahSocket.ConnLoopback
399 */
400 long long set_timeout(long long timeout_ms);
401 /**
402 * The local TCP port this fd is bound to (see the free local_port()).
403 * @return the port, or -1 on error.
404 * @complexity O(1) syscall.
405 * @alloc none.
406 * @test CheatahSocket.ConnLoopback
407 */
408 long long local_port() const;
409 /**
410 * Half-close both directions WITHOUT releasing the fd (see the free shutdown()) — wakes a
411 * blocked reader for a clean shutdown; still call close() (or let the destructor) afterward.
412 * @return 0 on success, -1 on error.
413 * @complexity O(1) syscall.
414 * @alloc none.
415 * @test CheatahSocket.ConnLoopback
416 */
417 long long shutdown();
418 /**
419 * Close the fd now (idempotent — the destructor will not close it again).
420 * @return 0 on success, -1 if already closed.
421 * @complexity O(1) syscall.
422 * @alloc none.
423 * @test CheatahSocket.ConnGuardClosesOnScopeExit
424 */
425 long long close();
426
427private:
428 long long fd_ = -1;
429};

431/**
432 * @brief An owning listening socket; accept() yields owning Conn clients; the destructor closes it.
433 *
434 * The server-side RAII guard: `with socket.serve(host, port, backlog) as server { … }` keeps the
435 * listening fd for the block and closes it on exit. Each accept() returns an owning Conn, so a
436 * whole server loop leaks neither the listener nor its clients. Move-only, like Conn.
437 */
438class Listener {
439public:
440 /**
441 * Construct a closed listener (owns no fd).
442 * @complexity O(1).
443 * @alloc none.
444 * @test CheatahSocket.ListenerDefaultIsClosed
445 */
446 Listener() = default;
447 /**
448 * Adopt an already-listening fd (e.g. from tcp_listen()); the `Listener` now owns it.
449 * @param fd a listening fd to take ownership of (-1 for a closed listener).
450 * @complexity O(1).
451 * @alloc none.
452 * @test CheatahSocket.ListenerLoopback
453 */
454 explicit Listener(long long fd) : fd_(fd) {}
455 Listener(const Listener&) = delete;
456 Listener& operator=(const Listener&) = delete;
457 /**
458 * Move-construct, taking over @p other's fd (the moved-from `Listener` becomes closed).
459 * @param other the listener to move from.
460 * @complexity O(1).
461 * @alloc none.
462 * @test CheatahSocket.ListenerLoopback
463 */
464 Listener(Listener&& other) noexcept : fd_(other.fd_) { other.fd_ = -1; }
465 /**
466 * Move-assign, closing this fd first, then taking over @p other's (which becomes closed).
467 * @param other the listener to move from.
468 * @return reference to this listener.
469 * @complexity O(1).
470 * @alloc none.
471 * @test CheatahSocket.ListenerLoopback
472 */
473 Listener& operator=(Listener&& other) noexcept;
474 /**
475 * Close the listening fd if still open.
476 * @complexity O(1) syscall.
477 * @alloc none.
478 * @test CheatahSocket.ListenerLoopback
479 */
480 ~Listener();
481
482 /**
483 * Is the listener open?
484 * @return true iff this owns an open listening fd.
485 * @complexity O(1).
486 * @alloc none.
487 * @test CheatahSocket.ListenerDefaultIsClosed
488 */
489 bool is_open() const { return fd_ >= 0; }
490 /**
491 * The raw listening fd.
492 * @return the owned fd, or -1 when closed.
493 * @complexity O(1).
494 * @alloc none.
495 * @test CheatahSocket.ListenerLoopback
496 */
497 long long fd() const { return fd_; }
498 /**
499 * Accept one pending connection, returned as an owning Conn (the listener stays open).
500 * @return an owning Conn for the client (its is_open() is false on error).
501 * @complexity O(1) syscall (blocks until a client arrives).
502 * @alloc none.
503 * @test CheatahSocket.ListenerLoopback
504 */
505 Conn accept();
506 /**
507 * The local TCP port this listener is bound to (useful after binding to port 0).
508 * @return the port, or -1 on error.
509 * @complexity O(1) syscall.
510 * @alloc none.
511 * @test CheatahSocket.ListenerLoopback
512 */
513 long long local_port() const;
514 /**
515 * Close the listening fd now (idempotent — the destructor will not close it again).
516 * @return 0 on success, -1 if already closed.
517 * @complexity O(1) syscall.
518 * @alloc none.
519 * @test CheatahSocket.ListenerLoopback
520 */
521 long long close();
522
523private:
524 long long fd_ = -1;
525};
