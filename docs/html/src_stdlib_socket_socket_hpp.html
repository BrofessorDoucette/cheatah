3/**
4 * @file socket.hpp
5 * @brief cheatah `socket` — a small wrapper around BSD/POSIX TCP sockets,
6 * in the spirit of Python's `socket`. `import socket` to use it.
7 *
8 * The recommended API is the owning `socket::Conn` / `socket::Listener` guards (from
9 * `socket.open(host, port)` / `socket.serve(host, port, backlog)`), which close their fd at
10 * scope exit. A **flat, file-descriptor-based** API is also available for hand-built servers:
11 * pass the integer fd from `socket()` / `tcp_listen()` / `accept()` to the other calls (an
12 * unclosed fd there is a resource leak — prefer the guards). IPv4 + TCP only; host names are
13 * resolved with `getaddrinfo` (so `"localhost"`, `"127.0.0.1"`, and DNS names all work).
14 * Errors are a negative return (or empty string for `recv`); `last_error()` gives the `errno` text.
15 *
16 * `import socket` includes this header AND links `libcheatah_socket`. Unit tests:
17 * `stdlib/tests/socket_test.cpp`; the suite runs under AddressSanitizer (the
18 * `asan` preset) and Valgrind (`security/run-valgrind.sh`) on every QA-gate run.
19 *
20 * @note Every call is a thin wrapper over one or two syscalls. Only `recv` and
21 * `last_error` allocate (their returned `std::string`); the fd/status calls
22 * return a `long long` and do not allocate.
23 */
24#include <string>

30/**
31 * Create a TCP socket bound to @p host:@p port and put it in the listening state (sets
32 * `SO_REUSEADDR`).
33 *
34 * Performs socket()/set_reuseaddr/bind/listen in one shot; on any failure it closes the
35 * partially-created socket and returns -1, so the caller never leaks an fd. On success the
36 * returned fd is owned by the caller and must be passed to close() when done.
37 * @param host interface to bind ("127.0.0.1", "0.0.0.0", …).
38 * @param port TCP port (0 = let the OS pick — read it back with local_port()).
39 * @param backlog pending-connection queue length.
40 * @return the listening fd, or -1 on error.
41 * @complexity O(1) (a few syscalls).
42 * @alloc none.
43 * @test CheatahSocket.Loopback
44 * @crtest SocketCompileRun.TcpListen
45 * @systest StdlibE2E.Socket
46 */
47long long tcp_listen(const std::string& host, long long port, long long backlog);
48
49/**
50 * Create a TCP socket and connect it to @p host:@p port.
51 *
52 * The host is resolved via `getaddrinfo`, so names, "localhost", and dotted IPs all work; the
53 * connect blocks until the handshake completes or fails. On failure the socket is closed and
54 * -1 is returned; on success the caller owns the connected fd and must close() it.
55 * @param host destination host (name or IP).
56 * @param port destination port.
57 * @return the connected fd, or -1 on error.
58 * @complexity O(1) + DNS resolution.
59 * @alloc none.
60 * @test CheatahSocket.Loopback
61 * @crtest SocketCompileRun.TcpConnect
62 * @systest StdlibE2E.Socket
63 */
64long long tcp_connect(const std::string& host, long long port);

68/**
69 * Accept one pending connection.
70 *
71 * Blocks until a client connects, then returns a new fd for that one connection (the listening
72 * fd stays open for further accepts). The returned client fd is owned by the caller and must be
73 * closed separately; the peer address is discarded.
74 * @param fd a listening fd.
75 * @return the connected client fd, or -1 on error.
76 * @complexity O(1) syscall (blocks until a client arrives).
77 * @alloc none.
78 * @test CheatahSocket.Loopback
79 * @crtest SocketCompileRun.Accept
80 * @systest StdlibE2E.Socket
81 */
82long long accept(long long fd);
83
84/**
85 * Receive up to @p bufsize bytes.
86 *
87 * Blocks for one `recv` and returns whatever bytes arrive (possibly fewer than @p bufsize); the
88 * result is binary-safe, so a returned string may contain embedded NULs and its length is the
89 * true byte count. A clean EOF (peer closed) and an error both yield "", so check last_error()
90 * to tell them apart; @p bufsize <= 0 also returns "" without touching the socket.
91 * @param fd a connected fd.
92 * @param bufsize maximum bytes to read.
93 * @return the bytes read (binary-safe), or "" on EOF/error.
94 * @complexity O(@p bufsize).
95 * @alloc allocates the returned string.
96 * @test CheatahSocket.Loopback
97 * @crtest SocketCompileRun.Recv
98 * @systest StdlibE2E.Socket
99 */
100std::string recv(long long fd, long long bufsize);
101
102/**
103 * Send some of @p data (one `send`).
104 *
105 * Issues a single `send`, which may transmit fewer bytes than supplied (a partial send); the
106 * caller is responsible for re-sending the remainder, or use sendall() to loop automatically.
107 * @param fd a connected fd.
108 * @param data bytes to send.
109 * @return bytes actually sent, or -1 on error.
110 * @complexity O(n).
111 * @alloc none (`MSG_NOSIGNAL`, so a broken pipe never raises `SIGPIPE`).
112 * @test CheatahSocket.Sendall
113 * @crtest SocketCompileRun.Send
114 * @systest StdlibE2E.Socket
115 */
116long long send(long long fd, const std::string& data);
117
118/**
119 * Send @p data in full, looping until all bytes are written.
120 *
121 * Repeatedly calls `send` on the unsent remainder until every byte is written, so unlike send()
122 * there are no partial sends to handle; it aborts with -1 the moment a `send` returns <= 0
123 * (error or peer hang-up), in which case some bytes may already have been transmitted.
124 * @param fd a connected fd.
125 * @param data bytes to send.
126 * @return 0 on success, -1 on error.
127 * @complexity O(n).
128 * @alloc none.
129 * @test CheatahSocket.Sendall
130 * @crtest SocketCompileRun.Sendall
131 * @systest StdlibE2E.Socket
132 */
133long long sendall(long long fd, const std::string& data);
134
135/**
136 * Close a socket.
137 *
138 * Releases the fd back to the OS; after this the fd is invalid and must not be reused. Closing an
139 * already-closed or never-opened fd fails with -1 (EBADF), which is how the BadFd test exercises
140 * the error path.
141 * @param fd the fd to close.
142 * @return 0 on success, -1 on error.
143 * @complexity O(1) syscall.
144 * @alloc none.
145 * @test CheatahSocket.BadFd
146 * @crtest SocketCompileRun.Close
147 * @systest StdlibE2E.Socket
148 */
149long long close(long long fd);
150
151/**
152 * Half-close @p fd in both directions (::shutdown SHUT_RDWR) WITHOUT releasing it.
153 * A blocking recv() on another thread returns immediately (EOF) — the safe way to
154 * wake a reader for a clean shutdown. The fd is still owned by the caller and must
155 * be close()d afterwards.
156 * @param fd the fd to half-close.
157 * @return 0 on success, -1 on error.
158 * @complexity O(1) syscall.
159 * @alloc none.
160 */
161long long shutdown(long long fd);

211/**
212 * Mark @p fd as a passive (listening) socket.
213 *
214 * Switches an already-bound socket into the listening state so accept() can pull connections from
215 * it; @p backlog caps how many fully-established connections may queue before new ones are refused.
216 * @param fd the socket.
217 * @param backlog queue length.
218 * @return 0 on success, -1 on error.
219 * @complexity O(1).
220 * @alloc none.
221 * @test CheatahSocket.ListenLowLevel
222 * @crtest SocketCompileRun.Listen

225long long listen(long long fd, long long backlog);

227/**
228 * Connect @p fd to @p host:@p port.
229 *
230 * Resolves @p host and blocks until the TCP handshake succeeds or fails; a refused connection
231 * returns -1 with errno ECONNREFUSED (see the ConnectRefused test). Unlike tcp_connect() it does
232 * not close the fd on failure — the caller still owns @p fd.
233 * @param fd the socket.
234 * @param host destination.
235 * @param port destination port.
236 * @return 0 on success, -1 on error.
237 * @complexity O(1) + DNS.
238 * @alloc none.
239 * @test CheatahSocket.ConnectRefused
240 * @crtest SocketCompileRun.Connect
241 * @systest StdlibE2E.Socket
242 */
243long long connect(long long fd, const std::string& host, long long port);

245/**
246 * The local TCP port @p fd is bound to (useful after binding to port 0).
247 *
248 * Reads the address actually assigned via `getsockname` and returns its port in host byte order;
249 * this is the way to discover the ephemeral port the OS chose when you bound to port 0.
250 * @param fd a bound socket.
251 * @return the port, or -1 on error.
252 * @complexity O(1) syscall.
253 * @alloc none.
254 * @test CheatahSocket.Loopback
255 * @crtest SocketCompileRun.LocalPort
256 * @systest StdlibE2E.Socket
257 */
258long long local_port(long long fd);
259
260/**
261 * Bound both blocking directions of @p fd by @p timeout_ms (SO_RCVTIMEO + SO_SNDTIMEO), so a
262 * silent peer cannot hang a recv/send forever. A recv that times out returns "" (check
263 * last_error() to distinguish from EOF). @p timeout_ms <= 0 clears the timeouts (block forever).
264 *
265 * @param fd a socket.
266 * @param timeout_ms the per-operation bound in milliseconds.
267 * @return 0 on success, -1 on error.
268 * @complexity O(1) (two setsockopt calls).
269 * @alloc none.
270 * @test CheatahSocket.SetTimeout
271 * @crtest SocketCompileRun.SetTimeout
272 * @systest StdlibE2E.Socket
273 */
274long long set_timeout(long long fd, long long timeout_ms);

276/**
277 * The message for the current `errno`.
278 *
279 * Returns the human-readable text for the thread's current `errno`; call it right after a function
280 * reports failure (a -1 return, or "" from recv), since any later syscall may overwrite `errno`.
281 * @return `strerror(errno)`.
282 * @complexity O(1).
283 * @alloc allocates the returned string.
284 * @test CheatahSocket.ConnectRefused
285 * @crtest SocketCompileRun.LastError
286 * @systest StdlibE2E.Socket
287 */
288std::string last_error();

290// ---- owning RAII connections (the `with`-friendly, leak-proof API) ----
291
292/**
293 * @brief An owning TCP connection — a socket fd whose destructor closes it.
294 *
295 * The RAII counterpart to the fd-based calls above, and the C++/cheatah analog of a
296 * Python socket used in a `with` block. A `Conn` owns exactly one fd; when it is
297 * destroyed (scope exit, including a `return`/`break`/exception out of a `with` body)
298 * or explicitly close()d, the fd is released — so a connection opened with
299 * `with socket.open(host, port) as c { … }` cannot leak. Move-only: copying would give
300 * two owners of one fd and double-close it, so the copy operations are deleted and a
301 * moved-from `Conn` is left closed.
302 */
303class Conn {
304public:
305 /**
306 * Construct a closed connection (owns no fd).
307 * @complexity O(1).
308 * @alloc none.
309 * @test CheatahSocket.ConnDefaultIsClosed
310 */
311 Conn() = default;
312 /**
313 * Adopt an already-connected fd (e.g. from tcp_connect()/accept()); the `Conn` now owns it.
314 * @param fd a connected fd to take ownership of (-1 for a closed connection).
315 * @complexity O(1).
316 * @alloc none.
317 * @test CheatahSocket.ConnGuardClosesOnScopeExit
318 */
319 explicit Conn(long long fd) : fd_(fd) {}
320 Conn(const Conn&) = delete;
321 Conn& operator=(const Conn&) = delete;
322 /**
323 * Move-construct, taking over @p other's fd (the moved-from `Conn` becomes closed).
324 * @param other the connection to move from.
325 * @complexity O(1).
326 * @alloc none.
327 * @test CheatahSocket.ConnMoveTransfersOwnership
328 */
329 Conn(Conn&& other) noexcept : fd_(other.fd_) { other.fd_ = -1; }
330 /**
331 * Move-assign, closing this fd first, then taking over @p other's (which becomes closed).
332 * @param other the connection to move from.
333 * @return reference to this connection.
334 * @complexity O(1).
335 * @alloc none.
336 * @test CheatahSocket.ConnMoveTransfersOwnership
337 */
338 Conn& operator=(Conn&& other) noexcept;
339 /**
340 * Close the fd if still open.
341 * @complexity O(1) syscall.
342 * @alloc none.
343 * @test CheatahSocket.ConnGuardClosesOnScopeExit
344 */
345 ~Conn();
346
347 /**
348 * Is a connection open?
349 * @return true iff this owns an open fd.
350 * @complexity O(1).
351 * @alloc none.
352 * @test CheatahSocket.ConnDefaultIsClosed
353 */
354 bool is_open() const { return fd_ >= 0; }
355 /**
356 * The raw fd, for the low-level calls or to hand to tls.open(conn.fd(), …).
357 * @return the owned fd, or -1 when closed.
358 * @complexity O(1).
359 * @alloc none.
360 * @test CheatahSocket.ConnGuardClosesOnScopeExit
361 */
362 long long fd() const { return fd_; }
363 /**
364 * Send some of @p data (one send(); see the free send()).
365 * @param data bytes to send.
366 * @return bytes actually sent, or -1 on error.
367 * @complexity O(n).
368 * @alloc none.
369 * @test CheatahSocket.ConnLoopback
370 */
371 long long send(const std::string& data);
372 /**
373 * Send @p data in full, looping until every byte is written (see the free sendall()).
374 * @param data bytes to send.
375 * @return 0 on success, -1 on error.
376 * @complexity O(n).
377 * @alloc none.
378 * @test CheatahSocket.ConnLoopback
379 */
380 long long sendall(const std::string& data);
381 /**
382 * Receive up to @p bufsize bytes (see the free recv()).
383 * @param bufsize maximum bytes to read.
384 * @return the bytes read (binary-safe), or "" on EOF/error.
385 * @complexity O(@p bufsize).
386 * @alloc allocates the returned string.
387 * @test CheatahSocket.ConnLoopback
388 */
389 std::string recv(long long bufsize);
390 /**
391 * Bound both blocking directions by @p timeout_ms (see the free set_timeout()).
392 * @param timeout_ms per-operation bound in milliseconds (<= 0 clears it).
393 * @return 0 on success, -1 on error.
394 * @complexity O(1).
395 * @alloc none.
396 * @test CheatahSocket.ConnLoopback
397 */
398 long long set_timeout(long long timeout_ms);
399 /**
400 * The local TCP port this fd is bound to (see the free local_port()).
401 * @return the port, or -1 on error.
402 * @complexity O(1) syscall.
403 * @alloc none.
404 * @test CheatahSocket.ConnLoopback
405 */
406 long long local_port() const;
407 /**
408 * Half-close both directions WITHOUT releasing the fd (see the free shutdown()) — wakes a
409 * blocked reader for a clean shutdown; still call close() (or let the destructor) afterward.
410 * @return 0 on success, -1 on error.
411 * @complexity O(1) syscall.
412 * @alloc none.
413 * @test CheatahSocket.ConnLoopback
414 */
415 long long shutdown();
416 /**
417 * Close the fd now (idempotent — the destructor will not close it again).
418 * @return 0 on success, -1 if already closed.
419 * @complexity O(1) syscall.
420 * @alloc none.
421 * @test CheatahSocket.ConnGuardClosesOnScopeExit
422 */
423 long long close();
424
425private:
426 long long fd_ = -1;
427};

429/**
430 * @brief An owning listening socket; accept() yields owning Conn clients; the destructor closes it.
431 *
432 * The server-side RAII guard: `with socket.serve(host, port, backlog) as server { … }` keeps the
433 * listening fd for the block and closes it on exit. Each accept() returns an owning Conn, so a
434 * whole server loop leaks neither the listener nor its clients. Move-only, like Conn.
435 */
436class Listener {
437public:
438 /**
439 * Construct a closed listener (owns no fd).
440 * @complexity O(1).
441 * @alloc none.
442 * @test CheatahSocket.ListenerDefaultIsClosed
443 */
444 Listener() = default;
445 /**
446 * Adopt an already-listening fd (e.g. from tcp_listen()); the `Listener` now owns it.
447 * @param fd a listening fd to take ownership of (-1 for a closed listener).
448 * @complexity O(1).
449 * @alloc none.
450 * @test CheatahSocket.ListenerLoopback
451 */
452 explicit Listener(long long fd) : fd_(fd) {}
453 Listener(const Listener&) = delete;
454 Listener& operator=(const Listener&) = delete;
455 /**
456 * Move-construct, taking over @p other's fd (the moved-from `Listener` becomes closed).
457 * @param other the listener to move from.
458 * @complexity O(1).
459 * @alloc none.
460 * @test CheatahSocket.ListenerLoopback
461 */
462 Listener(Listener&& other) noexcept : fd_(other.fd_) { other.fd_ = -1; }
463 /**
464 * Move-assign, closing this fd first, then taking over @p other's (which becomes closed).
465 * @param other the listener to move from.
466 * @return reference to this listener.
467 * @complexity O(1).
468 * @alloc none.
469 * @test CheatahSocket.ListenerLoopback
470 */
471 Listener& operator=(Listener&& other) noexcept;
472 /**
473 * Close the listening fd if still open.
474 * @complexity O(1) syscall.
475 * @alloc none.
476 * @test CheatahSocket.ListenerLoopback
477 */
478 ~Listener();
479
480 /**
481 * Is the listener open?
482 * @return true iff this owns an open listening fd.
483 * @complexity O(1).
484 * @alloc none.
485 * @test CheatahSocket.ListenerDefaultIsClosed
486 */
487 bool is_open() const { return fd_ >= 0; }
488 /**
489 * The raw listening fd.
490 * @return the owned fd, or -1 when closed.
491 * @complexity O(1).
492 * @alloc none.
493 * @test CheatahSocket.ListenerLoopback
494 */
495 long long fd() const { return fd_; }
496 /**
497 * Accept one pending connection, returned as an owning Conn (the listener stays open).
498 * @return an owning Conn for the client (its is_open() is false on error).
499 * @complexity O(1) syscall (blocks until a client arrives).
500 * @alloc none.
501 * @test CheatahSocket.ListenerLoopback
502 */
503 Conn accept();
504 /**
505 * The local TCP port this listener is bound to (useful after binding to port 0).
506 * @return the port, or -1 on error.
507 * @complexity O(1) syscall.
508 * @alloc none.
509 * @test CheatahSocket.ListenerLoopback
510 */
511 long long local_port() const;
512 /**
513 * Close the listening fd now (idempotent — the destructor will not close it again).
514 * @return 0 on success, -1 if already closed.
515 * @complexity O(1) syscall.
516 * @alloc none.
517 * @test CheatahSocket.ListenerLoopback
518 */
519 long long close();
520
521private:
522 long long fd_ = -1;
523};
