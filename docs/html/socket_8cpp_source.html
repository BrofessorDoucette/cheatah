14// Resolve host:port to an IPv4 TCP address. Returns true and fills `out`/`len` on
15// success. Used by bind/connect so "localhost", dotted IPs, and DNS names all work.
16bool resolve(const std::string& host, long long port, sockaddr_storage& out, socklen_t& len) {
17 addrinfo hints{};
18 hints.ai_family = AF_INET;
19 hints.ai_socktype = SOCK_STREAM;
20 hints.ai_protocol = IPPROTO_TCP;
21 const std::string service = std::to_string(port);
22 addrinfo* res = nullptr;
23 if (::getaddrinfo(host.c_str(), service.c_str(), &hints, &res) != 0 || res == nullptr) {
24 errno = EADDRNOTAVAIL;
25 return false;
26 }
27 std::memcpy(&out, res->ai_addr, res->ai_addrlen);
28 len = static_cast<socklen_t>(res->ai_addrlen);
29 ::freeaddrinfo(res);
30 return true;
31}

68long long local_port(long long fd) {

104long long tcp_listen(const std::string& host, long long port, long long backlog) {
105 // If socket() fails (-1), the bind below fails too and we fall through to the
106 // error path — no separate early return needed.
107 long long fd = socket();
108 set_reuseaddr(fd);
109 if (bind(fd, host, port) != 0 || listen(fd, backlog) != 0) {
110 close(fd);
111 return -1;
112 }
113 return fd;
114}
