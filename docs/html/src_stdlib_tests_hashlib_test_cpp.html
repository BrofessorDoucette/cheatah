28TEST(CheatahHashlib, EmbeddedNulIsHashed) {
29 // std::string_view carries the length, so an embedded NUL is part of the input.

31 EXPECT_NE(hl::sha256(with_nul), hl::sha256("a"));

57TEST(CheatahHashlib, RawDigestMatchesHex) {
58 // The raw digest is exactly the bytes the hex form spells (Python digest()/hexdigest()).
59 static const char hexd[] = "0123456789abcdef";
60 auto to_hex = [](const std::string& raw) {
61 std::string out;
62 for (unsigned char b : raw) { out.push_back(hexd[b >> 4]); out.push_back(hexd[b & 0xF]); }
63 return out;
64 };
65 EXPECT_EQ(hl::sha256_digest("abc").size(), 32u);
66 EXPECT_EQ(hl::sha512_digest("abc").size(), 64u);
67 EXPECT_EQ(to_hex(hl::sha256_digest("abc")), hl::sha256("abc"));
68 EXPECT_EQ(to_hex(hl::sha512_digest("abc")), hl::sha512("abc"));
69 EXPECT_EQ(to_hex(hl::sha256_digest("")), hl::sha256(""));
70 EXPECT_EQ(to_hex(hl::sha512_digest("")), hl::sha512(""));
