3/**
4 * @file random.hpp
5 * @brief cheatah `random` — pseudo-random numbers, mirroring the core of
6 * Python's `random` module. Backed by a seedable Mersenne Twister
7 * (`std::mt19937_64`); `gauss` gives normal deviates for Monte Carlo.
8 * `import random` to use it.
9 *
10 * Unit tests: `stdlib/tests/random_test.cpp`; the suite runs under
11 * AddressSanitizer (the `asan` preset) and Valgrind
12 * (`security/run-valgrind.sh`) on every QA-gate run.
13 *
14 * Doc convention (see also the other stdlib headers): each function documents
15 * its runtime complexity with @complexity, its heap allocation with @alloc, and
16 * the @test that covers it.
17 */
18#include <cstddef>
19#include <ranges>
20
21namespace cheatah::random {
22
23/**
24 * Seed the shared engine, making the stream reproducible.
25 *
26 * Reseeds the single process-wide Mersenne Twister; all `random`/`uniform`/
27 * `randint`/`gauss`/`choice` calls draw from it, so two runs seeded with the same
28 * value produce identical sequences. Until `seed` is called the engine is seeded
29 * non-deterministically from `std::random_device`.
30 * @param s the seed.
31 * @complexity O(1) time.
32 * @alloc none.
33 * @test CheatahRandom.SeedMakesTheStreamReproducible
34 * @crtest RandomCompileRun.Seed
35 * @systest StdlibE2E.Random
36 */
37void seed(unsigned long long s);
38/**
39 * Uniform random double.
40 *
41 * Draws from the shared engine with a uniform real distribution over the
42 * half-open unit interval, so 0.0 can occur but 1.0 cannot.
43 * @return a value in [0, 1).
44 * @complexity O(1) time.
45 * @alloc none.
46 * @test CheatahRandom.RandomInUnitInterval
47 * @crtest RandomCompileRun.Random
48 * @systest StdlibE2E.Random
49 */
50double random();
51/**
52 * Uniform random double in a range.
53 *
54 * Scales a uniform real distribution to span the given bounds; the caller is
55 * expected to pass @p a ≤ @p b (the bounds are not reordered or validated).
56 * @param a,b the bounds.
57 * @return a value in [@p a, @p b].
58 * @complexity O(1) time.
59 * @alloc none.
60 * @test CheatahRandom.UniformInRange
61 * @crtest RandomCompileRun.Uniform
62 * @systest StdlibE2E.Random
63 */
64double uniform(double a, double b);
65/**
66 * Uniform random integer.
67 *
68 * Returns each integer in the closed range with equal probability; both @p a and
69 * @p b are attainable, and @p a == @p b always yields that value.
70 * @param a,b inclusive bounds.
71 * @return an integer in [@p a, @p b].
72 * @complexity O(1) time.
73 * @alloc none.
74 * @test CheatahRandom.RandintInclusiveRange
75 * @crtest RandomCompileRun.Randint
76 * @systest StdlibE2E.Random
77 */
78long long randint(long long a, long long b);
79/**
80 * Normal (Gaussian) deviate.
81 *
82 * Samples the normal distribution N(@p mu, @p sigma²) from the shared engine; the
83 * result is unbounded and can fall on either side of the mean.
84 * @param mu mean.
85 * @param sigma standard deviation.
86 * @return a normal sample.
87 * @complexity O(1) time.
88 * @alloc none.
89 * @test CheatahRandom.GaussIsFiniteAndReproducible
90 * @crtest RandomCompileRun.Gauss
91 * @systest StdlibE2E.Random
92 */
93double gauss(double mu, double sigma);
94
95/**
96 * Random element of a random-access sequence (list/array).
97 *
98 * Picks a uniformly random index in [0, size) via @ref randint and returns a copy
99 * of that element. The sequence must be non-empty — an empty @p seq passes an
100 * inverted range to @ref randint, which is undefined.
101 * @param seq the sequence to pick from (must be non-empty).
102 * @return a copy of a uniformly chosen element.
103 * @complexity O(1) time.
104 * @alloc none (copies one element; via @ref randint).
105 * @test CheatahRandom.Choice
106 * @crtest RandomCompileRun.Choice
107 * @systest StdlibE2E.Random
108 */
109template <std::ranges::random_access_range R>
110std::ranges::range_value_t<R> choice(const R& seq) {
111 const auto n = static_cast<long long>(std::ranges::size(seq));
112 return seq[static_cast<std::size_t>(randint(0, n - 1))];
113}
114
115} // namespace cheatah::random
