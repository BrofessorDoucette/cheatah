5/**
6 * @file random.hpp
7 * @brief cheatah `random` — pseudo-random numbers, mirroring the core of
8 * Python's `random` module. Backed by a seedable Mersenne Twister
9 * (`std::mt19937_64`); `gauss` gives normal deviates for Monte Carlo.
10 * `import random` to use it.
11 *
12 * The engine is PER-THREAD (`thread_local`): concurrent draws from `thread.spawn`ed workers
13 * never race, and each thread's stream is independent. A new thread self-seeds from
14 * `std::random_device` on first use; `seed(s)` seeds the CALLING thread's engine only, so a
15 * worker that wants a reproducible stream calls `seed` itself.
16 *
17 * Unit tests: `stdlib/tests/random_test.cpp`; the suite runs under
18 * AddressSanitizer (the `asan` preset) and Valgrind
19 * (`security/run-valgrind.sh`) on every QA-gate run.
20 *
21 * Doc convention (see also the other stdlib headers): each function documents
22 * its runtime complexity with @complexity, its heap allocation with @alloc, and
23 * the @test that covers it.
24 */
25#include <cstddef>
26#include <ranges>
27
28namespace cheatah::random {
29
30/**
31 * Seed the calling thread's engine, making its stream reproducible.
32 *
33 * Reseeds THIS thread's Mersenne Twister; all `random`/`uniform`/`randint`/`gauss`/`choice`
34 * calls on the same thread draw from it, so two runs seeded with the same value produce
35 * identical sequences. Until `seed` is called a thread's engine is seeded
36 * non-deterministically from `std::random_device` — a `thread.spawn`ed worker that wants
37 * reproducibility calls `seed` itself (the main thread's seed does not reach it).
38 * @param s the seed.
39 * @complexity O(1) time.
40 * @alloc none.
41 * @test CheatahRandom.SeedMakesTheStreamReproducible
42 * @test CheatahRandom.EngineIsPerThread
43 * @crtest RandomCompileRun.Seed
44 * @systest StdlibE2E.Random
45 */
46void seed(unsigned long long s);
47/**
48 * Uniform random double.
49 *
50 * Draws from the shared engine with a uniform real distribution over the
51 * half-open unit interval, so 0.0 can occur but 1.0 cannot.
52 * @return a value in [0, 1).
53 * @complexity O(1) time.
54 * @alloc none.
55 * @test CheatahRandom.RandomInUnitInterval
56 * @crtest RandomCompileRun.Random
57 * @systest StdlibE2E.Random
58 */
59double random();
60/**
61 * Uniform random double in a range.
62 *
63 * Scales a uniform real distribution to span the given bounds; the caller is
64 * expected to pass @p a ≤ @p b (the bounds are not reordered or validated).
65 * @param a,b the bounds.
66 * @return a value in [@p a, @p b].
67 * @complexity O(1) time.
68 * @alloc none.
69 * @test CheatahRandom.UniformInRange
70 * @crtest RandomCompileRun.Uniform
71 * @systest StdlibE2E.Random
72 */
73double uniform(double a, double b);
74/**
75 * Uniform random integer.
76 *
77 * Returns each integer in the closed range with equal probability; both @p a and
78 * @p b are attainable, and @p a == @p b always yields that value.
79 * @param a,b inclusive bounds.
80 * @return an integer in [@p a, @p b].
81 * @complexity O(1) time.
82 * @alloc none.
83 * @test CheatahRandom.RandintInclusiveRange
84 * @crtest RandomCompileRun.Randint
85 * @systest StdlibE2E.Random
86 */
87long long randint(long long a, long long b);
88/**
89 * Normal (Gaussian) deviate.
90 *
91 * Samples the normal distribution N(@p mu, @p sigma²) from the shared engine; the
92 * result is unbounded and can fall on either side of the mean.
93 * @param mu mean.
94 * @param sigma standard deviation.
95 * @return a normal sample.
96 * @complexity O(1) time.
97 * @alloc none.
98 * @test CheatahRandom.GaussIsFiniteAndReproducible
99 * @crtest RandomCompileRun.Gauss
100 * @systest StdlibE2E.Random
101 */
102double gauss(double mu, double sigma);
103
104/**
105 * Random element of a random-access sequence (list/array).
106 *
107 * Picks a uniformly random index in [0, size) via @ref randint and returns a copy
108 * of that element. The sequence must be non-empty — an empty @p seq passes an
109 * inverted range to @ref randint, which is undefined.
110 * @param seq the sequence to pick from (must be non-empty).
111 * @return a copy of a uniformly chosen element.
112 * @complexity O(1) time.
113 * @alloc none (copies one element; via @ref randint).
114 * @test CheatahRandom.Choice
115 * @crtest RandomCompileRun.Choice
116 * @systest StdlibE2E.Random
117 */
118template <std::ranges::random_access_range R>
119std::ranges::range_value_t<R> choice(const R& seq) {
120 const auto n = static_cast<long long>(std::ranges::size(seq));
121 return seq[static_cast<std::size_t>(randint(0, n - 1))];
122}

124} // namespace cheatah::random
