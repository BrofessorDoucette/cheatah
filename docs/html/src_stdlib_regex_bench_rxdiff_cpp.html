2// Original work; see ACKNOWLEDGMENTS.md for the open-source ideas we build upon.
3//
4// rxdiff — differential tests: cheatah::regex vs Google RE2 as a correctness oracle.
5//
6// RE2 is configured with longest_match + Latin-1 (`eng::Re2Longest`), which matches cheatah's
7// documented leftmost-LONGEST byte semantics exactly, so `search`, `full_match` and `find`
8// (offsets AND matched text) must agree byte-for-byte on every pattern in the supported subset.
9//
10// Documented cheatah deviations are folded into the ORACLE pattern, never skipped silently:
11// - a negated class excludes '\n' in cheatah, so `[^ ]` is checked against RE2's `[^ \n]`;
12// - cheatah's `\s` includes '\v' (Python-style) and RE2's does not, so `\s`/`\S` rows filter
13// '\v' out of the generated inputs.
14// Unsupported syntax ({n,m}, \b, mid-pattern anchors, (?...)) is out of corpus by construction.
15//
16// Every failure prints a hex-escaped reproducer that pastes directly into a unit test.
17

30struct Pat {
31 const char* ch; // the cheatah pattern
32 const char* re2 = nullptr; // oracle rewrite when a documented deviation applies (else same)
33 bool strip_vtab = false; // pattern uses \s or \S: drop '\v' from inputs (RE2's \s lacks it)
34 const char* seeds[3] = {nullptr, nullptr, nullptr}; // strings worth embedding in inputs
35};

113std::uint64_t fnv1a(std::string_view s) {
114 std::uint64_t h = 1469598103934665603ull;
115 for (unsigned char c : s) { h ^= c; h *= 1099511628211ull; }
116 return h;
117}

119std::string hex_escape(std::string_view s) {
120 std::string out;
121 out.reserve(s.size() * 4);
122 for (unsigned char c : s) {
123 if (c >= 0x20 && c < 0x7F && c != '"' && c != '\\') {
124 out += static_cast<char>(c);
125 } else {
126 char buf[8];
127 snprintf(buf, sizeof buf, "\\x%02X", c);
128 out += buf;
129 }
130 }
131 return out;
132}

134// A biased ASCII alphabet that actually exercises the corpus (letters the patterns use, digits,
135// separators) plus the odd metacharacter-looking literal.
136std::string rand_ascii(std::mt19937& rng, std::size_t len) {
137 static constexpr char kAlpha[] = "aabbccxyz0123 .@=_-|INFOWARN\"!k=v";
138 std::string s(len, ' ');
139 for (char& c : s) c = kAlpha[rng() % (sizeof kAlpha - 1)];
140 return s;
141}
142
143std::string rand_bytes(std::mt19937& rng, std::size_t len) {
144 std::string s(len, '\0');
145 for (char& c : s) c = static_cast<char>(rng() & 0xFF);
146 return s;
147}

149std::vector<std::string> gen_inputs(const Pat& p) {
150 std::mt19937 rng(static_cast<std::uint32_t>(0xC0FFEEu ^ fnv1a(p.ch)));
151 std::vector<std::string> out = {"", "a", "b", "ab", "ba", "abc", "xaby",
152 "baa", "bbb", "xa", "ax", "aab", "abb", "\n",
153 "a\nb", "\na", "a\n"};
154 for (const char* s : p.seeds)
155 if (s) out.push_back(s);
156 for (int i = 0; i < 14; ++i) out.push_back(rand_ascii(rng, 1 + rng() % 48));
157 for (int i = 0; i < 14; ++i) out.push_back(rand_bytes(rng, 1 + rng() % 48));
158 for (int i = 0; i < 6; ++i) { // newline-heavy
159 std::string s = rand_ascii(rng, 32);
160 for (char& c : s)
161 if (rng() % 5 == 0) c = '\n';
162 out.push_back(s);
163 }
164 for (const char* s0 : p.seeds) { // seed embeddings, doublings, and 1-byte near-misses
165 if (!s0) continue;
166 const std::string mid = s0;
167 std::string s = rand_ascii(rng, rng() % 8) + mid + rand_ascii(rng, rng() % 8);
168 out.push_back(s);
169 out.push_back(mid + mid);
170 out.push_back(mid + "\n" + mid);
171 for (int k = 0; k < 5 && !s.empty(); ++k) {
172 std::string t = s;
173 t[rng() % t.size()] = static_cast<char>('a' + rng() % 26);
174 out.push_back(t);
175 }
176 }
177 if (p.strip_vtab)
178 for (std::string& s : out)
179 for (char& c : s)
180 if (c == '\v') c = 'x';
181 return out;
182}

186TEST(RxDiff, AgreesWithRe2OnCorpus) {
187 for (const Pat& p : kCorpus) {
188 auto ch = eng::try_compile<eng::Cheatah>(p.ch);
189 auto oracle = eng::try_compile<eng::Re2Longest>(p.re2 ? p.re2 : p.ch);
190 ASSERT_TRUE(ch != nullptr) << "cheatah rejects corpus pattern: " << p.ch;
191 ASSERT_TRUE(oracle != nullptr) << "RE2 rejects oracle pattern for: " << p.ch;
192 for (const std::string& in : gen_inputs(p)) {
193 SCOPED_TRACE("pattern \"" + std::string(p.ch) + "\" input \"" + hex_escape(in) + "\"");
194 const std::string_view t(in);
195 ASSERT_EQ(eng::Cheatah::search(*ch, t), eng::Re2Longest::search(*oracle, t));
196 ASSERT_EQ(eng::Cheatah::full(*ch, t), eng::Re2Longest::full(*oracle, t));
197 std::size_t cb = 0, ce = 0, ob = 0, oe = 0;
198 const bool cf = eng::Cheatah::find(*ch, t, cb, ce);
199 const bool of = eng::Re2Longest::find(*oracle, t, ob, oe);
200 ASSERT_EQ(cf, of);
201 if (cf) {
202 EXPECT_EQ(cb, ob);
203 EXPECT_EQ(ce, oe);
204 auto m = cheatah::regex::find(*ch, t);
205 EXPECT_EQ(m.text, in.substr(ob, oe - ob));
206 }
207 }
208 }
209}

211TEST(RxDiff, AdversarialLargeAgreesWithRe2) {
212 // The hostile shapes at half a megabyte, cross-checked against the oracle: search AND
213 // full_match booleans plus find offsets must agree on inputs this large too (the large-
214 // input fast paths — reverse pass, single forward pass, candidate budget — all fire).
215 std::mt19937 rng(0xBADCAFE);
216 std::string as(512 * 1024, 'a');
217 std::string xs(512 * 1024, 'x');
218 std::string ab(512 * 1024, 'a');
219 for (char& ch : ab)
220 if (rng() & 1) ch = 'b';
221 const std::string* inputs[] = {&as, &xs, &ab};
222 const char* pats[] = {"(a+)+$", "(a|a)*c", "c[ab]*$", "a*$", "[ab]+$", "x[0-9]", "(x+x+)+y"};
223 for (const char* pat : pats) {
224 auto ch = eng::try_compile<eng::Cheatah>(pat);
225 auto oracle = eng::try_compile<eng::Re2Longest>(pat);
226 ASSERT_TRUE(ch && oracle) << pat;
227 for (const std::string* in : inputs) {
228 SCOPED_TRACE(std::string(pat) + " over " + std::to_string(in->size()) + " bytes");
229 const std::string_view t(*in);
230 ASSERT_EQ(eng::Cheatah::search(*ch, t), eng::Re2Longest::search(*oracle, t));
231 ASSERT_EQ(eng::Cheatah::full(*ch, t), eng::Re2Longest::full(*oracle, t));
232 std::size_t cb = 0, ce = 0, ob = 0, oe = 0;
233 const bool cf = eng::Cheatah::find(*ch, t, cb, ce);
234 const bool of = eng::Re2Longest::find(*oracle, t, ob, oe);
235 ASSERT_EQ(cf, of);
236 if (cf) {
237 EXPECT_EQ(cb, ob);
238 EXPECT_EQ(ce, oe);
239 }
240 }
241 }
242}

244TEST(RxDiff, CompileErrorAgreement) {
245 // Patterns BOTH engines must reject (the shared malformed set).
246 const char* bad[] = {"(", "a(b", "((a)", "[a-", "[abc", "a\\", "*a", "+a", "?a", ")", "a)"};
247 for (const char* p : bad) {
248 EXPECT_FALSE(cheatah::regex::compile(p).ok) << p;
249 EXPECT_TRUE(eng::try_compile<eng::Re2Longest>(p) == nullptr) << p;
250 }
251 // And every corpus pattern must be accepted by both (the loop above ASSERTs it too, but a
252 // compile-only sweep localizes a corpus typo instantly).
253 for (const Pat& p : kCorpus) {
254 EXPECT_TRUE(cheatah::regex::compile(p.ch).ok) << p.ch;
255 EXPECT_TRUE(eng::try_compile<eng::Re2Longest>(p.re2 ? p.re2 : p.ch) != nullptr) << p.ch;
256 }
