1// System-level test for the cheatah stdlib `math` module: ONE cohesive program
2// that exercises EVERY public function and constant declared in
3// stdlib/math/math.hpp (sqrt, cbrt, fabs, floor, ceil, trunc, round, exp, log,
4// log2, log10, sin, cos, tan, asin, acos, atan, atan2, hypot, fmod, copysign,
5// degrees, radians, isnan, isinf, isfinite, gcd, factorial, and the math-related
6// built-ins abs/min/max/pow, plus the constants pi/e/tau/inf/nan).
7//
8// Unlike the per-function compile-run tests (math_cr_test.cpp), this drives a
9// small geometry / number-theory pipeline that wires the functions together
10// (round_to uses round+pow; unit_point uses cos/sin/hypot; the combined section
11// reduces a fraction by gcd then norms it via pow/sqrt) and prints a single
12// deterministic multi-line report.
13//
14// The expected stdout below was captured by compiling and running the program
15// with the debug-build purrc + cheatah and confirming the output is identical
16// across repeated runs (md5 stable) before hardcoding it. Nothing here is
17// time/RNG dependent, so it is reproducible byte-for-byte.
18

30# Round a double to `digits` decimal places (uses round/pow).
31fn round_to(x, digits) {
32 let scale = math.pow(10, digits)
33 return math.round(x * scale) / scale
34}
35
36# Polar -> cartesian distance check using trig + hypot.
37fn unit_point(theta) {
38 # returns hypot(cos, sin) which is 1 for any theta
39 return math.hypot(math.cos(theta), math.sin(theta))
40}

45io.print("inf", math.inf, "nan", math.nan)
