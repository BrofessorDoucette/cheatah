1// Folded sibling C++ (foo.purr + foo.hpp / foo.cpp): purrc weaves a hand-written C++ base into a
2// module's generated output, inside `namespace cheatah::<m>`, so the .purr and the C++ are one
3// module — the .purr calls the C++ by bare name, no import bridge. See docs and codegen.cpp.
4
5#include <sys/stat.h>
6
7#include <cstdio>
8#include <fstream>
9#include <sstream>
10#include <string>
11
12#include <gtest/gtest.h>

14#ifndef PURRC_PATH
15#define PURRC_PATH ""
16#endif
17#ifndef CHEATAH_RUNTIME_PATH
18#define CHEATAH_RUNTIME_PATH ""
19#endif
20#ifndef PURR_TEST_TMP
21#define PURR_TEST_TMP "."
22#endif

24namespace {
25bool exists(const std::string& p) {
26 struct stat st;
27 return ::stat(p.c_str(), &st) == 0;
28}
29std::string read_file(const std::string& p) {
30 std::ifstream f(p);
31 std::ostringstream s;
32 s << f.rdbuf();
33 return s.str();
34}
35void write_file(const std::string& p, const std::string& c) {
36 std::ofstream f(p);
37 f << c;
38}
39int run(const std::string& cmd) { return std::system((cmd + " >/dev/null 2>&1").c_str()); }
40std::string capture(const std::string& cmd) {
41 FILE* pipe = popen((cmd + " 2>/dev/null").c_str(), "r");
42 if (!pipe) return "";
43 std::string out;
44 char buf[256];
45 while (std::fgets(buf, sizeof buf, pipe)) out += buf;
46 pclose(pipe);
47 return out;
48}
49// A fresh per-test package dir <tmp>/folded_<tag>/<name>/ with <name>.purr in it.
50std::string make_dir(const std::string& tag, const std::string& name) {
51 const std::string root = std::string(PURR_TEST_TMP) + "/folded_" + tag;
52 const std::string dir = root + "/" + name;
53 run("rm -rf " + root + " && mkdir -p " + dir);
54 return dir;
55}
56const std::string kPurrc = PURRC_PATH;

58// A hand-written base header fragment: a naked C++ helper the .purr will call. Returns an `n`-byte
59// little-endian string so the .purr can observe its length.
60const char* kBaseHpp =
61 "// A hand-written C++ fragment folded into the module. No namespace, no bridge.\n"
62 "#include <string>\n"
63 "[[nodiscard]] inline std::string le2(long long n) {\n"
64 " return std::string{ static_cast<char>(n & 0xFF), static_cast<char>((n >> 8) & 0xFF) };\n"
65 "}\n";
66} // namespace

68// Case 1 — the proof: auto-detected sibling folded, the .purr calls it, and it runs end to end.
69TEST(FoldedBase, SiblingFoldedAndCalledEndToEnd) {
70 const std::string dir = make_dir("call", "pk");
71 const std::string base = dir + "/pk.hpp";
72 write_file(base, kBaseHpp);
73 write_file(dir + "/pk.purr", "fn width(n: int) -> int {\n return len(le2(n))\n}\n");
74 const std::string base_before = read_file(base);
75
76 // No flags: the same-stem pk.hpp is auto-detected and folded.
77 ASSERT_EQ(run(kPurrc + " --emit-library --transparent " + dir + "/pk.purr --import-root " + dir), 0);
78
79 // The generated output is pk.gen.hpp; the hand-written pk.hpp is left byte-for-byte untouched.
80 EXPECT_TRUE(exists(dir + "/pk.gen.hpp"));
81 EXPECT_FALSE(exists(dir + "/pk.hpp.gen.hpp"));
82 EXPECT_EQ(read_file(base), base_before);
83
84 const std::string hdr = read_file(dir + "/pk.gen.hpp");
85 const std::size_t inc = hdr.find("#include <string>");
86 const std::size_t ns = hdr.find("namespace cheatah::pk");
87 const std::size_t def = hdr.find("inline std::string le2");
88 const std::size_t fn = hdr.find("width"); // the transpiled function
89 ASSERT_NE(inc, std::string::npos);
90 ASSERT_NE(ns, std::string::npos);
91 ASSERT_NE(def, std::string::npos);
92 ASSERT_NE(fn, std::string::npos);
93 EXPECT_LT(inc, ns); // the #include is hoisted ABOVE the namespace (can't live inside one)
94 EXPECT_LT(ns, def); // the base body is INSIDE the namespace
95 EXPECT_LT(def, fn); // ...and BEFORE the transpiled function, so the .purr can call it
96
97 // A consumer imports the module, calls the .purr fn (which calls the folded C++), and prints 2.
98 const std::string prog = dir + "/../use.purr", mod = dir + "/../use.so";
99 write_file(prog, "import io\nimport pk\nio.print(pk.width(258))\n");
100 ASSERT_EQ(run(kPurrc + " " + prog + " -o " + mod + " --import-root " + dir), 0);
101 EXPECT_EQ(capture(std::string(CHEATAH_RUNTIME_PATH) + " " + mod), "2\n");
102}

104// Case 2 — the off-switch: --no-adjacent ignores the sibling; output is a plain .hpp with no fold.
105TEST(FoldedBase, NoAdjacentDisablesTheMagic) {
106 const std::string dir = make_dir("off", "pk");
107 write_file(dir + "/pk.hpp", kBaseHpp);
108 write_file(dir + "/pk.purr", "fn width(n: int) -> int {\n return len(le2(n))\n}\n");
109
110 ASSERT_EQ(run(kPurrc + " --emit-library --transparent --no-adjacent " + dir + "/pk.purr -o " +
111 dir + "/out.hpp --import-root " + dir), 0);
112 const std::string hdr = read_file(dir + "/out.hpp");
113 EXPECT_EQ(hdr.find("inline std::string le2"), std::string::npos); // the DEFINITION was not folded
114 EXPECT_NE(hdr.find("le2("), std::string::npos); // the call remains (debug view)
115 EXPECT_FALSE(exists(dir + "/pk.gen.hpp"));
116}

118// Case 3 — the safety guard: a sibling that is itself a purrc-generated file is never folded.
119TEST(FoldedBase, GeneratedSiblingIsSkipped) {
120 const std::string dir = make_dir("guard", "th");
121 write_file(dir + "/th.hpp",
122 "// Generated by purrc — do not edit.\ninline int stale() { return 9; }\n");
123 write_file(dir + "/th.purr", "fn f() -> int {\n return 1\n}\n");
124
125 ASSERT_EQ(run(kPurrc + " --emit-library --transparent " + dir + "/th.purr -o " + dir +
126 "/th.gen.hpp --import-root " + dir), 0);
127 // No fold happened, so no `stale` in the output (and the auto-detect left th.hpp alone).
128 EXPECT_EQ(read_file(dir + "/th.gen.hpp").find("stale"), std::string::npos);
129}

144// Case 5 — the fixed gap: a top-level `cpp{}` in a library module now survives into the header.
145TEST(FoldedBase, CppBlockSurvivesInLibraryHeader) {
146 const std::string dir = make_dir("cpp", "gr");
147 write_file(dir + "/gr.purr",
148 "cpp {\n#include <string>\ninline std::string bang(const std::string& s) { return s + \"!\"; }\n}\n\n"
149 "fn shout(s: str) -> str {\n return bang(s)\n}\n");
150 ASSERT_EQ(run(kPurrc + " --emit-library --transparent " + dir + "/gr.purr -o " + dir +
151 "/gr.hpp --import-root " + dir), 0);
152 const std::string hdr = read_file(dir + "/gr.hpp");
153 EXPECT_NE(hdr.find("inline std::string bang"), std::string::npos); // the cpp{} body is emitted
154 EXPECT_LT(hdr.find("inline std::string bang"), hdr.find("namespace cheatah::gr")); // at file scope
155}

157// Case 6 — the accessor pattern: a folded C++ getter READS a .purr struct's field, and the .purr
158// calls it. This is the point of the after-types splice: the header holds the accessor layer, the
159// .purr holds the logic. The base is emitted AFTER the struct (so it can see it) and BEFORE the fn.
160TEST(FoldedBase, FoldedGetterReadsPurrStruct) {
161 const std::string dir = make_dir("getter", "pt");
162 write_file(dir + "/pt.hpp",
163 "// a getter over the .purr struct Point — organization in the header\n"
164 "[[nodiscard]] inline long long manhattan(const Point& p) {\n"
165 " return (p.x < 0 ? -p.x : p.x) + (p.y < 0 ? -p.y : p.y);\n}\n");
166 write_file(dir + "/pt.purr",
167 "struct Point {\n x: int\n y: int\n}\n"
168 "fn make(x: int, y: int) -> Point {\n return Point({ .x = x, .y = y })\n}\n"
169 "fn dist(p: Point) -> int {\n return manhattan(p)\n}\n");
170
171 ASSERT_EQ(run(kPurrc + " --emit-library --transparent " + dir + "/pt.purr --import-root " + dir), 0);
172 const std::string hdr = read_file(dir + "/pt.gen.hpp");
173 const std::size_t st = hdr.find("struct Point");
174 const std::size_t get = hdr.find("manhattan(const Point&");
175 const std::size_t fn = hdr.find("dist");
176 ASSERT_NE(st, std::string::npos);
177 ASSERT_NE(get, std::string::npos);
178 ASSERT_NE(fn, std::string::npos);
179 EXPECT_LT(st, get); // the struct is defined BEFORE the getter (so the getter can read it)
180 EXPECT_LT(get, fn); // the getter is defined BEFORE the .purr fn that calls it
181
182 // And it runs: the .purr calls the C++ getter over a .purr struct it built.
183 const std::string prog = dir + "/../use.purr", mod = dir + "/../use.so";
184 write_file(prog, "import io\nimport pt\nio.print(pt.dist(pt.make(-3, 4)))\n");
185 ASSERT_EQ(run(kPurrc + " " + prog + " -o " + mod + " --import-root " + dir), 0);
186 EXPECT_EQ(capture(std::string(CHEATAH_RUNTIME_PATH) + " " + mod), "7\n");
187}
