1// System tests for the `websocket` module: a REAL RFC 6455 WebSocket handshake +
2// echo against Node's `ws` library (the reference WebSocket implementation) behind
3// Node's built-in TLS — test infrastructure only. The client side is pure cheatah:
4// the from-scratch tls 1.3 client (x25519, ChaCha20-Poly1305, HKDF, leaf-cert
5// verify) plus this module's from-scratch RFC 6455 framing. No WebSocket or TLS
6// protocol code is mirrored in our tree — `ws` frames, Node does TLS, openssl mints
7// the cert. This mirrors tls_sys_test.cpp's OpensslServer pattern.

35// Generate a throwaway self-signed leaf cert (@p newkey selects the key algorithm —
36// "ec -pkeyopt ec_paramgen_curve:prime256v1", "rsa:2048", "ed25519") and launch the
37// Node `ws` wss echo server (tests/fixtures/wss_echo_server.js) on @p port. The echo
38// server is the reference RFC 6455 peer; cheatah's client is what's under test. Waits
39// for the server's "READY" marker (bound), then is pkill'd in the destructor.
40// @complexity O(1) (two subprocesses) @alloc the command strings + a ready-file path
41class WssEchoServer {
42public:
43 explicit WssEchoServer(long long port,
44 const std::string& newkey = "ec -pkeyopt ec_paramgen_curve:prime256v1",
45 const std::string& mode = "")
46 : port_(port) {
47 const std::string tmp = PURR_TEST_TMP;
48 const std::string tag = std::to_string(port_);
49 cert_ = tmp + "/ws_test_cert_" + tag + ".pem";
50 key_ = tmp + "/ws_test_key_" + tag + ".pem";
51 ready_ = tmp + "/ws_test_ready_" + tag + ".log";
52 std::remove(ready_.c_str());
53 const std::string gen = "openssl req -x509 -newkey " + newkey + " -keyout '" + key_ +
54 "' -out '" + cert_ +
55 "' -days 2 -nodes -subj /CN=localhost 2>/dev/null";
56 cert_ok_ = std::system(gen.c_str()) == 0;
57 if (!cert_ok_) return;
58 const std::string script = std::string(WEBSOCKET_FIXTURE_DIR) + "/wss_echo_server.js";
59 const std::string serve = "'" + std::string(NODE_EXECUTABLE) + "' '" + script + "' '" +
60 cert_ + "' '" + key_ + "' " + tag + " " + mode + " >'" +
61 ready_ + "' 2>&1 &";
62 node_ok_ = std::system(serve.c_str()) == 0;
63 // Wait (up to ~5s) for the server to print READY, i.e. it has bound the port.
64 for (int i = 0; i < 100; ++i) {
65 if (ready_marker_seen()) { bound_ = true; break; }
66 std::this_thread::sleep_for(std::chrono::milliseconds(50));
67 }
68 }
69 ~WssEchoServer() {
70 const std::string kill =
71 "pkill -f 'wss_echo_server.js .* " + std::to_string(port_) + "'";
72 std::system(kill.c_str());
73 }
74 [[nodiscard]] bool ok() const { return cert_ok_ && node_ok_ && bound_; }
75 [[nodiscard]] long long port() const { return port_; }
76 [[nodiscard]] std::string url() const {
77 return "wss://localhost:" + std::to_string(port_) + "/";
78 }
79
80private:
81 [[nodiscard]] bool ready_marker_seen() const {
82 std::FILE* f = std::fopen(ready_.c_str(), "rb");
83 if (f == nullptr) return false;
84 char buf[256];
85 const std::size_t n = std::fread(buf, 1, sizeof(buf) - 1, f);
86 std::fclose(f);
87 buf[n] = '\0';
88 return std::string(buf).find("READY") != std::string::npos;
89 }
90 long long port_;
91 std::string cert_, key_, ready_;
92 bool cert_ok_ = false, node_ok_ = false, bound_ = false;
93};

95// node + the ws library must be present: the websocket handshake/framing paths can

101 "' >/dev/null 2>&1 && test -d '" +
102 std::string(WEBSOCKET_FIXTURE_DIR) + "/node_modules/ws'";
103 return std::system(check.c_str()) == 0;
104 }();
105 ASSERT_TRUE(has_node)
106 << "node + the `ws` package are required test infrastructure (run: cd tests/fixtures "
107 "&& npm install ws). Coverage runs require them.";
108}

112// The full path: TLS 1.3 handshake + RFC 6455 upgrade + a text echo round trip +
113// clean close, against the real Node `ws` server. This is the @systest anchor.
114TEST(WebSocketSys, EchoRoundTrip) {
115 require_node();
116 WssEchoServer server(48951);

126 EXPECT_EQ(ws::close(s), 0);

161// The RAII Client guard round trip: open_url / send_text / recv / is_open / id, then
162// move-construct, move-assign, shutdown and explicit close — mirrors tls's ConnGuard test.
163TEST(WebSocketSys, ClientGuardRoundTrip) {
164 require_node();
165 WssEchoServer server(48954);
166 ASSERT_TRUE(server.ok()) << "could not start node ws echo server (test infrastructure)";
167
168 ws::Client c = ws::open_url(server.url());
169 ASSERT_TRUE(c.is_open());
170 EXPECT_GT(c.id(), 0);
171 EXPECT_EQ(c.send_text("guarded"), 7);
172 EXPECT_EQ(c.recv(), "guarded");
173
174 ws::Client active(std::move(c)); // move-construct: transfer ownership
175 EXPECT_FALSE(c.is_open());
176 EXPECT_TRUE(active.is_open());
177 EXPECT_EQ(active.send_text("moved"), 5);
178 EXPECT_EQ(active.recv(), "moved");
179
180 ws::Client sink; // default-constructed: closed
181 EXPECT_FALSE(sink.is_open());
182 sink = std::move(active); // move-assign onto a closed guard
183 EXPECT_FALSE(active.is_open());
184 EXPECT_TRUE(sink.is_open());
185
186 EXPECT_EQ(sink.shutdown(), 0); // half-close the socket (wake any reader)
187 EXPECT_EQ(sink.close(), 0);
188 EXPECT_EQ(sink.close(), -1); // idempotent: already closed
189}

213// The server-initiated clean close: sending "close" makes the ws server send a real
214// RFC 6455 close frame. cheatah's recv sees opcode 0x8, echoes a close frame back, marks
215// the session closed and returns "" (EOF). A SECOND recv returns "" via the s->closed
216// short-circuit. Exercises recv's close-frame branch AND the closed-session fast path.
217TEST(WebSocketSys, ServerCloseYieldsEmptyRecv) {
218 require_node();
219 WssEchoServer server(48956);
220 ASSERT_TRUE(server.ok()) << "could not start node ws echo server (test infrastructure)";
221
222 const long long s = ws::connect_url(server.url());
223 ASSERT_GE(s, 0);
224 EXPECT_EQ(ws::send_text(s, "ping-echo"), 9);
225 EXPECT_EQ(ws::recv(s), "ping-echo");
226 ws::send_text(s, "close"); // ask the server for a clean close handshake
227 EXPECT_EQ(ws::recv(s), ""); // opcode 0x8 -> echo close, mark closed, return ""
228 EXPECT_EQ(ws::recv(s), ""); // s->closed short-circuit (no I/O)
229 EXPECT_EQ(ws::close(s), 0); // close() after a peer close: !closed is false, just teardown
230}

278// TLS handshake failure: connect to a peer that accepts TCP but is not a TLS server,
279// so tls::client_connect fails -> connect() must throw and close the fd (142-144).
280TEST(WebSocketSys, RefusesTlsHandshakeFailure) {
281 const long long listen_fd = sock::tcp_listen("127.0.0.1", 0, 4);
282 ASSERT_GE(listen_fd, 0);
283 const long long port = sock::local_port(listen_fd);
284 std::thread peer([listen_fd]() {
285 const long long client = sock::accept(listen_fd);
286 if (client >= 0) {
287 sock::sendall(client, "definitely not a TLS server\r\n");
288 sock::close(client);
289 }
290 });
291 EXPECT_THROW(ws::connect("127.0.0.1", port, "/", "localhost"), std::runtime_error);
292 peer.join();
293 sock::close(listen_fd);
294}
