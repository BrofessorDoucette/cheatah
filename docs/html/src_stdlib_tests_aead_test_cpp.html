1// Unit tests for the `aead` module against the RFC 8439 §2.8.2 AEAD test vector, plus
2// round-trip, tamper-rejection, and malformed-input behavior.
3#include <gtest/gtest.h>
4
5#include <string>
6
7#include "aead.hpp"
8
9namespace a = cheatah::aead;
10
11namespace {
12const std::string kKey = "808182838485868788898a8b8c8d8e8f909192939495969798999a9b9c9d9e9f";
13const std::string kNonce = "070000004041424344454647";
14const std::string kAad = std::string("\x50\x51\x52\x53\xc0\xc1\xc2\xc3\xc4\xc5\xc6\xc7", 12);
15const char* kPlain =
16 "Ladies and Gentlemen of the class of '99: If I could offer you "
17 "only one tip for the future, sunscreen would be it.";
18// RFC 8439 §2.8.2 expected tag (the ciphertext bytes are checked via round-trip + length).
19const std::string kTagHex = "1ae10b594f09e26a7e902ecbd0600691";
20
21std::string hex_of(std::string_view raw) {
22 static constexpr char kHex[] = "0123456789abcdef";
23 std::string out;
24 for (const char ch : raw) {
25 out.push_back(kHex[static_cast<unsigned char>(ch) >> 4]);
26 out.push_back(kHex[static_cast<unsigned char>(ch) & 0xF]);
27 }
28 return out;
29}
30} // namespace

32// The RFC 8439 §2.8.2 vector: ciphertext prefix + tag must match the spec exactly.
33TEST(CheatahAead, Rfc8439Encrypt) {
34 const std::string ct = a::chacha20poly1305_encrypt(kKey, kNonce, kAad, kPlain);
35 ASSERT_EQ(ct.size(), std::string(kPlain).size() + 16);
36 EXPECT_EQ(hex_of(ct.substr(0, 16)), "d31a8d34648e60db7b86afbc53ef7ec2"); // first CT block
37 EXPECT_EQ(hex_of(ct.substr(ct.size() - 16)), kTagHex); // the Poly1305 tag
38}

40// Decrypt of the spec ciphertext returns the spec plaintext (and the round trip holds).
41TEST(CheatahAead, Rfc8439Decrypt) {
42 const std::string ct = a::chacha20poly1305_encrypt(kKey, kNonce, kAad, kPlain);
43 EXPECT_EQ(a::chacha20poly1305_decrypt(kKey, kNonce, kAad, ct), kPlain);
44}

46// Any tampering — ciphertext byte, tag byte, or different aad — must yield "" (rejected).
47TEST(CheatahAead, RejectsTamper) {
48 std::string ct = a::chacha20poly1305_encrypt(kKey, kNonce, kAad, kPlain);
49 std::string flipped = ct;
50 flipped[3] = static_cast<char>(flipped[3] ^ 0x01);
51 EXPECT_EQ(a::chacha20poly1305_decrypt(kKey, kNonce, kAad, flipped), "");
52 std::string bad_tag = ct;
53 bad_tag[bad_tag.size() - 1] = static_cast<char>(bad_tag.back() ^ 0x80);
54 EXPECT_EQ(a::chacha20poly1305_decrypt(kKey, kNonce, kAad, bad_tag), "");
55 EXPECT_EQ(a::chacha20poly1305_decrypt(kKey, kNonce, "other aad", ct), "");
56 EXPECT_EQ(a::chacha20poly1305_decrypt(kKey, kNonce, kAad, "short"), "");
57 EXPECT_EQ(a::chacha20poly1305_encrypt("zz", kNonce, kAad, kPlain), "");
58}

60// Empty plaintext and empty aad are valid AEAD inputs (TLS uses empty aad in places).
61TEST(CheatahAead, EmptyInputs) {
62 const std::string ct = a::chacha20poly1305_encrypt(kKey, kNonce, "", "");
63 ASSERT_EQ(ct.size(), std::size_t{16}); // tag only
64 EXPECT_EQ(a::chacha20poly1305_decrypt(kKey, kNonce, "", ct), "");
65 // "" is also the FAILURE value; distinguish via a 1-byte round trip
66 const std::string one = a::chacha20poly1305_encrypt(kKey, kNonce, "", "x");
67 EXPECT_EQ(a::chacha20poly1305_decrypt(kKey, kNonce, "", one), "x");
68}

70// ---- AES-128-GCM (TLS_AES_128_GCM_SHA256) against the NIST/GCM (McGrew & Viega) test vectors ----
71namespace {
72std::string unhex(std::string_view h) {
73 auto v = [](char c) { return c <= '9' ? c - '0' : (c | 0x20) - 'a' + 10; };
74 std::string o;
75 for (std::size_t i = 0; i + 1 < h.size(); i += 2)
76 o.push_back(static_cast<char>((v(h[i]) << 4) | v(h[i + 1])));
77 return o;
78}
79} // namespace

112// Round trip + tamper rejection (ciphertext, tag, aad) + malformed key/nonce/short input.
113TEST(CheatahAead, AesGcmRejectsTamperAndMalformed) {
114 const std::string key = "000102030405060708090a0b0c0d0e0f", iv = "000102030405060708090a0b";
115 const std::string aad = "header", p = "the quick brown fox";
116 std::string ct = a::aes128gcm_encrypt(key, iv, aad, p);
117 ASSERT_EQ(ct.size(), p.size() + 16);
118 EXPECT_EQ(a::aes128gcm_decrypt(key, iv, aad, ct), p); // round trip
119 std::string flip_ct = ct;
120 flip_ct[2] = static_cast<char>(flip_ct[2] ^ 0x01);
121 EXPECT_EQ(a::aes128gcm_decrypt(key, iv, aad, flip_ct), "");
122 std::string flip_tag = ct;
123 flip_tag[flip_tag.size() - 1] = static_cast<char>(flip_tag.back() ^ 0x80);
124 EXPECT_EQ(a::aes128gcm_decrypt(key, iv, aad, flip_tag), "");
125 EXPECT_EQ(a::aes128gcm_decrypt(key, iv, "other aad", ct), "");
126 EXPECT_EQ(a::aes128gcm_decrypt(key, iv, aad, "short"), "");
127 EXPECT_EQ(a::aes128gcm_encrypt("ababab", iv, aad, p), ""); // key not 16 bytes
128 EXPECT_EQ(a::aes128gcm_encrypt(key, "00", aad, p), ""); // nonce not 12 bytes
129}

131// The AES-NI/PCLMULQDQ fast path and the portable scalar reference must agree byte-for-byte,
132// across sizes that exercise the 4-wide CTR loop (>=64), its tail (non-multiple of 64), the
133// single sub-block, and empty. Also keeps the scalar reference covered on AES-NI hardware.
134TEST(CheatahAead, AesGcmPortableMatchesHardware) {
135 const std::string key = "000102030405060708090a0b0c0d0e0f";
136 const std::string nonce = "101112131415161718191a1b";
137 const std::string aad = "associated-data-header";
138 for (std::size_t n : {std::size_t(0), std::size_t(13), std::size_t(16), std::size_t(64),
139 std::size_t(100), std::size_t(255)}) {
140 std::string pt(n, '\0');
141 for (std::size_t i = 0; i < n; ++i) pt[i] = static_cast<char>(i * 7 + 1);
142
143 a::set_force_portable_crypto(false);
144 const std::string hw = a::aes128gcm_encrypt(key, nonce, aad, pt);
145 a::set_force_portable_crypto(true);
146 const std::string sw = a::aes128gcm_encrypt(key, nonce, aad, pt);
147 a::set_force_portable_crypto(false);
148
149 ASSERT_EQ(hw.size(), n + 16);
150 EXPECT_EQ(hw, sw) << "hardware vs portable AES-GCM differ at size " << n;
151 EXPECT_EQ(a::aes128gcm_decrypt(key, nonce, aad, hw), pt); // hardware decrypt
152 a::set_force_portable_crypto(true);
153 EXPECT_EQ(a::aes128gcm_decrypt(key, nonce, aad, hw), pt); // portable decrypt
154 a::set_force_portable_crypto(false);
155 }
156 // A tampered tag is rejected on BOTH paths.
157 std::string ct = a::aes128gcm_encrypt(key, nonce, aad, std::string(80, 'z'));
158 ct[ct.size() - 1] = static_cast<char>(ct.back() ^ 0x01);
159 EXPECT_EQ(a::aes128gcm_decrypt(key, nonce, aad, ct), "");
160 a::set_force_portable_crypto(true);
161 EXPECT_EQ(a::aes128gcm_decrypt(key, nonce, aad, ct), "");
162 EXPECT_EQ(a::aes128gcm_decrypt(key, nonce, aad, "short"), ""); // portable: too short
163 a::set_force_portable_crypto(false);
164
165 // Large AAD (>= 128 bytes) exercises the 8-wide GHASH aggregation over the AAD too.
166 {
167 const std::string big_aad(200, 'A');
168 const std::string pt = "payload for the large-aad cross-check";
169 a::set_force_portable_crypto(false);
170 const std::string hw = a::aes128gcm_encrypt(key, nonce, big_aad, pt);
171 a::set_force_portable_crypto(true);
172 const std::string sw = a::aes128gcm_encrypt(key, nonce, big_aad, pt);
173 a::set_force_portable_crypto(false);
174 EXPECT_EQ(hw, sw);
175 EXPECT_EQ(a::aes128gcm_decrypt(key, nonce, big_aad, hw), pt);
176 }

179 EXPECT_FALSE(
180 a::aes128gcm_encrypt("000102030405060708090A0B0C0D0E0F", nonce, aad, "x").empty());
