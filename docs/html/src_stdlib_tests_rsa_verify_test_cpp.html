1// Unit tests for tls/rsa_verify.hpp — RSA-PSS (rsa_pss_rsae_sha256) signature verification used by the
2// TLS 1.3 CertificateVerify path. A FIXED vector (a 2048-bit RSA public key + a PSS-SHA256 signature
3// over a known message, produced once with OpenSSL) checks the accept path; the rest check that every
4// way a signature can be wrong is rejected. Complements tls_sys_test.cpp's TlsSys.HandshakeRsaCertificate
5// (which covers the accept path end-to-end against a real server) with the REJECTION paths.
6#include <gtest/gtest.h>
7
8#include <string>
9
10#include "rsa_verify.hpp"
11

14namespace {
15std::string unhex(std::string_view h) {
16 auto v = [](char c) { return c <= '9' ? c - '0' : (c | 0x20) - 'a' + 10; };
17 std::string o;
18 for (std::size_t i = 0; i + 1 < h.size(); i += 2)
19 o.push_back(static_cast<char>((v(h[i]) << 4) | v(h[i + 1])));
20 return o;
21}

42// The genuine OpenSSL-produced signature verifies against the public key + message.
43TEST(CheatahRsaVerify, AcceptsValidSignature) {
44 EXPECT_TRUE(rsa::verify_pss_sha256(unhex(kSpkiHex), kMsg, unhex(kSigHex)));
45}

47// A single flipped signature byte must be rejected.
48TEST(CheatahRsaVerify, RejectsTamperedSignature) {
49 std::string sig = unhex(kSigHex);
50 sig[100] = static_cast<char>(sig[100] ^ 0x01);
51 EXPECT_FALSE(rsa::verify_pss_sha256(unhex(kSpkiHex), kMsg, sig));
52}
53
54// The right signature over the WRONG message must be rejected.
55TEST(CheatahRsaVerify, RejectsWrongMessage) {
56 EXPECT_FALSE(rsa::verify_pss_sha256(unhex(kSpkiHex), kMsg + "!", unhex(kSigHex)));
57 EXPECT_FALSE(rsa::verify_pss_sha256(unhex(kSpkiHex), "", unhex(kSigHex)));
58}
59
60// A signature whose length != the modulus length is rejected outright.
61TEST(CheatahRsaVerify, RejectsWrongLengthSignature) {
62 std::string sig = unhex(kSigHex);
63 EXPECT_FALSE(rsa::verify_pss_sha256(unhex(kSpkiHex), kMsg, sig.substr(0, sig.size() - 1)));
64 EXPECT_FALSE(rsa::verify_pss_sha256(unhex(kSpkiHex), kMsg, sig + std::string(1, '\0')));
65}

67// A blob with no rsaEncryption SubjectPublicKeyInfo (not an RSA cert) yields no key -> reject.
68TEST(CheatahRsaVerify, RejectsNonRsaCert) {
69 EXPECT_FALSE(rsa::verify_pss_sha256("not a certificate", kMsg, unhex(kSigHex)));
70 EXPECT_FALSE(rsa::verify_pss_sha256("", kMsg, unhex(kSigHex)));
71}

73// A signature representative s that is NOT strictly less than the modulus n is out of
74// range and rejected (RFC 8017 §5.2.2). Using s == n exactly drives the big-integer
75// comparator's equality (all-limbs-equal) return path, and s == n+1 (via the top byte)
76// the greater-than path. The modulus is the 256-byte INTEGER inside kSpkiHex, just
77// after the `0282010100` (INTEGER, 257 bytes incl. sign byte) tag.
78TEST(CheatahRsaVerify, RejectsSignatureNotLessThanModulus) {
79 const std::string spki(kSpkiHex);
80 const std::string marker = "0282010100";
81 const std::size_t at = spki.find(marker) + marker.size();
82 const std::string mod_hex = spki.substr(at, 512); // 256 bytes
83 const std::string n = unhex(mod_hex);
84 ASSERT_EQ(n.size(), 256u);
85 // s == n -> cmp(s, n) == 0 -> out of range.
86 EXPECT_FALSE(rsa::verify_pss_sha256(unhex(kSpkiHex), kMsg, n));
87 // s == n with a bumped low byte (> n) -> cmp(s, n) > 0 -> out of range.
88 std::string bigger = n;
89 // n's least-significant byte is 0x91; incrementing keeps length and makes s > n.
90 bigger.back() = static_cast<char>(static_cast<unsigned char>(bigger.back()) + 1);
91 EXPECT_FALSE(rsa::verify_pss_sha256(unhex(kSpkiHex), kMsg, bigger));
92}
