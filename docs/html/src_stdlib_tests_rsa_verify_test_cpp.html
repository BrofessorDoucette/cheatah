3// Unit tests for tls/rsa_verify.hpp — RSA-PSS (rsa_pss_rsae_sha256) signature verification used by the
4// TLS 1.3 CertificateVerify path. A FIXED vector (a 2048-bit RSA public key + a PSS-SHA256 signature
5// over a known message, produced once with OpenSSL) checks the accept path; the rest check that every
6// way a signature can be wrong is rejected. Complements tls_sys_test.cpp's TlsSys.HandshakeRsaCertificate
7// (which covers the accept path end-to-end against a real server) with the REJECTION paths.

17std::string unhex(std::string_view h) {
18 auto v = [](char c) { return c <= '9' ? c - '0' : (c | 0x20) - 'a' + 10; };
19 std::string o;
20 for (std::size_t i = 0; i + 1 < h.size(); i += 2)
21 o.push_back(static_cast<char>((v(h[i]) << 4) | v(h[i + 1])));
22 return o;
23}

44// The genuine OpenSSL-produced signature verifies against the public key + message.
45TEST(CheatahRsaVerify, AcceptsValidSignature) {
46 EXPECT_TRUE(rsa::verify_pss_sha256(unhex(kSpkiHex), kMsg, unhex(kSigHex)));
47}

49// A single flipped signature byte must be rejected.
50TEST(CheatahRsaVerify, RejectsTamperedSignature) {
51 std::string sig = unhex(kSigHex);
52 sig[100] = static_cast<char>(sig[100] ^ 0x01);
53 EXPECT_FALSE(rsa::verify_pss_sha256(unhex(kSpkiHex), kMsg, sig));
54}

56// The right signature over the WRONG message must be rejected.
57TEST(CheatahRsaVerify, RejectsWrongMessage) {
58 EXPECT_FALSE(rsa::verify_pss_sha256(unhex(kSpkiHex), kMsg + "!", unhex(kSigHex)));
59 EXPECT_FALSE(rsa::verify_pss_sha256(unhex(kSpkiHex), "", unhex(kSigHex)));
60}

62// A signature whose length != the modulus length is rejected outright.
63TEST(CheatahRsaVerify, RejectsWrongLengthSignature) {
64 std::string sig = unhex(kSigHex);
65 EXPECT_FALSE(rsa::verify_pss_sha256(unhex(kSpkiHex), kMsg, sig.substr(0, sig.size() - 1)));
66 EXPECT_FALSE(rsa::verify_pss_sha256(unhex(kSpkiHex), kMsg, sig + std::string(1, '\0')));
67}

69// SECURITY: a key with a dangerous public exponent is rejected before any modexp. With e = 1
70// an attacker forges a signature trivially (m = s^1 = the PSS encoding they can compute), so a
71// real verifier refuses e < 3 and even e. We rewrite the exponent TLV in the SPKI (it ends with
72// `02 03 01 00 01` = 65537) to e = 1 and to e = 65536 (even) and confirm both are refused.
73TEST(CheatahRsaVerify, RejectsUnsafeExponent) {
74 const std::string spki(kSpkiHex);
75 const std::size_t at = spki.rfind("0203010001"); // the trailing exponent INTEGER
76 ASSERT_NE(at, std::string::npos);
77
78 std::string e1 = spki;
79 e1.replace(at, 10, "0203000001"); // exponent value 00 00 01 -> e = 1
80 EXPECT_FALSE(rsa::verify_pss_sha256(unhex(e1), kMsg, unhex(kSigHex)));
81
82 std::string e_even = spki;
83 e_even.replace(at, 10, "0203010000"); // exponent value 01 00 00 -> e = 65536 (even)
84 EXPECT_FALSE(rsa::verify_pss_sha256(unhex(e_even), kMsg, unhex(kSigHex)));
85
86 // Sanity: the untouched key (e = 65537) with its genuine signature still verifies.
87 EXPECT_TRUE(rsa::verify_pss_sha256(unhex(kSpkiHex), kMsg, unhex(kSigHex)));
88}

90// A blob with no rsaEncryption SubjectPublicKeyInfo (not an RSA cert) yields no key -> reject.
91TEST(CheatahRsaVerify, RejectsNonRsaCert) {
92 EXPECT_FALSE(rsa::verify_pss_sha256("not a certificate", kMsg, unhex(kSigHex)));
93 EXPECT_FALSE(rsa::verify_pss_sha256("", kMsg, unhex(kSigHex)));
94}

96// A signature representative s that is NOT strictly less than the modulus n is out of
97// range and rejected (RFC 8017 §5.2.2). Using s == n exactly drives the big-integer
98// comparator's equality (all-limbs-equal) return path, and s == n+1 (via the top byte)
99// the greater-than path. The modulus is the 256-byte INTEGER inside kSpkiHex, just
100// after the `0282010100` (INTEGER, 257 bytes incl. sign byte) tag.
101TEST(CheatahRsaVerify, RejectsSignatureNotLessThanModulus) {
102 const std::string spki(kSpkiHex);
103 const std::string marker = "0282010100";
104 const std::size_t at = spki.find(marker) + marker.size();
105 const std::string mod_hex = spki.substr(at, 512); // 256 bytes
106 const std::string n = unhex(mod_hex);
107 ASSERT_EQ(n.size(), 256u);
108 // s == n -> cmp(s, n) == 0 -> out of range.
109 EXPECT_FALSE(rsa::verify_pss_sha256(unhex(kSpkiHex), kMsg, n));
110 // s == n with a bumped low byte (> n) -> cmp(s, n) > 0 -> out of range.
111 std::string bigger = n;
112 // n's least-significant byte is 0x91; incrementing keeps length and makes s > n.
113 bigger.back() = static_cast<char>(static_cast<unsigned char>(bigger.back()) + 1);
114 EXPECT_FALSE(rsa::verify_pss_sha256(unhex(kSpkiHex), kMsg, bigger));
115}
