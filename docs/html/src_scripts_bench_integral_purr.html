1# Copyright (c) 2026 BigBrain LLC. MIT-licensed (see LICENSE).
2# Original work; see ACKNOWLEDGMENTS.md for the open-source ideas we build upon.
3import io
4import time
5import math
6let a = 0.0
7let b = 50.0
8let n = 20000000
9let h = (b - a) / n
10let s = 0.5 * (math.sin(a) * math.exp(-0.01 * a) + math.sin(b) * math.exp(-0.01 * b))
11let t0 = time.monotonic()
12for i in range(1, n) {
13 let x = a + h * i
14 s = s + math.sin(x) * math.exp(-0.01 * x)
15}
16let t1 = time.monotonic()
17io.print(s * h)
18io.print(t1 - t0)
