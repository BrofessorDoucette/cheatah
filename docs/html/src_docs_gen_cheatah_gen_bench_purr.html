docs/gen-cheatah/gen_bench.purr

1# Copyright (c) 2026 BigBrain LLC. MIT-licensed (see LICENSE).
2# Original work; see ACKNOWLEDGMENTS.md for the open-source ideas we build upon.
3# gen_bench.purr — times the pure-cheatah doc-render work (parse every Doxygen namespace
4# XML + build each module's member HTML), for a like-for-like comparison against the same
5# task in Python (docs/gen-cheatah/gen_bench.py). Compute-only: files are read up front.
6import io
7import os
8import time
9import string
10import parsers.xml
11
12fn child_text(d, id, tag) { return parsers.xml.text(d, parsers.xml.find(d, id, tag)) }
13
14fn esc(s) {
15 let r = string.replace(s, "&", "&amp;")
16 r = string.replace(r, "<", "&lt;")
17 return string.replace(r, ">", "&gt;")
18}
19
20fn render(src) {
21 let d = parsers.xml.parse(src)
22 let r = parsers.xml.root(d)
23 let out = ""
24 for cd in parsers.xml.iter(d, r, "compounddef") {
25 out = out + "<h1><code>" + esc(child_text(d, cd, "compoundname")) + "</code></h1>\n"
26 for md in parsers.xml.iter(d, cd, "memberdef") {
27 let brief = esc(child_text(d, md, "briefdescription"))
28 out = out + "<section class=\"member\"><span class=\"badge\">"
29 out = out + parsers.xml.attr(d, md, "kind") + "</span> <code>"
30 out = out + esc(child_text(d, md, "name")) + esc(child_text(d, md, "argsstring"))
31 out = out + "</code><p>" + brief + "</p></section>\n"
32 }
33 }
34 return out
35}
36
37fn main() {
38 let dir = "docs/xml"
39 let srcs: list<str> = []
40 for f in os.listdir(dir) {
41 if string.startswith(f, "namespacecheatah_") and string.endswith(f, ".xml") {
42 srcs.append(io.read_file(os.path.join(dir, f)))
43 }
44 }
45 # ONE timed pass per invocation. gen_bench_compare.purr alternates this process with the
46 # CPython one round by round, so neither side is measured as a consecutive block while the
47 # machine drifts. It also prints every sample rather than a best-of-N minimum: a minimum
48 # has no dispersion, so a steady case and a wildly variable one look identical.
49 let t0 = time.perf_counter()
50 let bytes = 0
51 for s in srcs { bytes = bytes + len(render(s)) }
52 let dt = time.perf_counter() - t0
53 io.print("sample_ms=" + io.str(dt * 1000.0))
54 io.print("cheatah files=" + io.str(len(srcs)) + " out_bytes=" + io.str(bytes))
55}
56
57main()
