55std::string to_hex(const std::uint8_t* p, std::size_t n) {
56 static const char hexd[] = "0123456789abcdef";
57 std::string out;
58 out.reserve(n * 2);
59 for (std::size_t i = 0; i < n; ++i) {
60 out.push_back(hexd[p[i] >> 4]);
61 out.push_back(hexd[p[i] & 0xF]);
62 }
63 return out;
64}

66// SHA-256 core: returns the raw 32-byte digest.
67std::array<std::uint8_t, 32> sha256_raw(std::string_view data) {
68 std::uint32_t h[8] = {0x6a09e667, 0xbb67ae85, 0x3c6ef372, 0xa54ff53a,
69 0x510e527f, 0x9b05688c, 0x1f83d9ab, 0x5be0cd19};
70
71 // Pad: append 0x80, then zeros, then the 64-bit big-endian bit length.
72 const std::uint64_t bitlen = static_cast<std::uint64_t>(data.size()) * 8;
73 std::vector<std::uint8_t> msg(data.begin(), data.end());
74 msg.push_back(0x80);
75 while (msg.size() % 64 != 56) msg.push_back(0x00);
76 for (int i = 7; i >= 0; --i) msg.push_back(static_cast<std::uint8_t>(bitlen >> (i * 8)));
77
78 for (std::size_t off = 0; off < msg.size(); off += 64) {
79 std::uint32_t w[64];
80 for (int i = 0; i < 16; ++i) {
81 w[i] = (static_cast<std::uint32_t>(msg[off + i * 4]) << 24) |
82 (static_cast<std::uint32_t>(msg[off + i * 4 + 1]) << 16) |
83 (static_cast<std::uint32_t>(msg[off + i * 4 + 2]) << 8) |
84 (static_cast<std::uint32_t>(msg[off + i * 4 + 3]));
85 }
86 for (int i = 16; i < 64; ++i) {
87 const std::uint32_t s0 = rotr(w[i - 15], 7) ^ rotr(w[i - 15], 18) ^ (w[i - 15] >> 3);
88 const std::uint32_t s1 = rotr(w[i - 2], 17) ^ rotr(w[i - 2], 19) ^ (w[i - 2] >> 10);
89 w[i] = w[i - 16] + s0 + w[i - 7] + s1;
90 }
91
92 std::uint32_t a = h[0], b = h[1], c = h[2], d = h[3];
93 std::uint32_t e = h[4], f = h[5], g = h[6], hh = h[7];
94 for (int i = 0; i < 64; ++i) {
95 const std::uint32_t S1 = rotr(e, 6) ^ rotr(e, 11) ^ rotr(e, 25);
96 const std::uint32_t ch = (e & f) ^ (~e & g);
97 const std::uint32_t t1 = hh + S1 + ch + K[i] + w[i];
98 const std::uint32_t S0 = rotr(a, 2) ^ rotr(a, 13) ^ rotr(a, 22);
99 const std::uint32_t maj = (a & b) ^ (a & c) ^ (b & c);
100 const std::uint32_t t2 = S0 + maj;
101 hh = g; g = f; f = e; e = d + t1;
102 d = c; c = b; b = a; a = t1 + t2;
103 }
104 h[0] += a; h[1] += b; h[2] += c; h[3] += d;
105 h[4] += e; h[5] += f; h[6] += g; h[7] += hh;
106 }
107
108 std::array<std::uint8_t, 32> out{};
109 for (int i = 0; i < 8; ++i)
110 for (int j = 0; j < 4; ++j) out[i * 4 + j] = static_cast<std::uint8_t>(h[i] >> ((3 - j) * 8));
111 return out;
112}

114// SHA-512 core: returns the raw 64-byte digest.
115std::array<std::uint8_t, 64> sha512_raw(std::string_view data) {
116 std::uint64_t h[8] = {0x6a09e667f3bcc908ULL, 0xbb67ae8584caa73bULL, 0x3c6ef372fe94f82bULL,
117 0xa54ff53a5f1d36f1ULL, 0x510e527fade682d1ULL, 0x9b05688c2b3e6c1fULL,
118 0x1f83d9abfb41bd6bULL, 0x5be0cd19137e2179ULL};
119
120 // Pad: 0x80, zeros to a 128-byte block leaving 16 bytes, then the 128-bit big-endian
121 // bit length (the high 64 bits are always 0 for our inputs).
122 const std::uint64_t bitlen = static_cast<std::uint64_t>(data.size()) * 8;
123 std::vector<std::uint8_t> msg(data.begin(), data.end());
124 msg.push_back(0x80);
125 while (msg.size() % 128 != 112) msg.push_back(0x00);
126 for (int i = 0; i < 8; ++i) msg.push_back(0x00); // high 64 bits of the length
127 for (int i = 7; i >= 0; --i) msg.push_back(static_cast<std::uint8_t>(bitlen >> (i * 8)));
128
129 for (std::size_t off = 0; off < msg.size(); off += 128) {
130 std::uint64_t w[80];
131 for (int i = 0; i < 16; ++i) {
132 w[i] = 0;
133 for (int j = 0; j < 8; ++j)
134 w[i] = (w[i] << 8) | msg[off + i * 8 + j];
135 }
136 for (int i = 16; i < 80; ++i) {
137 const std::uint64_t s0 = rotr64(w[i - 15], 1) ^ rotr64(w[i - 15], 8) ^ (w[i - 15] >> 7);
138 const std::uint64_t s1 = rotr64(w[i - 2], 19) ^ rotr64(w[i - 2], 61) ^ (w[i - 2] >> 6);
139 w[i] = w[i - 16] + s0 + w[i - 7] + s1;
140 }
141
142 std::uint64_t a = h[0], b = h[1], c = h[2], d = h[3];
143 std::uint64_t e = h[4], f = h[5], g = h[6], hh = h[7];
144 for (int i = 0; i < 80; ++i) {
145 const std::uint64_t S1 = rotr64(e, 14) ^ rotr64(e, 18) ^ rotr64(e, 41);
146 const std::uint64_t ch = (e & f) ^ (~e & g);
147 const std::uint64_t t1 = hh + S1 + ch + K512[i] + w[i];
148 const std::uint64_t S0 = rotr64(a, 28) ^ rotr64(a, 34) ^ rotr64(a, 39);
149 const std::uint64_t maj = (a & b) ^ (a & c) ^ (b & c);
150 const std::uint64_t t2 = S0 + maj;
151 hh = g; g = f; f = e; e = d + t1;
152 d = c; c = b; b = a; a = t1 + t2;
153 }
154 h[0] += a; h[1] += b; h[2] += c; h[3] += d;
155 h[4] += e; h[5] += f; h[6] += g; h[7] += hh;
156 }
157
158 std::array<std::uint8_t, 64> out{};
159 for (int i = 0; i < 8; ++i)
160 for (int j = 0; j < 8; ++j) out[i * 8 + j] = static_cast<std::uint8_t>(h[i] >> ((7 - j) * 8));
161 return out;
162}
