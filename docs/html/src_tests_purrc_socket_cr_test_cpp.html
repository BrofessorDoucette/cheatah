1// Compile-run unit tests for the `socket` module: one test per function. Each
2// writes a tiny .purr that calls a single socket function, compiles it with
3// purrc, runs it under the cheatah runtime, and asserts the exact stdout.
4// Complements the in-process unit tests (stdlib/tests/socket_test.cpp) and the
5// per-module system-level test (StdlibE2E.Socket).
6//
7// Real networking has no fixed output, so each program asserts DETERMINISTIC
8// PROPERTIES as booleans (io.print emits "True"/"False"): bind to 127.0.0.1
9// port 0 (OS-assigned) so nothing depends on a fixed port or external network,
10// connect to port 1 (nothing listens there) for the refusal paths, and use a
11// bad fd (-1) for the error paths. Functions that block waiting on a peer are
12// exercised only through their non-blocking bad-fd error path.
13#include "e2e_harness.hpp"
14
15// tcp_listen: binding a listener on 127.0.0.1:0 yields a valid (>= 0) fd.
16TEST(SocketCompileRun, TcpListen) {
17 e2e::expect_e2e("socket_tcp_listen", R"PURR(import io
18import socket
19let fd = socket.tcp_listen("127.0.0.1", 0, 1)
20io.print(fd >= 0)
21socket.close(fd)
22)PURR", "True\n");
23}
