1// Copyright (c) 2026 BigBrain LLC. MIT-licensed (see LICENSE).
2// Original work; see ACKNOWLEDGMENTS.md for the open-source ideas we build upon.
3// Compile-run unit tests for the `socket` module: one test per function. Each
4// writes a tiny .purr that calls a single socket function, compiles it with
5// purrc, runs it under the cheatah runtime, and asserts the exact stdout.
6// Complements the in-process unit tests (stdlib/tests/socket_test.cpp) and the
7// per-module system-level test (StdlibE2E.Socket).
8//
9// Real networking has no fixed output, so each program asserts DETERMINISTIC
10// PROPERTIES as booleans (io.print emits "True"/"False"): bind to 127.0.0.1
11// port 0 (OS-assigned) so nothing depends on a fixed port or external network,
12// connect to port 1 (nothing listens there) for the refusal paths, and use a
13// bad fd (-1) for the error paths. Functions that block waiting on a peer are
14// exercised only through their non-blocking bad-fd error path.
15#include "e2e_harness.hpp"
16
17// tcp_listen: binding a listener on 127.0.0.1:0 yields a valid (>= 0) fd.
18TEST(SocketCompileRun, TcpListen) {
19 e2e::expect_e2e("socket_tcp_listen", R"PURR(import io
20import socket
21let fd = socket.tcp_listen("127.0.0.1", 0, 1)
22io.print(fd >= 0)
23socket.close(fd)
24)PURR", "True\n");
25}

64// set_reuseaddr: enabling SO_REUSEADDR on a fresh socket succeeds (returns 0).
65// udp_socket / sendto / recvfrom: a datagram round trip from purr, the sender's address reported.
66TEST(SocketCompileRun, UdpLoopback) {
67 e2e::expect_e2e("socket_udp_loopback", R"PURR(import io
68import socket
69let rx = socket.udp_socket()
70socket.bind(rx, "127.0.0.1", 0)
71socket.set_timeout(rx, 500)
72let port = socket.local_port(rx)
73let tx = socket.udp_socket()
74let sent = socket.sendto(tx, "127.0.0.1", port, "pose 7")
75let host = ""
76let from_port = 0
77let got = socket.recvfrom(rx, 64, host, from_port)
78io.print(sent == 6 and got == "pose 7" and host == "127.0.0.1" and from_port > 0)
79socket.close(tx)
80socket.close(rx)
81)PURR", "True\n");
82}
