80/// StringViewable<T>: a `std::string_view` can be built from T — what join() needs.
81template <typename T>
82concept StringViewable = requires(const T& v) { std::string_view(v); };
83
84/** Join @p parts with @p sep. @param sep separator. @param parts any range of string-like values. @return the joined string. @note O(total length); allocates the result. @test CheatahString.Transform */
85template <std::ranges::input_range Range>
86 requires StringViewable<std::ranges::range_value_t<Range>>
87std::string join(std::string_view sep, const Range& parts) {
88 std::string out;
89 bool first = true;
90 for (const auto& part : parts) {
91 if (!first) {
92 out += sep;
93 }
94 out += std::string_view(part);
95 first = false;
96 }
97 return out;
98}
