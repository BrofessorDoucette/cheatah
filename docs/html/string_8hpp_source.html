206/// StringViewable<T>: a `std::string_view` can be built from T — what join() needs.
207template <typename T>
208concept StringViewable = requires(const T& v) { std::string_view(v); };
209
210/**
211 * Join @p parts with @p sep.
212 * @param sep separator.
213 * @param parts any range of string-like values.
214 * @return the joined string.
215 * @note O(total length); allocates the result.
216 * @test CheatahString.Transform
217 */
218template <std::ranges::input_range Range>
219 requires StringViewable<std::ranges::range_value_t<Range>>
220std::string join(std::string_view sep, const Range& parts) {
221 std::string out;
222 bool first = true;
223 for (const auto& part : parts) {
224 if (!first) {
225 out += sep;
226 }
227 out += std::string_view(part);
228 first = false;
229 }
230 return out;
231}
