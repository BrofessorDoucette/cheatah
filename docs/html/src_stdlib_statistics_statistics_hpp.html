3/**
4 * @file statistics.hpp
5 * @brief cheatah `statistics` — descriptive statistics over numeric sequences,
6 * mirroring Python's `statistics` module. `import statistics` to use it.
7 *
8 * Header-only: every function is a template constrained by the `NumericRange`
9 * concept, so it accepts any iterable of arithmetic values (`list[float]`,
10 * `array[int]`, …) and is instantiated at the call site. Unit tests:
11 * `stdlib/tests/statistics_test.cpp`; the suite runs under AddressSanitizer (the
12 * `asan` preset) and Valgrind (`security/run-valgrind.sh`) on every QA-gate run.
13 *
14 * @note `n` below is the element count. Single-pass reductions are O(n) and
15 * allocation-free; `median` is the exception (it materializes and sorts a
16 * copy). Results are scalar `double`/`std::size_t` — no heap on return.
17 */
18#include <algorithm>
19#include <cmath>
20#include <cstddef>
21#include <ranges>

27/// NumericRange<R>: an iterable of arithmetic values (list[float], array[int], …).
28template <typename R>
29concept NumericRange =
30 std::ranges::input_range<R> && std::is_arithmetic_v<std::ranges::range_value_t<R>>;
31
32/**
33 * Sum of the elements.
34 *
35 * Accumulates every element into a `double`, so an empty range sums to 0.0 and
36 * integer inputs are widened before adding (no integer overflow).
37 * @param data the numeric range.
38 * @return Σ@p data as `double`.
39 * @complexity O(n) single pass.
40 * @alloc none.
41 * @test CheatahStatistics.SumCountMean
42 * @crtest StatisticsCompileRun.Sum
43 * @systest StdlibE2E.Statistics
44 */
45template <NumericRange R>
46double sum(const R& data) {
47 double s = 0.0;
48 for (const auto& x : data) s += static_cast<double>(x);
49 return s;
50}
51
52/**
53 * Element count.
54 * @param data the numeric range.
55 * @return the number of elements.
56 * @complexity O(n) single pass.
57 * @alloc none.
58 * @test CheatahStatistics.SumCountMean
59 * @crtest StatisticsCompileRun.Count
60 * @systest StdlibE2E.Statistics
61 */
62template <NumericRange R>
63std::size_t count(const R& data) {
64 std::size_t n = 0;
65 for (const auto& x : data) {
66 (void)x;
67 ++n;
68 }
69 return n;
70}

72/**
73 * Arithmetic mean.
74 *
75 * Computes sum/count, but guards division by zero: an empty range returns 0.0
76 * rather than NaN.
77 * @param data the numeric range.
78 * @return the mean, or 0.0 if empty.
79 * @complexity O(n) (two passes: count + sum).
80 * @alloc none.
81 * @test CheatahStatistics.SumCountMean
82 * @crtest StatisticsCompileRun.Mean
83 * @systest StdlibE2E.Statistics
84 */
85template <NumericRange R>
86double mean(const R& data) {
87 const std::size_t n = count(data);
88 return n == 0 ? 0.0 : sum(data) / static_cast<double>(n);
89}

91/**
92 * Population variance (divide by N).
93 *
94 * Mean of the squared deviations from the mean, dividing by N (treats @p data as
95 * the entire population). Returns 0.0 for an empty range; a single element yields
96 * 0.0.
97 * @param data the numeric range.
98 * @return the variance, or 0.0 if empty.
99 * @complexity O(n).
100 * @alloc none.
101 * @test CheatahStatistics.PopulationVarianceAndStdev
102 * @crtest StatisticsCompileRun.Pvariance
103 * @systest StdlibE2E.Statistics
104 */
105template <NumericRange R>
106double pvariance(const R& data) {
107 const double m = mean(data);
108 double s = 0.0;
109 std::size_t n = 0;
110 for (const auto& x : data) {
111 const double d = static_cast<double>(x) - m;
112 s += d * d;
113 ++n;
114 }
115 return n == 0 ? 0.0 : s / static_cast<double>(n);
116}
117/**
118 * Population standard deviation.
119 *
120 * Square root of @ref pvariance, so it is 0.0 for empty or single-element ranges
121 * and never negative.
122 * @param data the numeric range.
123 * @return √pvariance(@p data).
124 * @complexity O(n).
125 * @alloc none.
126 * @test CheatahStatistics.PopulationVarianceAndStdev
127 * @crtest StatisticsCompileRun.Pstdev
128 * @systest StdlibE2E.Statistics
129 */
130template <NumericRange R>
131double pstdev(const R& data) { return std::sqrt(pvariance(data)); }
132
133/**
134 * Sample variance (divide by N−1).
135 *
136 * Sum of squared deviations from the mean divided by N−1 (Bessel's correction,
137 * estimating the variance of the wider population from a sample). Requires at
138 * least two elements; an empty or single-element range returns 0.0 rather than
139 * dividing by zero.
140 * @param data the numeric range.
141 * @return the variance, or 0.0 if fewer than 2 elements.
142 * @complexity O(n).
143 * @alloc none.
144 * @test CheatahStatistics.SampleVarianceAndStdev
145 * @crtest StatisticsCompileRun.Variance
146 * @systest StdlibE2E.Statistics
147 */
148template <NumericRange R>
149double variance(const R& data) {
150 const double m = mean(data);
151 double s = 0.0;
152 std::size_t n = 0;
153 for (const auto& x : data) {
154 const double d = static_cast<double>(x) - m;
155 s += d * d;
156 ++n;
157 }
158 return n > 1 ? s / static_cast<double>(n - 1) : 0.0;
159}
160/**
161 * Sample standard deviation.
162 *
163 * Square root of @ref variance, so it is 0.0 when there are fewer than two
164 * elements.
165 * @param data the numeric range.
166 * @return √variance(@p data).
167 * @complexity O(n).
168 * @alloc none.
169 * @test CheatahStatistics.SampleVarianceAndStdev
170 * @crtest StatisticsCompileRun.Stdev
171 * @systest StdlibE2E.Statistics
172 */
173template <NumericRange R>
174double stdev(const R& data) { return std::sqrt(variance(data)); }

176/**
177 * Median (mean of the two middle values when the count is even).
178 *
179 * Copies the elements into a `double` vector and sorts ascending: with an odd
180 * count it returns the single middle value, and with an even count it averages
181 * the two central values. Returns 0.0 for an empty range.
182 * @param data the numeric range.
183 * @return the median, or 0.0 if empty.
184 * @complexity O(n log n) — copies the elements into a vector and sorts.
185 * @alloc allocates a temporary `std::vector<double>`.
186 * @test CheatahStatistics.MedianOddAndEven
187 * @crtest StatisticsCompileRun.Median
188 * @systest StdlibE2E.Statistics
189 */
190template <NumericRange R>
191double median(const R& data) {
192 std::vector<double> v;
193 for (const auto& x : data) v.push_back(static_cast<double>(x));
194 if (v.empty()) return 0.0;
195 std::sort(v.begin(), v.end());
196 const std::size_t n = v.size();
197 return (n % 2 == 1) ? v[n / 2] : (v[n / 2 - 1] + v[n / 2]) / 2.0;
198}
