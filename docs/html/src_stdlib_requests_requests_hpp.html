136/**
137 * Percent-encode `text` for a query string.
138 *
139 * RFC 3986 unreserved characters pass through; everything else (including space)
140 * becomes %XX with uppercase hex digits.
141 * @param text the raw query-parameter name or value.
142 * @return the percent-encoded form, safe to place in a request-target.
143 * @complexity O(n) over the input length.
144 * @alloc allocates the result.
145 * @systest RequestsSys.QueryParams
146 */
147auto percent_encode(builtins::Value auto&& text) {
148 auto unreserved = std::string("ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789-._~");
149 auto hexdigits = std::string("0123456789ABCDEF");
150 auto out = std::string("");
151 for (auto& ch : text) {
152 if (string::contains(unreserved, ch)) {
153 out += ch;
154 } else {
155 auto v = builtins::ord(ch);
156 ((out += "%") += builtins::index(hexdigits, builtins::floordiv(v, 16LL))) += builtins::index(hexdigits, builtins::mod(v, 16LL));
157 }
158 }
159 return out;
160}

163/**
164 * Parse one hex chunk-size token (e.g. "1aF").
165 *
166 * @param text the hex token, both digit cases accepted, no prefix.
167 * @return the decoded size, or -1 on a malformed digit. An empty token decodes to 0,
168 * which the chunked framing treats as the final chunk.
169 * @complexity O(n) over the token length.
170 * @alloc none.
171 * @systest RequestsSys.Chunked
172 */
173auto parse_hex(builtins::Value auto&& text) {
174 auto n = 0LL;
175 for (auto& ch : text) {
176 auto v = builtins::ord(ch);
177 if (((v >= 48LL) && (v <= 57LL))) {
178 n = ((n * 16LL) + (v - 48LL));
179 } else {
180 if (((v >= 97LL) && (v <= 102LL))) {
181 n = ((n * 16LL) + (v - 87LL));
182 } else {
183 if (((v >= 65LL) && (v <= 70LL))) {
184 n = ((n * 16LL) + (v - 55LL));
185 } else {
186 return (-1LL);
187 }
188 }
189 }
190 }
191 return n;
192}

195/**
196 * Decode an HTTP/1.1 chunked body.
197 *
198 * Framing: hex-size line (chunk extensions after `;` ignored), chunk bytes, CRLF,
199 * repeated until the 0 chunk. Trailers after the 0 chunk are simply unread.
200 * @param raw everything after the response head (the body bytes as received).
201 * @param r the Response under construction; `r.error` is set on malformed framing.
202 * @return the decoded body, or "" when framing was malformed (see `r.error`).
203 * @complexity O(n) over the body length.
204 * @alloc allocates the decoded body.
205 * @systest RequestsSys.Chunked
206 */
207auto dechunk(builtins::Value auto&& raw, builtins::Value auto&& r) {
208 auto body = std::string("");
209 auto pos = 0LL;
210 while (true) {
211 auto line_end = string::find(builtins::slice(raw, pos, builtins::slice_end), std::string("\r\n"));
212 if ((line_end < 0LL)) {
213 r.error = std::string("connection closed inside chunked body");
214 return std::string("");
215 }
216 auto size_token = builtins::slice(raw, pos, (pos + line_end));
217 auto semi = string::find(size_token, std::string(";"));
218 if ((semi >= 0LL)) {
219 size_token = builtins::slice(size_token, 0LL, semi);
220 }
221 auto size = parse_hex(string::strip(size_token));
222 if ((size < 0LL)) {
223 r.error = std::string("malformed chunk size");
224 return std::string("");
225 }
226 (pos += line_end) += 2LL;
227 if ((size == 0LL)) {
228 return body;
229 }
230 if ((builtins::len(raw) < ((pos + size) + 2LL))) {
231 r.error = std::string("connection closed inside chunked body");
232 return std::string("");
233 }
234 body += builtins::slice(raw, pos, (pos + size));
235 (pos += size) += 2LL;
236 }
237}

241 * Read the whole response from a connected socket (or TLS session for https).
242 *
243 * Reads every byte until the peer closes — or the socket timeout set by get_once
244 * fires, which recv reports as "" (indistinguishable from close by design: we always
245 * request `Connection: close`, so EOF IS the end of the response). When @p conn is open
246 * the bytes are read (and decrypted) through its TLS session, mirroring the send path;
247 * a closed @p conn reads the plain socket.
248 * @param fd a connected socket descriptor from socket.tcp_connect.
249 * @param conn the owning tls.Conn for https (open), or a closed Conn for plain http.
250 * @return all bytes received.
251 * @complexity O(n) over the response size.
252 * @alloc allocates the received buffer.
253 * @systest RequestsSys.BasicGet
254 */
255auto read_all(builtins::Value auto&& fd, builtins::Value auto&& conn) {
256 auto raw = std::string("");
257 while (true) {
258 auto chunk = std::string("");
259 if (conn.is_open()) {
260 chunk = conn.recv(65536LL);
261 } else {
262 chunk = socket::recv(fd, 65536LL);
263 }
264 if ((builtins::len(chunk) == 0LL)) {
265 break;
266 }
267 raw += chunk;
268 }
269 return raw;
270}

273/**
274 * Parse a raw HTTP/1.1 response (status line + headers + body) into `r`.
275 *
276 * Headers land in `r.headers` with LOWERCASED names and stripped values. Body framing
277 * precedence: chunked when declared, else Content-Length, else everything to EOF (we
278 * always send `Connection: close`). Malformed input sets `r.error` and returns early.
279 * @param raw the complete response bytes as received.
280 * @param r the Response under construction (status/headers/body/error are filled in).
281 * @return `r`, completed or carrying `error`.
282 * @complexity O(n) over the response size.
283 * @alloc allocates the parsed headers and body.
284 * @systest RequestsSys.EofFraming
285 * @systest RequestsSys.HeaderLookup
286 */
287auto parse_response(builtins::Value auto&& raw, builtins::Value auto&& r) {
288 auto head_end = string::find(raw, std::string("\r\n\r\n"));
289 if ((head_end < 0LL)) {
290 r.error = std::string("connection closed before a complete response head");
291 return r;
292 }
293 if ((string::find(raw, std::string("HTTP/")) != 0LL)) {
294 r.error = std::string("malformed response head");
295 return r;
296 }
297 auto head = builtins::slice(raw, 0LL, head_end);
298 auto line_end = string::find(head, std::string("\r\n"));
299 if ((line_end < 0LL)) {
300 line_end = builtins::len(head);
301 }
302 auto space = string::find(head, std::string(" "));
303 if (((space < 0LL) || ((space + 4LL) > line_end))) {
304 r.error = std::string("malformed status line");
305 return r;
306 }
307 r.status = builtins::to_int(builtins::slice(head, (space + 1LL), (space + 4LL)));
308 auto rest = builtins::slice(head, (line_end + 2LL), builtins::slice_end);
309 while ((builtins::len(rest) > 0LL)) {
310 auto eol = string::find(rest, std::string("\r\n"));
311 if ((eol < 0LL)) {
312 eol = builtins::len(rest);
313 }
314 auto line = builtins::slice(rest, 0LL, eol);
315 auto colon = string::find(line, std::string(":"));
316 if ((colon > 0LL)) {
317 auto name = string::lower(string::strip(builtins::slice(line, 0LL, colon)));
318 r.headers[name] = string::strip(builtins::slice(line, (colon + 1LL), builtins::slice_end));
319 }
320 if ((eol == builtins::len(rest))) {
321 break;
322 }
323 rest = builtins::slice(rest, (eol + 2LL), builtins::slice_end);
324 }
325 auto body = builtins::slice(raw, (head_end + 4LL), builtins::slice_end);
326 if ((r.header(std::string("transfer-encoding")) == std::string("chunked"))) {
327 r.body = dechunk(body, r);
328 return r;
329 }
330 auto cl = r.header(std::string("content-length"));
331 if ((cl != std::string(""))) {
332 auto n = builtins::to_int(cl);
333 if ((builtins::len(body) < n)) {
334 r.error = std::string("connection closed before the complete body arrived");
335 return r;
336 }
337 r.body = builtins::slice(body, 0LL, n);
338 } else {
339 r.body = body;
340 }
341 return r;
342}

345/**
346 * One GET exchange against an already-parsed URL (no redirect handling).
347 *
348 * Connects, applies the timeout, appends `o.params` percent-encoded to the request-target,
349 * sends the request (custom headers, default User-Agent, `Connection: close`), then reads
350 * and parses the full response. Transport failures come back in `r.error`.
351 * @param u the parsed URL (scheme/host/port/target) from parsers.url.
352 * @param o per-request options (timeout, params, headers).
353 * @param r the Response under construction.
354 * @return the completed Response (a non-2xx status is a completed exchange, not an error).
355 * @alloc allocates the request and response buffers.
356 * @systest RequestsSys.BasicGet
357 */
358auto get_once(builtins::Value auto&& u, builtins::Value auto&& o, builtins::Value auto&& r) {
359 auto fd = socket::tcp_connect(u.host, u.port);
360 if ((fd < 0LL)) {
361 r.error = (((((std::string("connect to ") + builtins::str(u.host)) + std::string(":")) + builtins::str(u.port)) + std::string(" failed: ")) + builtins::str(socket::last_error()));
362 return r;
363 }
364 auto timeout = o.timeout_ms;
365 if ((timeout <= 0LL)) {
366 timeout = 30000LL;
367 }
368 socket::set_timeout(fd, timeout);
369 auto conn = tls::Conn();
370 if ((u.scheme == std::string("https"))) {
371 conn = tls::open(fd, u.host);
372 if ((!conn.is_open())) {
373 socket::close(fd);
374 r.error = (std::string("tls: ") + builtins::str(tls::last_error()));
375 return r;
376 }
377 }
378 auto target = u.target;
379 auto sep = std::string("?");
380 if ((string::find(target, std::string("?")) >= 0LL)) {
381 sep = std::string("&");
382 }
383 for (auto& kv : o.params) {
384 (((target += sep) += percent_encode(kv.first)) += "=") += percent_encode(kv.second);
385 sep = std::string("&");
386 }
387 auto req = ((((((std::string("GET ") + builtins::str(target)) + std::string(" HTTP/1.1\r\nHost: ")) + builtins::str(u.host)) + std::string(":")) + builtins::str(u.port)) + std::string("\r\n"));
388 for (auto& kv : o.headers) {
389 (((req += kv.first) += ": ") += kv.second) += "\r\n";
390 }
391 if ((!(builtins::contains(o.headers, std::string("user-agent")) || builtins::contains(o.headers, std::string("User-Agent"))))) {
392 req += "User-Agent: cheatah-requests/0.2\r\n";
393 }
394 req += "Connection: close\r\nAccept: */*\r\n\r\n";
395 auto sent = 0LL;
396 if (conn.is_open()) {
397 sent = conn.send(req);
398 } else {
399 sent = socket::sendall(fd, req);
400 }
401 if ((sent != 0LL)) {
402 conn.close();
403 socket::close(fd);
404 r.error = std::string("send failed");
405 return r;
406 }
407 auto raw = read_all(fd, conn);
408 conn.close();
409 socket::close(fd);
410 return parse_response(raw, r);
411}

414/**
415 * Perform an HTTP GET, following up to max_redirects 3xx hops.
416 *
417 * Never raises for network conditions: every failure comes back as a Response with
418 * `error` set (and status 0). Redirects (301/302/303/307/308) follow absolute and
419 * host-relative Location targets; `https://` is reported in `error` until cheatah TLS
420 * lands. Call with a URL alone, or pass Options for params/headers/timeout/redirects.
421 * @param url the absolute `http://host[:port]/path[?query]` URL.
422 * @param o per-request options; defaults to a 30 s timeout and 5 redirect hops.
423 * @return the final Response — check `ok()`, then `status`/`headers`/`body`.
424 * @systest RequestsSys.BasicGet
425 * @systest RequestsSys.Redirect
426 * @systest RequestsSys.RedirectLoop
427 */
428auto get(builtins::Value auto&& url, builtins::Value auto&& o) {
429 auto max_hops = o.max_redirects;
430 if ((max_hops <= 0LL)) {
431 max_hops = 5LL;
432 }
433 auto current = url;
434 auto hop = 0LL;
435 while ((hop <= max_hops)) {
436 auto r = Response{.url = static_cast<std::string>(current)};
437 auto parser = parsers::url::Parser();
438 auto u = parsers::url::Url();
439 if ((!parser.parse(current, u))) {
440 r.error = (std::string("malformed URL: ") + builtins::str(current));
441 return r;
442 }
443 r = get_once(u, o, r);
444 auto redirect = ((r.error == std::string("")) && (((((r.status == 301LL) || (r.status == 302LL)) || (r.status == 303LL)) || (r.status == 307LL)) || (r.status == 308LL)));
445 if ((!redirect)) {
446 return r;
447 }
448 auto loc = r.header(std::string("location"));
449 if ((loc == std::string(""))) {
450 r.error = ((std::string("redirect (") + builtins::str(r.status)) + std::string(") without a Location header"));
451 return r;
452 }
453 if ((string::startswith(loc, std::string("http://")) || string::startswith(loc, std::string("https://")))) {
454 current = loc;
455 } else {
456 if (string::startswith(loc, std::string("/"))) {
457 current = (((((builtins::str(u.scheme) + std::string("://")) + builtins::str(u.host)) + std::string(":")) + builtins::str(u.port)) + builtins::str(loc));
458 } else {
459 r.error = (std::string("unsupported relative redirect Location: ") + builtins::str(loc));
460 return r;
461 }
462 }
463 hop += 1LL;
464 }
465 auto r = Response{.url = static_cast<std::string>(current)};
466 r.error = ((std::string("too many redirects (max_redirects = ") + builtins::str(max_hops)) + std::string(")"));
467 return r;
468}
