277 * Does `text` contain a byte that would break out of the line it is written on?
278 *
279 * The request is built by concatenating a request-target and header values into a CRLF-framed
280 * message, so a CR or LF reaching either one lets a caller inject headers or split the request
281 * entirely. That matters most when the value is not the caller's own: a URL taken from a fetched
282 * document or a `Location` header is attacker-controlled data, and nothing else on the path
283 * re-checks it. Refusing here means no consumer of this module can be made to emit a forged
284 * request, whatever it was handed.
285 *
286 * @param text a request-target or header value about to be written to the wire.
287 * @return true when `text` carries CR or LF and must not be sent.
288 * @complexity O(n) over the input length.
289 * @alloc none.
290 * @systest CheatahRequests.CrlfInjectionRefused
291 */
292inline auto has_control_bytes(builtins::Value auto&& text) {
293 return ((string::find(text, std::string("\r")) >= 0LL) || (string::find(text, std::string("\n")) >= 0LL));
294}

297/**
298 * Percent-encode `text` for a query string or form body.
299 *
300 * RFC 3986 unreserved characters pass through; everything else (including space)
301 * becomes %XX with uppercase hex digits.
302 * @param text the raw name or value.
303 * @return the percent-encoded form, safe to place in a request-target or form body.
304 * @complexity O(n) over the input length.
305 * @alloc allocates the result.
306 * @systest RequestsSys.QueryParams
307 */
308inline auto percent_encode(builtins::Value auto&& text) {
309 auto unreserved = std::string("ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789-._~");
310 auto hexdigits = std::string("0123456789ABCDEF");
311 auto out = std::string("");
312 for (auto& ch : text) {
313 if (string::contains(unreserved, ch)) {
314 out += ch;
315 } else {
316 auto v = builtins::ord(ch);
317 ((out += "%") += builtins::index(hexdigits, builtins::floordiv(v, 16LL))) += builtins::index(hexdigits, builtins::mod(v, 16LL));
318 }
319 }
320 return out;
321}

413 * Parse one hex chunk-size token (e.g. "1aF").
414 *
415 * @param text the hex token, both digit cases accepted, no prefix.
416 * @return the decoded size, or -1 on a malformed digit. An empty token decodes to 0,
417 * which the chunked framing treats as the final chunk.
418 * @complexity O(n) over the token length.
419 * @alloc none.
420 * @systest RequestsSys.Chunked
421 */
422inline auto parse_hex(builtins::Value auto&& text) {
423 auto n = 0LL;
424 for (auto& ch : text) {
425 if ((n > 1099511627776LL)) {
426 return (-1LL);
427 }
428 auto v = builtins::ord(ch);
429 if (((v >= 48LL) && (v <= 57LL))) {
430 n = ((n * 16LL) + (v - 48LL));
431 } else {
432 if (((v >= 97LL) && (v <= 102LL))) {
433 n = ((n * 16LL) + (v - 87LL));
434 } else {
435 if (((v >= 65LL) && (v <= 70LL))) {
436 n = ((n * 16LL) + (v - 55LL));
437 } else {
438 return (-1LL);
439 }
440 }
441 }
442 }
443 return n;

477/**
478 * Decode an HTTP/1.1 chunked body.
479 *
480 * Framing: hex-size line (chunk extensions after `;` ignored), chunk bytes, CRLF,
481 * repeated until the 0 chunk. Trailers after the 0 chunk are simply unread.
482 * @param raw everything after the response head (the body bytes as received).
483 * @param r the Response under construction; `r.error` is set on malformed framing.
484 * @return the decoded body, or "" when framing was malformed (see `r.error`).
485 * @complexity O(n) over the body length.
486 * @alloc allocates the decoded body.
487 * @systest RequestsSys.Chunked
488 */
489inline auto dechunk(builtins::Value auto&& raw, builtins::Value auto&& r) {
490 auto body = std::string("");
491 auto pos = 0LL;
492 while (true) {
493 auto crlf = string::find(raw, std::string("\r\n"), pos);
494 if ((crlf < 0LL)) {
495 r.error = std::string("connection closed inside chunked body");
496 return std::string("");
497 }
498 auto line_end = (crlf - pos);
499 auto size_token = builtins::slice(raw, pos, (pos + line_end));
500 auto semi = string::find(size_token, std::string(";"));
501 if ((semi >= 0LL)) {
502 size_token = builtins::slice(size_token, 0LL, semi);
503 }
504 auto size = parse_hex(string::strip(size_token));
505 if ((size < 0LL)) {
506 r.error = std::string("malformed chunk size");
507 return std::string("");
508 }
509 (pos += line_end) += 2LL;
510 if ((size == 0LL)) {
511 return body;
512 }
513 if ((builtins::len(raw) < ((pos + size) + 2LL))) {
514 r.error = std::string("connection closed inside chunked body");
515 return std::string("");
516 }
517 body += builtins::slice(raw, pos, (pos + size));
518 (pos += size) += 2LL;
519 }
520}

523/**
524 * Read the whole response from a connected socket (or TLS session for https).
525 *
526 * Reads every byte until the peer closes — or the socket timeout set by request_once
527 * fires, which recv reports as "" (indistinguishable from close by design: we always
528 * request `Connection: close`, so EOF IS the end of the response). When @p conn is open
529 * the bytes are read (and decrypted) through its TLS session, mirroring the send path;
530 * a closed @p conn reads the plain socket.
531 * Stops once more than @p limit bytes have arrived so a malicious/compromised peer cannot
532 * stream an unbounded body and exhaust memory; the caller treats an over-limit read as an error.
533 * @param fd a connected socket descriptor from socket.tcp_connect.
534 * @param conn the owning tls.Conn for https (open), or a closed Conn for plain http.
535 * @param limit the maximum number of body bytes to accept before bailing out.
536 * @return the bytes received, bounded to at most one 64 KiB chunk beyond @p limit.
537 * @complexity O(n) over the response size (bounded by @p limit).
538 * @alloc allocates the received buffer (bounded by @p limit).
539 * @systest RequestsSys.BasicGet
540 */
541inline auto read_all(builtins::Value auto&& fd, builtins::Value auto&& conn, builtins::Value auto&& limit) {
542 auto raw = std::string("");
543 while (true) {
544 auto chunk = std::string("");
545 if (conn.is_open()) {
546 chunk = conn.recv(65536LL);
547 } else {
548 chunk = socket::recv(fd, 65536LL);
549 }
550 if ((builtins::len(chunk) == 0LL)) {
551 break;
552 }
553 raw += chunk;
554 if ((builtins::len(raw) > limit)) {
555 break;
556 }
557 }
558 return raw;
559}

572 * of throwing out of the request path; a Content-Length beyond @p limit is rejected outright.
573 * @param raw the complete response bytes as received.
574 * @param r the Response under construction (status_code/reason/headers/cookies/body/error).
575 * @param method the request method (so HEAD skips body framing).
576 * @param limit the maximum acceptable body size in bytes.
577 * @return `r`, completed or carrying `error`.
578 * @complexity O(n) over the response size.
579 * @alloc allocates the parsed headers and body.
580 * @systest RequestsSys.EofFraming
581 * @systest RequestsSys.HeaderLookup
582 */
583inline auto parse_response(builtins::Value auto&& raw, builtins::Value auto&& r, builtins::Value auto&& method, builtins::Value auto&& limit) {
584 auto head_end = string::find(raw, std::string("\r\n\r\n"));
585 if ((head_end < 0LL)) {
586 r.error = std::string("connection closed before a complete response head");
587 return r;
588 }
589 if ((string::find(raw, std::string("HTTP/")) != 0LL)) {
590 r.error = std::string("malformed response head");
591 return r;
592 }
593 auto head_block = builtins::slice(raw, 0LL, head_end);
594 auto line_end = string::find(head_block, std::string("\r\n"));
595 if ((line_end < 0LL)) {
596 line_end = builtins::len(head_block);
597 }
598 auto space = string::find(head_block, std::string(" "));
599 if (((space < 0LL) || ((space + 4LL) > line_end))) {
600 r.error = std::string("malformed status line");
601 return r;
602 }
603 auto code = parse_uint(builtins::slice(head_block, (space + 1LL), (space + 4LL)));
604 if ((code < 0LL)) {
605 r.error = std::string("malformed status code");
606 return r;
607 }
608 r.status_code = code;
609 if (((space + 5LL) <= line_end)) {
610 r.reason = string::strip(builtins::slice(head_block, (space + 5LL), line_end));
611 }
612 auto hend = builtins::len(head_block);
613 auto hpos = (line_end + 2LL);
614 while ((hpos < hend)) {
615 auto eol = string::find(head_block, std::string("\r\n"), hpos);
616 if ((eol < 0LL)) {
617 eol = hend;
618 }
619 auto line = builtins::slice(head_block, hpos, eol);
620 auto colon = string::find(line, std::string(":"));
621 if ((colon > 0LL)) {
622 auto name = string::lower(string::strip(builtins::slice(line, 0LL, colon)));
623 auto value = string::strip(builtins::slice(line, (colon + 1LL), builtins::slice_end));
624 r.headers[name] = value;
625 if ((name == std::string("set-cookie"))) {
626 auto semi = string::find(value, std::string(";"));
627 auto pair = value;
628 if ((semi >= 0LL)) {
629 pair = builtins::slice(value, 0LL, semi);
630 }
631 auto eq = string::find(pair, std::string("="));
632 if ((eq > 0LL)) {
633 r.cookies[string::strip(builtins::slice(pair, 0LL, eq))] = string::strip(builtins::slice(pair, (eq + 1LL), builtins::slice_end));
634 }
635 }
636 }
637 if ((eol == hend)) {
638 break;
639 }
640 hpos = (eol + 2LL);
641 }
642 if ((method == std::string("HEAD"))) {
643 r.body = std::string("");
644 return r;
645 }
646 auto body = builtins::slice(raw, (head_end + 4LL), builtins::slice_end);
647 if ((r.header(std::string("transfer-encoding")) == std::string("chunked"))) {
648 r.body = dechunk(body, r);
649 return r;
650 }
651 auto cl = r.header(std::string("content-length"));
652 if ((cl != std::string(""))) {
653 auto n = parse_uint(cl);
654 if ((n < 0LL)) {
655 r.error = std::string("invalid Content-Length");
656 return r;
657 }
658 if ((n > limit)) {
659 r.error = std::string("Content-Length exceeds max_bytes");
660 return r;
661 }
662 if ((builtins::len(body) < n)) {
663 r.error = std::string("connection closed before the complete body arrived");
664 return r;
665 }
666 r.body = builtins::slice(body, 0LL, n);
667 } else {
668 r.body = body;
669 }
670 return r;
671}

674/**
675 * One request exchange against an already-parsed URL (no redirect handling).
676 *
677 * Connects, applies the timeout, appends `o.params` percent-encoded to the request-target,
678 * builds the body (json_body -> data -> body; GET/HEAD send none), sends the request (custom
679 * headers, auto Content-Type/Content-Length/Authorization, default User-Agent, `Connection:
680 * close`), then reads and parses the full response. Transport failures come back in `r.error`.
681 * @param method the HTTP method ("GET", "POST", …).
682 * @param u the parsed URL (scheme/host/port/target) from parsers.url.
683 * @param o per-request options (timeout, params, headers, body, auth).
684 * @param r the Response under construction.
685 * @return the completed Response (a non-2xx status is a completed exchange, not an error).
686 * @complexity O(request + response bytes) (+ one TLS handshake for https).
687 * @alloc allocates the request and response buffers (+ a `tls` session for https).
688 * @concurrency blocking — connect/handshake/read are all bounded by the socket timeout
689 * (`o.timeout_ms`, default 30 s).
690 * @systest RequestsSys.PostJson
691 */
692inline auto request_once(builtins::Value auto&& method, builtins::Value auto&& u, builtins::Value auto&& o, builtins::Value auto&& r) {
693 auto fd = socket::tcp_connect(u.host, u.port);
694 if ((fd < 0LL)) {
695 r.error = (((((std::string("connect to ") + builtins::str(u.host)) + std::string(":")) + builtins::str(u.port)) + std::string(" failed: ")) + builtins::str(socket::last_error()));
696 return r;
697 }
698 auto timeout = o.timeout_ms;
699 if ((timeout <= 0LL)) {
700 timeout = 30000LL;
701 }
702 socket::set_timeout(fd, timeout);
703 auto conn = tls::Conn();
704 if ((u.scheme == std::string("https"))) {
705 conn = tls::open(fd, u.host, o.insecure, o.ca_file);
706 if ((!conn.is_open())) {
707 socket::close(fd);
708 r.error = (std::string("tls: ") + builtins::str(tls::last_error()));
709 return r;
710 }
711 }
712 auto body = std::string("");
713 auto ctype = std::string("");
714 if (((method != std::string("GET")) && (method != std::string("HEAD")))) {
715 if ((o.json_body != std::string(""))) {
716 body = o.json_body;
717 ctype = std::string("application/json");
718 } else {
719 if ((builtins::len(o.data) > 0LL)) {
720 auto sep = std::string("");
721 for (auto& kv : o.data) {
722 (((body += sep) += percent_encode(kv.first)) += "=") += percent_encode(kv.second);
723 sep = std::string("&");
724 }
725 ctype = std::string("application/x-www-form-urlencoded");
726 } else {
727 if ((o.body != std::string(""))) {
728 body = o.body;
729 }
730 }
731 }
732 }
733 auto target = u.target;
734 auto sep = std::string("?");
735 if ((string::find(target, std::string("?")) >= 0LL)) {
736 sep = std::string("&");
737 }
738 for (auto& kv : o.params) {
739 (((target += sep) += percent_encode(kv.first)) += "=") += percent_encode(kv.second);
740 sep = std::string("&");
741 }
742 if ((has_control_bytes(target) || has_control_bytes(u.host))) {
743 conn.close();
744 socket::close(fd);
745 r.error = std::string("refused: control bytes in the request target");
746 return r;
747 }
748 auto req = (((((((builtins::str(method) + std::string(" ")) + builtins::str(target)) + std::string(" HTTP/1.1\r\nHost: ")) + builtins::str(u.host)) + std::string(":")) + builtins::str(u.port)) + std::string("\r\n"));
749 for (auto& kv : o.headers) {
750 if ((has_control_bytes(kv.first) || has_control_bytes(kv.second))) {
751 conn.close();
752 socket::close(fd);
753 r.error = std::string("refused: control bytes in a request header");
754 return r;
755 }
756 (((req += kv.first) += ": ") += kv.second) += "\r\n";
757 }
758 if ((!has_header(o.headers, std::string("User-Agent")))) {
759 req += "User-Agent: cheatah-requests/1.2\r\n";
760 }
761 if (((o.auth_user != std::string("")) && (!has_header(o.headers, std::string("Authorization"))))) {
762 ((req += "Authorization: Basic ") += hashlib::base64_encode(((builtins::str(o.auth_user) + std::string(":")) + builtins::str(o.auth_pass)))) += "\r\n";
763 }
764 if (((ctype != std::string("")) && (!has_header(o.headers, std::string("Content-Type"))))) {
765 ((req += "Content-Type: ") += ctype) += "\r\n";
766 }
767 if ((!has_header(o.headers, std::string("Content-Length")))) {
768 if (((((builtins::len(body) > 0LL) || (method == std::string("POST"))) || (method == std::string("PUT"))) || (method == std::string("PATCH")))) {
769 ((req += "Content-Length: ") += builtins::str(builtins::len(body))) += "\r\n";
770 }
771 }
772 (req += "Connection: close\r\nAccept: */*\r\n\r\n") += body;
773 auto sent = 0LL;
774 if (conn.is_open()) {
775 sent = conn.send(req);
776 } else {
777 sent = socket::sendall(fd, req);
778 }
779 if ((sent != 0LL)) {
780 conn.close();
781 socket::close(fd);
782 r.error = std::string("send failed");
783 return r;
784 }
785 auto limit = o.max_bytes;
786 if ((limit <= 0LL)) {
787 limit = 104857600LL;
788 }
789 auto raw = read_all(fd, conn, limit);
790 conn.close();
791 socket::close(fd);
792 if ((builtins::len(raw) > limit)) {
793 r.error = ((std::string("response body exceeds max_bytes (") + builtins::str(limit)) + std::string(")"));
794 return r;
795 }
796 return parse_response(raw, r, method, limit);
797}

800/**
801 * Strip credentials and cookies from `o` — used when a redirect crosses to a different host so
802 * secrets scoped to the original host are never sent to another (the classic cross-origin
803 * redirect credential leak). Clears Basic auth and drops any `Authorization`/`Cookie` header.
804 *
805 * @param o the (per-request, already-copied) options to sanitize in place.
806 * @return nothing — @p o is modified in place (auth cleared, sensitive headers dropped).
807 * @complexity O(k) over the number of headers.
808 * @alloc allocates the rebuilt header map.
809 * @test CheatahRequests.CrossHostRedirectStripsCredentials
810 * @test CheatahRequests.SameHostRedirectKeepsCredentials
811 */
812inline auto strip_sensitive(builtins::Value auto&& o) {
813 o.auth_user = std::string("");
814 o.auth_pass = std::string("");
815 std::unordered_map<std::string, std::string> clean;
816 for (auto& kv : o.headers) {
817 auto lname = string::lower(kv.first);
818 if (((lname != std::string("authorization")) && (lname != std::string("cookie")))) {
819 clean[kv.first] = kv.second;
820 }
821 }
822 o.headers = clean;
823}

826/**
827 * Perform an HTTP request, following up to max_redirects 3xx hops unless no_redirect.
828 *
829 * On a redirect to a DIFFERENT host, Basic-auth credentials and any `Authorization`/`Cookie`
830 * header are stripped before the next hop, so secrets are never leaked to another origin.
831 * Never raises for network conditions: every failure comes back as a Response with `error`
832 * set (and status_code 0). Redirects (301/302/303/307/308) follow absolute and host-relative
833 * Location targets, recording each hop in the returned Response's `history`; 303 (and 301/302
834 * on a POST) switch the method to GET and drop the body, matching Python. Set
835 * `o.no_redirect = true` to return the 3xx response directly.
836 * @param method the HTTP method ("GET", "POST", …).
837 * @param url the absolute `http(s)://host[:port]/path[?query]` URL.
838 * @param o per-request options; defaults to a 30 s timeout and 5 redirect hops (redirects followed unless no_redirect).
839 * @return the final Response — check `ok()`, then `status_code`/`headers`/`body`.
840 * @complexity one full exchange (request_once) per hop, at most 1 + max_redirects hops.
841 * @alloc allocates each hop's request/response buffers, the recorded `history`, and a
842 * private copy of @p o (so redirect-time credential stripping never mutates the caller's).
843 * @concurrency blocking, with every hop's socket I/O bounded by `timeout_ms`; no shared
844 * state — concurrent requests from separate threads are independent.
845 * @systest RequestsSys.BasicGet
846 * @systest RequestsSys.Redirect
847 * @systest RequestsSys.RedirectLoop
848 */
849inline auto request(builtins::Value auto&& method, builtins::Value auto&& url, builtins::Value auto&& o) {
850 auto opts = o;
851 auto max_hops = opts.max_redirects;
852 if ((max_hops <= 0LL)) {
853 max_hops = 5LL;
854 }
855 auto current = url;
856 auto cur_method = method;
857 auto hop = 0LL;
858 auto origin_host = std::string("");
859 std::vector<Response> hist;
860 while ((hop <= max_hops)) {
861 auto r = Response{.url = static_cast<std::string>(current)};
862 auto parser = parsers::url::Parser();
863 auto u = parsers::url::Url();
864 if ((!parser.parse(current, u))) {
865 r.error = (std::string("malformed URL: ") + builtins::str(current));
866 r.history = hist;
867 return r;
868 }
869 if ((origin_host == std::string(""))) {
870 origin_host = u.host;
871 } else {
872 if ((u.host != origin_host)) {
873 strip_sensitive(opts);
874 }
875 }
876 r = request_once(cur_method, u, opts, r);
877 auto redirect = ((r.error == std::string("")) && (((((r.status_code == 301LL) || (r.status_code == 302LL)) || (r.status_code == 303LL)) || (r.status_code == 307LL)) || (r.status_code == 308LL)));
878 if ((opts.no_redirect || (!redirect))) {
879 r.history = hist;
880 return r;
881 }
882 auto loc = r.header(std::string("location"));
883 if ((loc == std::string(""))) {
884 r.error = ((std::string("redirect (") + builtins::str(r.status_code)) + std::string(") without a Location header"));
885 r.history = hist;
886 return r;
887 }
888 auto next = std::string("");
889 if ((string::startswith(loc, std::string("http://")) || string::startswith(loc, std::string("https://")))) {
890 next = loc;
891 } else {
892 if (string::startswith(loc, std::string("/"))) {
893 next = (((((builtins::str(u.scheme) + std::string("://")) + builtins::str(u.host)) + std::string(":")) + builtins::str(u.port)) + builtins::str(loc));
894 } else {
895 r.error = (std::string("unsupported relative redirect Location: ") + builtins::str(loc));
896 r.history = hist;
897 return r;
898 }
899 }
900 builtins::append(hist, r);
901 {
902 auto __match_1 = r.status_code;
903 switch (__match_1) {
904 case 303LL: {
905 cur_method = std::string("GET");
906 break;
907 }
908 case 301LL: {
909 if ((cur_method == std::string("POST"))) {
910 cur_method = std::string("GET");
911 }
912 break;
913 }
914 case 302LL: {
915 if ((cur_method == std::string("POST"))) {
916 cur_method = std::string("GET");
917 }
918 break;
919 }
920 default: {
921 break;
922 }
923 }
924 }
925 current = next;
926 hop += 1LL;
927 }
928 auto r = Response{.url = static_cast<std::string>(current)};
929 r.error = ((std::string("too many redirects (max_redirects = ") + builtins::str(max_hops)) + std::string(")"));
930 r.history = hist;
931 return r;
932}
